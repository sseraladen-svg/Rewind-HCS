   490→  // Sprocket holes — right side
   491→  drawSprocketHoles(
   492→    ctx,
   493→    stripW - 28,
   494→    filmBorder,
   495→    canvas.height - filmBorder,
   496→    20,
   497→    sprocketH,
   498→    sprocketGap,
   499→    '#2a2018',
   500→  );
   501→
   502→  // Draw each photo frame
   503→  for (let i = 0; i < photos.length; i++) {
   504→    const y = filmBorder + i * (photoH + frameGap);
   505→
   506→    // Frame number
   507→    ctx.save();
   508→    ctx.globalAlpha = 0.35;
   509→    ctx.font = '9px monospace';
   510→    ctx.fillStyle = '#d4a574';
   511→    ctx.textAlign = 'right';
   512→    ctx.fillText(`${String(i + 1).padStart(2, '0')}A`, stripW - 6, y + photoH - 4);
   513→    ctx.restore();
   514→
   515→    // Photo
   516→    const tmp = document.createElement('canvas');
   517→    tmp.width = photoW;
   518→    tmp.height = photoH;
   519→    const tmpCtx = tmp.getContext('2d')!;
   520→
   521→    const img = await loadImage(photos[i].url);
   522→    const scale = Math.max(photoW / img.width, photoH / img.height);
   523→    const sw = photoW / scale;
   524→    const sh = photoH / scale;
   525→    const sx = (img.width - sw) / 2;
   526→    const sy = (img.height - sh) / 2;
   527→    tmpCtx.drawImage(img, sx, sy, sw, sh, 0, 0, photoW, photoH);
   528→
   529→    // Apply kodak filter
   530→    processFrame(tmpCtx, photoW, photoH, 'kodak');
   531→
   532→    ctx.drawImage(tmp, filmBorder, y, photoW, photoH);
   533→
   534→    // Frame divider line
   535→    if (i < photos.length - 1) {
   536→      ctx.save();
   537→      ctx.strokeStyle = 'rgba(200,180,140,0.15)';
   538→      ctx.lineWidth = 1;
   539→      ctx.beginPath();
   540→      ctx.moveTo(filmBorder, y + photoH + frameGap / 2);
   541→      ctx.lineTo(stripW - filmBorder, y + photoH + frameGap / 2);
   542→      ctx.stroke();
   543→      ctx.restore();
   544→    }
   545→  }
   546→
   547→  drawWatermark(ctx, stripW, canvas.height);
   548→  return canvas;
   549→}
