   177→interface MatchCard {
   178→  id: number;
   179→  emoji: string;
   180→  label: string;
   181→  isFlipped: boolean;
   182→  isMatched: boolean;
   183→}
   184→
   185→function shuffle<T>(arr: T[]): T[] {
   186→  const a = [...arr];
   187→  for (let i = a.length - 1; i > 0; i--) {
   188→    const j = Math.floor(Math.random() * (i + 1));
   189→    [a[i], a[j]] = [a[j], a[i]];
   190→  }
   191→  return a;
   192→}
   193→
   194→function createMatchCards(data: DecadeData): MatchCard[] {
   195→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   196→  const picked = shuffle(pool).slice(0, 6);
   197→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   198→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   199→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   200→  ]);
   201→  return shuffle(initial);
   202→}

   204→function MemoryMatchGame({ data }: { data: DecadeData }) {
   205→  const [cards, setCards] = useState<MatchCard[]>([]);
   206→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   207→  const [moves, setMoves] = useState(0);
   208→  const [matches, setMatches] = useState(0);
   209→  const [gameWon, setGameWon] = useState(false);
   210→  const lockRef = useRef(false);
   211→
   212→  useEffect(() => {
   213→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   214→    return () => cancelAnimationFrame(id);
   215→  }, [data]);
   216→
   217→  const resetGame = useCallback(() => {
   218→    setCards(createMatchCards(data));
   219→    setFlippedIds([]);
   220→    setMoves(0);
   221→    setMatches(0);
   222→    setGameWon(false);
   223→    lockRef.current = false;
   224→  }, [data]);
   225→
   226→  const handleFlip = useCallback((id: number) => {
   227→    if (lockRef.current) return;
   228→    setCards(prev => {
   229→      const card = prev.find(c => c.id === id);
   230→      if (!card || card.isFlipped || card.isMatched) return prev;
   231→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   232→    });
   233→
   234→    setFlippedIds(prev => {
   235→      if (prev.includes(id) || prev.length >= 2) return prev;
   236→      const newFlipped = [...prev, id];
   237→
   238→      if (newFlipped.length === 2) {
   239→        setMoves(m => m + 1);
   240→        lockRef.current = true;
   241→
   242→        // Use a timeout to read the latest cards state
   243→        setTimeout(() => {
   244→          setCards(currentCards => {
   245→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   246→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   247→
   248→            if (first.label === second.label) {
   249→              // Match!
   250→              const updated = currentCards.map(c =>
   251→                c.id === first.id || c.id === second.id
   252→                  ? { ...c, isMatched: true, isFlipped: true }
   253→                  : c
   254→              );
   255→              setMatches(m => {
   256→                const newM = m + 1;
   257→                if (newM === 6) setGameWon(true);
   258→                return newM;
   259→              });
   260→              setFlippedIds([]);
   261→              lockRef.current = false;
   262→              return updated;
   263→            } else {
   264→              // No match — flip back
   265→              setTimeout(() => {
   266→                setCards(cc => cc.map(c =>
   267→                  c.id === first.id || c.id === second.id
   268→                    ? { ...c, isFlipped: false }
   269→                    : c
   270→                ));
   271→                setFlippedIds([]);
   272→                lockRef.current = false;
   273→              }, 700);
   274→              return currentCards;
   275→            }
   276→          });
   277→        }, 50);
   278→      }
   279→
   280→      return newFlipped;
   281→    });
   282→  }, []);
   283→
   284→  return (
   285→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   286→      <div className="flex items-center justify-between mb-4">
   287→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   288→          <span className="text-xl">🃏</span> Memory Match
   289→        </h4>
   290→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   291→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   292→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   293→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   294→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   295→          </button>
   296→        </div>
   297→      </div>
   298→
   299→      {gameWon && (
   300→        <motion.div
   301→          initial={{ scale: 0.8, opacity: 0 }}
   302→          animate={{ scale: 1, opacity: 1 }}
   303→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   304→        >
   305→          <span className="text-5xl mb-2">🎉</span>
   306→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   307→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   308→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   309→            Play Again
   310→          </button>
   311→        </motion.div>
   312→      )}
   313→
   314→      <div className="grid grid-cols-4 gap-2">
   315→        {cards.map((card) => {
   316→          const showFront = card.isFlipped || card.isMatched;
   317→          return (
   318→            <button
   319→              key={card.id}
   320→              onClick={() => handleFlip(card.id)}
   321→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   322→              disabled={card.isMatched}
   323→              style={{
   324→                background: showFront
   325→                  ? '#faf6eb'
   326→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   327→                borderColor: showFront
   328→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')
   329→                  : '#5c4033',
   330→                boxShadow: showFront
   331→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   332→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   333→                opacity: card.isMatched ? 0.6 : 1,
   334→              }}
   335→            >
   336→              {showFront ? (
   337→                <>
   338→                  <span className="text-2xl md:text-3xl">{card.emoji}</span>
   339→                  <span className="old-typewriter text-[7px] md:text-[9px] font-bold text-[#5c4033] mt-0.5 text-center leading-tight" style={{ maxWidth: '100%', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{card.label}</span>
   340→                </>
   341→              ) : (
   342→                <span className="text-xl text-[#d4a574] font-bold old-title">?</span>
   343→              )}
   344→            </button>
   345→          );
   346→        })}
   347→      </div>
   348→    </div>
   349→  );
   350→}

   352→/* ═══════════════════════════════════════════════════
   353→   GAME 2: GUESS THE YEAR
   354→   ═══════════════════════════════════════════════════ */
   355→interface TriviaQ {
   356→  question: string;
   357→  emoji: string;
   358→  correct: number;
   359→  options: string[];
   360→}
   361→
   362→function buildTrivia(data: DecadeData): TriviaQ[] {
   363→  const decadeStart = parseInt(data.years.split('–')[0]);
   364→  const qs: TriviaQ[] = [];
   365→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   366→  const shuffled = shuffle(all).slice(0, 8);
   367→
   368→  for (const item of shuffled) {
   369→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   370→    const wrongYears = [
   371→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   372→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   373→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   374→    ].filter(y => y !== realYear).slice(0, 3);
   375→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   376→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   377→    qs.push({
   378→      question: `When did "${item.title}" first become popular?`,
   379→      emoji: item.emoji,
   380→      correct: options.indexOf(realYear),
   381→      options: options.map(String),
   382→    });
   383→  }
   384→  return qs;
   385→}

   387→function GuessTheYearGame({ data }: { data: DecadeData }) {
   388→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   389→  const [qIdx, setQIdx] = useState(0);
   390→  const [score, setScore] = useState(0);
   391→  const [selected, setSelected] = useState<number | null>(null);
   392→  const [showResult, setShowResult] = useState(false);
   393→  const [done, setDone] = useState(false);
   394→
   395→  useEffect(() => {
   396→    setQuestions(buildTrivia(data));
   397→  }, [data]);
   398→
   399→  const q = questions[qIdx];
   400→
   401→  const handleAnswer = (optIdx: number) => {
   402→    if (showResult || !q) return;
   403→    setSelected(optIdx);
   404→    setShowResult(true);
   405→    if (optIdx === q.correct) setScore(s => s + 1);
   406→    setTimeout(() => {
   407→      if (qIdx + 1 < questions.length) {
   408→        setQIdx(i => i + 1);
   409→        setSelected(null);
   410→        setShowResult(false);
   411→      } else {
   412→        setDone(true);
   413→      }
   414→    }, 1500);
   415→  };
   416→
   417→  const resetGame = () => {
   418→    const fresh = buildTrivia(data);
   419→    setQuestions(fresh);
   420→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   421→  };
   422→
   423→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   424→
   425→  if (done) {
   426→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   427→    return (
   428→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   429→        <div className="text-center py-4">
   430→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   431→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
   432→          <p className="old-body text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   433→          <p className="old-quote text-xs text-[#8b7355] mt-2">
   434→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   435→          </p>
   436→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   437→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   438→          </button>
   439→        </div>
   440→      </div>
   441→    );
   442→  }
   443→
   444→  return (
   445→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   446→      <div className="flex items-center justify-between mb-4">
   447→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   448→          <span className="text-xl">📅</span> Guess the Year
   449→        </h4>
   450→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   451→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{qIdx + 1}/{questions.length}</span>
   452→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Score: {score}</span>
   453→        </div>
   454→      </div>
   455→
   456→      {/* Progress bar */}
   457→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   458→        <div
   459→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   460→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   461→        />
   462→      </div>
   463→
   464→      <div className="text-center mb-4">
   465→        <span className="text-5xl block mb-2">{q.emoji}</span>
   466→        <p className="old-body text-sm font-semibold text-[#5c4033]">{q.question}</p>
   467→      </div>
   468→
   469→      <div className="grid grid-cols-2 gap-2">
   470→        {q.options.map((opt, i) => {
   471→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 old-typewriter ';
   472→          if (showResult) {
   473→            if (i === q.correct) cls += 'border-green-500 bg-green-50 sk-sticker';
   474→            else if (i === selected) cls += 'border-red-400 bg-red-50 opacity-60';
   475→            else cls += 'border-[#d4c5a9] opacity-40';
   476→          } else {
   477→            cls += 'border-[#d4c5a9] hover:border-[#8b5e3c] hover:shadow-md active:translate-y-0.5';
   478→          }
   479→          return (
   480→            <button key={`${qIdx}-${i}`} onClick={() => handleAnswer(i)} className={cls} disabled={showResult}>
   481→              {opt}
   482→            </button>
   483→          );
   484→        })}
   485→      </div>
   486→
   487→      {showResult && (
   488→        <motion.div
   489→          initial={{ opacity: 0, y: 5 }}
   490→          animate={{ opacity: 1, y: 0 }}

   505→   GAME 3: TAP THE MEMORY (Reaction Game)
   506→   ═══════════════════════════════════════════════════ */
   507→function WhackGame({ data }: { data: DecadeData }) {
   508→  const GRID = 9;
   509→  const allItems = useMemo(() => [
   510→    ...data.toys, ...data.snacks, ...data.cartoons, ...data.games,
   511→  ], [data]);
   512→
   513→  const [grid, setGrid] = useState<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   514→  const [score, setScore] = useState(0);
   515→  const [timeLeft, setTimeLeft] = useState(20);
   516→  const [gameActive, setGameActive] = useState(false);
   517→  const [gameOver, setGameOver] = useState(false);
   518→  const [feedback, setFeedback] = useState<string | null>(null);
   519→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   520→  const spawnRef = useRef<ReturnType<typeof setInterval> | null>(null);
   521→  const gridRef = useRef<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   522→  const activeRef = useRef(false);
   523→
   524→  const spawnItem = useCallback(() => {
   525→    if (!activeRef.current) return;
   526→    const empty = gridRef.current.reduce<number[]>((acc, v, i) => v === null ? [...acc, i] : acc, []);
   527→    if (empty.length === 0) return;
   528→    const pos = empty[Math.floor(Math.random() * empty.length)];
   529→    const item = allItems[Math.floor(Math.random() * allItems.length)];
   530→    const newGrid = [...gridRef.current];
   531→    newGrid[pos] = item;
   532→    gridRef.current = newGrid;
   533→    setGrid([...newGrid]);
   534→    setTimeout(() => {
   535→      if (gridRef.current[pos] === item) {
   536→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);
   537→      }
   538→    }, 1200);
   539→  }, [allItems]);
   540→
   541→  const stopGame = useCallback(() => {
   542→    activeRef.current = false;
   543→    if (timerRef.current) { clearInterval(timerRef.current); timerRef.current = null; }
   544→    if (spawnRef.current) { clearInterval(spawnRef.current); spawnRef.current = null; }
   545→  }, []);
   546→
   547→  const startGame = useCallback(() => {
   548→    // Clear any previous intervals
   549→    stopGame();
   550→
   551→    const fresh = Array(GRID).fill(null);
   552→    gridRef.current = fresh;
   553→    setGrid([...fresh]);
   554→    setScore(0); setTimeLeft(20); setGameOver(false); setGameActive(true);
   555→    activeRef.current = true;
   556→
   557→    timerRef.current = setInterval(() => {
   558→      if (!activeRef.current) return;
   559→      setTimeLeft(t => {
   560→        if (t <= 1) {
   561→          stopGame();
   562→          setGameActive(false);
   563→          setGameOver(true);
   564→          return 0;
   565→        }
   566→        return t - 1;
   567→      });
   568→    }, 1000);
   569→    spawnRef.current = setInterval(spawnItem, 700);
   570→  }, [spawnItem, stopGame]);
   571→
   572→  useEffect(() => {
   573→    return () => {
   574→      activeRef.current = false;
   575→      if (timerRef.current) clearInterval(timerRef.current);
   576→      if (spawnRef.current) clearInterval(spawnRef.current);
   577→    };
   578→  }, []);
   579→
   580→  const handleWhack = (idx: number) => {
   581→    if (!gameActive || !grid[idx]) return;
   582→    setScore(s => s + 10);
   583→    setFeedback('+10');
   584→    const g = [...gridRef.current]; g[idx] = null; gridRef.current = g; setGrid([...g]);
   585→    setTimeout(() => setFeedback(null), 300);
   586→  };
   587→
   588→  const handleMiss = () => {
   589→    if (!gameActive) return;
   590→    setFeedback('Miss!');
   591→    setTimeout(() => setFeedback(null), 300);
   592→  };
   593→
   594→  return (
   595→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   596→      <div className="flex items-center justify-between mb-4">
   597→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   598→          <span className="text-xl">🔨</span> Tap the Memory
   599→        </h4>
   600→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   601→          <span className="sk-inset px-2 py-1 rounded old-typewriter flex items-center gap-1">
   602→            <Timer className="h-3 w-3" />{timeLeft}s
   603→          </span>
   604→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{score}pts</span>
   605→        </div>
   606→      </div>
   607→
   608→      {!gameActive && !gameOver && (
   609→        <div className="text-center py-6">
   610→          <p className="old-body text-sm text-[#8b7355] mb-4">Tap nostalgic items as they pop up! You have 20 seconds.</p>
   611→          <button onClick={startGame} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   612→            <Play className="h-5 w-5" /> Start Game
   613→          </button>
   614→        </div>
   615→      )}
   616→
   617→      {gameOver && (
   618→        <div className="text-center py-4">
   619→          <span className="text-5xl mb-2 block">{score >= 80 ? '🏆' : score >= 40 ? '🎉' : '💪'}</span>
   620→          <p className="old-title text-lg font-bold text-[#5c4033]">Time&apos;s Up!</p>
   621→          <p className="old-body text-sm text-[#8b7355] mt-1">Score: {score} points</p>
   622→          <button onClick={startGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   623→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   624→          </button>
   625→        </div>
   626→      )}
   627→
   628→      {gameActive && (
   629→        <div className="grid grid-cols-3 gap-2 select-none">
   630→          {grid.map((item, i) => (
   631→            <button
   632→              key={i}
   633→              onClick={() => item ? handleWhack(i) : handleMiss()}
   634→              className="aspect-square rounded-lg transition-all duration-150 cursor-pointer border-2 flex items-center justify-center"
   635→              style={{
   636→                background: item
   637→                  ? `linear-gradient(135deg, ${data.color}18, ${data.color}30)`
   638→                  : 'linear-gradient(135deg, #e8dcc8, #d4c5a9)',
   639→                borderColor: item ? `${data.color}60` : '#c4b594',
   640→                boxShadow: item
   641→                  ? `0 2px 8px ${data.color}30, inset 0 1px 0 rgba(255,255,255,0.3)`
   642→                  : 'inset 0 2px 4px rgba(0,0,0,0.08)',
   643→                transform: item ? 'scale(1.05)' : 'scale(1)',
   644→              }}
   645→            >
   646→              {item && (
   647→                <motion.span
   648→                  initial={{ scale: 0 }}
   649→                  animate={{ scale: 1 }}
   650→                  className="text-3xl md:text-4xl"
   651→                >
   652→                  {item.emoji}
   653→                </motion.span>
   654→              )}
   655→            </button>
   656→          ))}
   657→        </div>
   658→      )}
   659→
   660→      {feedback && (
   661→        <motion.div
   662→          key={feedback}
   663→          initial={{ opacity: 0, y: 10 }}
   664→          animate={{ opacity: 1, y: 0 }}
   665→          className="absolute bottom-3 left-1/2 -translate-x-1/2"
   666→        >
   667→          <span className={`text-xs font-bold px-3 py-1 rounded-full sk-btn old-typewriter ${feedback === 'Miss!' ? 'opacity-50' : ''}`}>
   668→            {feedback}
   669→          </span>
   670→        </motion.div>
