   132→   132→// useStoredState — localStorage-backed state via useSyncExternalStore
   133→   133→// ═══════════════════════════════════════════════════════════════
   134→   134→
   135→   135→function useStoredState<T>(
   136→   136→  key: string,
   137→   137→  initial: T,
   138→   138→): [T, (updater: T | ((prev: T) => T)) => void] {
   139→   139→  const subscribe = useCallback(
   140→   140→    (onStoreChange: () => void) => {
   141→   141→      const handler = (e: StorageEvent) => {
   142→   142→        if (e.key === key || e.key === null) onStoreChange();
   143→   143→      };
   144→   144→      window.addEventListener('storage', handler);
   145→   145→      return () => window.removeEventListener('storage', handler);
   146→   146→    },
   147→   147→    [key],
   148→   148→  );
   149→   149→
   150→   150→  const getSnapshot = useCallback(() => {
   151→   151→    return localStorage.getItem(key);
   152→   152→  }, [key]);
   153→   153→
   154→   154→  const getServerSnapshot = useCallback(() => null, []);
   155→   155→
   156→   156→  const raw = useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
   157→   157→
   158→   158→  const value = useMemo<T>(() => {
   159→   159→    if (raw === null) return initial;
   160→   160→    try {
   161→   161→      return JSON.parse(raw) as T;
   162→   162→    } catch {
   163→   163→      return initial;
   164→   164→    }
   165→   165→  }, [raw, initial]);
   166→   166→
   167→   167→  const setValue = useCallback(
   168→   168→    (updater: T | ((prev: T) => T)) => {
   169→   169→      try {
   170→   170→        const currentRaw = localStorage.getItem(key);
   171→   171→        const current: T =
   172→   172→          currentRaw !== null ? (JSON.parse(currentRaw) as T) : initial;
   173→   173→        const next =
   174→   174→          typeof updater === 'function'
   175→   175→            ? (updater as (prev: T) => T)(current)
   176→   176→            : updater;
   177→   177→        localStorage.setItem(key, JSON.stringify(next));
   178→   178→        window.dispatchEvent(new StorageEvent('storage', { key }));
   179→   179→      } catch {
   180→   180→        /* ignore storage errors */
   181→   181→      }
   182→   182→    },
   183→   183→    [key, initial],
   184→   184→  );
   185→   185→
   186→   186→  return [value, setValue];
   187→   187→}

   190→   190→// useVisitCounter — auto-incrementing visit counter
   191→   191→// ═══════════════════════════════════════════════════════════════
   192→   192→
   193→   193→// Only called when isClient is true, so localStorage is available.
   194→   194→// The useState lazy initializer runs exactly once on first mount.
   195→   195→function useVisitCounter(): number {
   196→   196→  const [count] = useState(() => {
   197→   197→    const stored = localStorage.getItem('rewind_visit_count');
   198→   198→    let c = 1;
   199→   199→    if (stored) {
   200→   200→      const parsed = parseInt(stored, 10);
   201→   201→      if (!isNaN(parsed)) c = parsed + 1;
   202→   202→    }
   203→   203→    localStorage.setItem('rewind_visit_count', String(c));
   204→   204→    return c;
   205→   205→  });
   206→   206→  return count;
   207→   207→}
