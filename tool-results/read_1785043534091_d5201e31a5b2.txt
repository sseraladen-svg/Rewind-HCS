   726→      <motion.div
   727→        variants={staggerItem}
   728→        className="flex items-center gap-3 mt-5 w-full max-w-xs"
   729→      >
   730→        <motion.button
   731→          onClick={onBack}
   732→          whileHover={{ scale: 1.05 }}
   733→          whileTap={{ scale: 0.95 }}
   734→          className="sk-btn old-btn rounded-lg px-4 py-2.5 text-sm cursor-pointer flex items-center gap-1"
   735→        >
   736→          <ChevronLeft className="h-4 w-4" />
   737→          Back
   738→        </motion.button>
   739→        <motion.button
   740→          whileHover={{ scale: 1.03, y: -1 }}
   741→          whileTap={{ scale: 0.97 }}
   742→          onClick={onNext}
   743→          className="sk-leather old-btn flex-1 text-[#faf6eb] font-bold text-base py-2.5 rounded-lg cursor-pointer transition-all duration-200 hover:shadow-[0_4px_12px_rgba(0,0,0,0.3),inset_0_1px_0_rgba(255,255,255,0.15)] active:translate-y-0.5 active:shadow-[inset_0_2px_4px_rgba(0,0,0,0.3)] flex items-center justify-center gap-2"
   744→        >
   745→          <PartyPopper className="h-4 w-4" />
   746→          Start Rewinding!
   747→        </motion.button>
   748→      </motion.div>
   749→    </motion.div>
   750→  );
   751→}
   752→
   753→/* ═══════════════════════════════════════════════════════════════
   754→   Main OnboardingFlow Component
   755→   ═══════════════════════════════════════════════════════════════ */
   756→
   757→export default function OnboardingFlow({ onComplete }: OnboardingFlowProps) {
   758→  const [currentStep, setCurrentStep] = useState(0);
   759→  const [direction, setDirection] = useState<number>(STEP_DIRECTION.forward);
   760→  const [birthYear, setBirthYearState] = useState('');
   761→  const [selectedEra, setSelectedEra] = useState<string | null>(null);
   762→  const [currentFeature, setCurrentFeature] = useState(0);
   763→  const [isLoaded, setIsLoaded] = useState(false);
   764→  const [isCompleting, setIsCompleting] = useState(false);
   765→
   766→  /* ───── Check IndexedDB on mount ───── */
   767→  useEffect(() => {
   768→    let cancelled = false;
   769→
   770→    (async () => {
   771→      try {
   772→        const db = await getDB();
   773→        if (cancelled) return;
   774→
   775→        const record: any = await db.get('era_skin', ONBOARDING_ID);
   776→        if (cancelled) return;
   777→
   778→        if (record?.complete) {
   779→          // Already onboarded — restore state and skip
   780→          const savedBirthYear = record.birthYear ?? null;
   781→          const savedEra = record.eraSkin ?? null;
   782→
   783→          if (savedBirthYear && savedEra) {
   784→            onComplete({ birthYear: savedBirthYear, eraSkin: savedEra });
   785→            return;
   786→          }
   787→        }
   788→        setIsLoaded(true);
   789→      } catch {
   790→        // If IDB fails, still show onboarding
   791→        setIsLoaded(true);
   792→      }
   793→    })();
   794→
   795→    return () => { cancelled = true; };
   796→  }, [onComplete]);
   797→
   798→  /* ───── Persistence helpers ───── */
   799→  const saveOnboardingData = useCallback(
   800→    async (year: number, era: string) => {
   801→      const db = await getDB();
   802→      const record: OnboardingRecord = {
   803→        id: ONBOARDING_ID,
   804→        birthYear: year,
   805→        eraSkin: era,
   806→        complete: true,
   807→      };
   808→      await db.put('era_skin', record as any);
   809→      // Also set the era skin in the standard location
   810→      if (era !== 'mixed') {
   811→        await setEraSkinId(era);
   812→      }
   813→    },
   814→    [],
   815→  );
   816→
   817→  /* ───── Navigation helpers ───── */
   818→  const goNext = useCallback(() => {
   819→    setDirection(STEP_DIRECTION.forward);
   820→    setCurrentStep((s) => Math.min(s + 1, TOTAL_STEPS - 1));
   821→  }, []);
   822→
   823→  const goBack = useCallback(() => {
   824→    setDirection(STEP_DIRECTION.backward);
   825→    setCurrentStep((s) => Math.max(s - 1, 0));
   826→  }, []);
   827→
   828→  const setBirthYear = useCallback((val: string) => {
   829→    setBirthYearState(val);
   830→  }, []);
   831→
   832→  const canContinueBirthYear = useMemo(() => {
   833→    const y = parseInt(birthYear);
   834→    return y >= 1970 && y <= 2015;
   835→  }, [birthYear]);
   836→
   837→  /* ───── Handle final completion ───── */
   838→  const handleComplete = useCallback(() => {
   839→    const year = parseInt(birthYear);
   840→    const era = selectedEra || 'mixed';
   841→    setIsCompleting(true);
   842→
   843→    // Small delay for the exit animation to feel natural
   844→    setTimeout(async () => {
   845→      try {
   846→        await saveOnboardingData(year, era);
   847→      } catch (err) {
   848→        console.warn('[Onboarding] Failed to persist:', err);
   849→      }
   850→      onComplete({ birthYear: year, eraSkin: era });
   851→    }, 600);
   852→  }, [birthYear, selectedEra, saveOnboardingData, onComplete]);
   853→
   854→  /* ───── Loading state while checking IDB ───── */
   855→  if (!isLoaded) {
   856→    return (
   857→      <div className="min-h-screen flex items-center justify-center bg-[#f5edd6]">
   858→        <motion.div
   859→          initial={{ opacity: 0 }}
   860→          animate={{ opacity: 1 }}
   861→          className="text-center"
   862→        >
   863→          <div className="text-4xl mb-3">📼</div>
   864→          <p className="old-typewriter text-sm text-[#8b7355]">Loading...</p>
   865→        </motion.div>
   866→      </div>
   867→    );
   868→  }
   869→
   870→  /* ───── Render current step ───── */
   871→  const renderStep = () => {
   872→    switch (currentStep) {
   873→      case 0:
   874→        return <WelcomeStep onNext={goNext} />;
   875→      case 1:
   876→        return (
   877→          <BirthYearStep
   878→            birthYear={birthYear}
   879→            setBirthYear={setBirthYear}
   880→            onNext={goNext}
   881→            onBack={goBack}
   882→            canContinue={canContinueBirthYear}
   883→          />
   884→        );
   885→      case 2:
   886→        return (
   887→          <EraSkinStep
   888→            selectedEra={selectedEra}
   889→            setSelectedEra={setSelectedEra}
   890→            onNext={goNext}
   891→            onBack={goBack}
   892→          />
   893→        );
   894→      case 3:
   895→        return (
   896→          <FeatureTourStep
   897→            currentFeature={currentFeature}
   898→            setCurrentFeature={setCurrentFeature}
   899→            onNext={handleComplete}
   900→            onBack={goBack}
   901→          />
   902→        );
   903→      default:
   904→        return null;
   905→    }
   906→  };
   907→
   908→  return (
   909→    <AnimatePresence mode="wait">
   910→      {!isCompleting ? (
   911→        <motion.div
   912→          key="onboarding-container"
   913→          initial={{ opacity: 0 }}
   914→          animate={{ opacity: 1 }}
   915→          exit={{ opacity: 0, scale: 0.95, y: -20 }}
   916→          transition={{ duration: 0.6, ease: 'easeInOut' }}
   917→          className="min-h-screen flex flex-col"
   918→          style={{
   919→            background:
   920→              'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)',
   921→          }}
   922→        >
   923→          {/* Main step area */}
   924→          <div className="flex-1 flex items-center justify-center overflow-hidden">
   925→            <AnimatePresence custom={direction} mode="wait">
   926→              <motion.div
   927→                key={currentStep}
   928→                custom={direction}
   929→                variants={slideVariants}
   930→                initial="enter"
   931→                animate="center"
   932→                exit="exit"
   933→                transition={slideTransition}
   934→                className="w-full"
   935→              >
   936→                {renderStep()}
   937→              </motion.div>
   938→            </AnimatePresence>
   939→          </div>
   940→
   941→          {/* Progress dots — fixed at bottom */}
   942→          <div className="pb-8 pt-4">
   943→            <ProgressDots current={currentStep} total={TOTAL_STEPS} />
   944→          </div>
   945→        </motion.div>
   946→      ) : (
   947→        /* Completion celebration overlay */
   948→        <motion.div
   949→          key="onboarding-complete"
   950→          initial={{ opacity: 0 }}
   951→          animate={{ opacity: 1 }}
   952→          className="min-h-screen flex items-center justify-center bg-[#f5edd6]"
   953→        >
   954→          <motion.div
   955→            initial={{ scale: 0, rotate: -20 }}
   956→            animate={{ scale: 1, rotate: 0 }}
   957→            transition={{ type: 'spring', stiffness: 200, damping: 15 }}
   958→            className="text-center px-6"
   959→          >
   960→            <motion.div
   961→              className="text-6xl mb-4"
   962→              animate={{ rotate: [0, 10, -10, 0] }}
   963→              transition={{ duration: 0.8, repeat: 2 }}
   964→            >
   965→              🎉
   966→            </motion.div>
   967→            <h2 className="old-title text-3xl font-black text-[#5c4033] mb-2">
   968→              All Set!
   969→            </h2>
   970→            <p className="old-body text-[#8b7355]">
   971→              Rewinding to your era...
   972→            </p>
   973→            <motion.div
   974→              className="mt-6 mx-auto"
   975→              style={{ width: 120, height: 4, borderRadius: 2 }}
   976→              initial={{ opacity: 0.3 }}
   977→              animate={{ opacity: [0.3, 1, 0.3] }}
   978→              transition={{ duration: 1.2, repeat: Infinity }}
   979→            >
   980→              <div
   981→                className="h-full rounded-full"
   982→                style={{
   983→                  width: '100%',
   984→                  background: 'linear-gradient(90deg, #8b5e3c, #c4956a, #8b5e3c)',
   985→                  backgroundSize: '200% 100%',
   986→                  animation: 'rewind-shimmer 1.5s ease-in-out infinite',
   987→                }}
   988→              />
   989→            </motion.div>
   990→          </motion.div>
   991→        </motion.div>
   992→      )}
   993→    </AnimatePresence>
   994→  );
   995→}
