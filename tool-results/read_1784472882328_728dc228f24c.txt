    17→    17→/* ═══════ TYPES ═══════ */
    18→    18→interface CapturedItem {
    19→    19→  id: string;
    20→    20→  type: 'photo' | 'video';
    21→    21→  url: string;
    22→    22→  timestamp: number;
    23→    23→  blob?: Blob;
    24→    24→}
    25→    25→
    26→    26→type FilterPreset = 'none' | 'sepia70s' | 'faded80s' | 'warm90s' | 'bw60s' | 'polaroid' | 'vhs' | 'kodak';
    27→    27→
    28→    28→/* Real-time preview resolution (low for performance) */
    29→    29→const PREVIEW_W = 360;
    30→    30→const PREVIEW_H = 202;
    31→    31→const TARGET_FPS = 24;
    32→    32→const FRAME_INTERVAL = 1000 / TARGET_FPS;
    33→    33→
    34→    34→/* ═══════ VINTAGE CAMERA COMPONENT ═══════ */
    35→    35→export default function VintageCamera({ onClose }: { onClose: () => void }) {
    36→    36→  const videoRef = useRef<HTMLVideoElement>(null);
    37→    37→  const captureCanvasRef = useRef<HTMLCanvasElement>(null);   // full-res for capture
    38→    38→  const previewCanvasRef = useRef<HTMLCanvasElement>(null);   // live processed preview
    39→    39→  const offscreenRef = useRef<HTMLCanvasElement | null>(null);
    40→    40→  const streamRef = useRef<MediaStream | null>(null);
    41→    41→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
    42→    42→  const recordedChunksRef = useRef<Blob[]>([]);
    43→    43→  const flashRef = useRef<HTMLDivElement>(null);
    44→    44→  const rafRef = useRef<number>(0);
    45→    45→  const lastFrameTimeRef = useRef<number>(0);
    46→    46→  const frameCountRef = useRef<number>(0);
    47→    47→  const flickerEvRef = useRef<number>(0);
    48→    48→
    49→    49→  const [cameraOpen, setCameraOpen] = useState(false);
    50→    50→  const [facingMode, setFacingMode] = useState<'user' | 'environment'>('environment');
    51→    51→  const [mode, setMode] = useState<'photo' | 'video'>('photo');
    52→    52→  const [filter, setFilter] = useState<FilterPreset>('sepia70s');
    53→    53→  const [recording, setRecording] = useState(false);
    54→    54→  const [recordTime, setRecordTime] = useState(0);
    55→    55→  const [flash, setFlash] = useState(false);
    56→    56→  const [gallery, setGallery] = useState<CapturedItem[]>([]);
    57→    57→  const [viewItem, setViewItem] = useState<CapturedItem | null>(null);
    58→    58→  const [cameraError, setCameraError] = useState<string | null>(null);
    59→    59→  const [showFilters, setShowFilters] = useState(false);
    60→    60→  const [shutterAnim, setShutterAnim] = useState(false);
    61→    61→  const [processing, setProcessing] = useState(false);
    62→    62→
    63→    63→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    64→    64→
    65→    65→  /* ── Get current preset config ── */
    66→    66→  const currentConfig: PipelineConfig = PIPELINE_PRESETS[filter] || PIPELINE_PRESETS['none'];
    67→    67→
    68→    68→  /* ── Start Camera ── */
    69→    69→  const startCamera = useCallback(async () => {
    70→    70→    try {
    71→    71→      if (streamRef.current) {
    72→    72→        streamRef.current.getTracks().forEach(t => t.stop());
    73→    73→      }
    74→    74→      const stream = await navigator.mediaDevices.getUserMedia({
    75→    75→        video: { facingMode, width: { ideal: 1280 }, height: { ideal: 720 } },
    76→    76→        audio: mode === 'video',
    77→    77→      });
    78→    78→      streamRef.current = stream;
    79→    79→      if (videoRef.current) {
    80→    80→        videoRef.current.srcObject = stream;
    81→    81→        await videoRef.current.play();
    82→    82→      }
    83→    83→      setCameraOpen(true);
    84→    84→      setCameraError(null);
    85→    85→    } catch (err) {
    86→    86→      console.error('Camera error:', err);
    87→    87→      setCameraError('Camera access denied or not available. Please allow camera permissions.');
    88→    88→    }
    89→    89→  }, [facingMode, mode]);
    90→    90→
    91→    91→  /* ── Stop Camera ── */
    92→    92→  const stopCamera = useCallback(() => {
    93→    93→    if (rafRef.current) {
    94→    94→      cancelAnimationFrame(rafRef.current);
    95→    95→      rafRef.current = 0;
    96→    96→    }
    97→    97→    if (streamRef.current) {
    98→    98→      streamRef.current.getTracks().forEach(t => t.stop());
    99→    99→      streamRef.current = null;
   100→   100→    }
   101→   101→    setCameraOpen(false);
   102→   102→    if (recordTimerRef.current) {
   103→   103→      clearInterval(recordTimerRef.current);
   104→   104→      recordTimerRef.current = null;
   105→   105→    }
   106→   106→  }, []);
   107→   107→
   108→   108→  /* ── Switch Camera ── */
   109→   109→  const handleSwitchCamera = useCallback(() => {
   110→   110→    setFacingMode(prev => prev === 'user' ? 'environment' : 'user');
   111→   111→  }, []);
   112→   112→
   113→   113→  /* ── Restart on facing/mode change ── */
   114→   114→  useEffect(() => {
   115→   115→    if (cameraOpen) {
   116→   116→      startCamera();
   117→   117→    }
   118→   118→    return () => {};
   119→   119→  }, [facingMode, mode, cameraOpen, startCamera]);
   120→   120→
   121→   121→  /* ── Real-time preview loop ── */
   122→   122→  useEffect(() => {
   123→   123→    if (!cameraOpen || !videoRef.current || !previewCanvasRef.current) return;
   124→   124→
   125→   125→    // Create offscreen canvas for drawing video frames
   126→   126→    if (!offscreenRef.current) {
   127→   127→      offscreenRef.current = document.createElement('canvas');
   128→   128→    }
   129→   129→    offscreenRef.current.width = PREVIEW_W;
   130→   130→    offscreenRef.current.height = PREVIEW_H;
   131→   131→    const offCtx = offscreenRef.current.getContext('2d', { willReadFrequently: true });
   132→   132→    if (!offCtx) return;
   133→   133→
   134→   134→    const previewCanvas = previewCanvasRef.current;
   135→   135→    previewCanvas.width = PREVIEW_W;
   136→   136→    previewCanvas.height = PREVIEW_H;
   137→   137→    const previewCtx = previewCanvas.getContext('2d', { willReadFrequently: true });
   138→   138→    if (!previewCtx) return;
   139→   139→
   140→   140→    let running = true;
   141→   141→
   142→   142→    const renderLoop = (timestamp: number) => {
   143→   143→      if (!running) return;
   144→   144→
   145→   145→      const elapsed = timestamp - lastFrameTimeRef.current;
   146→   146→      if (elapsed >= FRAME_INTERVAL) {
   147→   147→        lastFrameTimeRef.current = timestamp - (elapsed % FRAME_INTERVAL);
   148→   148→        frameCountRef.current++;
   149→   149→
   150→   150→        const video = videoRef.current;
   151→   151→        if (!video || video.readyState < 2) {
   152→   152→          rafRef.current = requestAnimationFrame(renderLoop);
   153→   153→          return;
   154→   154→        }
   155→   155→
   156→   156→        // Draw video frame to offscreen canvas (mirrored if front camera)
   157→   157→        offCtx.save();
   158→   158→        if (facingMode === 'user') {
   159→   159→          offCtx.translate(PREVIEW_W, 0);
   160→   160→          offCtx.scale(-1, 1);
   161→   161→        }
   162→   162→        offCtx.drawImage(video, 0, 0, PREVIEW_W, PREVIEW_H);
   163→   163→        offCtx.restore();
   164→   164→
   165→   165→        // Apply lightweight preview pipeline
   166→   166→        try {
   167→   167→          processPreviewFrame(offCtx, PREVIEW_W, PREVIEW_H, filter, frameCountRef.current);
   168→   168→
   169→   169→          // Gate flicker for video mode
   170→   170→          if (mode === 'video' && currentConfig.gateFlicker.enabled) {
   171→   171→            flickerEvRef.current = getGateFlickerEV(currentConfig.gateFlicker.intensity);
   172→   172→            const imgData = offCtx.getImageData(0, 0, PREVIEW_W, PREVIEW_H);
   173→   173→            const d = imgData.data;
   174→   174→            const mult = 1 + flickerEvRef.current * 0.5;
   175→   175→            for (let i = 0; i < d.length; i += 4) {
   176→   176→              d[i] = Math.max(0, Math.min(255, Math.round(d[i] * mult)));
   177→   177→              d[i + 1] = Math.max(0, Math.min(255, Math.round(d[i + 1] * mult)));
   178→   178→              d[i + 2] = Math.max(0, Math.min(255, Math.round(d[i + 2] * mult)));
   179→   179→            }
   180→   180→            offCtx.putImageData(imgData, 0, 0);
   181→   181→          }
   182→   182→        } catch {
   183→   183→          // silently skip if processing errors (e.g., tab not visible)
   184→   184→        }
   185→   185→
   186→   186→        // Copy to visible preview canvas
   187→   187→        previewCtx.drawImage(offscreenRef.current, 0, 0);
   188→   188→      }
   189→   189→
   190→   190→      rafRef.current = requestAnimationFrame(renderLoop);
   191→   191→    };
   192→   192→
   193→   193→    rafRef.current = requestAnimationFrame(renderLoop);
   194→   194→
   195→   195→    return () => {
   196→   196→      running = false;
   197→   197→      if (rafRef.current) {
   198→   198→        cancelAnimationFrame(rafRef.current);
   199→   199→        rafRef.current = 0;
   200→   200→      }
   201→   201→    };
   202→   202→  }, [cameraOpen, filter, facingMode, mode, currentConfig.gateFlicker.enabled, currentConfig.gateFlicker.intensity]);
   203→   203→
   204→   204→  /* ── Capture Photo (Full Pipeline) ── */
   205→   205→  const capturePhoto = useCallback(() => {
   206→   206→    if (!videoRef.current || !captureCanvasRef.current) return;
   207→   207→    const video = videoRef.current;
   208→   208→    const canvas = captureCanvasRef.current;
   209→   209→    const ctx = canvas.getContext('2d', { willReadFrequently: true });
   210→   210→    if (!ctx) return;
   211→   211→
   212→   212→    setProcessing(true);
   213→   213→
   214→   214→    // Use full resolution
   215→   215→    const w = video.videoWidth || 1280;
   216→   216→    const h = video.videoHeight || 720;
   217→   217→    canvas.width = w;
   218→   218→    canvas.height = h;
   219→   219→
   220→   220→    // Mirror if front camera
   221→   221→    if (facingMode === 'user') {
   222→   222→      ctx.translate(w, 0);
   223→   223→      ctx.scale(-1, 1);
   224→   224→    }
   225→   225→    ctx.drawImage(video, 0, 0, w, h);
   226→   226→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   227→   227→
   228→   228→    // Apply full pipeline
   229→   229→    try {
   230→   230→      processFrame(ctx, w, h, filter, { addStamp: true });
   231→   231→    } catch (e) {
   232→   232→      console.error('Pipeline error:', e);
   233→   233→    }
   234→   234→
   235→   235→    // Flash effect
   236→   236→    setFlash(true);
   237→   237→    setShutterAnim(true);
   238→   238→    setTimeout(() => setFlash(false), 200);
   239→   239→    setTimeout(() => setShutterAnim(false), 300);
   240→   240→
   241→   241→    // Save to gallery
   242→   242→    canvas.toBlob((blob) => {
   243→   243→      if (!blob) { setProcessing(false); return; }
   244→   244→      const url = URL.createObjectURL(blob);
   245→   245→      setGallery(prev => [{
   246→   246→        id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   247→   247→        type: 'photo',
   248→   248→        url,
   249→   249→        timestamp: Date.now(),
   250→   250→        blob,
   251→   251→      }, ...prev]);
   252→   252→      setProcessing(false);
   253→   253→    }, 'image/jpeg', 0.92);
   254→   254→  }, [filter, facingMode]);
   255→   255→
   256→   256→  /* ── Start/Stop Video Recording ── */
   257→   257→  const toggleRecording = useCallback(() => {
   258→   258→    if (!recording && streamRef.current) {
   259→   259→      const options = { mimeType: 'video/webm;codecs=vp9' };
   260→   260→      const tryStart = (mimeType: string) => {
   261→   261→        try {
   262→   262→          const mediaRecorder = new MediaRecorder(streamRef.current!, { mimeType });
   263→   263→          recordedChunksRef.current = [];
   264→   264→          mediaRecorder.ondataavailable = (e) => {
   265→   265→            if (e.data.size > 0) recordedChunksRef.current.push(e.data);
   266→   266→          };
   267→   267→          mediaRecorder.onstop = () => {
   268→   268→            const blob = new Blob(recordedChunksRef.current, { type: 'video/webm' });
   269→   269→            const url = URL.createObjectURL(blob);
   270→   270→            setGallery(prev => [{
   271→   271→              id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   272→   272→              type: 'video',
   273→   273→              url,
   274→   274→              timestamp: Date.now(),
   275→   275→              blob,
   276→   276→            }, ...prev]);
   277→   277→            recordedChunksRef.current = [];
   278→   278→          };
   279→   279→          mediaRecorder.start(100);
   280→   280→          mediaRecorderRef.current = mediaRecorder;
   281→   281→          setRecording(true);
   282→   282→          setRecordTime(0);
   283→   283→          recordTimerRef.current = setInterval(() => {
   284→   284→            setRecordTime(prev => prev + 1);
   285→   285→          }, 1000);
   286→   286→        } catch {
   287→   287→          if (mimeType.includes('vp9')) {
   288→   288→            tryStart('video/webm;codecs=vp8');
   289→   289→          } else {
   290→   290→            setCameraError('Video recording not supported in this browser.');
   291→   291→          }
   292→   292→        }
   293→   293→      };
   294→   294→      tryStart(options.mimeType);
   295→   295→    } else if (recording && mediaRecorderRef.current) {
   296→   296→      mediaRecorderRef.current.stop();
   297→   297→      mediaRecorderRef.current = null;
   298→   298→      setRecording(false);
   299→   299→      if (recordTimerRef.current) {
   300→   300→        clearInterval(recordTimerRef.current);
   301→   301→        recordTimerRef.current = null;
   302→   302→      }
   303→   303→    }
   304→   304→  }, [recording]);
   305→   305→
   306→   306→  /* ── Delete from gallery ── */
   307→   307→  const deleteItem = useCallback((id: string) => {
   308→   308→    setGallery(prev => {
   309→   309→      const item = prev.find(i => i.id === id);
   310→   310→      if (item) URL.revokeObjectURL(item.url);
   311→   311→      return prev.filter(i => i.id !== id);
   312→   312→    });
   313→   313→  }, []);
   314→   314→
   315→   315→  /* ── Download ── */
   316→   316→  const downloadItem = useCallback((item: CapturedItem) => {
   317→   317→    const a = document.createElement('a');
   318→   318→    a.href = item.url;
   319→   319→    a.download = `rewind_${item.type}_${new Date(item.timestamp).toISOString().replace(/[:.]/g, '-')}.${item.type === 'photo' ? 'jpg' : 'webm'}`;
   320→   320→    document.body.appendChild(a);
   321→   321→    a.click();
   322→   322→    document.body.removeChild(a);
   323→   323→  }, []);
   324→   324→
   325→   325→  /* ── Cleanup on close ── */
   326→   326→  useEffect(() => {
   327→   327→    return () => {
   328→   328→      stopCamera();
   329→   329→      gallery.forEach(item => URL.revokeObjectURL(item.url));
   330→   330→    };
   331→   331→  // eslint-disable-next-line react-hooks/exhaustive-deps
   332→   332→  }, []);
   333→   333→
   334→   334→  /* ── Format time ── */
   335→   335→  const formatTime = (s: number) => `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
   336→   336→
   337→   337→  /* ── Pipeline info for HUD ── */
   338→   338→  const pipelineStages = [
   339→   339→    currentConfig.chromaticAberration.enabled && 'CA',
   340→   340→    currentConfig.toneCurve.blackLift > 0 && 'MATTE',
   341→   341→    currentConfig.contrastReduce > 0 && 'FLAT',
   342→   342→    currentConfig.antiClarity.enabled && 'SOFT',
   343→   343→    currentConfig.grain.enabled && 'GRAIN',
   344→   344→    currentConfig.halation.enabled && 'GLOW',
   345→   345→    currentConfig.vignette.intensity > 0 && 'VIG',
   346→   346→  ].filter(Boolean);
   347→   347→
   348→   348→  return (
   349→   349→    <>
   350→   350→      {/* Hidden canvases */}
   351→   351→      <canvas ref={captureCanvasRef} className="hidden" />
   352→   352→
   353→   353→      <AnimatePresence>
   354→   354→        <motion.div
   355→   355→          initial={{ opacity: 0 }}
   356→   356→          animate={{ opacity: 1 }}
   357→   357→          exit={{ opacity: 0 }}
   358→   358→          className="fixed inset-0 z-50 flex flex-col"
   359→   359→          style={{ background: '#1a1208' }}
   360→   360→        >
   361→   361→          {/* ═══ HEADER - Camera Top Bar ═══ */}
   362→   362→          <div className="sk-leather px-3 py-2.5 flex items-center justify-between z-10">
   363→   363→            <button onClick={() => { stopCamera(); onClose(); }} className="sk-btn old-btn px-3 py-1.5 rounded-lg text-xs font-bold cursor-pointer flex items-center gap-1 text-[#faf6eb]">
   364→   364→              <X className="h-4 w-4" /> Close
   365→   365→            </button>
   366→   366→            <h2 className="old-title font-black text-base text-[#faf6eb] tracking-wide" style={{ textShadow: '0 1px 2px rgba(0,0,0,0.3)' }}>
   367→   367→              📷 Vintage Camera
   368→   368→            </h2>
   369→   369→            <div className="w-16" />
   370→   370→          </div>
   371→   371→
   372→   372→          {/* ═══ VIEWFINDER ═══ */}
   373→   373→          <div className="flex-1 relative overflow-hidden flex items-center justify-center">
   374→   374→            {!cameraOpen && !cameraError && (
   375→   375→              <div className="text-center px-6">
   376→   376→                <motion.div
   377→   377→                  animate={{ rotate: [0, 5, -5, 0] }}
   378→   378→                  transition={{ duration: 4, repeat: Infinity }}
   379→   379→                  className="text-6xl mb-4"
   380→   380→                >📸</motion.div>
   381→   381→                <p className="old-body text-[#d4a574] mb-2 text-sm">Open the camera to capture memories in vintage style</p>
   382→   382→                <p className="old-typewriter text-[10px] text-[#8b7355] mb-6">
   383→   383→                  Full image processing pipeline: chromatic aberration, tone curves, Perlin grain, halation glow
   384→   384→                </p>
   385→   385→                <button onClick={startCamera} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   386→   386→                  <Camera className="h-5 w-5" /> Open Camera
   387→   387→                </button>
   388→   388→              </div>
   389→   389→            )}
   390→   390→
   391→   391→            {cameraError && (
   392→   392→              <div className="text-center px-6">
   393→   393→                <div className="text-5xl mb-4">😵</div>
   394→   394→                <p className="old-body text-[#d4a574] mb-4 text-sm">{cameraError}</p>
   395→   395→                <button onClick={startCamera} className="sk-btn old-btn px-5 py-2.5 rounded-lg font-bold cursor-pointer flex items-center gap-2 mx-auto">
   396→   396→                  <RotateCcw className="h-4 w-4" /> Try Again
   397→   397→                </button>
   398→   398→              </div>
   399→   399→            )}
   400→   400→
   401→   401→            {cameraOpen && (
   402→   402→              <>
   403→   403→                {/* Hidden video source (never shown directly) */}
   404→   404→                <video
   405→   405→                  ref={videoRef}
   406→   406→                  className="hidden"
   407→   407→                  playsInline
   408→   408→                  muted
   409→   409→                />
   410→   410→
   411→   411→                {/* Real-time processed canvas preview */}
   412→   412→                <canvas
   413→   413→                  ref={previewCanvasRef}
   414→   414→                  className="absolute inset-0 w-full h-full object-cover"
   415→   415→                  style={{
   416→   416→                    imageRendering: 'auto', // smooth upscaling for that vintage soft look
   417→   417→                  }}
   418→   418→                />
   419→   419→
   420→   420→                {/* Halation glow overlay (CSS-based, adds dreamy ambient light) */}
   421→   421→                {currentConfig.halation.enabled && (
   422→   422→                  <div
   423→   423→                    className="absolute inset-0 pointer-events-none mix-blend-screen"
   424→   424→                    style={{
   425→   425→                      background: `radial-gradient(ellipse at 50% 50%, rgba(255,240,200,${currentConfig.halation.intensity * 0.06}) 0%, transparent 70%)`,
   426→   426→                    }}
   427→   427→                  />
   428→   428→                )}
   429→   429→
   430→   430→                {/* Scanline overlay (VHS era feel) */}
   431→   431→                <div className="absolute inset-0 pointer-events-none opacity-[0.07]" style={{
   432→   432→                  background: 'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(0,0,0,0.2) 2px, rgba(0,0,0,0.2) 4px)',
   433→   433→                }} />
   434→   434→
   435→   435→                {/* Film border overlay - viewfinder frame */}
   436→   436→                <div className="absolute inset-0 pointer-events-none border-[6px] border-[#1a1208] rounded-none" />
   437→   437→
   438→   438→                {/* Corner brackets */}
