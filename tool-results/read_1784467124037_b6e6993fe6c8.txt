    59→function formatDate(iso: string): string {
    60→  const d = new Date(iso);
    61→  const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    62→  const day = d.getDate();
    63→  const suffix = day === 1 || day === 21 || day === 31 ? 'st' : day === 2 || day === 22 ? 'nd' : day === 3 || day === 23 ? 'rd' : 'th';
    64→  return `${months[d.getMonth()]} ${day}${suffix}, ${d.getFullYear()}`;
    65→}
    66→
    67→function formatTimeAgo(ts: number): string {
    68→  const diff = Date.now() - ts;
    69→  const mins = Math.floor(diff / 60000);
    70→  if (mins < 1) return 'Just now';
    71→  if (mins < 60) return `${mins}m ago`;
    72→  const hrs = Math.floor(mins / 60);
    73→  if (hrs < 24) return `${hrs}h ago`;
    74→  const days = Math.floor(hrs / 24);
    75→  if (days < 7) return `${days}d ago`;
    76→  return formatDate(new Date(ts).toISOString());
    77→}

   595→/* ═══════ MAIN JOURNAL COMPONENT ═══════ */
   596→export default function OldJournal() {
   597→  const [entries, setEntries] = useState<JournalEntry[]>([]);
   598→  const [view, setView] = useState<'cover' | 'list' | 'write' | 'read'>('cover');
   599→  const [selectedId, setSelectedId] = useState<string | null>(null);
   600→  const [editEntry, setEditEntry] = useState<JournalEntry | undefined>(undefined);
   601→  const [confirmDelete, setConfirmDelete] = useState(false);
   602→
   603→  // Load entries from localStorage
   604→  useEffect(() => {
   605→    setEntries(loadEntries());
   606→  }, []);
   607→
   608→  const handleSave = useCallback((entry: Omit<JournalEntry, 'id' | 'createdAt'>) => {
   609→    setEntries(prev => {
   610→      let updated: JournalEntry[];
   611→      if (editEntry) {
   612→        updated = prev.map(e => e.id === editEntry.id ? { ...e, ...entry } : e);
   613→      } else {
   614→        const newEntry: JournalEntry = {
   615→          ...entry,
   616→          id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   617→          createdAt: Date.now(),
   618→        };
   619→        updated = [newEntry, ...prev];
   620→      }
   621→      saveEntries(updated);
   622→      return updated;
   623→    });
   624→    setEditEntry(undefined);
   625→    setView('read');
   626→  }, [editEntry]);
   627→
   628→  const handleDelete = useCallback(() => {
   629→    if (!selectedId) return;
   630→    if (!confirmDelete) {
   631→      setConfirmDelete(true);
   632→      setTimeout(() => setConfirmDelete(false), 3000);
   633→      return;
   634→    }
   635→    setEntries(prev => {
   636→      const updated = prev.filter(e => e.id !== selectedId);
   637→      saveEntries(updated);
   638→      return updated;
   639→    });
   640→    setSelectedId(null);
   641→    setView('list');
   642→    setConfirmDelete(false);
   643→  }, [selectedId, confirmDelete]);
   644→
   645→  const handleEdit = useCallback(() => {
   646→    if (!selectedId) return;
   647→    const entry = entries.find(e => e.id === selectedId);
   648→    if (entry) {
   649→      setEditEntry(entry);
   650→      setView('write');
   651→    }
   652→  }, [selectedId, entries]);
   653→
   654→  const currentEntry = entries.find(e => e.id === selectedId);
   655→  const currentIndex = entries.findIndex(e => e.id === selectedId);
   656→
   657→  return (
   658→    <div className="relative min-h-full pb-4">
   659→      <AnimatePresence mode="wait">
   660→        {/* ═══ COVER VIEW ═══ */}
   661→        {view === 'cover' && (
   662→          <JournalCover
   663→            key="cover"
   664→            onOpen={() => entries.length > 0 ? setView('list') : setView('write')}
   665→            entryCount={entries.length}
   666→          />
   667→        )}
   668→
   669→        {/* ═══ ENTRIES LIST ═══ */}
   670→        {view === 'list' && (
   671→          <EntriesList
   672→            key="list"
   673→            entries={entries}
   674→            onSelect={id => { setSelectedId(id); setView('read'); }}
   675→            onNew={() => { setEditEntry(undefined); setView('write'); }}
   676→            onBack={() => setView('cover')}
   677→          />
   678→        )}
   679→
   680→        {/* ═══ WRITE/EDIT ═══ */}
   681→        {view === 'write' && (
   682→          <EntryEditor
   683→            key={`write-${editEntry?.id ?? 'new'}`}
   684→            onSave={handleSave}
   685→            onCancel={() => {
   686→              setEditEntry(undefined);
   687→              setView(entries.length > 0 ? 'list' : 'cover');
   688→            }}
   689→            initialEntry={editEntry}
   690→          />
   691→        )}
   692→
   693→        {/* ═══ READ ENTRY ═══ */}
   694→        {view === 'read' && currentEntry && (
   695→          <EntryViewer
   696→            key={`read-${currentEntry.id}`}
   697→            entry={currentEntry}
   698→            onEdit={handleEdit}
   699→            onDelete={handleDelete}
   700→            onClose={() => setView('list')}
   701→            onPrev={() => {
   702→              if (currentIndex > 0) setSelectedId(entries[currentIndex - 1].id);
   703→            }}
   704→            onNext={() => {
   705→              if (currentIndex < entries.length - 1) setSelectedId(entries[currentIndex + 1].id);
   706→            }}
   707→            hasPrev={currentIndex > 0}
   708→            hasNext={currentIndex < entries.length - 1}
   709→            index={currentIndex}
   710→            total={entries.length}
   711→          />
   712→        )}
   713→      </AnimatePresence>
   714→
   715→      {/* Delete confirmation toast */}
   716→      <AnimatePresence>
   717→        {confirmDelete && (
   718→          <motion.div
   719→            initial={{ opacity: 0, y: 20 }}
   720→            animate={{ opacity: 1, y: 0 }}
   721→            exit={{ opacity: 0, y: 20 }}
   722→            className="fixed bottom-20 left-1/2 -translate-x-1/2 z-50 sk-paper rounded-lg px-4 py-2 shadow-xl"
   723→          >
   724→            <p className="old-handwritten text-xs text-[#c0392b]">
   725→              Tap &quot;Discard&quot; again to confirm deletion
   726→            </p>
   727→          </motion.div>
   728→        )}
   729→      </AnimatePresence>
   730→    </div>
   731→  );
   732→}
