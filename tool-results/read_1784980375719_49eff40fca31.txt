    83→    83→// ─── Helpers ────────────────────────────────────────

    85→    85→function getWeekStart(date?: Date): Date {
    86→    86→  const d = date ?? new Date();
    87→    87→  const day = d.getDay();
    88→    88→  const diff = d.getDate() - day + (day === 0 ? -6 : 1); // Monday
    89→    89→  return new Date(d.getFullYear(), d.getMonth(), diff);
    90→    90→}
    91→    91→
    92→    92→function getWeekNumber(): number {
    93→    93→  const now = new Date();
    94→    94→  const start = new Date(now.getFullYear(), 0, 1);
    95→    95→  const diff = now.getTime() - start.getTime();
    96→    96→  const oneWeek = 604800000;
    97→    97→  return Math.floor(diff / oneWeek);
    98→    98→}
    99→    99→
   100→   100→function startOfDay(d: Date): number {
   101→   101→  return new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
   102→   102→}
   103→   103→
   104→   104→function isSameDay(ts1: number, ts2: number): boolean {
   105→   105→  const d1 = new Date(ts1);
   106→   106→  const d2 = new Date(ts2);
   107→   107→  return d1.getFullYear() === d2.getFullYear() &&
   108→   108→    d1.getMonth() === d2.getMonth() &&
   109→   109→    d1.getDate() === d2.getDate();
   110→   110→}
   111→   111→
   112→   112→function isYesterday(ts: number): boolean {
   113→   113→  const yesterday = new Date();
   114→   114→  yesterday.setDate(yesterday.getDate() - 1);
   115→   115→  return isSameDay(ts, yesterday.getTime());
   116→   116→}
   117→   117→
   118→   118→function isToday(ts: number): boolean {
   119→   119→  return isSameDay(ts, Date.now());
   120→   120→}

   146→   146→// ─── Visit Streak Logic ──────────────────────────────
   147→   147→
   148→   148→async function getVisitStreak(): Promise<VisitStreak> {
   149→   149→  try {
   150→   150→    const db = await getDB();
   151→   151→    const record = await db.get('era_skin', 'visit_streak');
   152→   152→    if (record) {
   153→   153→      try {
   154→   154→        const data = JSON.parse(record.skin_id);
   155→   155→        return { days: data.days ?? 0, lastVisit: data.lastVisit ?? 0 };
   156→   156→      } catch {
   157→   157→        return { days: 0, lastVisit: 0 };
   158→   158→      }
   159→   159→    }
   160→   160→  } catch {
   161→   161→    // silent
   162→   162→  }
   163→   163→  return { days: 0, lastVisit: 0 };
   164→   164→}
   165→   165→
   166→   166→async function updateVisitStreak(): Promise<VisitStreak> {
   167→   167→  try {
   168→   168→    const db = await getDB();
   169→   169→    const record = await db.get('era_skin', 'visit_streak');
   170→   170→    let streak: VisitStreak;
   171→   171→
   172→   172→    if (record) {
   173→   173→      try {
   174→   174→        const data = JSON.parse(record.skin_id);
   175→   175→        streak = { days: data.days ?? 0, lastVisit: data.lastVisit ?? 0 };
   176→   176→      } catch {
   177→   177→        streak = { days: 0, lastVisit: 0 };
   178→   178→      }
   179→   179→    } else {
   180→   180→      streak = { days: 0, lastVisit: 0 };
   181→   181→    }
   182→   182→
   183→   183→    const now = Date.now();
   184→   184→
   185→   185→    if (isToday(streak.lastVisit)) {
   186→   186→      // Already visited today, no change
   187→   187→      return streak;
   188→   188→    } else if (isYesterday(streak.lastVisit)) {
   189→   189→      // Consecutive day
   190→   190→      streak.days += 1;
   191→   191→      streak.lastVisit = now;
   192→   192→    } else {
   193→   193→      // Streak broken or first visit
   194→   194→      streak.days = 1;
   195→   195→      streak.lastVisit = now;
   196→   196→    }
   197→   197→
   198→   198→    await db.put('era_skin', {
   199→   199→      id: 'visit_streak',
   200→   200→      skin_id: JSON.stringify({ days: streak.days, lastVisit: streak.lastVisit }),
   201→   201→    });
   202→   202→
   203→   203→    return streak;
   204→   204→  } catch {
   205→   205→    return { days: 0, lastVisit: 0 };
   206→   206→  }
   207→   207→}

   740→   740→        </div>
