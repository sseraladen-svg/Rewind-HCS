     8→/* ──────────── PERLIN NOISE (2D) ──────────── */
     9→const PERM = new Uint8Array(512);
    10→const GRAD2: [number, number][] = [
    11→  [1, 1], [-1, 1], [1, -1], [-1, -1],
    12→  [1, 0], [-1, 0], [0, 1], [0, -1],
    13→];
    14→
    15→function initPermutation(seed: number = 42) {
    16→  const p = new Uint8Array(256);
    17→  for (let i = 0; i < 256; i++) p[i] = i;
    18→  // Fisher-Yates shuffle with seed
    19→  let s = seed;
    20→  for (let i = 255; i > 0; i--) {
    21→    s = (s * 16807 + 0) % 2147483647;
    22→    const j = s % (i + 1);
    23→    [p[i], p[j]] = [p[j], p[i]];
    24→  }
    25→  for (let i = 0; i < 512; i++) PERM[i] = p[i & 255];
    26→}
    27→
    28→function fade(t: number): number {
    29→  return t * t * t * (t * (t * 6 - 15) + 10);
    30→}
    31→
    32→function lerp(a: number, b: number, t: number): number {
    33→  return a + t * (b - a);
    34→}
    35→
    36→function dot2(g: [number, number], x: number, y: number): number {
    37→  return g[0] * x + g[1] * y;
    38→}
    39→
    40→export function perlin2D(x: number, y: number): number {
    41→  const X = Math.floor(x) & 255;
    42→  const Y = Math.floor(y) & 255;
    43→  const xf = x - Math.floor(x);
    44→  const yf = y - Math.floor(y);
    45→  const u = fade(xf);
    46→  const v = fade(yf);
    47→
    48→  const aa = PERM[PERM[X] + Y] % 8;
    49→  const ab = PERM[PERM[X] + Y + 1] % 8;
    50→  const ba = PERM[PERM[X + 1] + Y] % 8;
    51→  const bb = PERM[PERM[X + 1] + Y + 1] % 8;
    52→
    53→  const x1 = lerp(dot2(GRAD2[aa], xf, yf), dot2(GRAD2[ba], xf - 1, yf), u);
    54→  const x2 = lerp(dot2(GRAD2[ab], xf, yf - 1), dot2(GRAD2[bb], xf - 1, yf - 1), u);
    55→
    56→  return lerp(x1, x2, v); // -1..1
    57→}

    60→initPermutation(Date.now());

   247→/** Stage 1: Chromatic Aberration — radial RGB channel displacement */
   248→function applyChromaticAberration(
   249→  src: Uint8ClampedArray, w: number, h: number,
   250→  redShift: number, blueShift: number
   251→): Uint8ClampedArray {
   252→  if (redShift === 0 && blueShift === 0) return src;
   253→  const out = new Uint8ClampedArray(src.length);
   254→  const cx = w / 2, cy = h / 2;
   255→  const maxDist = Math.sqrt(cx * cx + cy * cy);
   256→
   257→  for (let y = 0; y < h; y++) {
   258→    for (let x = 0; x < w; x++) {
   259→      const idx = (y * w + x) * 4;
   260→      const dx = x - cx, dy = y - cy;
   261→      const dist = Math.sqrt(dx * dx + dy * dy);
   262→      const normDist = dist / maxDist;
   263→
   264→      // Red channel — shift outward from center
   265→      const rShift = redShift * normDist;
   266→      const rx = Math.round(x + (dx / dist) * rShift);
   267→      const ry = Math.round(y + (dy / dist) * rShift);
   268→      const rIdx = (clamp(ry, 0, h - 1) * w + clamp(rx, 0, w - 1)) * 4;
   269→      out[idx] = src[rIdx];
   270→
   271→      // Green channel — no shift (anchor)
   272→      out[idx + 1] = src[idx + 1];
   273→
   274→      // Blue channel — shift inward
   275→      const bShift = blueShift * normDist;
   276→      const bx = Math.round(x - (dx / dist) * bShift);
   277→      const by = Math.round(y - (dy / dist) * bShift);
   278→      const bIdx = (clamp(by, 0, h - 1) * w + clamp(bx, 0, w - 1)) * 4;
   279→      out[idx + 2] = src[bIdx];
   280→
   281→      out[idx + 3] = 255;
   282→    }
   283→  }
   284→  return out;
   285→}

   287→/** Stage 2: Tone Curve Distortion — "Matte Effect" (lift blacks, crush whites) */
   288→function applyToneCurve(
   289→  data: Uint8ClampedArray,
   290→  blackLift: number, whiteCrush: number
   291→): void {
   292→  // Build a 256-entry LUT for speed
   293→  const lut = new Uint8Array(256);
   294→  for (let i = 0; i < 256; i++) {
   295→    // Normalize input 0..255 → 0..1
   296→    const t = i / 255;
   297→    // Apply tone curve: map black→blackLift, white→whiteCrush
   298→    const lifted = blackLift + t * (whiteCrush - blackLift);
   299→    lut[i] = clamp(Math.round(lifted), 0, 255);
   300→  }
   301→  for (let i = 0; i < data.length; i += 4) {
   302→    data[i]     = lut[data[i]];
   303→    data[i + 1] = lut[data[i + 1]];
   304→    data[i + 2] = lut[data[i + 2]];
   305→  }
   306→}

   308→/** Stage 3: Contrast Flattening — reduce global contrast */
   309→function applyContrastReduction(
   310→  data: Uint8ClampedArray, reduction: number
   311→): void {
   312→  if (reduction <= 0) return;
   313→  // factor: 1.0 = no change, 0.65 = 35% reduction
   314→  const factor = 1 - reduction;
   315→  for (let i = 0; i < data.length; i += 4) {
   316→    data[i]     = clamp(Math.round(128 + (data[i] - 128) * factor), 0, 255);
   317→    data[i + 1] = clamp(Math.round(128 + (data[i + 1] - 128) * factor), 0, 255);
   318→    data[i + 2] = clamp(Math.round(128 + (data[i + 2] - 128) * factor), 0, 255);
   319→  }
   320→}
   321→
   322→/** Stage 4: Chrominance Shift — amber/yellow + green tint in shadows */
   323→function applyChrominanceShift(
   324→  data: Uint8ClampedArray,
   325→  amberYellow: number, greenTint: number
   326→): void {
   327→  if (amberYellow <= 0 && greenTint <= 0) return;
   328→  for (let i = 0; i < data.length; i += 4) {
   329→    let r = data[i], g = data[i + 1], b = data[i + 2];
   330→    const lum = (r + g + b) / 3;
   331→    const shadowFactor = 1 - (lum / 255); // 1 in shadows, 0 in highlights
   332→
   333→    // Amber/yellow shift: boost R, slightly boost G, reduce B
   334→    r = r + amberYellow * 40;
   335→    g = g + amberYellow * 15;
   336→    b = b - amberYellow * 25;
   337→
   338→    // Green tint in midtones and shadows
   339→    g = g + greenTint * shadowFactor * 30;
   340→    r = r - greenTint * shadowFactor * 8;
   341→
   342→    data[i]     = clamp(Math.round(r), 0, 255);
   343→    data[i + 1] = clamp(Math.round(g), 0, 255);
   344→    data[i + 2] = clamp(Math.round(b), 0, 255);
   345→  }
   346→}
   347→
   348→/** Stage 5: Saturation control */
   349→function applySaturation(data: Uint8ClampedArray, sat: number): void {
   350→  if (sat === 1) return;
   351→  for (let i = 0; i < data.length; i += 4) {
   352→    const r = data[i], g = data[i + 1], b = data[i + 2];
   353→    const gray = 0.299 * r + 0.587 * g + 0.114 * b;
   354→    data[i]     = clamp(Math.round(gray + (r - gray) * sat), 0, 255);
   355→    data[i + 1] = clamp(Math.round(gray + (g - gray) * sat), 0, 255);
   356→    data[i + 2] = clamp(Math.round(gray + (b - gray) * sat), 0, 255);
   357→  }
   358→}
   359→
   360→/** Stage 5b: B&W mode */
   361→function applyBlackAndWhite(data: Uint8ClampedArray): void {
   362→  for (let i = 0; i < data.length; i += 4) {
   363→    const gray = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2];
   364→    data[i] = data[i + 1] = data[i + 2] = clamp(Math.round(gray), 0, 255);
   365→  }
   366→}

   368→/** Stage 6: Anti-Clarity — simplified bilateral blur (softens fine texture) */
   369→function applyAntiClarity(
   370→  data: Uint8ClampedArray, w: number, h: number,
   371→  radius: number, sigmaSpace: number, sigmaColor: number
   372→): void {
   373→  if (radius <= 0) return;
   374→  const src = new Uint8ClampedArray(data);
   375→  const twoSigmaSpace2 = 2 * sigmaSpace * sigmaSpace;
   376→  const twoSigmaColor2 = 2 * sigmaColor * sigmaColor;
   377→
   378→  for (let y = 0; y < h; y++) {
   379→    for (let x = 0; x < w; x++) {
   380→      const idx = (y * w + x) * 4;
   381→      let sumR = 0, sumG = 0, sumB = 0, sumW = 0;
   382→      const cR = src[idx], cG = src[idx + 1], cB = src[idx + 2];
   383→
   384→      for (let dy = -radius; dy <= radius; dy++) {
   385→        const ny = clamp(y + dy, 0, h - 1);
   386→        for (let dx = -radius; dx <= radius; dx++) {
   387→          const nx = clamp(x + dx, 0, w - 1);
   388→          const nIdx = (ny * w + nx) * 4;
   389→          const nR = src[nIdx], nG = src[nIdx + 1], nB = src[nIdx + 2];
   390→
   391→          // Spatial weight
   392→          const spatialDist2 = dx * dx + dy * dy;
   393→          const wSpatial = Math.exp(-spatialDist2 / twoSigmaSpace2);
   394→
   395→          // Color/range weight
   396→          const colorDist2 = (nR - cR) ** 2 + (nG - cG) ** 2 + (nB - cB) ** 2;
   397→          const wColor = Math.exp(-colorDist2 / twoSigmaColor2);
   398→
   399→          const weight = wSpatial * wColor;
   400→          sumR += nR * weight;
   401→          sumG += nG * weight;
   402→          sumB += nB * weight;
   403→          sumW += weight;
   404→        }
   405→      }
   406→
   407→      data[idx]     = clamp(Math.round(sumR / sumW), 0, 255);
   408→      data[idx + 1] = clamp(Math.round(sumG / sumW), 0, 255);
   409→      data[idx + 2] = clamp(Math.round(sumB / sumW), 0, 255);
   410→    }
   411→  }
   412→}

   414→/** Stage 7: Procedural Film Grain — Perlin noise, luminance-dependent */
   415→function applyFilmGrain(
   416→  data: Uint8ClampedArray, w: number, h: number,
   417→  scale: number, intensity: number, luminanceWeight: number,
   418→  frameSeed: number = 0
   419→): void {
   420→  for (let y = 0; y < h; y++) {
   421→    for (let x = 0; x < w; x++) {
   422→      const idx = (y * w + x) * 4;
   423→      const r = data[idx], g = data[idx + 1], b = data[idx + 2];
   424→      const lum = (r + g + b) / (3 * 255); // 0..1
   425→
   426→      // Perlin noise sample — multiple octaves for organic feel
   427→      const nx = x * scale + frameSeed * 0.1;
   428→      const ny = y * scale + frameSeed * 0.07;
   429→      let noise = perlin2D(nx, ny);
   430→      noise += 0.5 * perlin2D(nx * 2.3, ny * 2.3);
   431→      noise += 0.25 * perlin2D(nx * 4.7, ny * 4.7);
   432→      noise /= 1.75; // normalize to -1..1
   433→
   434→      // Dynamic opacity: higher in midtones/shadows, lower in highlights
   435→      const grainOpacity = intensity * (0.3 + 0.7 * luminanceWeight * (1 - lum));
   436→      const grainValue = noise * grainOpacity * 60; // scale to pixel range
   437→
   438→      data[idx]     = clamp(Math.round(r + grainValue), 0, 255);
   439→      data[idx + 1] = clamp(Math.round(g + grainValue), 0, 255);
   440→      data[idx + 2] = clamp(Math.round(b + grainValue), 0, 255);
   441→    }
   442→  }
   443→}

   445→/** Stage 8: Halation — dreamy glow around highlights */
   446→function applyHalation(
   447→  data: Uint8ClampedArray, w: number, h: number,
   448→  threshold: number, blurRadius: number, blendIntensity: number
   449→): void {
   450→  if (blurRadius <= 0) return;
   451→
   452→  // Extract highlight mask
   453→  const highlightMask = new Float32Array(w * h);
   454→  for (let i = 0; i < w * h; i++) {
   455→    const idx = i * 4;
   456→    const lum = (data[idx] + data[idx + 1] + data[idx + 2]) / (3 * 255);
   457→    if (lum > threshold) {
   458→      highlightMask[i] = (lum - threshold) / (1 - threshold); // 0..1
   459→    }
   460→  }
   461→
   462→  // Box blur the highlight mask (3 passes approximates Gaussian)
   463→  const blurred = new Float32Array(highlightMask);
   464→  for (let pass = 0; pass < 3; pass++) {
   465→    boxBlur1D(blurred, w, h, blurRadius, true);  // horizontal
   466→    boxBlur1D(blurred, w, h, blurRadius, false); // vertical
   467→  }
   468→
   469→  // Screen blend: result = 1 - (1 - base) * (1 - blend)
   470→  for (let i = 0; i < w * h; i++) {
   471→    const idx = i * 4;
   472→    const glow = blurred[i] * blendIntensity;
   473→    // Screen blend mode
   474→    data[idx]     = clamp(Math.round(255 - (255 - data[idx]) * (255 - glow * 255)), 0, 255);
   475→    data[idx + 1] = clamp(Math.round(255 - (255 - data[idx + 1]) * (255 - glow * 255)), 0, 255);
   476→    data[idx + 2] = clamp(Math.round(255 - (255 - data[idx + 2]) * (255 - glow * 255)), 0, 255);
   477→  }
   478→}
   479→
   480→/** 1D box blur pass (horizontal or vertical) */
   481→function boxBlur1D(
   482→  data: Float32Array, w: number, h: number,
   483→  radius: number, horizontal: boolean
   484→): void {
   485→  const temp = new Float32Array(w * h);
   486→  const r = Math.round(radius);
   487→
   488→  for (let y = 0; y < h; y++) {
   489→    for (let x = 0; x < w; x++) {
   490→      let sum = 0, count = 0;
   491→      if (horizontal) {
   492→        for (let dx = -r; dx <= r; dx++) {
   493→          const nx = clamp(x + dx, 0, w - 1);
   494→          sum += data[y * w + nx];
   495→          count++;
   496→        }
   497→      } else {
   498→        for (let dy = -r; dy <= r; dy++) {
   499→          const ny = clamp(y + dy, 0, h - 1);
   500→          sum += data[ny * w + x];
   501→          count++;
   502→        }
   503→      }
   504→      temp[y * w + x] = sum / count;
   505→    }
   506→  }
   507→
   508→  // Copy back
   509→  for (let i = 0; i < data.length; i++) {
   510→    data[i] = temp[i];
   511→  }
   512→}

   514→/** Stage 9: Vignette — radial darkening + corner softness */
   515→function applyVignette(
   516→  ctx: CanvasRenderingContext2D, w: number, h: number,
   517→  intensity: number, cornerBlurFraction: number, cornerBlurRadius: number
   518→): void {
   519→  // Radial darkening
   520→  if (intensity > 0) {
   521→    const gradient = ctx.createRadialGradient(
   522→      w / 2, h / 2, w * 0.25,
   523→      w / 2, h / 2, w * 0.78
   524→    );
   525→    gradient.addColorStop(0, 'rgba(0,0,0,0)');
   526→    gradient.addColorStop(0.6, 'rgba(0,0,0,0)');
   527→    gradient.addColorStop(1, `rgba(0,0,0,${intensity})`);
   528→    ctx.fillStyle = gradient;
   529→    ctx.fillRect(0, 0, w, h);
   530→  }
   531→
   532→  // Corner softness — radial Gaussian blur on outer region
   533→  if (cornerBlurFraction > 0 && cornerBlurRadius > 0) {
   534→    const imageData = ctx.getImageData(0, 0, w, h);
   535→    const data = imageData.data;
   536→    const cx = w / 2, cy = h / 2;
   537→    const maxDist = Math.sqrt(cx * cx + cy * cy);
   538→    const blurStart = (1 - cornerBlurFraction) * maxDist;
   539→    const r = Math.round(cornerBlurRadius);
   540→
   541→    // Simple box blur only on corner pixels
   542→    const src = new Uint8ClampedArray(data);
   543→    for (let y = 0; y < h; y++) {
   544→      for (let x = 0; x < w; x++) {
   545→        const dx = x - cx, dy = y - cy;
   546→        const dist = Math.sqrt(dx * dx + dy * dy);
   547→        if (dist < blurStart) continue;
   548→
   549→        // Blend factor: 0 at blurStart, 1 at edges
   550→        const blurFactor = Math.min(1, (dist - blurStart) / (maxDist - blurStart));
   551→        const idx = (y * w + x) * 4;
   552→        let sumR = 0, sumG = 0, sumB = 0, count = 0;
   553→
   554→        for (let dy2 = -r; dy2 <= r; dy2++) {
   555→          for (let dx2 = -r; dx2 <= r; dx2++) {
   556→            const nx = clamp(x + dx2, 0, w - 1);
   557→            const ny = clamp(y + dy2, 0, h - 1);
   558→            const nIdx = (ny * w + nx) * 4;
   559→            sumR += src[nIdx];
   560→            sumG += src[nIdx + 1];
   561→            sumB += src[nIdx + 2];
   562→            count++;
   563→          }
   564→        }
   565→
   566→        // Blend original with blurred
   567→        data[idx]     = clamp(Math.round(src[idx] * (1 - blurFactor) + (sumR / count) * blurFactor), 0, 255);
   568→        data[idx + 1] = clamp(Math.round(src[idx + 1] * (1 - blurFactor) + (sumG / count) * blurFactor), 0, 255);
   569→        data[idx + 2] = clamp(Math.round(src[idx + 2] * (1 - blurFactor) + (sumB / count) * blurFactor), 0, 255);
   570→      }
   571→    }
   572→    ctx.putImageData(imageData, 0, 0);
   573→  }
   574→}

   576→/** Gate Flicker — per-frame micro-exposure variation for video */
   577→export function getGateFlickerEV(intensity: number): number {
   578→  return (Math.random() - 0.5) * 2 * intensity;
   579→}
   580→
   581→/** Apply gate flicker as brightness adjustment */
   582→export function applyGateFlicker(
   583→  data: Uint8ClampedArray, ev: number
   584→): void {
   585→  // Convert EV to multiplier: 2^EV, but for small EV use linear approx
   586→  const multiplier = 1 + ev * 0.5; // linear approx for small EV
   587→  for (let i = 0; i < data.length; i += 4) {
   588→    data[i]     = clamp(Math.round(data[i] * multiplier), 0, 255);
   589→    data[i + 1] = clamp(Math.round(data[i + 1] * multiplier), 0, 255);
   590→    data[i + 2] = clamp(Math.round(data[i + 2] * multiplier), 0, 255);
   591→  }
   592→}

   594→/** Add vintage date stamp (like old disposable cameras) */
   595→function addDateStamp(ctx: CanvasRenderingContext2D, w: number, h: number): void {
   596→  const now = new Date();
   597→  const dateStr = `${now.getFullYear()}'${String(now.getMonth() + 1).padStart(2, '0')}'${String(now.getDate()).padStart(2, '0')}`;
   598→  ctx.save();
   599→  ctx.font = 'bold 14px "Courier New", monospace';
   600→  ctx.fillStyle = 'rgba(255, 80, 80, 0.75)';
   601→  ctx.shadowColor = 'rgba(0,0,0,0.5)';
   602→  ctx.shadowBlur = 1;
   603→  ctx.textAlign = 'right';
   604→  ctx.fillText(dateStr, w - 12, h - 12);
   605→  ctx.restore();
   606→}

   608→/** Add VHS tracking lines artifact */
   609→function addVHSArtifacts(
   610→  ctx: CanvasRenderingContext2D, w: number, h: number
   611→): void {
   612→  // Random horizontal tracking glitch lines
   613→  ctx.save();
   614→  const numLines = 2 + Math.floor(Math.random() * 4);
   615→  for (let i = 0; i < numLines; i++) {
   616→    const y = Math.floor(Math.random() * h);
   617→    const lineH = 1 + Math.floor(Math.random() * 3);
   618→    const shift = (Math.random() - 0.5) * 20;
   619→
   620→    // Shift a horizontal strip
   621→    const strip = ctx.getImageData(0, y, w, lineH);
   622→    ctx.putImageData(strip, shift, y);
   623→  }
   624→
   625→  // Occasional color bleed bar
   626→  if (Math.random() > 0.7) {
   627→    const barY = Math.floor(Math.random() * h);
   628→    const barH = 2 + Math.floor(Math.random() * 8);
   629→    ctx.fillStyle = `rgba(${Math.random() > 0.5 ? '255,0,100' : '0,200,255'}, 0.08)`;
   630→    ctx.fillRect(0, barY, w, barH);
   631→  }
   632→  ctx.restore();
   633→}

   635→/* ═══════════════════════════════════════════════════════════
   636→   MAIN PIPELINE — Full processing for photo capture
   637→   ═══════════════════════════════════════════════════════════ */
   638→
   639→export function processFrame(
   640→  ctx: CanvasRenderingContext2D,
   641→  w: number,
   642→  h: number,
   643→  presetKey: string,
   644→  options?: {
   645→    isVideoFrame?: boolean;
   646→    frameIndex?: number;
   647→    addStamp?: boolean;
   648→  }
   649→): void {
   650→  const config = PIPELINE_PRESETS[presetKey];
   651→  if (!config) return;
   652→
   653→  const { isVideoFrame = false, frameIndex = 0, addStamp = false } = options || {};
   654→
   655→  let imageData = ctx.getImageData(0, 0, w, h);
   656→  let data = imageData.data;
   657→
   658→  // ── Stage 1: Chromatic Aberration ──
   659→  if (config.chromaticAberration.enabled) {
   660→    data = applyChromaticAberration(
   661→      data, w, h,
   662→      config.chromaticAberration.redShift,
   663→      config.chromaticAberration.blueShift
   664→    );
   665→    imageData = new ImageData(data, w, h);
   666→  }
   667→
   668→  // ── Stage 2: Tone Curve Distortion (Matte Effect) ──
   669→  applyToneCurve(data, config.toneCurve.blackLift, config.toneCurve.whiteCrush);
   670→
   671→  // ── Stage 3: Contrast Flattening ──
   672→  applyContrastReduction(data, config.contrastReduce);
   673→
   674→  // ── Stage 4: Chrominance Shift ──
   675→  applyChrominanceShift(data, config.chrominanceShift.amberYellow, config.chrominanceShift.greenTint);
   676→
   677→  // ── Stage 5: Saturation / B&W ──
   678→  if (config.extraPerPreset === 'bw') {
   679→    applyBlackAndWhite(data);
   680→  } else {
   681→    applySaturation(data, config.saturation);
   682→  }
   683→
   684→  // ── Stage 6: Anti-Clarity (Bilateral Blur) ──
   685→  if (config.antiClarity.enabled && config.antiClarity.radius > 0) {
   686→    // For performance, only apply bilateral blur at reduced resolution or skip on large images
   687→    const pixelCount = w * h;
   688→    if (pixelCount <= 921600) { // up to ~1280x720
   689→      applyAntiClarity(
   690→        data, w, h,
   691→        config.antiClarity.radius,
   692→        config.antiClarity.sigmaSpace,
   693→        config.antiClarity.sigmaColor
   694→      );
   695→    }
   696→  }
   697→
   698→  ctx.putImageData(imageData, 0, 0);
   699→
   700→  // ── Stage 7: Film Grain (Perlin) ──
   701→  if (config.grain.enabled) {
   702→    imageData = ctx.getImageData(0, 0, w, h);
   703→    applyFilmGrain(
   704→      imageData.data, w, h,
   705→      config.grain.scale,
   706→      config.grain.intensity,
   707→      config.grain.luminanceWeight,
   708→      isVideoFrame ? frameIndex : 0
   709→    );
   710→    ctx.putImageData(imageData, 0, 0);
   711→  }
   712→
   713→  // ── Stage 8: Halation (Glow) ──
   714→  if (config.halation.enabled) {
   715→    imageData = ctx.getImageData(0, 0, w, h);
   716→    applyHalation(
   717→      imageData.data, w, h,
   718→      config.halation.threshold,
   719→      config.halation.blurRadius,
   720→      config.halation.intensity
   721→    );
   722→    ctx.putImageData(imageData, 0, 0);
   723→  }
   724→
   725→  // ── Stage 9: Vignette + Corner Softness ──
   726→  applyVignette(ctx, w, h, config.vignette.intensity, config.vignette.cornerBlur, config.vignette.cornerBlurRadius);
   727→
   728→  // ── Gate Flicker (video mode) ──
   729→  if (isVideoFrame && config.gateFlicker.enabled) {
   730→    const ev = getGateFlickerEV(config.gateFlicker.intensity);
   731→    imageData = ctx.getImageData(0, 0, w, h);
   732→    applyGateFlicker(imageData.data, ev);
   733→    ctx.putImageData(imageData, 0, 0);
   734→  }
   735→
   736→  // ── VHS Artifacts ──
   737→  if (presetKey === 'vhs') {
   738→    addVHSArtifacts(ctx, w, h);
   739→  }
   740→
   741→  // ── Date Stamp ──
   742→  if (addStamp || presetKey === 'polaroid' || presetKey === 'sepia70s') {
   743→    addDateStamp(ctx, w, h);
   744→  }
   745→}
   746→
   747→/* ═══════════════════════════════════════════════════════════
   748→   LIGHTWEIGHT PREVIEW — for real-time canvas preview
   749→   Skips heavy stages (bilateral blur, halation, corner blur)
   750→   ═══════════════════════════════════════════════════════════ */
   751→
   752→export function processPreviewFrame(
   753→  ctx: CanvasRenderingContext2D,
   754→  w: number,
   755→  h: number,
   756→  presetKey: string,
   757→  frameIndex: number = 0
   758→): void {
   759→  const config = PIPELINE_PRESETS[presetKey];
   760→  if (!config) return;
   761→
   762→  let imageData = ctx.getImageData(0, 0, w, h);
   763→  let data = imageData.data;
   764→
   765→  // Chromatic aberration (reduced)
   766→  if (config.chromaticAberration.enabled) {
   767→    data = applyChromaticAberration(
   768→      data, w, h,
   769→      config.chromaticAberration.redShift * 0.6,
   770→      config.chromaticAberration.blueShift * 0.6
   771→    );
   772→    imageData = new ImageData(data, w, h);
   773→  }
   774→
   775→  // Tone curve
   776→  applyToneCurve(data, config.toneCurve.blackLift, config.toneCurve.whiteCrush);
   777→
   778→  // Contrast reduction
   779→  applyContrastReduction(data, config.contrastReduce);
   780→
   781→  // Chrominance shift
   782→  applyChrominanceShift(data, config.chrominanceShift.amberYellow, config.chrominanceShift.greenTint);
   783→
   784→  // Saturation / B&W
   785→  if (config.extraPerPreset === 'bw') {
   786→    applyBlackAndWhite(data);
   787→  } else {
   788→    applySaturation(data, config.saturation);
   789→  }
   790→
   791→  ctx.putImageData(imageData, 0, 0);
   792→
   793→  // Film grain (simplified — faster scale)
   794→  if (config.grain.enabled) {
   795→    imageData = ctx.getImageData(0, 0, w, h);
   796→    applyFilmGrain(
   797→      imageData.data, w, h,
   798→      config.grain.scale * 0.7,
   799→      config.grain.intensity * 0.8,
   800→      config.grain.luminanceWeight,
   801→      frameIndex
   802→    );
   803→    ctx.putImageData(imageData, 0, 0);
   804→  }
   805→
   806→  // Simple vignette (no corner blur — too expensive for preview)
   807→  if (config.vignette.intensity > 0) {
   808→    const gradient = ctx.createRadialGradient(
   809→      w / 2, h / 2, w * 0.25,
   810→      w / 2, h / 2, w * 0.78
   811→    );
   812→    gradient.addColorStop(0, 'rgba(0,0,0,0)');
   813→    gradient.addColorStop(0.6, 'rgba(0,0,0,0)');
   814→    gradient.addColorStop(1, `rgba(0,0,0,${config.vignette.intensity})`);
   815→    ctx.fillStyle = gradient;
   816→    ctx.fillRect(0, 0, w, h);
   817→  }
   818→}

   820→/* ──────────── HELPERS ──────────── */
   821→function clamp(v: number, min: number, max: number): number {
   822→  return v < min ? min : v > max ? max : v;
   823→}
