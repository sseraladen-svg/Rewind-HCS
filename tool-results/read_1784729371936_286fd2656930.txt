    32→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    33→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    34→  const [bootLines, setBootLines] = useState<string[]>([]);
    35→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    36→
    37→  const allBootLines = useMemo(() => [
    38→    'BIOS v2.4 — Nostalgia Systems Inc.',
    39→    'POST: Memory test... 640K OK',
    40→    'Detecting floppy drives... A: OK  B: OK',
    41→    'Loading REWIND.EXE...',
    42→    'Mounting childhood memories... 47% complete',
    43→    'Indexing nostalgia database...',
    44→    'Rewinding time...',
    45→    'Loading good old days... 83% complete',
    46→    'Ready.',
    47→  ], []);
    48→
    49→  useEffect(() => {
    50→    const timers: ReturnType<typeof setTimeout>[] = [];
    51→    // Reveal boot lines one by one
    52→    allBootLines.forEach((line, i) => {
    53→      timers.push(setTimeout(() => {
    54→        setBootLines(prev => [...prev, line]);
    55→        if (i === allBootLines.length - 1) {
    56→          // Last line — show title after a beat
    57→          timers.push(setTimeout(() => setPhase(1), 1000));
    58→        }
    59→      }, 500 + i * 500));
    60→    });
    61→
    62→    // Auto-dismiss after title phase
    63→    fadeTimerRef.current = setTimeout(() => {
    64→      setPhase(2);
    65→      setTimeout(onDone, 800);
    66→    }, 7000);
    67→
    68→    return () => {
    69→      timers.forEach(clearTimeout);
    70→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    71→    };
    72→  }, [allBootLines, onDone]);
    73→
    74→  return (
    75→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    76→      <div className="crt-scanlines" />
    77→      <div className="vhs-noise" />
    78→      <div className="vhs-tracking" />
    79→      <div className="vhs-tracking" style={{ animationDelay: '1.2s', opacity: 0.5 }} />
    80→
    81→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
    82→        {/* Boot lines */}
    83→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
    84→          {bootLines.map((line, i) => (
    85→            <div
    86→              key={i}
    87→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    88→              style={{ animationDelay: '0s' }}
    89→            >
    90→              {line}
    91→              {i === bootLines.length - 1 && (
    92→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
    93→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
    94→                />
    95→              )}
    96→            </div>
    97→          ))}
    98→        </div>
    99→
   100→        {/* Title */}
   101→        {phase >= 1 && (
   102→          <motion.div
   103→            initial={{ opacity: 0, scale: 0.9 }}
   104→            animate={{ opacity: 1, scale: 1 }}
   105→            transition={{ duration: 1.0 }}
   106→            className="text-center mt-1 sm:mt-2 w-full"
   107→          >
   108→            <h1 className="text-3xl sm:text-4xl md:text-5xl font-black crt-glow old-title tracking-wider" style={{ color: '#c4956a' }}>
   109→              REWIND
   110→            </h1>
   111→            <div className="loading-bar-track sm:loading-bar-track-md mt-3 sm:mt-4 mx-auto">
   112→              <div className="loading-bar-fill" style={{ animationDuration: '3.5s' }} />
   113→            </div>
   114→            <p className="old-typewriter text-[8px] sm:text-[10px] mt-2 sm:mt-3" style={{ color: 'rgba(196,149,106,0.5)' }}>
   115→              <span style={{ animation: 'blink-cursor 0.8s step-end infinite' }}>_</span>
   116→              {' '}press any key to continue_
   117→            </p>
   118→          </motion.div>
   119→        )}
   120→      </div>
   121→
   122→      {/* VHS timestamp corner */}
   123→      <div className="absolute bottom-2 sm:bottom-4 left-2 sm:left-4 z-10">
   124→        <span className="old-typewriter text-[8px] sm:text-[10px]" style={{ color: 'rgba(196,149,106,0.35)' }}>

   141→function DustMotes() {
   142→  const [mounted, setMounted] = useState(false);
   143→
   144→  useEffect(() => {
   145→    setMounted(true);
   146→  }, []);
   147→

   149→    if (!mounted) return [];

   151→    return Array.from({ length: 25 }, (_, i) => ({
   152→      id: i,
   153→      w: 2 + Math.random() * 6,
   154→      h: 2 + Math.random() * 6,
   155→      l: Math.random() * 100,
   156→      t: Math.random() * 100,
   157→      bg: colors[i % 5],
   158→      delay: Math.random() * 25,
   159→      dur: 20 + Math.random() * 20,
   160→    }));
   161→  }, [mounted]);
   162→
   163→  return (
   164→    <div className="pointer-events-none fixed inset-0 overflow-hidden z-0" aria-hidden>
   165→      {motes.map((m) => (
   166→        <span
   167→          key={m.id}
   168→          className="absolute rounded-full opacity-15 animate-[float_25s_infinite_linear]"
   169→          style={{
   170→            width: `${m.w}px`,
   171→            height: `${m.h}px`,
   172→            left: `${m.l}%`,
   173→            top: `${m.t}%`,
   174→            background: m.bg,
   175→            animationDelay: `${m.delay}s`,
   176→            animationDuration: `${m.dur}s`,

   331→      >
