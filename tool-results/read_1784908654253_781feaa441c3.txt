    32→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    33→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    34→  const [bootLines, setBootLines] = useState<string[]>([]);
    35→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    36→
    37→  const allBootLines = useMemo(() => [
    38→    'BIOS v2.4 — Nostalgia Systems Inc.',
    39→    'POST: Memory test... 640K OK',
    40→    'Detecting floppy drives... A: OK  B: OK',
    41→    'Loading REWIND.EXE...',
    42→    'Mounting childhood memories... 47% complete',
    43→    'Indexing nostalgia database...',
    44→    'Rewinding time...',
    45→    'Loading good old days... 83% complete',
    46→    'Ready.',
    47→  ], []);
    48→
    49→  useEffect(() => {
    50→    const timers: ReturnType<typeof setTimeout>[] = [];
    51→    // Reveal boot lines one by one
    52→    allBootLines.forEach((line, i) => {
    53→      timers.push(setTimeout(() => {
    54→        setBootLines(prev => [...prev, line]);
    55→        if (i === allBootLines.length - 1) {
    56→          // Last line — show title after a beat
    57→          timers.push(setTimeout(() => setPhase(1), 1000));
    58→        }
    59→      }, 500 + i * 500));
    60→    });
    61→
    62→    // Auto-dismiss after title phase
    63→    fadeTimerRef.current = setTimeout(() => {
    64→      setPhase(2);
    65→      setTimeout(onDone, 800);
    66→    }, 7000);
    67→
    68→    return () => {
    69→      timers.forEach(clearTimeout);
    70→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    71→    };
    72→  }, [allBootLines, onDone]);
    73→
    74→  return (
    75→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    76→      <div className="crt-scanlines" />
    77→      <div className="vhs-noise" />
    78→      <div className="vhs-tracking" />
    79→      <div className="vhs-tracking" style={{ animationDelay: '1.2s', opacity: 0.5 }} />
    80→
    81→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
    82→        {/* Boot lines */}
    83→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
    84→          {bootLines.map((line, i) => (
    85→            <div
    86→              key={i}
    87→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    88→              style={{ animationDelay: '0s' }}
    89→            >
    90→              {line}
    91→              {i === bootLines.length - 1 && (
    92→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
    93→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
    94→                />
    95→              )}
    96→            </div>
    97→          ))}
    98→        </div>
    99→
   100→        {/* Title */}
   101→        {phase >= 1 && (
   102→          <motion.div
   103→            initial={{ opacity: 0, scale: 0.9 }}
   104→            animate={{ opacity: 1, scale: 1 }}
   105→            transition={{ duration: 1.0 }}
   106→            className="text-center mt-1 sm:mt-2 w-full"

   112→              <div className="loading-bar-fill" style={{ animationDuration: '3.5s' }} />
   113→            </div>
   114→            <p className="old-typewriter text-[8px] sm:text-[10px] mt-2 sm:mt-3" style={{ color: 'rgba(196,149,106,0.5)' }}>

   116→              {' '}press any key to continue_

   164→    <div className="pointer-events-none fixed inset-0 overflow-hidden z-0" aria-hidden>

   166→        <span
   167→          key={m.id}

   315→function QuoteBanner({ quotes }: { quotes: string[] }) {
   316→  const [idx, setIdx] = useState(0);
   317→  useEffect(() => {
   318→    const t = setInterval(() => setIdx((i) => (i + 1) % quotes.length), 8000);
   319→    return () => clearInterval(t);
   320→  }, [quotes.length]);
   321→
   322→  return (
   323→    <AnimatePresence mode="wait">
   324→      <motion.div
   325→        key={idx}
   326→        initial={{ opacity: 0, y: 10 }}
   327→        animate={{ opacity: 1, y: 0 }}
   328→        exit={{ opacity: 0, y: -10 }}
   329→        transition={{ duration: 0.5 }}
   330→        className="sk-lined p-4 rounded-lg old-quote text-[#5c4033] text-sm md:text-base text-center"
   331→      >
   332→        &ldquo;{quotes[idx]}&rdquo;
   333→      </motion.div>
   334→    </AnimatePresence>
   335→  );
   336→}

   384→function getDecadeTheme(birthYear: number): string {
   385→  const decade = Math.floor(birthYear / 10) * 10;
   386→  if (decade === 1980) return '80s';
   387→  if (decade === 1990) return '90s';
   388→  if (decade === 2000) return '2000s';
   389→  return '';
   390→}

   393→  '80s': '#0a0a0a',
   394→  '90s': '#c0c0c0',

   396→};

   662→export default function Home() {
   663→  const [birthYear, setBirthYear] = useState<number | null>(null);
   664→  const [loading, setLoading] = useState(true);
   665→
   666→  // Run one-time localStorage → IndexedDB migration on first load
   667→  useEffect(() => {
   668→    if (typeof window !== 'undefined') {
   669→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   670→    }
   671→  }, []);
   672→
   673→  const handleLoadingDone = useCallback(() => {
   674→    setLoading(false);
   675→  }, []);
   676→
   677→  const handleSubmit = useCallback((year: number) => {
   678→    triggerOldLoader('Tuning into your era...');
   679→    setTimeout(() => {
   680→      setBirthYear(year);
   681→      window.scrollTo({ top: 0, behavior: 'smooth' });
   682→      hideOldLoader();
   683→    }, 1100);
   684→  }, []);
   685→
   686→  const handleBack = useCallback(() => {
   687→    triggerOldLoader('Rewinding...');
   688→    setTimeout(() => {
   689→      setBirthYear(null);
   690→      hideOldLoader();
   691→    }, 850);
   692→  }, []);
   693→
   694→  return (
   695→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   696→      {/* CRT Boot Loading Screen */}
   697→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   698→
   699→      <AnimatePresence mode="wait">
   700→        {birthYear === null ? (
   701→          <motion.div
   702→            key="hero"
   703→            initial={{ opacity: 0 }}
   704→            animate={{ opacity: 1 }}
   705→            exit={{ opacity: 0, y: -20 }}
   706→            transition={{ duration: 0.4 }}
   707→            style={{ opacity: loading ? 0 : 1 }}
   708→          >
   709→            <HeroSection onSubmit={handleSubmit} />
   710→          </motion.div>
   711→        ) : (
   712→          <motion.div
   713→            key="dashboard"
   714→            initial={{ opacity: 0, y: 20 }}
   715→            animate={{ opacity: 1, y: 0 }}
   716→            exit={{ opacity: 0 }}
   717→            transition={{ duration: 0.5 }}
   718→            className="flex flex-col min-h-screen"
   719→          >
   720→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   721→          </motion.div>
   722→        )}
   723→      </AnimatePresence>
   724→    </div>
   725→  );
   726→}
