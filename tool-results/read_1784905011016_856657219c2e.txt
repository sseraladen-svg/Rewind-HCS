    24→    24→function formatTime(seconds: number): string {
    25→    25→  const m = Math.floor(seconds / 60);
    26→    26→  const s = Math.floor(seconds % 60);
    27→    27→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    28→    28→}
    29→    29→
    30→    30→function generateId(): string {
    31→    31→  return `vm_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
    32→    32→}
    33→    33→
    34→    34→function formatDate(iso: string): string {
    35→    35→  const d = new Date(iso);
    36→    36→  const month = d.toLocaleString('en-US', { month: 'short' }).toUpperCase();
    37→    37→  const day = d.getDate();
    38→    38→  const hour = d.getHours();
    39→    39→  const ampm = hour >= 12 ? 'PM' : 'AM';
    40→    40→  const h12 = hour % 12 || 12;
    41→    41→  const min = String(d.getMinutes()).padStart(2, '0');
    42→    42→  return `${month} ${day}, ${h12}:${min} ${ampm}`;
    43→    43→}

    45→    45→/** Play a short beep using Web Audio API */
    46→    46→function playBeep(frequency = 1000, duration = 0.15, type: OscillatorType = 'square') {
    47→    47→  if (typeof window === 'undefined') return;
    48→    48→  try {
    49→    49→    const ctx = new AudioContext();
    50→    50→    const osc = ctx.createOscillator();
    51→    51→    const gain = ctx.createGain();
    52→    52→    osc.type = type;
    53→    53→    osc.frequency.value = frequency;
    54→    54→    gain.gain.value = 0.12;
    55→    55→    osc.connect(gain);
    56→    56→    gain.connect(ctx.destination);
    57→    57→    osc.start(ctx.currentTime);
    58→    58→    osc.stop(ctx.currentTime + duration);
    59→    59→    osc.onended = () => ctx.close();
    60→    60→  } catch {
    61→    61→    // silently ignore audio errors
    62→    62→  }
    63→    63→}

    66→    66→function generateWaveformBars(count = 32): number[] {
    67→    67→  return Array.from({ length: count }, () => Math.random() * 0.7 + 0.15);
    68→    68→}

   129→   129→        audioElRef.current.pause();
   130→   130→        audioElRef.current = null;
   131→   131→      }
   132→   132→      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);

   134→   134→        URL.revokeObjectURL(objectUrlRef.current);
   135→   135→        objectUrlRef.current = null;

   287→   287→          URL.revokeObjectURL(objectUrlRef.current);
   288→   288→          objectUrlRef.current = null;

   294→   294→        setStatus('ready');

   297→   297→          objectUrlRef.current = null;
   298→   298→        }
