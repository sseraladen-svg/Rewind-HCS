   140→function GuestbookSection({ visitorCount }: { visitorCount: number }) {
   141→  const [entries, setEntries] = useState<GuestbookEntry[]>([]);
   142→  const [name, setName] = useState('');
   143→  const [message, setMessage] = useState('');
   144→  const [isLoaded, setIsLoaded] = useState(false);
   145→
   146→  useEffect(() => {
   147→    const stored = localStorage.getItem('rewind_guestbook');
   148→    if (stored) {
   149→      try {
   150→        setEntries(JSON.parse(stored));
   151→      } catch {
   152→        setEntries([]);
   153→      }
   154→    }
   155→    setIsLoaded(true);
   156→  }, []);
   157→
   158→  const handleSubmit = useCallback(() => {
   159→    const trimmedName = name.trim();
   160→    const trimmedMessage = message.trim();
   161→    if (!trimmedName || !trimmedMessage) return;
   162→
   163→    const newEntry: GuestbookEntry = {
   164→      id: generateId(),
   165→      name: trimmedName,
   166→      message: trimmedMessage,
   167→      date: new Date().toISOString(),
   168→      emoji: getRandomEmoji(),
   169→    };
   170→
   171→    const updated = [newEntry, ...entries].slice(0, MAX_GUESTBOOK_ENTRIES);
   172→    setEntries(updated);
   173→    localStorage.setItem('rewind_guestbook', JSON.stringify(updated));
   174→    setName('');
   175→    setMessage('');
   176→  }, [name, message, entries]);
   177→
   178→  const rotations = [-2, 1.5, -1, 2, -0.5, 1, -1.5, 0.5];
   179→
   180→  return (
   181→    <section id="guestbook" className="scroll-mt-8">
   182→      <SectionHeader
   183→        icon={BookOpen}
   184→        title="Guestbook"
   185→        subtitle="Sign my guestbook! Just like the good old GeoCities days."
   186→      />
   187→
   188→      {/* Visitor Counter Badge */}
   189→      <motion.div
   190→        initial={{ scale: 0.8, opacity: 0 }}
   191→        animate={isLoaded ? { scale: 1, opacity: 1 } : {}}
   192→        transition={{ delay: 0.3, type: 'spring', stiffness: 200 }}
   193→        className="flex justify-center mb-8"
   194→      >
   195→        <div className="sk-metal rounded-lg px-5 py-2.5 flex items-center gap-3">
   196→          <Eye className="w-4 h-4 text-[#5c4033]" />
   197→          <span className="old-label text-[#3d2b1f]">You are visitor #</span>
   198→          <span className="old-typewriter text-lg font-bold text-[#8b5e3c] tracking-widest">
   199→            {String(visitorCount).padStart(6, '0')}
   200→          </span>
   201→        </div>
   202→      </motion.div>
   203→
   204→      {/* Sign Guestbook Form */}
   205→      <motion.div
   206→        initial={{ opacity: 0, y: 20 }}
   207→        animate={isLoaded ? { opacity: 1, y: 0 } : {}}
   208→        transition={{ delay: 0.4, duration: 0.5 }}
   209→        className="max-w-lg mx-auto mb-10"
   210→      >
   211→        <div className="sk-paper rounded-xl p-6 relative sk-tape">
   212→          <h3 className="old-heading text-xl text-foreground mb-4 text-center">
   213→            ✍️ Leave a Note
   214→          </h3>
   215→          <div className="space-y-3">
   216→            <div className="flex items-center gap-2">
   217→              <User className="w-4 h-4 text-muted-foreground shrink-0" />
   218→              <Input
   219→                value={name}
   220→                onChange={(e) => setName(e.target.value)}
   221→                placeholder="Your name..."
   222→                maxLength={40}
   223→                className="sk-inset old-typewriter text-sm border-0"
   224→              />
   225→            </div>
   226→            <div className="flex items-start gap-2">
   227→              <MessageSquare className="w-4 h-4 text-muted-foreground shrink-0 mt-2" />
   228→              <Textarea
   229→                value={message}
   230→                onChange={(e) => setMessage(e.target.value)}
   231→                placeholder="Write something nice..."
   232→                maxLength={200}
   233→                rows={3}
   234→                className="sk-inset old-typewriter text-sm border-0 resize-none"
   235→              />
   236→            </div>
   237→            <div className="flex justify-end">
   238→              <Button
   239→                onClick={handleSubmit}
   240→                disabled={!name.trim() || !message.trim()}
   241→                className="sk-btn old-btn text-xs px-5 py-2 rounded-lg hover:opacity-90 disabled:opacity-40"
   242→              >
   243→                <Send className="w-3.5 h-3.5 mr-1.5" />
   244→                Sign!
   245→              </Button>
   246→            </div>
   247→          </div>
   248→        </div>
   249→      </motion.div>
   250→
   251→      {/* Guestbook Entries */}
   252→      <div className="max-w-2xl mx-auto">
   253→        <AnimatePresence mode="popLayout">
   254→          {entries.length === 0 && isLoaded && (
   255→            <motion.p
   256→              initial={{ opacity: 0 }}
   257→              animate={{ opacity: 1 }}
   258→              className="old-handwritten text-center text-muted-foreground text-lg py-8"
   259→            >
   260→              Be the first to sign the guestbook! 📝
   261→            </motion.p>
   262→          )}
   263→          {entries.map((entry, index) => (
   264→            <motion.div
   265→              key={entry.id}
   266→              layout
   267→              initial={{ opacity: 0, scale: 0.8, rotate: rotations[index % rotations.length] - 5 }}
   268→              animate={{ opacity: 1, scale: 1, rotate: rotations[index % rotations.length] }}
   269→              exit={{ opacity: 0, scale: 0.7, y: -20 }}
   270→              transition={{ duration: 0.4, type: 'spring', stiffness: 150 }}
   271→              className="sk-paper-aged rounded-lg p-4 mb-4 relative sk-pin cursor-default"
   272→              style={{ '--rotate': `${rotations[index % rotations.length]}deg` } as React.CSSProperties}
   273→            >
   274→              <div className="flex items-start gap-3">
   275→                <span className="text-2xl shrink-0 mt-0.5">{entry.emoji}</span>
   276→                <div className="flex-1 min-w-0">
   277→                  <div className="flex items-center justify-between mb-1">
   278→                    <span className="old-heading text-sm font-semibold text-foreground truncate">
   279→                      {entry.name}
   280→                    </span>
   281→                    <span className="old-label text-[10px] shrink-0 ml-2">{formatDate(entry.date)}</span>
   282→                  </div>
   283→                  <p className="old-body text-sm text-foreground/80 leading-relaxed">
   284→                    {entry.message}
   285→                  </p>
   286→                </div>
   287→              </div>
   288→            </motion.div>
   289→          ))}
   290→        </AnimatePresence>
   291→      </div>
   292→    </section>
   293→  );
   294→}

   300→function MixtapeSection() {
   301→  const [to, setTo] = useState('');
   302→  const [from, setFrom] = useState('');
   303→  const [side, setSide] = useState<'A' | 'B'>('A');
   304→  const [tracks, setTracks] = useState<string[]>(Array(MIXTAPE_TRACK_SLOTS).fill(''));
   305→  const [savedMixtapes, setSavedMixtapes] = useState<Mixtape[]>([]);
   306→  const [isLoaded, setIsLoaded] = useState(false);
   307→  const [showSaved, setShowSaved] = useState(false);
   308→
   309→  useEffect(() => {
   310→    const stored = localStorage.getItem('rewind_mixtapes');
   311→    if (stored) {
   312→      try {
   313→        setSavedMixtapes(JSON.parse(stored));
   314→      } catch {
   315→        setSavedMixtapes([]);
   316→      }
   317→    }
   318→    setIsLoaded(true);
   319→  }, []);
   320→
   321→  const handleTrackChange = useCallback((index: number, value: string) => {
   322→    setTracks((prev) => {
   323→      const updated = [...prev];
   324→      updated[index] = value;
   325→      return updated;
   326→    });
   327→  }, []);
   328→
   329→  const handleSave = useCallback(() => {
   330→    const filledTracks = tracks.filter((t) => t.trim() !== '');
   331→    if (!to.trim() || filledTracks.length === 0) return;
   332→
   333→    const mixtape: Mixtape = {
   334→      id: generateId(),
   335→      to: to.trim(),
   336→      from: from.trim() || 'Anonymous',
   337→      side,
   338→      tracks: filledTracks.map((t) => t.trim()),
   339→      createdAt: new Date().toISOString(),
   340→    };
   341→
   342→    const updated = [mixtape, ...savedMixtapes];
   343→    setSavedMixtapes(updated);
   344→    localStorage.setItem('rewind_mixtapes', JSON.stringify(updated));
   345→    setTo('');
   346→    setFrom('');
   347→    setTracks(Array(MIXTAPE_TRACK_SLOTS).fill(''));
   348→  }, [to, from, side, tracks, savedMixtapes]);
   349→
   350→  const handleDelete = useCallback(
   351→    (id: string) => {
   352→      const updated = savedMixtapes.filter((m) => m.id !== id);
   353→      setSavedMixtapes(updated);
   354→      localStorage.setItem('rewind_mixtapes', JSON.stringify(updated));
   355→    },
   356→    [savedMixtapes],
   357→  );
   358→
   359→  return (
   360→    <section id="mixtape" className="scroll-mt-8">
   361→      <SectionHeader
   362→        icon={Music}
   363→        title="Mixtape Creator"
   364→        subtitle="Make a mixtape for that special someone, just like the old days."
   365→      />
   366→

   368→        {/* Cassette Tape Visual */}
   369→        <motion.div
   370→          initial={{ opacity: 0, y: 30 }}
   371→          animate={isLoaded ? { opacity: 1, y: 0 } : {}}
   372→          transition={{ duration: 0.6, type: 'spring' }}
   373→          className="relative"
   374→        >
   375→          <div className="sk-metal rounded-2xl p-1">
   376→            {/* Cassette body */}
   377→            <div className="bg-gradient-to-b from-[#2a2a2a] to-[#1a1a1a] rounded-xl p-5 relative overflow-hidden">
   378→              {/* Noise overlay */}
   379→              <div className="absolute inset-0 sk-noise pointer-events-none" />
   380→
   381→              {/* Top section: reels */}
   382→              <div className="flex items-center justify-center gap-10 mb-4 relative z-10">
   383→                {/* Left reel */}
   384→                <div className="relative">
   385→                  <div className="w-20 h-20 rounded-full border-4 border-[#444] bg-[#222] flex items-center justify-center">
   386→                    <div className="w-14 h-14 rounded-full border-2 border-[#555] bg-gradient-to-br from-[#333] to-[#1a1a1a] flex items-center justify-center">
   387→                      <div className="w-4 h-4 rounded-full bg-[#555] border border-[#666]" />
   388→                    </div>
   389→                  </div>
   390→                  {/* Spinning spokes */}
   391→                  <motion.div
   392→                    className="absolute inset-0 flex items-center justify-center"
   393→                    animate={side === 'A' ? { rotate: 360 } : { rotate: 0 }}
   394→                    transition={
   395→                      side === 'A'
   396→                        ? { duration: 3, repeat: Infinity, ease: 'linear' }
   397→                        : { duration: 0 }
   398→                    }
   399→                  >
   400→                    <div className="w-16 h-16">
   401→                      {[0, 60, 120, 180, 240, 300].map((deg) => (
   402→                        <div
   403→                          key={deg}
   404→                          className="absolute top-1/2 left-1/2 w-px h-7 bg-[#444]"
