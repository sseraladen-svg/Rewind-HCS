   121→// ═══════════════════════════════════════════════════════════════
   122→// useStoredState — localStorage-backed state via useSyncExternalStore
   123→// ═══════════════════════════════════════════════════════════════
   124→
   125→function useStoredState<T>(
   126→  key: string,
   127→  initial: T,
   128→): [T, (updater: T | ((prev: T) => T)) => void] {
   129→  const subscribe = useCallback(
   130→    (onStoreChange: () => void) => {
   131→      const handler = (e: StorageEvent) => {
   132→        if (e.key === key || e.key === null) onStoreChange();
   133→      };
   134→      window.addEventListener('storage', handler);
   135→      return () => window.removeEventListener('storage', handler);
   136→    },
   137→    [key],
   138→  );
   139→
   140→  const getSnapshot = useCallback(() => {
   141→    return localStorage.getItem(key);
   142→  }, [key]);
   143→
   144→  const getServerSnapshot = useCallback(() => null, []);
   145→
   146→  const raw = useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
   147→
   148→  const value = useMemo<T>(() => {
   149→    if (raw === null) return initial;
   150→    try {
   151→      return JSON.parse(raw) as T;
   152→    } catch {
   153→      return initial;
   154→    }
   155→  }, [raw, initial]);
   156→
   157→  const setValue = useCallback(
   158→    (updater: T | ((prev: T) => T)) => {
   159→      try {
   160→        const currentRaw = localStorage.getItem(key);
   161→        const current: T =
   162→          currentRaw !== null ? (JSON.parse(currentRaw) as T) : initial;
   163→        const next =
   164→          typeof updater === 'function'
   165→            ? (updater as (prev: T) => T)(current)
   166→            : updater;
   167→        localStorage.setItem(key, JSON.stringify(next));
   168→        window.dispatchEvent(new StorageEvent('storage', { key }));
   169→      } catch {
   170→        /* ignore storage errors */
   171→      }
   172→    },
   173→    [key, initial],
   174→  );
   175→
   176→  return [value, setValue];
   177→}

   181→// ═══════════════════════════════════════════════════════════════

   183→// Only called when isClient is true, so localStorage is available.
   184→// The useState lazy initializer runs exactly once on first mount.
   185→function useVisitCounter(): number {
   186→  const [count] = useState(() => {
   187→    const stored = localStorage.getItem('rewind_visit_count');
   188→    let c = 1;
   189→    if (stored) {
   190→      const parsed = parseInt(stored, 10);
   191→      if (!isNaN(parsed)) c = parsed + 1;
   192→    }
   193→    localStorage.setItem('rewind_visit_count', String(c));
   194→    return c;
   195→  });
   196→  return count;
   197→}
