   754→/* ═══════════════════════════════════════════════════════════════
   755→   Main OnboardingFlow Component
   756→   ═══════════════════════════════════════════════════════════════ */
   757→
   758→export default function OnboardingFlow({ onComplete }: OnboardingFlowProps) {
   759→  const [currentStep, setCurrentStep] = useState(0);
   760→  const [direction, setDirection] = useState<number>(STEP_DIRECTION.forward);
   761→  const [birthYear, setBirthYearState] = useState('');
   762→  const [selectedEra, setSelectedEra] = useState<string | null>(null);
   763→  const [currentFeature, setCurrentFeature] = useState(0);
   764→  const [isLoaded, setIsLoaded] = useState(false);
   765→  const [isCompleting, setIsCompleting] = useState(false);
   766→
   767→  /* ───── Check IndexedDB on mount ───── */
   768→  useEffect(() => {
   769→    let cancelled = false;
   770→
   771→    (async () => {
   772→      try {
   773→        const db = await getDB();
   774→        if (cancelled) return;
   775→
   776→        const record: any = await db.get('era_skin', ONBOARDING_ID);
   777→        if (cancelled) return;
   778→
   779→        if (record?.complete) {
   780→          // Already onboarded — restore state and skip
   781→          const savedBirthYear = record.birthYear ?? null;
   782→          const savedEra = record.eraSkin ?? null;
   783→
   784→          if (savedBirthYear && savedEra) {
   785→            onComplete({ birthYear: savedBirthYear, eraSkin: savedEra });
   786→            return;
   787→          }
   788→        }
   789→        setIsLoaded(true);
   790→      } catch {
   791→        // If IDB fails, still show onboarding
   792→        setIsLoaded(true);
   793→      }
   794→    })();
   795→
   796→    return () => { cancelled = true; };
   797→  }, [onComplete]);
   798→
   799→  /* ───── Persistence helpers ───── */
   800→  const saveOnboardingData = useCallback(
   801→    async (year: number, era: string) => {
   802→      const db = await getDB();
   803→      const record: OnboardingRecord = {
   804→        id: ONBOARDING_ID,
   805→        birthYear: year,
   806→        eraSkin: era,
   807→        complete: true,
   808→      };
   809→      await db.put('era_skin', record as any);
   810→      // Also set the era skin in the standard location
   811→      if (era !== 'mixed') {
   812→        await setEraSkinId(era);
   813→      }
   814→    },
   815→    [],
   816→  );
   817→
   818→  /* ───── Navigation helpers ───── */
   819→  const goNext = useCallback(() => {
   820→    soundButtonClick();
   821→    setDirection(STEP_DIRECTION.forward);
   822→    setCurrentStep((s) => Math.min(s + 1, TOTAL_STEPS - 1));
   823→  }, []);
   824→
   825→  const goBack = useCallback(() => {
   826→    soundButtonClick();
   827→    setDirection(STEP_DIRECTION.backward);
   828→    setCurrentStep((s) => Math.max(s - 1, 0));
   829→  }, []);
   830→
   831→  const setBirthYear = useCallback((val: string) => {
   832→    setBirthYearState(val);
   833→  }, []);
   834→
   835→  const canContinueBirthYear = useMemo(() => {
   836→    const y = parseInt(birthYear);
   837→    return y >= 1970 && y <= 2015;
   838→  }, [birthYear]);
   839→
   840→  /* ───── Handle final completion ───── */
   841→  const handleComplete = useCallback(() => {
   842→    soundChime();
   843→    const year = parseInt(birthYear);
   844→    const era = selectedEra || 'mixed';
   845→    setIsCompleting(true);
   846→
   847→    // Small delay for the exit animation to feel natural
   848→    setTimeout(async () => {
   849→      try {
   850→        await saveOnboardingData(year, era);
   851→      } catch (err) {
   852→        console.warn('[Onboarding] Failed to persist:', err);
   853→      }
   854→      onComplete({ birthYear: year, eraSkin: era });
   855→    }, 600);
   856→  }, [birthYear, selectedEra, saveOnboardingData, onComplete]);
   857→
   858→  /* ───── Loading state while checking IDB ───── */
   859→  if (!isLoaded) {
   860→    return (
   861→      <div className="min-h-screen flex items-center justify-center bg-[#f5edd6]">
   862→        <motion.div
   863→          initial={{ opacity: 0 }}
   864→          animate={{ opacity: 1 }}
   865→          className="text-center"
   866→        >
   867→          <div className="text-4xl mb-3">📼</div>
   868→          <p className="old-typewriter text-sm text-[#8b7355]">Loading...</p>
   869→        </motion.div>
   870→      </div>
   871→    );
   872→  }
   873→
   874→  /* ───── Render current step ───── */
   875→  const renderStep = () => {
   876→    switch (currentStep) {
   877→      case 0:
   878→        return <WelcomeStep onNext={goNext} />;
   879→      case 1:
   880→        return (
   881→          <BirthYearStep
   882→            birthYear={birthYear}
   883→            setBirthYear={setBirthYear}
   884→            onNext={goNext}
   885→            onBack={goBack}
   886→            canContinue={canContinueBirthYear}
   887→          />
   888→        );
   889→      case 2:
   890→        return (
   891→          <EraSkinStep
   892→            selectedEra={selectedEra}
   893→            setSelectedEra={setSelectedEra}
   894→            onNext={goNext}
   895→            onBack={goBack}
   896→          />
   897→        );
   898→      case 3:
   899→        return (
   900→          <FeatureTourStep
   901→            currentFeature={currentFeature}
   902→            setCurrentFeature={setCurrentFeature}
   903→            onNext={handleComplete}
   904→            onBack={goBack}
   905→          />
   906→        );
   907→      default:
   908→        return null;
   909→    }
   910→  };
   911→
   912→  return (
   913→    <AnimatePresence mode="wait">
   914→      {!isCompleting ? (
   915→        <motion.div
   916→          key="onboarding-container"
   917→          initial={{ opacity: 0 }}
   918→          animate={{ opacity: 1 }}
   919→          exit={{ opacity: 0, scale: 0.95, y: -20 }}
   920→          transition={{ duration: 0.6, ease: 'easeInOut' }}
   921→          className="min-h-screen flex flex-col"
   922→          style={{
   923→            background:
   924→              'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)',
   925→          }}
   926→        >
   927→          {/* Main step area */}
   928→          <div className="flex-1 flex items-center justify-center overflow-hidden">
   929→            <AnimatePresence custom={direction} mode="wait">
   930→              <motion.div
   931→                key={currentStep}
   932→                custom={direction}
   933→                variants={slideVariants}
   934→                initial="enter"
   935→                animate="center"
   936→                exit="exit"
   937→                transition={slideTransition}
   938→                className="w-full"
   939→              >
   940→                {renderStep()}
   941→              </motion.div>
   942→            </AnimatePresence>
   943→          </div>
   944→
   945→          {/* Progress dots — fixed at bottom */}
   946→          <div className="pb-8 pt-4">
   947→            <ProgressDots current={currentStep} total={TOTAL_STEPS} />
   948→          </div>
   949→        </motion.div>
   950→      ) : (
   951→        /* Completion celebration overlay */
   952→        <motion.div
   953→          key="onboarding-complete"
   954→          initial={{ opacity: 0 }}
   955→          animate={{ opacity: 1 }}
   956→          className="min-h-screen flex items-center justify-center bg-[#f5edd6]"
   957→        >
   958→          <motion.div
   959→            initial={{ scale: 0, rotate: -20 }}
   960→            animate={{ scale: 1, rotate: 0 }}
   961→            transition={{ type: 'spring', stiffness: 200, damping: 15 }}
   962→            className="text-center px-6"
   963→          >
   964→            <motion.div
   965→              className="text-6xl mb-4"
   966→              animate={{ rotate: [0, 10, -10, 0] }}
   967→              transition={{ duration: 0.8, repeat: 2 }}
   968→            >
   969→              🎉
   970→            </motion.div>
   971→            <h2 className="old-title text-3xl font-black text-[#5c4033] mb-2">
   972→              All Set!
   973→            </h2>
   974→            <p className="old-body text-[#8b7355]">
   975→              Rewinding to your era...
   976→            </p>
   977→            <motion.div
   978→              className="mt-6 mx-auto"
   979→              style={{ width: 120, height: 4, borderRadius: 2 }}
   980→              initial={{ opacity: 0.3 }}
   981→              animate={{ opacity: [0.3, 1, 0.3] }}
   982→              transition={{ duration: 1.2, repeat: Infinity }}
   983→            >
   984→              <div
   985→                className="h-full rounded-full"
   986→                style={{
   987→                  width: '100%',
   988→                  background: 'linear-gradient(90deg, #8b5e3c, #c4956a, #8b5e3c)',
   989→                  backgroundSize: '200% 100%',
   990→                  animation: 'rewind-shimmer 1.5s ease-in-out infinite',
   991→                }}
   992→              />
   993→            </motion.div>
   994→          </motion.div>
   995→        </motion.div>
   996→      )}
   997→    </AnimatePresence>
   998→  );
   999→}
