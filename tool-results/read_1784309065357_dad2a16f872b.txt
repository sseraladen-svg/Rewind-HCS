    74→function shuffle<T>(arr: T[]): T[] {
    75→  const a = [...arr];
    76→  for (let i = a.length - 1; i > 0; i--) {
    77→    const j = Math.floor(Math.random() * (i + 1));
    78→    [a[i], a[j]] = [a[j], a[i]];
    79→  }
    80→  return a;
    81→}

    83→function MemoryMatchGame({ data }: { data: DecadeData }) {
    84→  const [cards, setCards] = useState<MatchCard[]>([]);
    85→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
    86→  const [moves, setMoves] = useState(0);
    87→  const [matches, setMatches] = useState(0);
    88→  const [gameWon, setGameWon] = useState(false);
    89→  const lockRef = useRef(false);
    90→
    91→  // Initialize cards client-side only to avoid SSR hydration mismatch from Math.random()
    92→  useEffect(() => {
    93→    const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
    94→    const picked = shuffle(pool).slice(0, 6);
    95→    const initial: MatchCard[] = picked.flatMap((item, i) => [
    96→      { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
    97→      { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
    98→    ]);
    99→    // Deferred init to avoid calling setState synchronously in effect body
   100→    const id = requestAnimationFrame(() => setCards(shuffle(initial)));
   101→    return () => cancelAnimationFrame(id);
   102→  }, [data]);
   103→
   104→  const resetGame = useCallback(() => {
   105→    const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   106→    const picked = shuffle(pool).slice(0, 6);
   107→    const fresh: MatchCard[] = picked.flatMap((item, i) => [
   108→      { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   109→      { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   110→    ]);
   111→    setCards(shuffle(fresh));
   112→    setFlippedIds([]);
   113→    setMoves(0);
   114→    setMatches(0);
   115→    setGameWon(false);
   116→    lockRef.current = false;
   117→  }, [data]);
   118→
   119→  const handleFlip = useCallback((id: number) => {
   120→    if (lockRef.current) return;
   121→    const card = cards.find(c => c.id === id);
   122→    if (!card || card.isFlipped || card.isMatched) return;
   123→    if (flippedIds.includes(id)) return;
   124→    if (flippedIds.length >= 2) return;
   125→
   126→    const newFlipped = [...flippedIds, id];
   127→    setCards(prev => prev.map(c => c.id === id ? { ...c, isFlipped: true } : c));
   128→    setFlippedIds(newFlipped);
   129→
   130→    if (newFlipped.length === 2) {
   131→      setMoves(m => m + 1);
   132→      lockRef.current = true;
   133→      const first = cards.find(c => c.id === newFlipped[0])!;
   134→      const second = cards.find(c => c.id === newFlipped[1])!;
   135→
   136→      if (first.label === second.label) {
   137→        setTimeout(() => {
   138→          setCards(prev => prev.map(c =>
   139→            c.id === first.id || c.id === second.id ? { ...c, isMatched: true, isFlipped: true } : c
   140→          ));
   141→          setMatches(m => {
   142→            const newM = m + 1;
   143→            if (newM === 6) setGameWon(true);
   144→            return newM;
   145→          });
   146→          setFlippedIds([]);
   147→          lockRef.current = false;
   148→        }, 400);
   149→      } else {
   150→        setTimeout(() => {
   151→          setCards(prev => prev.map(c =>
   152→            c.id === first.id || c.id === second.id ? { ...c, isFlipped: false } : c
   153→          ));
   154→          setFlippedIds([]);
   155→          lockRef.current = false;
   156→        }, 900);
   157→      }
   158→    }
   159→  }, [cards, flippedIds]);
   160→
   161→  return (
   162→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   163→      <div className="flex items-center justify-between mb-4">
   164→        <h4 className="font-bold text-base text-[#5c4033] flex items-center gap-2">
   165→          <span className="text-xl">🃏</span> Memory Match
   166→        </h4>
   167→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   168→          <span className="sk-inset px-2 py-1 rounded">Moves: {moves}</span>
   169→          <span className="sk-inset px-2 py-1 rounded">Pairs: {matches}/6</span>
   170→          <button onClick={resetGame} className="sk-btn px-3 py-1 rounded text-xs cursor-pointer">
   171→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   172→          </button>
   173→        </div>
   174→      </div>
   175→
   176→      {gameWon && (
   177→        <motion.div
   178→          initial={{ scale: 0.8, opacity: 0 }}
   179→          animate={{ scale: 1, opacity: 1 }}
   180→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   181→        >
   182→          <span className="text-5xl mb-2">🎉</span>
   183→          <p className="text-lg font-bold text-[#5c4033]">You found them all!</p>
   184→          <p className="text-sm text-[#8b7355] mb-3">Completed in {moves} moves</p>
   185→          <button onClick={resetGame} className="sk-btn px-5 py-2 rounded-lg font-semibold cursor-pointer">
   186→            Play Again
   187→          </button>
   188→        </motion.div>
   189→      )}
   190→
   191→      <div className="grid grid-cols-4 gap-2">
   192→        {cards.map((card) => {
   193→          const showFront = card.isFlipped || card.isMatched;
   194→          return (
   195→            <button
   196→              key={card.id}
   197→              onClick={() => handleFlip(card.id)}
   198→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   199→              disabled={card.isMatched}
   200→              style={{
   201→                background: showFront
   202→                  ? '#faf6eb'
   203→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   204→                borderColor: showFront
   205→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')
   206→                  : '#5c4033',
   207→                boxShadow: showFront
   208→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   209→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   210→                opacity: card.isMatched ? 0.6 : 1,
   211→              }}
   212→            >
   213→              {showFront ? (
   214→                <>
   215→                  <span className="text-2xl md:text-3xl">{card.emoji}</span>

   219→                <span className="text-xl text-[#d4a574] font-bold">?</span>

   225→    </div>

   229→/* ═══════════════════════════════════════════════════
   230→   GAME 2: GUESS THE YEAR

   232→interface TriviaQ {
   233→  question: string;
   234→  emoji: string;
   235→  correct: number;
   236→  options: string[];
   237→}
   238→
   239→function buildTrivia(data: DecadeData): TriviaQ[] {
   240→  const decadeStart = parseInt(data.years.split('–')[0]);
   241→  const qs: TriviaQ[] = [];
   242→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   243→  const shuffled = shuffle(all).slice(0, 8);
   244→
   245→  for (const item of shuffled) {
   246→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   247→    const wrongYears = [
   248→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   249→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   250→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   251→    ].filter(y => y !== realYear).slice(0, 3);
   252→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   253→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   254→    qs.push({
   255→      question: `When did "${item.title}" first become popular?`,
   256→      emoji: item.emoji,
   257→      correct: options.indexOf(realYear),
   258→      options: options.map(String),
   259→    });
   260→  }
   261→  return qs;
   262→}

   264→function GuessTheYearGame({ data }: { data: DecadeData }) {
   265→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   266→  const [qIdx, setQIdx] = useState(0);
   267→  const [score, setScore] = useState(0);
   268→  const [selected, setSelected] = useState<number | null>(null);
   269→  const [showResult, setShowResult] = useState(false);
   270→  const [done, setDone] = useState(false);
   271→
   272→  useEffect(() => {
   273→    setQuestions(buildTrivia(data));
   274→  }, [data]);
   275→
   276→  const q = questions[qIdx];
   277→
   278→  const handleAnswer = (optIdx: number) => {
   279→    if (showResult || !q) return;
   280→    setSelected(optIdx);
   281→    setShowResult(true);
   282→    if (optIdx === q.correct) setScore(s => s + 1);
   283→    setTimeout(() => {
   284→      if (qIdx + 1 < questions.length) {
   285→        setQIdx(i => i + 1);
   286→        setSelected(null);
   287→        setShowResult(false);
   288→      } else {
   289→        setDone(true);
   290→      }
   291→    }, 1500);
   292→  };
   293→
   294→  const resetGame = () => {
   295→    const fresh = buildTrivia(data);
   296→    setQuestions(fresh);
   297→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   298→  };
   299→
   300→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center text-sm text-[#8b7355]">Loading...</div>;
   301→
   302→  if (done) {
   303→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   304→    return (
   305→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   306→        <div className="text-center py-4">
   307→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   308→          <p className="text-lg font-bold text-[#5c4033]">Game Over!</p>
   309→          <p className="text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   310→          <p className="text-xs text-[#8b7355] mt-2 italic">
   311→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   312→          </p>
   313→          <button onClick={resetGame} className="sk-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   314→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   315→          </button>
   316→        </div>
   317→      </div>
   318→    );
   319→  }
   320→
   321→  return (
   322→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   323→      <div className="flex items-center justify-between mb-4">
   324→        <h4 className="font-bold text-base text-[#5c4033] flex items-center gap-2">
   325→          <span className="text-xl">📅</span> Guess the Year
   326→        </h4>
   327→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   328→          <span className="sk-inset px-2 py-1 rounded">{qIdx + 1}/{questions.length}</span>
   329→          <span className="sk-inset px-2 py-1 rounded">Score: {score}</span>
   330→        </div>
   331→      </div>
   332→
   333→      {/* Progress bar */}
   334→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   335→        <div
   336→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   337→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   338→        />
   339→      </div>
   340→
   341→      <div className="text-center mb-4">
   342→        <span className="text-5xl block mb-2">{q.emoji}</span>
   343→        <p className="text-sm font-semibold text-[#5c4033]">{q.question}</p>
   344→      </div>
   345→
   346→      <div className="grid grid-cols-2 gap-2">
   347→        {q.options.map((opt, i) => {
   348→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 ';
   349→          if (showResult) {
   350→            if (i === q.correct) cls += 'border-green-500 bg-green-50 sk-sticker';
   351→            else if (i === selected) cls += 'border-red-400 bg-red-50 opacity-60';
   352→            else cls += 'border-[#d4c5a9] opacity-40';
   353→          } else {
   354→            cls += 'border-[#d4c5a9] hover:border-[#8b5e3c] hover:shadow-md active:translate-y-0.5';
   355→          }
   356→          return (
   357→            <button key={`${qIdx}-${i}`} onClick={() => handleAnswer(i)} className={cls} disabled={showResult}>
   358→              {opt}
   359→            </button>
   360→          );
   361→        })}
   362→      </div>
   363→
   364→      {showResult && (
   365→        <motion.div
   366→          initial={{ opacity: 0, y: 5 }}
   367→          animate={{ opacity: 1, y: 0 }}
   368→          className="text-center mt-3 text-xs font-semibold flex items-center justify-center gap-1"

   370→          {selected === q.correct ? (
   371→            <><CheckCircle2 className="h-4 w-4 text-green-600" /><span className="text-green-700">Correct! You remembered!</span></>
   372→          ) : (
   373→            <><XCircle className="h-4 w-4 text-red-500" /><span className="text-red-600">It was {q.options[q.correct]}</span></>
   374→          )}
   375→        </motion.div>
   376→      )}
   377→    </div>

   381→/* ═══════════════════════════════════════════════════
   382→   GAME 3: TAP THE MEMORY (Reaction Game)
   383→   ═══════════════════════════════════════════════════ */
   384→function WhackGame({ data }: { data: DecadeData }) {
   385→  const GRID = 9;
   386→  const allItems = useMemo(() => [
   387→    ...data.toys, ...data.snacks, ...data.cartoons, ...data.games,
   388→  ], [data]);
   389→
   390→  const [grid, setGrid] = useState<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   391→  const [score, setScore] = useState(0);
   392→  const [timeLeft, setTimeLeft] = useState(20);
   393→  const [gameActive, setGameActive] = useState(false);
   394→  const [gameOver, setGameOver] = useState(false);
   395→  const [feedback, setFeedback] = useState<string | null>(null);
   396→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   397→  const spawnRef = useRef<ReturnType<typeof setInterval> | null>(null);
   398→  const gridRef = useRef<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   399→
   400→  const spawnItem = useCallback(() => {
   401→    const empty = gridRef.current.reduce<number[]>((acc, v, i) => v === null ? [...acc, i] : acc, []);
   402→    if (empty.length === 0) return;
   403→    const pos = empty[Math.floor(Math.random() * empty.length)];
   404→    const item = allItems[Math.floor(Math.random() * allItems.length)];
   405→    const newGrid = [...gridRef.current];
   406→    newGrid[pos] = item;
   407→    gridRef.current = newGrid;
   408→    setGrid([...newGrid]);
   409→    setTimeout(() => {
   410→      if (gridRef.current[pos] === item) {
   411→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);
   412→      }
   413→    }, 1200);
   414→  }, [allItems]);
   415→
   416→  const startGame = useCallback(() => {
   417→    const fresh = Array(GRID).fill(null);
   418→    gridRef.current = fresh;
   419→    setGrid([...fresh]);
   420→    setScore(0); setTimeLeft(20); setGameOver(false); setGameActive(true);
   421→
   422→    timerRef.current = setInterval(() => {
   423→      setTimeLeft(t => {
   424→        if (t <= 1) {
   425→          if (timerRef.current) clearInterval(timerRef.current);
   426→          if (spawnRef.current) clearInterval(spawnRef.current);
   427→          setGameActive(false); setGameOver(true);
   428→          return 0;
   429→        }
   430→        return t - 1;
   431→      });
   432→    }, 1000);
   433→    spawnRef.current = setInterval(spawnItem, 700);
   434→  }, [spawnItem]);
   435→
   436→  useEffect(() => {
   437→    return () => {
   438→      if (timerRef.current) clearInterval(timerRef.current);
   439→      if (spawnRef.current) clearInterval(spawnRef.current);
   440→    };
   441→  }, []);
   442→
   443→  const handleWhack = (idx: number) => {
   444→    if (!gameActive || !grid[idx]) return;
   445→    setScore(s => s + 10);
   446→    setFeedback('+10');
   447→    const g = [...gridRef.current]; g[idx] = null; gridRef.current = g; setGrid([...g]);
   448→    setTimeout(() => setFeedback(null), 300);
   449→  };
   450→
   451→  const handleMiss = () => {
   452→    if (!gameActive) return;
   453→    setFeedback('Miss!');
   454→    setTimeout(() => setFeedback(null), 300);
   455→  };
   456→
   457→  return (
   458→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   459→      <div className="flex items-center justify-between mb-4">
   460→        <h4 className="font-bold text-base text-[#5c4033] flex items-center gap-2">
   461→          <span className="text-xl">🔨</span> Tap the Memory
   462→        </h4>
   463→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   464→          <span className="sk-inset px-2 py-1 rounded flex items-center gap-1">
   465→            <Timer className="h-3 w-3" />{timeLeft}s
   466→          </span>
   467→          <span className="sk-inset px-2 py-1 rounded">{score}pts</span>
   468→        </div>
   469→      </div>
   470→
   471→      {!gameActive && !gameOver && (
   472→        <div className="text-center py-6">
   473→          <p className="text-sm text-[#8b7355] mb-4">Tap nostalgic items as they pop up! You have 20 seconds.</p>
   474→          <button onClick={startGame} className="sk-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   475→            <Play className="h-5 w-5" /> Start Game
   476→          </button>
   477→        </div>
   478→      )}
   479→
   480→      {gameOver && (
