    38→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    39→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    40→  const [bootLines, setBootLines] = useState<string[]>([]);
    41→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    42→
    43→  const allBootLines = useMemo(() => [
    44→    'BIOS v2.4 — Nostalgia Systems Inc.',
    45→    'POST: Memory test... 640K OK',
    46→    'Detecting floppy drives... A: OK  B: OK',
    47→    'Loading REWIND.EXE...',
    48→    'Mounting childhood memories... 47% complete',
    49→    'Indexing nostalgia database...',
    50→    'Rewinding time...',
    51→    'Loading good old days... 83% complete',
    52→    'Ready.',
    53→  ], []);
    54→
    55→  useEffect(() => {
    56→    const timers: ReturnType<typeof setTimeout>[] = [];
    57→    // Reveal boot lines one by one
    58→    allBootLines.forEach((line, i) => {
    59→      timers.push(setTimeout(() => {
    60→        setBootLines(prev => [...prev, line]);
    61→        if (i === allBootLines.length - 1) {
    62→          // Last line — show title after a beat
    63→          timers.push(setTimeout(() => setPhase(1), 1000));
    64→        }
    65→      }, 500 + i * 500));
    66→    });
    67→
    68→    // Auto-dismiss after title phase
    69→    fadeTimerRef.current = setTimeout(() => {
    70→      setPhase(2);
    71→      setTimeout(onDone, 800);
    72→    }, 7000);
    73→
    74→    return () => {
    75→      timers.forEach(clearTimeout);
    76→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    77→    };
    78→  }, [allBootLines, onDone]);
    79→
    80→  return (
    81→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    82→      <div className="crt-scanlines" />
    83→      <div className="vhs-noise" />
    84→      <div className="vhs-tracking" />

    87→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
    88→        {/* Boot lines */}
    89→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
    90→          {bootLines.map((line, i) => (
    91→            <div
    92→              key={i}
    93→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    94→              style={{ animationDelay: '0s' }}
    95→            >
    96→              {line}
    97→              {i === bootLines.length - 1 && (
    98→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
    99→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
   100→                />
   101→              )}
   102→            </div>
   103→          ))}
   104→        </div>
   105→
   106→        {/* Title */}
   107→        {phase >= 1 && (
   108→          <motion.div
   109→            initial={{ opacity: 0, scale: 0.9 }}
   110→            animate={{ opacity: 1, scale: 1 }}
   111→            transition={{ duration: 1.0 }}

   329→    <AnimatePresence mode="wait">

   331→        key={idx}

   333→        animate={{ opacity: 1, y: 0 }}

   337→      >

   395→  return '';

   742→export default function Home() {
   743→  const [birthYear, setBirthYear] = useState<number | null>(null);
   744→  const [loading, setLoading] = useState(true);
   745→  const [showOnboarding, setShowOnboarding] = useState(false);
   746→  const [onboardingChecked, setOnboardingChecked] = useState(false);
   747→
   748→  // Run one-time localStorage → IndexedDB migration on first load
   749→  useEffect(() => {
   750→    if (typeof window !== 'undefined') {
   751→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   752→      // Check if onboarding was completed
   753→      (async () => {
   754→        try {
   755→          const db = await getDB();
   756→          const record: any = await db.get('era_skin', 'onboarding');
   757→          if (record?.complete) {
   758→            // Restore persisted birth year
   759→            if (record.birthYear) setBirthYear(record.birthYear);
   760→          } else {
   761→            setShowOnboarding(true);
   762→          }
   763→        } catch {
   764→          setShowOnboarding(true);
   765→        }
   766→        setOnboardingChecked(true);
   767→      })();
   768→    }
   769→  }, []);
   770→
   771→  const handleLoadingDone = useCallback(() => {
   772→    setLoading(false);
   773→  }, []);
   774→
   775→  const handleSubmit = useCallback((year: number) => {
   776→    triggerOldLoader('Tuning into your era...');
   777→    setTimeout(() => {
   778→      setBirthYear(year);
   779→      window.scrollTo({ top: 0, behavior: 'smooth' });
   780→      hideOldLoader();
   781→    }, 1100);
   782→  }, []);
   783→
   784→  const handleOnboardingComplete = useCallback((data: { birthYear: number; eraSkin: string }) => {
   785→    setShowOnboarding(false);
   786→    triggerOldLoader('Tuning into your era...');
   787→    setTimeout(() => {
   788→      setBirthYear(data.birthYear);
   789→      hideOldLoader();
   790→    }, 1100);
   791→  }, []);
   792→
   793→  const handleBack = useCallback(() => {
   794→    triggerOldLoader('Rewinding...');
   795→    setTimeout(() => {
   796→      setBirthYear(null);
   797→      hideOldLoader();
   798→    }, 850);
   799→  }, []);
   800→
   801→  return (
   802→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   803→      {/* CRT Boot Loading Screen */}
   804→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   805→
   806→      {/* Onboarding (first visit only) */}
   807→      <AnimatePresence>
   808→        {showOnboarding && onboardingChecked && (
   809→          <OnboardingFlow onComplete={handleOnboardingComplete} />
   810→        )}
   811→      </AnimatePresence>
   812→
   813→      <AnimatePresence mode="wait">
   814→        {!showOnboarding && birthYear === null && onboardingChecked ? (
   815→          <motion.div
   816→            key="hero"
   817→            initial={{ opacity: 0 }}
   818→            animate={{ opacity: 1 }}
   819→            exit={{ opacity: 0, y: -20 }}
   820→            transition={{ duration: 0.4 }}
   821→            style={{ opacity: loading ? 0 : 1 }}
   822→          >
   823→            <HeroSection onSubmit={handleSubmit} />
   824→          </motion.div>
   825→        ) : birthYear !== null ? (
   826→          <motion.div
   827→            key="dashboard"
   828→            initial={{ opacity: 0, y: 20 }}
   829→            animate={{ opacity: 1, y: 0 }}
   830→            exit={{ opacity: 0 }}
   831→            transition={{ duration: 0.5 }}
   832→            className="flex flex-col min-h-screen"
   833→          >
   834→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   835→          </motion.div>
   836→        ) : null}
   837→      </AnimatePresence>
   838→    </div>
   839→  );
