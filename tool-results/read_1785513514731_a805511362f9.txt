    80→function pickReply(userText: string): string {
    81→  const lower = userText.toLowerCase();
    82→  for (const [keys, replies] of Object.entries(KEYWORD_REPLIES)) {
    83→    if (keys === 'default') continue;
    84→    const patterns = keys.split('|');
    85→    if (patterns.some(p => lower.includes(p))) {
    86→      return replies[Math.floor(Math.random() * replies.length)];
    87→    }
    88→  }
    89→  const defaults = KEYWORD_REPLIES['default'];
    90→  return defaults[Math.floor(Math.random() * defaults.length)];
    91→}
