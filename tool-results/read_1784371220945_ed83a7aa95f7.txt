   208→interface MatchCard {
   209→  id: number;
   210→  emoji: string;
   211→  label: string;
   212→  isFlipped: boolean;
   213→  isMatched: boolean;
   214→}
   215→
   216→function shuffle<T>(arr: T[]): T[] {
   217→  const a = [...arr];
   218→  for (let i = a.length - 1; i > 0; i--) {
   219→    const j = Math.floor(Math.random() * (i + 1));
   220→    [a[i], a[j]] = [a[j], a[i]];
   221→  }
   222→  return a;
   223→}
   224→
   225→function createMatchCards(data: DecadeData): MatchCard[] {
   226→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   227→  const picked = shuffle(pool).slice(0, 6);
   228→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   229→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   230→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   231→  ]);
   232→  return shuffle(initial);
   233→}

   235→function MemoryMatchGame({ data }: { data: DecadeData }) {
   236→  const [cards, setCards] = useState<MatchCard[]>([]);
   237→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   238→  const [moves, setMoves] = useState(0);
   239→  const [matches, setMatches] = useState(0);
   240→  const [gameWon, setGameWon] = useState(false);
   241→  const lockRef = useRef(false);
   242→
   243→  useEffect(() => {
   244→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   245→    return () => cancelAnimationFrame(id);
   246→  }, [data]);
   247→
   248→  const resetGame = useCallback(() => {
   249→    setCards(createMatchCards(data));
   250→    setFlippedIds([]);
   251→    setMoves(0);
   252→    setMatches(0);
   253→    setGameWon(false);
   254→    lockRef.current = false;
   255→  }, [data]);
   256→
   257→  const handleFlip = useCallback((id: number) => {
   258→    if (lockRef.current) return;
   259→    setCards(prev => {
   260→      const card = prev.find(c => c.id === id);
   261→      if (!card || card.isFlipped || card.isMatched) return prev;
   262→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   263→    });
   264→
   265→    setFlippedIds(prev => {
   266→      if (prev.includes(id) || prev.length >= 2) return prev;
   267→      const newFlipped = [...prev, id];
   268→
   269→      if (newFlipped.length === 2) {
   270→        setMoves(m => m + 1);
   271→        lockRef.current = true;
   272→
   273→        // Use a timeout to read the latest cards state
   274→        setTimeout(() => {
   275→          setCards(currentCards => {
   276→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   277→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   278→
   279→            if (first.label === second.label) {
   280→              // Match!
   281→              const updated = currentCards.map(c =>
   282→                c.id === first.id || c.id === second.id
   283→                  ? { ...c, isMatched: true, isFlipped: true }
   284→                  : c
   285→              );
   286→              setMatches(m => {
   287→                const newM = m + 1;
   288→                if (newM === 6) setGameWon(true);
   289→                return newM;
   290→              });
   291→              setFlippedIds([]);
   292→              lockRef.current = false;
   293→              return updated;
   294→            } else {
   295→              // No match — flip back
   296→              setTimeout(() => {
   297→                setCards(cc => cc.map(c =>
   298→                  c.id === first.id || c.id === second.id
   299→                    ? { ...c, isFlipped: false }
   300→                    : c
   301→                ));
   302→                setFlippedIds([]);
   303→                lockRef.current = false;
   304→              }, 700);
   305→              return currentCards;
   306→            }
   307→          });
   308→        }, 50);
   309→      }
   310→
   311→      return newFlipped;
   312→    });
   313→  }, []);
   314→
   315→  return (
   316→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   317→      <div className="flex items-center justify-between mb-4">
   318→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   319→          <span className="text-xl">🃏</span> Memory Match
   320→        </h4>
   321→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   322→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   323→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   324→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   325→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   326→          </button>
   327→        </div>
   328→      </div>
   329→
   330→      {gameWon && (
   331→        <motion.div
   332→          initial={{ scale: 0.8, opacity: 0 }}
   333→          animate={{ scale: 1, opacity: 1 }}
   334→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   335→        >
   336→          <span className="text-5xl mb-2">🎉</span>
   337→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   338→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   339→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   340→            Play Again
   341→          </button>
   342→        </motion.div>
   343→      )}
   344→
   345→      <div className="grid grid-cols-4 gap-2">
   346→        {cards.map((card) => {
   347→          const showFront = card.isFlipped || card.isMatched;
   348→          return (
   349→            <button
   350→              key={card.id}
   351→              onClick={() => handleFlip(card.id)}
   352→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   353→              disabled={card.isMatched}
   354→              style={{
   355→                background: showFront
   356→                  ? '#faf6eb'
   357→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   358→                borderColor: showFront
   359→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')
   360→                  : '#5c4033',
   361→                boxShadow: showFront
   362→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   363→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   364→                opacity: card.isMatched ? 0.6 : 1,
   365→              }}
   366→            >
   367→              {showFront ? (
   368→                <>
   369→                  <span className="text-2xl md:text-3xl">{card.emoji}</span>
   370→                  <span className="old-typewriter text-[7px] md:text-[9px] font-bold text-[#5c4033] mt-0.5 text-center leading-tight" style={{ maxWidth: '100%', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{card.label}</span>
   371→                </>
   372→              ) : (
   373→                <span className="text-xl text-[#d4a574] font-bold old-title">?</span>
   374→              )}
   375→            </button>
   376→          );
   377→        })}
   378→      </div>
   379→    </div>
   380→  );
   381→}

   383→/* ═══════════════════════════════════════════════════

   386→interface TriviaQ {
   387→  question: string;
   388→  emoji: string;
   389→  correct: number;
   390→  options: string[];
   391→}
   392→
   393→function buildTrivia(data: DecadeData): TriviaQ[] {
   394→  const decadeStart = parseInt(data.years.split('–')[0]);
   395→  const qs: TriviaQ[] = [];
   396→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   397→  const shuffled = shuffle(all).slice(0, 8);
   398→
   399→  for (const item of shuffled) {
   400→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   401→    const wrongYears = [
   402→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   403→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   404→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   405→    ].filter(y => y !== realYear).slice(0, 3);
   406→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   407→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   408→    qs.push({
   409→      question: `When did "${item.title}" first become popular?`,
   410→      emoji: item.emoji,
   411→      correct: options.indexOf(realYear),
   412→      options: options.map(String),
   413→    });
   414→  }
   415→  return qs;
   416→}

   418→function GuessTheYearGame({ data }: { data: DecadeData }) {
   419→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   420→  const [qIdx, setQIdx] = useState(0);
   421→  const [score, setScore] = useState(0);
   422→  const [selected, setSelected] = useState<number | null>(null);
   423→  const [showResult, setShowResult] = useState(false);
   424→  const [done, setDone] = useState(false);
   425→
   426→  useEffect(() => {
   427→    setQuestions(buildTrivia(data));
   428→  }, [data]);
   429→
   430→  const q = questions[qIdx];
   431→
   432→  const handleAnswer = (optIdx: number) => {
   433→    if (showResult || !q) return;
   434→    setSelected(optIdx);
   435→    setShowResult(true);
   436→    if (optIdx === q.correct) setScore(s => s + 1);
   437→    setTimeout(() => {
   438→      if (qIdx + 1 < questions.length) {
   439→        setQIdx(i => i + 1);
   440→        setSelected(null);
   441→        setShowResult(false);
   442→      } else {
   443→        setDone(true);
   444→      }
   445→    }, 1500);
   446→  };
   447→
   448→  const resetGame = () => {
   449→    const fresh = buildTrivia(data);
   450→    setQuestions(fresh);
   451→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   452→  };
   453→
   454→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   455→
   456→  if (done) {
   457→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   458→    return (
   459→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   460→        <div className="text-center py-4">
   461→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   462→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
   463→          <p className="old-body text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   464→          <p className="old-quote text-xs text-[#8b7355] mt-2">
   465→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   466→          </p>
   467→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   468→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   469→          </button>
   470→        </div>
   471→      </div>
   472→    );
   473→  }
   474→
   475→  return (
   476→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   477→      <div className="flex items-center justify-between mb-4">
   478→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   479→          <span className="text-xl">📅</span> Guess the Year
   480→        </h4>
   481→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   482→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{qIdx + 1}/{questions.length}</span>
   483→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Score: {score}</span>
   484→        </div>
   485→      </div>
   486→
   487→      {/* Progress bar */}
   488→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   489→        <div
   490→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   491→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   492→        />
   493→      </div>
   494→
   495→      <div className="text-center mb-4">
   496→        <span className="text-5xl block mb-2">{q.emoji}</span>
   497→        <p className="old-body text-sm font-semibold text-[#5c4033]">{q.question}</p>
   498→      </div>
   499→
   500→      <div className="grid grid-cols-2 gap-2">
   501→        {q.options.map((opt, i) => {
   502→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 old-typewriter ';
   503→          if (showResult) {
   504→            if (i === q.correct) cls += 'border-green-500 bg-green-50 sk-sticker';
   505→            else if (i === selected) cls += 'border-red-400 bg-red-50 opacity-60';
   506→            else cls += 'border-[#d4c5a9] opacity-40';
   507→          } else {
   508→            cls += 'border-[#d4c5a9] hover:border-[#8b5e3c] hover:shadow-md active:translate-y-0.5';
   509→          }
   510→          return (
   511→            <button key={`${qIdx}-${i}`} onClick={() => handleAnswer(i)} className={cls} disabled={showResult}>
   512→              {opt}
   513→            </button>
   514→          );

   538→function WhackGame({ data }: { data: DecadeData }) {
   539→  const GRID = 9;
   540→  const allItems = useMemo(() => [
   541→    ...data.toys, ...data.snacks, ...data.cartoons, ...data.games,
   542→  ], [data]);
   543→
   544→  const [grid, setGrid] = useState<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   545→  const [score, setScore] = useState(0);
   546→  const [timeLeft, setTimeLeft] = useState(20);
   547→  const [gameActive, setGameActive] = useState(false);
   548→  const [gameOver, setGameOver] = useState(false);
   549→  const [feedback, setFeedback] = useState<string | null>(null);
   550→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   551→  const spawnRef = useRef<ReturnType<typeof setInterval> | null>(null);
   552→  const gridRef = useRef<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   553→  const activeRef = useRef(false);
   554→
   555→  const spawnItem = useCallback(() => {
   556→    if (!activeRef.current) return;
   557→    const empty = gridRef.current.reduce<number[]>((acc, v, i) => v === null ? [...acc, i] : acc, []);
   558→    if (empty.length === 0) return;
   559→    const pos = empty[Math.floor(Math.random() * empty.length)];
   560→    const item = allItems[Math.floor(Math.random() * allItems.length)];
   561→    const newGrid = [...gridRef.current];
   562→    newGrid[pos] = item;
   563→    gridRef.current = newGrid;
   564→    setGrid([...newGrid]);
   565→    setTimeout(() => {
   566→      if (gridRef.current[pos] === item) {
   567→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);
   568→      }
   569→    }, 1200);
   570→  }, [allItems]);
   571→
   572→  const stopGame = useCallback(() => {
   573→    activeRef.current = false;
   574→    if (timerRef.current) { clearInterval(timerRef.current); timerRef.current = null; }

   604→    return () => {

   606→      if (timerRef.current) clearInterval(timerRef.current);

   608→    };
   609→  }, []);

   616→    setTimeout(() => setFeedback(null), 300);
   617→  };

   623→  };

   626→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   627→      <div className="flex items-center justify-between mb-4">
   628→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   629→          <span className="text-xl">🔨</span> Tap the Memory
   630→        </h4>
   631→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   632→          <span className="sk-inset px-2 py-1 rounded old-typewriter flex items-center gap-1">
   633→            <Timer className="h-3 w-3" />{timeLeft}s
   634→          </span>
   635→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{score}pts</span>
   636→        </div>
   637→      </div>
   638→
   639→      {!gameActive && !gameOver && (
   640→        <div className="text-center py-6">
   641→          <p className="old-body text-sm text-[#8b7355] mb-4">Tap nostalgic items as they pop up! You have 20 seconds.</p>
   642→          <button onClick={startGame} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   643→            <Play className="h-5 w-5" /> Start Game
   644→          </button>
   645→        </div>

   647→
   648→      {gameOver && (
   649→        <div className="text-center py-4">
   650→          <span className="text-5xl mb-2 block">{score >= 80 ? '🏆' : score >= 40 ? '🎉' : '💪'}</span>
   651→          <p className="old-title text-lg font-bold text-[#5c4033]">Time&apos;s Up!</p>
   652→          <p className="old-body text-sm text-[#8b7355] mt-1">Score: {score} points</p>
