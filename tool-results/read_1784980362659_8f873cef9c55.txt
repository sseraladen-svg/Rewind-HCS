    83→// ─── Helpers ────────────────────────────────────────

    85→function getWeekStart(date?: Date): Date {
    86→  const d = date ?? new Date();
    87→  const day = d.getDay();
    88→  const diff = d.getDate() - day + (day === 0 ? -6 : 1); // Monday
    89→  return new Date(d.getFullYear(), d.getMonth(), diff);
    90→}
    91→
    92→function getWeekNumber(): number {
    93→  const now = new Date();
    94→  const start = new Date(now.getFullYear(), 0, 1);
    95→  const diff = now.getTime() - start.getTime();
    96→  const oneWeek = 604800000;
    97→  return Math.floor(diff / oneWeek);
    98→}
    99→
   100→function startOfDay(d: Date): number {
   101→  return new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
   102→}
   103→
   104→function isSameDay(ts1: number, ts2: number): boolean {
   105→  const d1 = new Date(ts1);
   106→  const d2 = new Date(ts2);
   107→  return d1.getFullYear() === d2.getFullYear() &&
   108→    d1.getMonth() === d2.getMonth() &&
   109→    d1.getDate() === d2.getDate();
   110→}
   111→
   112→function isYesterday(ts: number): boolean {
   113→  const yesterday = new Date();
   114→  yesterday.setDate(yesterday.getDate() - 1);
   115→  return isSameDay(ts, yesterday.getTime());
   116→}
   117→
   118→function isToday(ts: number): boolean {
   119→  return isSameDay(ts, Date.now());
   120→}

   146→// ─── Visit Streak Logic ──────────────────────────────
   147→
   148→async function getVisitStreak(): Promise<VisitStreak> {
   149→  try {
   150→    const db = await getDB();
   151→    const record = await db.get('era_skin', 'visit_streak');
   152→    if (record) {
   153→      try {
   154→        const data = JSON.parse(record.skin_id);
   155→        return { days: data.days ?? 0, lastVisit: data.lastVisit ?? 0 };
   156→      } catch {
   157→        return { days: 0, lastVisit: 0 };
   158→      }
   159→    }
   160→  } catch {
   161→    // silent
   162→  }
   163→  return { days: 0, lastVisit: 0 };
   164→}
   165→
   166→async function updateVisitStreak(): Promise<VisitStreak> {
   167→  try {
   168→    const db = await getDB();
   169→    const record = await db.get('era_skin', 'visit_streak');
   170→    let streak: VisitStreak;
   171→
   172→    if (record) {
   173→      try {
   174→        const data = JSON.parse(record.skin_id);
   175→        streak = { days: data.days ?? 0, lastVisit: data.lastVisit ?? 0 };
   176→      } catch {
   177→        streak = { days: 0, lastVisit: 0 };
   178→      }
   179→    } else {
   180→      streak = { days: 0, lastVisit: 0 };
   181→    }
   182→
   183→    const now = Date.now();
   184→
   185→    if (isToday(streak.lastVisit)) {
   186→      // Already visited today, no change
   187→      return streak;
   188→    } else if (isYesterday(streak.lastVisit)) {
   189→      // Consecutive day
   190→      streak.days += 1;
   191→      streak.lastVisit = now;
   192→    } else {
   193→      // Streak broken or first visit
   194→      streak.days = 1;
   195→      streak.lastVisit = now;
   196→    }
   197→
   198→    await db.put('era_skin', {
   199→      id: 'visit_streak',
   200→      skin_id: JSON.stringify({ days: streak.days, lastVisit: streak.lastVisit }),
   201→    });
   202→
   203→    return streak;
   204→  } catch {
   205→    return { days: 0, lastVisit: 0 };
   206→  }
   207→}

   710→
   711→function ComparisonBar({
   712→  label,
   713→  thisWeek,
   714→  lastWeek,
   715→  color,
   716→  delay,
   717→}: {
   718→  label: string;
   719→  thisWeek: number;
   720→  lastWeek: number;
   721→  color: string;
   722→  delay: number;
   723→}) {
   724→  const maxVal = Math.max(thisWeek, lastWeek, 1);
   725→  const thisPct = (thisWeek / maxVal) * 100;
   726→  const lastPct = (lastWeek / maxVal) * 100;
   727→
   728→  const TrendIcon =
   729→    thisWeek > lastWeek ? TrendingUp : thisWeek < lastWeek ? TrendingDown : Minus;
   730→
   731→  return (
   732→    <div className="mb-3 last:mb-0">
   733→      <div className="flex items-center justify-between mb-1">
   734→        <span className="old-typewriter text-[9px] text-[#5c4033] font-bold">{label}</span>
   735→        <div className="flex items-center gap-1.5">
   736→          <span className="old-typewriter text-[8px] text-[#8b7355]">
   737→            This: {thisWeek} | Last: {lastWeek}
   738→          </span>
   739→          <TrendIcon className="w-3 h-3" style={{ color: thisWeek >= lastWeek ? '#6b8e6b' : '#9e6b6b' }} />
   740→        </div>
   741→      </div>
   742→
   743→      {/* This week bar */}
   744→      <div className="h-2 rounded-full bg-[#e8dcc8] mb-1 overflow-hidden">
   745→        <motion.div
   746→          className="h-full rounded-full"
   747→          style={{ background: color }}
   748→          initial={{ width: 0 }}
   749→          animate={{ width: `${Math.max(thisPct, thisWeek > 0 ? 8 : 0)}%` }}
   750→          transition={{ delay, duration: 0.6, ease: 'easeOut' }}
   751→        />
   752→      </div>
   753→
   754→      {/* Last week bar (lighter) */}
   755→      <div className="h-1.5 rounded-full bg-[#e8dcc8] overflow-hidden">
   756→        <motion.div
   757→          className="h-full rounded-full opacity-40"
   758→          style={{ background: color }}
   759→          initial={{ width: 0 }}
   760→          animate={{ width: `${Math.max(lastPct, lastWeek > 0 ? 8 : 0)}%` }}
   761→          transition={{ delay: delay + 0.1, duration: 0.6, ease: 'easeOut' }}
   762→        />
   763→      </div>
   764→    </div>
   765→  );
   766→}
