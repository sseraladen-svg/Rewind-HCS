    20→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {

    22→  const [bootLines, setBootLines] = useState<string[]>([]);
    23→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    24→
    25→  const allBootLines = useMemo(() => [
    26→    'BIOS v2.4 — Nostalgia Systems Inc.',
    27→    'Memory test... 640K OK',
    28→    'Loading REWIND.EXE...',
    29→    'Mounting childhood memories...',
    30→    'Rewinding time...',
    31→    'Loading good old days...',
    32→    'Ready.',
    33→  ], []);
    34→
    35→  useEffect(() => {
    36→    const timers: ReturnType<typeof setTimeout>[] = [];
    37→    // Reveal boot lines one by one
    38→    allBootLines.forEach((line, i) => {
    39→      timers.push(setTimeout(() => {
    40→        setBootLines(prev => [...prev, line]);
    41→        if (i === allBootLines.length - 1) {
    42→          // Last line — show title after a beat
    43→          timers.push(setTimeout(() => setPhase(1), 600));
    44→        }
    45→      }, 300 + i * 350));
    46→    });

    48→    // Auto-dismiss after title phase
    49→    fadeTimerRef.current = setTimeout(() => {
    50→      setPhase(2);
    51→      setTimeout(onDone, 500);

    53→
    54→    return () => {
    55→      timers.forEach(clearTimeout);
    56→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    57→    };
    58→  }, [allBootLines, onDone]);
    59→

   170→/* ═══════════════════════════════════════════════════

   173→interface MatchCard {
   174→  id: number;
   175→  emoji: string;
   176→  label: string;
   177→  isFlipped: boolean;
   178→  isMatched: boolean;
   179→}
   180→
   181→function shuffle<T>(arr: T[]): T[] {
   182→  const a = [...arr];
   183→  for (let i = a.length - 1; i > 0; i--) {
   184→    const j = Math.floor(Math.random() * (i + 1));
   185→    [a[i], a[j]] = [a[j], a[i]];
   186→  }
   187→  return a;
   188→}
   189→
   190→function createMatchCards(data: DecadeData): MatchCard[] {
   191→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   192→  const picked = shuffle(pool).slice(0, 6);
   193→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   194→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   195→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   196→  ]);
   197→  return shuffle(initial);
   198→}

   200→function MemoryMatchGame({ data }: { data: DecadeData }) {
   201→  const [cards, setCards] = useState<MatchCard[]>([]);
   202→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   203→  const [moves, setMoves] = useState(0);
   204→  const [matches, setMatches] = useState(0);
   205→  const [gameWon, setGameWon] = useState(false);
   206→  const lockRef = useRef(false);
   207→
   208→  useEffect(() => {
   209→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   210→    return () => cancelAnimationFrame(id);
   211→  }, [data]);
   212→
   213→  const resetGame = useCallback(() => {
   214→    setCards(createMatchCards(data));
   215→    setFlippedIds([]);
   216→    setMoves(0);
   217→    setMatches(0);
   218→    setGameWon(false);
   219→    lockRef.current = false;
   220→  }, [data]);
   221→
   222→  const handleFlip = useCallback((id: number) => {
   223→    if (lockRef.current) return;
   224→    setCards(prev => {
   225→      const card = prev.find(c => c.id === id);
   226→      if (!card || card.isFlipped || card.isMatched) return prev;
   227→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   228→    });
   229→
   230→    setFlippedIds(prev => {
   231→      if (prev.includes(id) || prev.length >= 2) return prev;
   232→      const newFlipped = [...prev, id];
   233→
   234→      if (newFlipped.length === 2) {
   235→        setMoves(m => m + 1);
   236→        lockRef.current = true;
   237→
   238→        // Use a timeout to read the latest cards state
   239→        setTimeout(() => {
   240→          setCards(currentCards => {
   241→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   242→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   243→
   244→            if (first.label === second.label) {
   245→              // Match!
   246→              const updated = currentCards.map(c =>
   247→                c.id === first.id || c.id === second.id
   248→                  ? { ...c, isMatched: true, isFlipped: true }
   249→                  : c
   250→              );
   251→              setMatches(m => {
   252→                const newM = m + 1;
   253→                if (newM === 6) setGameWon(true);
   254→                return newM;
   255→              });
   256→              setFlippedIds([]);
   257→              lockRef.current = false;
   258→              return updated;
   259→            } else {
   260→              // No match — flip back
   261→              setTimeout(() => {
   262→                setCards(cc => cc.map(c =>
   263→                  c.id === first.id || c.id === second.id
   264→                    ? { ...c, isFlipped: false }
   265→                    : c
   266→                ));
   267→                setFlippedIds([]);
   268→                lockRef.current = false;
   269→              }, 700);
   270→              return currentCards;
   271→            }
   272→          });
   273→        }, 50);
   274→      }
   275→
   276→      return newFlipped;
   277→    });
   278→  }, []);
   279→
   280→  return (
   281→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   282→      <div className="flex items-center justify-between mb-4">
   283→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   284→          <span className="text-xl">🃏</span> Memory Match
   285→        </h4>
   286→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   287→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   288→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   289→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   290→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   291→          </button>
   292→        </div>
   293→      </div>
   294→
   295→      {gameWon && (
   296→        <motion.div
   297→          initial={{ scale: 0.8, opacity: 0 }}
   298→          animate={{ scale: 1, opacity: 1 }}
   299→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   300→        >
   301→          <span className="text-5xl mb-2">🎉</span>
   302→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   303→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   304→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   305→            Play Again
   306→          </button>
   307→        </motion.div>
   308→      )}
   309→
   310→      <div className="grid grid-cols-4 gap-2">
   311→        {cards.map((card) => {
   312→          const showFront = card.isFlipped || card.isMatched;
   313→          return (
   314→            <button
   315→              key={card.id}
   316→              onClick={() => handleFlip(card.id)}
   317→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   318→              disabled={card.isMatched}
   319→              style={{
   320→                background: showFront
   321→                  ? '#faf6eb'
   322→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   323→                borderColor: showFront
   324→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')
   325→                  : '#5c4033',
   326→                boxShadow: showFront
   327→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   328→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   329→                opacity: card.isMatched ? 0.6 : 1,
   330→              }}
   331→            >
   332→              {showFront ? (
   333→                <>
   334→                  <span className="text-2xl md:text-3xl">{card.emoji}</span>
   335→                  <span className="old-typewriter text-[7px] md:text-[9px] font-bold text-[#5c4033] mt-0.5 text-center leading-tight" style={{ maxWidth: '100%', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{card.label}</span>
   336→                </>
   337→              ) : (
   338→                <span className="text-xl text-[#d4a574] font-bold old-title">?</span>
   339→              )}
   340→            </button>
   341→          );
   342→        })}
   343→      </div>
   344→    </div>
   345→  );
   346→}

   348→/* ═══════════════════════════════════════════════════
   349→   GAME 2: GUESS THE YEAR
   350→   ═══════════════════════════════════════════════════ */
   351→interface TriviaQ {
   352→  question: string;
   353→  emoji: string;
   354→  correct: number;
   355→  options: string[];
   356→}
   357→
   358→function buildTrivia(data: DecadeData): TriviaQ[] {
   359→  const decadeStart = parseInt(data.years.split('–')[0]);
   360→  const qs: TriviaQ[] = [];
   361→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   362→  const shuffled = shuffle(all).slice(0, 8);
   363→
   364→  for (const item of shuffled) {
   365→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   366→    const wrongYears = [
   367→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   368→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   369→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   370→    ].filter(y => y !== realYear).slice(0, 3);
   371→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   372→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   373→    qs.push({
   374→      question: `When did "${item.title}" first become popular?`,
   375→      emoji: item.emoji,
   376→      correct: options.indexOf(realYear),
   377→      options: options.map(String),
   378→    });
   379→  }
   380→  return qs;
   381→}
   382→
   383→function GuessTheYearGame({ data }: { data: DecadeData }) {
   384→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   385→  const [qIdx, setQIdx] = useState(0);
   386→  const [score, setScore] = useState(0);
   387→  const [selected, setSelected] = useState<number | null>(null);
   388→  const [showResult, setShowResult] = useState(false);
   389→  const [done, setDone] = useState(false);
   390→
   391→  useEffect(() => {
   392→    setQuestions(buildTrivia(data));
   393→  }, [data]);
   394→
   395→  const q = questions[qIdx];
   396→
   397→  const handleAnswer = (optIdx: number) => {
   398→    if (showResult || !q) return;
   399→    setSelected(optIdx);
   400→    setShowResult(true);
   401→    if (optIdx === q.correct) setScore(s => s + 1);
   402→    setTimeout(() => {
   403→      if (qIdx + 1 < questions.length) {
   404→        setQIdx(i => i + 1);
   405→        setSelected(null);
   406→        setShowResult(false);
   407→      } else {
   408→        setDone(true);
   409→      }
   410→    }, 1500);
   411→  };
   412→
   413→  const resetGame = () => {
   414→    const fresh = buildTrivia(data);
   415→    setQuestions(fresh);
   416→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   417→  };
   418→
   419→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   420→
   421→  if (done) {
   422→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   423→    return (
   424→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   425→        <div className="text-center py-4">
   426→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   427→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
   428→          <p className="old-body text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   429→          <p className="old-quote text-xs text-[#8b7355] mt-2">
   430→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   431→          </p>
   432→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   433→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   434→          </button>
   435→        </div>
   436→      </div>
   437→    );
   438→  }
   439→
   440→  return (
   441→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   442→      <div className="flex items-center justify-between mb-4">
   443→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   444→          <span className="text-xl">📅</span> Guess the Year
   445→        </h4>
   446→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   447→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{qIdx + 1}/{questions.length}</span>
   448→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Score: {score}</span>
   449→        </div>
   450→      </div>
   451→
   452→      {/* Progress bar */}
   453→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   454→        <div
   455→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   456→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   457→        />
   458→      </div>
   459→
   460→      <div className="text-center mb-4">
   461→        <span className="text-5xl block mb-2">{q.emoji}</span>
   462→        <p className="old-body text-sm font-semibold text-[#5c4033]">{q.question}</p>
   463→      </div>
   464→
   465→      <div className="grid grid-cols-2 gap-2">
   466→        {q.options.map((opt, i) => {
   467→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 old-typewriter ';
   468→          if (showResult) {
   469→            if (i === q.correct) cls += 'border-green-500 bg-green-50 sk-sticker';
   470→            else if (i === selected) cls += 'border-red-400 bg-red-50 opacity-60';
   471→            else cls += 'border-[#d4c5a9] opacity-40';
   472→          } else {
   473→            cls += 'border-[#d4c5a9] hover:border-[#8b5e3c] hover:shadow-md active:translate-y-0.5';
   474→          }
   475→          return (
   476→            <button key={`${qIdx}-${i}`} onClick={() => handleAnswer(i)} className={cls} disabled={showResult}>
   477→              {opt}
   478→            </button>
   479→          );
   480→        })}
   481→      </div>
   482→
   483→      {showResult && (
   484→        <motion.div
   485→          initial={{ opacity: 0, y: 5 }}
   486→          animate={{ opacity: 1, y: 0 }}
   487→          className="text-center mt-3 text-xs font-semibold flex items-center justify-center gap-1 old-body"
   488→        >
   489→          {selected === q.correct ? (
   490→            <><CheckCircle2 className="h-4 w-4 text-green-600" /><span className="text-green-700">Correct! You remembered!</span></>
   491→          ) : (
   492→            <><XCircle className="h-4 w-4 text-red-500" /><span className="text-red-600">It was {q.options[q.correct]}</span></>
   493→          )}
   494→        </motion.div>
   495→      )}
   496→    </div>
   497→  );
   498→}

   501→   GAME 3: TAP THE MEMORY (Reaction Game)
   502→   ═══════════════════════════════════════════════════ */
   503→function WhackGame({ data }: { data: DecadeData }) {
   504→  const GRID = 9;
   505→  const allItems = useMemo(() => [
   506→    ...data.toys, ...data.snacks, ...data.cartoons, ...data.games,
   507→  ], [data]);
   508→
   509→  const [grid, setGrid] = useState<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   510→  const [score, setScore] = useState(0);
   511→  const [timeLeft, setTimeLeft] = useState(20);
   512→  const [gameActive, setGameActive] = useState(false);
   513→  const [gameOver, setGameOver] = useState(false);
   514→  const [feedback, setFeedback] = useState<string | null>(null);
   515→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   516→  const spawnRef = useRef<ReturnType<typeof setInterval> | null>(null);
   517→  const gridRef = useRef<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   518→  const activeRef = useRef(false);
   519→
   520→  const spawnItem = useCallback(() => {
   521→    if (!activeRef.current) return;
   522→    const empty = gridRef.current.reduce<number[]>((acc, v, i) => v === null ? [...acc, i] : acc, []);
   523→    if (empty.length === 0) return;
   524→    const pos = empty[Math.floor(Math.random() * empty.length)];
   525→    const item = allItems[Math.floor(Math.random() * allItems.length)];
   526→    const newGrid = [...gridRef.current];
   527→    newGrid[pos] = item;
   528→    gridRef.current = newGrid;
   529→    setGrid([...newGrid]);
   530→    setTimeout(() => {
   531→      if (gridRef.current[pos] === item) {
   532→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);
   533→      }
   534→    }, 1200);
   535→  }, [allItems]);
   536→
   537→  const stopGame = useCallback(() => {
   538→    activeRef.current = false;
   539→    if (timerRef.current) { clearInterval(timerRef.current); timerRef.current = null; }
   540→    if (spawnRef.current) { clearInterval(spawnRef.current); spawnRef.current = null; }
   541→  }, []);
   542→
   543→  const startGame = useCallback(() => {
   544→    // Clear any previous intervals
   545→    stopGame();
   546→
   547→    const fresh = Array(GRID).fill(null);
   548→    gridRef.current = fresh;
   549→    setGrid([...fresh]);
   550→    setScore(0); setTimeLeft(20); setGameOver(false); setGameActive(true);
   551→    activeRef.current = true;
   552→
   553→    timerRef.current = setInterval(() => {
   554→      if (!activeRef.current) return;
   555→      setTimeLeft(t => {
   556→        if (t <= 1) {
   557→          stopGame();
   558→          setGameActive(false);
   559→          setGameOver(true);
   560→          return 0;
   561→        }
   562→        return t - 1;
   563→      });
   564→    }, 1000);
   565→    spawnRef.current = setInterval(spawnItem, 700);
   566→  }, [spawnItem, stopGame]);
   567→
   568→  useEffect(() => {
   569→    return () => {
   570→      activeRef.current = false;
   571→      if (timerRef.current) clearInterval(timerRef.current);
   572→      if (spawnRef.current) clearInterval(spawnRef.current);
   573→    };
   574→  }, []);
   575→
   576→  const handleWhack = (idx: number) => {
   577→    if (!gameActive || !grid[idx]) return;
   578→    setScore(s => s + 10);
   579→    setFeedback('+10');
   580→    const g = [...gridRef.current]; g[idx] = null; gridRef.current = g; setGrid([...g]);
   581→    setTimeout(() => setFeedback(null), 300);
   582→  };
   583→
   584→  const handleMiss = () => {
   585→    if (!gameActive) return;
   586→    setFeedback('Miss!');
   587→    setTimeout(() => setFeedback(null), 300);
   588→  };
   589→
   590→  return (
   591→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   592→      <div className="flex items-center justify-between mb-4">
   593→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   594→          <span className="text-xl">🔨</span> Tap the Memory
   595→        </h4>
   596→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   597→          <span className="sk-inset px-2 py-1 rounded old-typewriter flex items-center gap-1">
   598→            <Timer className="h-3 w-3" />{timeLeft}s
   599→          </span>
   600→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{score}pts</span>
   601→        </div>
   602→      </div>
   603→
   604→      {!gameActive && !gameOver && (
   605→        <div className="text-center py-6">
   606→          <p className="old-body text-sm text-[#8b7355] mb-4">Tap nostalgic items as they pop up! You have 20 seconds.</p>
   607→          <button onClick={startGame} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   608→            <Play className="h-5 w-5" /> Start Game
   609→          </button>
   610→        </div>
   611→      )}
   612→
   613→      {gameOver && (
   614→        <div className="text-center py-4">
   615→          <span className="text-5xl mb-2 block">{score >= 80 ? '🏆' : score >= 40 ? '🎉' : '💪'}</span>
   616→          <p className="old-title text-lg font-bold text-[#5c4033]">Time&apos;s Up!</p>
   617→          <p className="old-body text-sm text-[#8b7355] mt-1">Score: {score} points</p>
   618→          <button onClick={startGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   619→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   620→          </button>
   621→        </div>
   622→      )}
   623→
   624→      {gameActive && (
   625→        <div className="grid grid-cols-3 gap-2 select-none">
   626→          {grid.map((item, i) => (
   627→            <button
   628→              key={i}
   629→              onClick={() => item ? handleWhack(i) : handleMiss()}
   630→              className="aspect-square rounded-lg transition-all duration-150 cursor-pointer border-2 flex items-center justify-center"
   631→              style={{
   632→                background: item
   633→                  ? `linear-gradient(135deg, ${data.color}18, ${data.color}30)`
   634→                  : 'linear-gradient(135deg, #e8dcc8, #d4c5a9)',
   635→                borderColor: item ? `${data.color}60` : '#c4b594',
   636→                boxShadow: item

   639→                transform: item ? 'scale(1.05)' : 'scale(1)',
   640→              }}
   641→            >
   642→              {item && (
   643→                <motion.span
   644→                  initial={{ scale: 0 }}
   645→                  animate={{ scale: 1 }}
   646→                  className="text-3xl md:text-4xl"
   647→                >
   648→                  {item.emoji}
   649→                </motion.span>
   650→              )}
   651→            </button>
   652→          ))}
   653→        </div>
   654→      )}
   655→
   656→      {feedback && (
   657→        <motion.div
   658→          key={feedback}
   659→          initial={{ opacity: 0, y: 10 }}
   660→          animate={{ opacity: 1, y: 0 }}
   661→          className="absolute bottom-3 left-1/2 -translate-x-1/2"

   663→          <span className={`text-xs font-bold px-3 py-1 rounded-full sk-btn old-typewriter ${feedback === 'Miss!' ? 'opacity-50' : ''}`}>

   665→          </span>
