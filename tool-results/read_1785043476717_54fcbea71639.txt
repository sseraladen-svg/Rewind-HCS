    50→    50→function formatDate(iso: string): string {
    51→    51→  const d = new Date(iso);
    52→    52→  const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    53→    53→  const day = d.getDate();
    54→    54→  const suffix = day === 1 || day === 21 || day === 31 ? 'st' : day === 2 || day === 22 ? 'nd' : day === 3 || day === 23 ? 'rd' : 'th';
    55→    55→  return `${months[d.getMonth()]} ${day}${suffix}, ${d.getFullYear()}`;
    56→    56→}
    57→    57→
    58→    58→function formatTimeAgo(ts: number): string {
    59→    59→  const diff = Date.now() - ts;
    60→    60→  const mins = Math.floor(diff / 60000);
    61→    61→  if (mins < 1) return 'Just now';
    62→    62→  if (mins < 60) return `${mins}m ago`;
    63→    63→  const hrs = Math.floor(mins / 60);
    64→    64→  if (hrs < 24) return `${hrs}h ago`;
    65→    65→  const days = Math.floor(hrs / 24);
    66→    66→  if (days < 7) return `${days}d ago`;
    67→    67→  return formatDate(new Date(ts).toISOString());
    68→    68→}

   586→   586→/* ═══════ MAIN JOURNAL COMPONENT ═══════ */
   587→   587→export default function OldJournal() {
   588→   588→  const [entries, setEntries] = useState<JournalEntry[]>([]);
   589→   589→  const [view, setView] = useState<'cover' | 'list' | 'write' | 'read'>('cover');
   590→   590→  const [selectedId, setSelectedId] = useState<string | null>(null);
   591→   591→  const [editEntry, setEditEntry] = useState<JournalEntry | undefined>(undefined);
   592→   592→  const [confirmDelete, setConfirmDelete] = useState(false);
   593→   593→
   594→   594→  // Load entries from IndexedDB
   595→   595→  const loadEntries = useMemo(() => loadEntriesFromDB, []);
   596→   596→  useEffect(() => {
   597→   597→    let cancelled = false;
   598→   598→    loadEntries().then(data => {
   599→   599→      if (!cancelled) setEntries(data);
   600→   600→    }).catch(() => {});
   601→   601→    return () => { cancelled = true; };
   602→   602→  }, [loadEntries]);
   603→   603→
   604→   604→  const handleSave = useCallback((entry: Omit<JournalEntry, 'id' | 'createdAt'>) => {
   605→   605→    if (editEntry) {
   606→   606→      // Update existing
   607→   607→      const updated = { ...editEntry, ...entry };
   608→   608→      setEntries(prev => prev.map(e => e.id === editEntry.id ? updated : e));
   609→   609→      putJournalEntry(updated).catch(() => {});
   610→   610→    } else {
   611→   611→      // Add new
   612→   612→      const newEntry: JournalEntry = {
   613→   613→        ...entry,
   614→   614→        id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   615→   615→        createdAt: Date.now(),
   616→   616→      };
   617→   617→      setEntries(prev => [newEntry, ...prev]);
   618→   618→      addJournalEntry(newEntry).catch(() => {});
   619→   619→    }
   620→   620→    setEditEntry(undefined);
   621→   621→    setView('read');
   622→   622→  }, [editEntry]);
   623→   623→
   624→   624→  const handleDelete = useCallback(() => {
   625→   625→    if (!selectedId) return;
   626→   626→    if (!confirmDelete) {
   627→   627→      setConfirmDelete(true);
   628→   628→      setTimeout(() => setConfirmDelete(false), 3000);
   629→   629→      return;
   630→   630→    }
   631→   631→    setEntries(prev => {
   632→   632→      const updated = prev.filter(e => e.id !== selectedId);
   633→   633→      deleteJournalEntry(selectedId).catch(() => {});
   634→   634→      return updated;
   635→   635→    });
   636→   636→    setSelectedId(null);
   637→   637→    setView('list');
   638→   638→    setConfirmDelete(false);
   639→   639→  }, [selectedId, confirmDelete]);
   640→   640→
   641→   641→  const handleEdit = useCallback(() => {
   642→   642→    if (!selectedId) return;
   643→   643→    const entry = entries.find(e => e.id === selectedId);
   644→   644→    if (entry) {
   645→   645→      setEditEntry(entry);
   646→   646→      setView('write');
   647→   647→    }
   648→   648→  }, [selectedId, entries]);
   649→   649→
   650→   650→  const currentEntry = entries.find(e => e.id === selectedId);
   651→   651→  const currentIndex = entries.findIndex(e => e.id === selectedId);
   652→   652→
   653→   653→  return (
   654→   654→    <div className="relative min-h-full pb-4">
   655→   655→      <AnimatePresence mode="wait">
   656→   656→        {/* ═══ COVER VIEW ═══ */}
   657→   657→        {view === 'cover' && (
   658→   658→          <JournalCover
   659→   659→            key="cover"
   660→   660→            onOpen={() => entries.length > 0 ? setView('list') : setView('write')}
   661→   661→            entryCount={entries.length}
   662→   662→          />
   663→   663→        )}
   664→   664→
   665→   665→        {/* ═══ ENTRIES LIST ═══ */}
   666→   666→        {view === 'list' && (
   667→   667→          <EntriesList
   668→   668→            key="list"
   669→   669→            entries={entries}
   670→   670→            onSelect={id => { setSelectedId(id); setView('read'); }}
   671→   671→            onNew={() => { setEditEntry(undefined); setView('write'); }}
   672→   672→            onBack={() => setView('cover')}
   673→   673→          />
   674→   674→        )}
   675→   675→
   676→   676→        {/* ═══ WRITE/EDIT ═══ */}
   677→   677→        {view === 'write' && (
   678→   678→          <EntryEditor
   679→   679→            key={`write-${editEntry?.id ?? 'new'}`}
   680→   680→            onSave={handleSave}
   681→   681→            onCancel={() => {
   682→   682→              setEditEntry(undefined);
   683→   683→              setView(entries.length > 0 ? 'list' : 'cover');
   684→   684→            }}
   685→   685→            initialEntry={editEntry}
   686→   686→          />
   687→   687→        )}
   688→   688→
   689→   689→        {/* ═══ READ ENTRY ═══ */}
   690→   690→        {view === 'read' && currentEntry && (
   691→   691→          <EntryViewer
   692→   692→            key={`read-${currentEntry.id}`}
   693→   693→            entry={currentEntry}
   694→   694→            onEdit={handleEdit}
   695→   695→            onDelete={handleDelete}
   696→   696→            onClose={() => setView('list')}
   697→   697→            onPrev={() => {
   698→   698→              if (currentIndex > 0) setSelectedId(entries[currentIndex - 1].id);
   699→   699→            }}
   700→   700→            onNext={() => {
   701→   701→              if (currentIndex < entries.length - 1) setSelectedId(entries[currentIndex + 1].id);
   702→   702→            }}
   703→   703→            hasPrev={currentIndex > 0}
   704→   704→            hasNext={currentIndex < entries.length - 1}
   705→   705→            index={currentIndex}
   706→   706→            total={entries.length}
   707→   707→          />
   708→   708→        )}
   709→   709→      </AnimatePresence>
   710→   710→
   711→   711→      {/* Delete confirmation toast */}
   712→   712→      <AnimatePresence>
   713→   713→        {confirmDelete && (
   714→   714→          <motion.div
   715→   715→            initial={{ opacity: 0, y: 20 }}
   716→   716→            animate={{ opacity: 1, y: 0 }}
   717→   717→            exit={{ opacity: 0, y: 20 }}
   718→   718→            className="fixed bottom-20 left-1/2 -translate-x-1/2 z-50 sk-paper rounded-lg px-4 py-2 shadow-xl"
   719→   719→          >
   720→   720→            <p className="old-handwritten text-xs text-[#c0392b]">
   721→   721→              Tap &quot;Discard&quot; again to confirm deletion
   722→   722→            </p>
   723→   723→          </motion.div>
   724→   724→        )}
   725→   725→      </AnimatePresence>
   726→   726→    </div>
   727→   727→  );
