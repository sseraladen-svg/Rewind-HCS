   137→   137→// useDBState — IndexedDB-backed state
   138→   138→// ═══════════════════════════════════════════════════════════════
   139→   139→
   140→   140→function useDBState<T>(
   141→   141→  loader: () => Promise<T>,
   142→   142→  saver: (val: T) => Promise<void>,
   143→   143→  initial: T,
   144→   144→): [T, (updater: T | ((prev: T) => T)) => void] {
   145→   145→  const [value, setValue] = useState<T>(initial);
   146→   146→
   147→   147→  useEffect(() => {
   148→   148→    loader().then((loaded) => setValue(loaded)).catch(() => {});
   149→   149→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
   150→   150→
   151→   151→  const set = useCallback(
   152→   152→    (updater: T | ((prev: T) => T)) => {
   153→   153→      setValue((prev) => {
   154→   154→        const next = typeof updater === 'function' ? (updater as (prev: T) => T)(prev) : updater;
   155→   155→        saver(next).catch(() => {});
   156→   156→        return next;
   157→   157→      });
   158→   158→    },
   159→   159→    [saver], // eslint-disable-line react-hooks/exhaustive-deps
   160→   160→  );
   161→   161→
   162→   162→  return [value, set];
   163→   163→}

   176→   176→}
