    48→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    49→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    50→  const [bootLines, setBootLines] = useState<string[]>([]);
    51→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    52→
    53→  const allBootLines = useMemo(() => [
    54→    'BIOS v2.4 — Nostalgia Systems Inc.',
    55→    'POST: Memory test... 640K OK',
    56→    'Detecting floppy drives... A: OK  B: OK',
    57→    'Loading REWIND.EXE...',
    58→    'Mounting childhood memories... 47% complete',
    59→    'Indexing nostalgia database...',
    60→    'Rewinding time...',
    61→    'Loading good old days... 83% complete',
    62→    'Ready.',
    63→  ], []);
    64→
    65→  useEffect(() => {
    66→    const timers: ReturnType<typeof setTimeout>[] = [];
    67→    // Initialize audio on first boot
    68→    timers.push(setTimeout(() => initAudio(), 100));
    69→    // Reveal boot lines one by one with beep sounds
    70→    allBootLines.forEach((line, i) => {
    71→      timers.push(setTimeout(() => {
    72→        setBootLines(prev => [...prev, line]);
    73→        soundBootBeep();
    74→        if (i === allBootLines.length - 1) {
    75→          // Last line — show title after a beat
    76→          timers.push(setTimeout(() => {
    77→            setPhase(1);
    78→            soundBootReady();
    79→          }, 1000));
    80→        }
    81→      }, 500 + i * 500));
    82→    });
    83→
    84→    // Auto-dismiss after title phase
    85→    fadeTimerRef.current = setTimeout(() => {
    86→      setPhase(2);
    87→      setTimeout(onDone, 800);
    88→    }, 7000);
    89→
    90→    return () => {
    91→      timers.forEach(clearTimeout);
    92→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    93→    };
    94→  }, [allBootLines, onDone]);
    95→
    96→  return (
    97→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    98→      <div className="crt-scanlines" />
    99→      <div className="vhs-noise" />
   100→      <div className="vhs-tracking" />
   101→      <div className="vhs-tracking" style={{ animationDelay: '1.2s', opacity: 0.5 }} />
   102→
   103→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
   104→        {/* Boot lines */}
   105→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
   106→          {bootLines.map((line, i) => (
   107→            <div
   108→              key={i}
   109→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
   110→              style={{ animationDelay: '0s' }}
   111→            >
   112→              {line}
   113→              {i === bootLines.length - 1 && (
   114→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
   115→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
   116→                />
   117→              )}
   118→            </div>
   119→          ))}
   120→        </div>
   121→
   122→        {/* Title */}
   123→        {phase >= 1 && (
   124→          <motion.div
   125→            initial={{ opacity: 0, scale: 0.9 }}

   809→export default function Home() {

   832→          setShowOnboarding(true);

   863→    setTimeout(() => {
   864→      setBirthYear(null);
   865→      hideOldLoader();
   866→    }, 850);
   867→  }, []);
