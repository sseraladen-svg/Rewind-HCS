   181→interface MatchCard {
   182→  id: number;
   183→  emoji: string;
   184→  label: string;
   185→  isFlipped: boolean;
   186→  isMatched: boolean;
   187→}
   188→
   189→function shuffle<T>(arr: T[]): T[] {
   190→  const a = [...arr];
   191→  for (let i = a.length - 1; i > 0; i--) {
   192→    const j = Math.floor(Math.random() * (i + 1));
   193→    [a[i], a[j]] = [a[j], a[i]];
   194→  }
   195→  return a;
   196→}
   197→
   198→function createMatchCards(data: DecadeData): MatchCard[] {
   199→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   200→  const picked = shuffle(pool).slice(0, 6);
   201→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   202→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   203→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   204→  ]);
   205→  return shuffle(initial);
   206→}

   208→function MemoryMatchGame({ data }: { data: DecadeData }) {
   209→  const [cards, setCards] = useState<MatchCard[]>([]);
   210→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   211→  const [moves, setMoves] = useState(0);
   212→  const [matches, setMatches] = useState(0);
   213→  const [gameWon, setGameWon] = useState(false);
   214→  const lockRef = useRef(false);
   215→
   216→  useEffect(() => {
   217→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   218→    return () => cancelAnimationFrame(id);
   219→  }, [data]);
   220→
   221→  const resetGame = useCallback(() => {
   222→    setCards(createMatchCards(data));
   223→    setFlippedIds([]);
   224→    setMoves(0);
   225→    setMatches(0);
   226→    setGameWon(false);
   227→    lockRef.current = false;
   228→  }, [data]);
   229→
   230→  const handleFlip = useCallback((id: number) => {
   231→    if (lockRef.current) return;
   232→    setCards(prev => {
   233→      const card = prev.find(c => c.id === id);
   234→      if (!card || card.isFlipped || card.isMatched) return prev;
   235→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   236→    });
   237→
   238→    setFlippedIds(prev => {
   239→      if (prev.includes(id) || prev.length >= 2) return prev;
   240→      const newFlipped = [...prev, id];
   241→
   242→      if (newFlipped.length === 2) {
   243→        setMoves(m => m + 1);
   244→        lockRef.current = true;
   245→
   246→        // Use a timeout to read the latest cards state
   247→        setTimeout(() => {
   248→          setCards(currentCards => {
   249→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   250→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   251→
   252→            if (first.label === second.label) {
   253→              // Match!
   254→              const updated = currentCards.map(c =>
   255→                c.id === first.id || c.id === second.id
   256→                  ? { ...c, isMatched: true, isFlipped: true }
   257→                  : c
   258→              );
   259→              setMatches(m => {
   260→                const newM = m + 1;
   261→                if (newM === 6) setGameWon(true);
   262→                return newM;
   263→              });
   264→              setFlippedIds([]);
   265→              lockRef.current = false;
   266→              return updated;
   267→            } else {
   268→              // No match — flip back
   269→              setTimeout(() => {
   270→                setCards(cc => cc.map(c =>
   271→                  c.id === first.id || c.id === second.id
   272→                    ? { ...c, isFlipped: false }
   273→                    : c
   274→                ));
   275→                setFlippedIds([]);
   276→                lockRef.current = false;
   277→              }, 700);
   278→              return currentCards;
   279→            }
   280→          });
   281→        }, 50);
   282→      }
   283→
   284→      return newFlipped;
   285→    });
   286→  }, []);
   287→
   288→  return (
   289→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   290→      <div className="flex items-center justify-between mb-4">
   291→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   292→          <span className="text-xl">🃏</span> Memory Match
   293→        </h4>
   294→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   295→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   296→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   297→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   298→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   299→          </button>
   300→        </div>
   301→      </div>
   302→
   303→      {gameWon && (
   304→        <motion.div
   305→          initial={{ scale: 0.8, opacity: 0 }}
   306→          animate={{ scale: 1, opacity: 1 }}
   307→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   308→        >
   309→          <span className="text-5xl mb-2">🎉</span>
   310→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   311→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   312→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   313→            Play Again
   314→          </button>
   315→        </motion.div>
   316→      )}
   317→
   318→      <div className="grid grid-cols-4 gap-2">
   319→        {cards.map((card) => {
   320→          const showFront = card.isFlipped || card.isMatched;
   321→          return (
   322→            <button
   323→              key={card.id}
   324→              onClick={() => handleFlip(card.id)}
   325→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   326→              disabled={card.isMatched}
   327→              style={{
   328→                background: showFront
   329→                  ? '#faf6eb'
   330→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   331→                borderColor: showFront
   332→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')
   333→                  : '#5c4033',
   334→                boxShadow: showFront
   335→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   336→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   337→                opacity: card.isMatched ? 0.6 : 1,
   338→              }}
   339→            >
   340→              {showFront ? (
   341→                <>
   342→                  <span className="text-2xl md:text-3xl">{card.emoji}</span>
   343→                  <span className="old-typewriter text-[7px] md:text-[9px] font-bold text-[#5c4033] mt-0.5 text-center leading-tight" style={{ maxWidth: '100%', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{card.label}</span>
   344→                </>
   345→              ) : (
   346→                <span className="text-xl text-[#d4a574] font-bold old-title">?</span>
   347→              )}
   348→            </button>
   349→          );
   350→        })}
   351→      </div>
   352→    </div>
   353→  );
   354→}

   356→/* ═══════════════════════════════════════════════════
   357→   GAME 2: GUESS THE YEAR
   358→   ═══════════════════════════════════════════════════ */
   359→interface TriviaQ {
   360→  question: string;
   361→  emoji: string;
   362→  correct: number;
   363→  options: string[];
   364→}
   365→
   366→function buildTrivia(data: DecadeData): TriviaQ[] {
   367→  const decadeStart = parseInt(data.years.split('–')[0]);
   368→  const qs: TriviaQ[] = [];
   369→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   370→  const shuffled = shuffle(all).slice(0, 8);
   371→
   372→  for (const item of shuffled) {
   373→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   374→    const wrongYears = [
   375→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   376→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   377→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   378→    ].filter(y => y !== realYear).slice(0, 3);
   379→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   380→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   381→    qs.push({
   382→      question: `When did "${item.title}" first become popular?`,
   383→      emoji: item.emoji,
   384→      correct: options.indexOf(realYear),
   385→      options: options.map(String),
   386→    });
   387→  }
   388→  return qs;
   389→}

   391→function GuessTheYearGame({ data }: { data: DecadeData }) {
   392→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   393→  const [qIdx, setQIdx] = useState(0);
   394→  const [score, setScore] = useState(0);
   395→  const [selected, setSelected] = useState<number | null>(null);
   396→  const [showResult, setShowResult] = useState(false);
   397→  const [done, setDone] = useState(false);
   398→
   399→  useEffect(() => {
   400→    setQuestions(buildTrivia(data));
   401→  }, [data]);
   402→
   403→  const q = questions[qIdx];
   404→
   405→  const handleAnswer = (optIdx: number) => {
   406→    if (showResult || !q) return;
   407→    setSelected(optIdx);
   408→    setShowResult(true);
   409→    if (optIdx === q.correct) setScore(s => s + 1);
   410→    setTimeout(() => {
   411→      if (qIdx + 1 < questions.length) {
   412→        setQIdx(i => i + 1);
   413→        setSelected(null);
   414→        setShowResult(false);
   415→      } else {
   416→        setDone(true);
   417→      }
   418→    }, 1500);
   419→  };
   420→
   421→  const resetGame = () => {
   422→    const fresh = buildTrivia(data);
   423→    setQuestions(fresh);
   424→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   425→  };
   426→
   427→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   428→
   429→  if (done) {
   430→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   431→    return (
   432→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   433→        <div className="text-center py-4">
   434→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   435→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
   436→          <p className="old-body text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   437→          <p className="old-quote text-xs text-[#8b7355] mt-2">
   438→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   439→          </p>
   440→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   441→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   442→          </button>
   443→        </div>
   444→      </div>
   445→    );
   446→  }
   447→
   448→  return (
   449→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   450→      <div className="flex items-center justify-between mb-4">
   451→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   452→          <span className="text-xl">📅</span> Guess the Year
   453→        </h4>
   454→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   455→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{qIdx + 1}/{questions.length}</span>
   456→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Score: {score}</span>
   457→        </div>
   458→      </div>
   459→
   460→      {/* Progress bar */}
   461→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   462→        <div
   463→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   464→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   465→        />
   466→      </div>
   467→
   468→      <div className="text-center mb-4">
   469→        <span className="text-5xl block mb-2">{q.emoji}</span>
   470→        <p className="old-body text-sm font-semibold text-[#5c4033]">{q.question}</p>
   471→      </div>
   472→
   473→      <div className="grid grid-cols-2 gap-2">
   474→        {q.options.map((opt, i) => {
   475→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 old-typewriter ';
   476→          if (showResult) {
   477→            if (i === q.correct) cls += 'border-green-500 bg-green-50 sk-sticker';
   478→            else if (i === selected) cls += 'border-red-400 bg-red-50 opacity-60';
   479→            else cls += 'border-[#d4c5a9] opacity-40';
   480→          } else {
   481→            cls += 'border-[#d4c5a9] hover:border-[#8b5e3c] hover:shadow-md active:translate-y-0.5';
   482→          }
   483→          return (
   484→            <button key={`${qIdx}-${i}`} onClick={() => handleAnswer(i)} className={cls} disabled={showResult}>
   485→              {opt}
   486→            </button>
   487→          );
   488→        })}
   489→      </div>

   511→function WhackGame({ data }: { data: DecadeData }) {
   512→  const GRID = 9;
   513→  const allItems = useMemo(() => [
   514→    ...data.toys, ...data.snacks, ...data.cartoons, ...data.games,
   515→  ], [data]);
   516→
   517→  const [grid, setGrid] = useState<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   518→  const [score, setScore] = useState(0);
   519→  const [timeLeft, setTimeLeft] = useState(20);
   520→  const [gameActive, setGameActive] = useState(false);
   521→  const [gameOver, setGameOver] = useState(false);
   522→  const [feedback, setFeedback] = useState<string | null>(null);
   523→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   524→  const spawnRef = useRef<ReturnType<typeof setInterval> | null>(null);
   525→  const gridRef = useRef<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   526→  const activeRef = useRef(false);
   527→
   528→  const spawnItem = useCallback(() => {
   529→    if (!activeRef.current) return;
   530→    const empty = gridRef.current.reduce<number[]>((acc, v, i) => v === null ? [...acc, i] : acc, []);
   531→    if (empty.length === 0) return;
   532→    const pos = empty[Math.floor(Math.random() * empty.length)];
   533→    const item = allItems[Math.floor(Math.random() * allItems.length)];
   534→    const newGrid = [...gridRef.current];
   535→    newGrid[pos] = item;
   536→    gridRef.current = newGrid;
   537→    setGrid([...newGrid]);
   538→    setTimeout(() => {
   539→      if (gridRef.current[pos] === item) {
   540→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);
   541→      }
   542→    }, 1200);
   543→  }, [allItems]);
   544→
   545→  const stopGame = useCallback(() => {
   546→    activeRef.current = false;
   547→    if (timerRef.current) { clearInterval(timerRef.current); timerRef.current = null; }
   548→    if (spawnRef.current) { clearInterval(spawnRef.current); spawnRef.current = null; }
   549→  }, []);
   550→
   551→  const startGame = useCallback(() => {
   552→    // Clear any previous intervals
   553→    stopGame();

   555→    const fresh = Array(GRID).fill(null);
   556→    gridRef.current = fresh;
   557→    setGrid([...fresh]);
   558→    setScore(0); setTimeLeft(20); setGameOver(false); setGameActive(true);
   559→    activeRef.current = true;
   560→

   562→      if (!activeRef.current) return;

   567→          setGameOver(true);
   568→          return 0;
   569→        }
   570→        return t - 1;
   571→      });
   572→    }, 1000);
   573→    spawnRef.current = setInterval(spawnItem, 700);
   574→  }, [spawnItem, stopGame]);
   575→
   576→  useEffect(() => {
   577→    return () => {
   578→      activeRef.current = false;
   579→      if (timerRef.current) clearInterval(timerRef.current);
   580→      if (spawnRef.current) clearInterval(spawnRef.current);
   581→    };
   582→  }, []);
   583→
   584→  const handleWhack = (idx: number) => {
   585→    if (!gameActive || !grid[idx]) return;
   586→    setScore(s => s + 10);
   587→    setFeedback('+10');
   588→    const g = [...gridRef.current]; g[idx] = null; gridRef.current = g; setGrid([...g]);
   589→    setTimeout(() => setFeedback(null), 300);
   590→  };
   591→
   592→  const handleMiss = () => {
   593→    if (!gameActive) return;
   594→    setFeedback('Miss!');
   595→    setTimeout(() => setFeedback(null), 300);
   596→  };
   597→
   598→  return (
   599→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   600→      <div className="flex items-center justify-between mb-4">
   601→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   602→          <span className="text-xl">🔨</span> Tap the Memory
   603→        </h4>
   604→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   605→          <span className="sk-inset px-2 py-1 rounded old-typewriter flex items-center gap-1">
   606→            <Timer className="h-3 w-3" />{timeLeft}s
   607→          </span>
   608→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{score}pts</span>
   609→        </div>
   610→      </div>
   611→
   612→      {!gameActive && !gameOver && (
   613→        <div className="text-center py-6">
   614→          <p className="old-body text-sm text-[#8b7355] mb-4">Tap nostalgic items as they pop up! You have 20 seconds.</p>
   615→          <button onClick={startGame} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   616→            <Play className="h-5 w-5" /> Start Game
   617→          </button>
   618→        </div>
   619→      )}
   620→
   621→      {gameOver && (
   622→        <div className="text-center py-4">
   623→          <span className="text-5xl mb-2 block">{score >= 80 ? '🏆' : score >= 40 ? '🎉' : '💪'}</span>
   624→          <p className="old-title text-lg font-bold text-[#5c4033]">Time&apos;s Up!</p>
   625→          <p className="old-body text-sm text-[#8b7355] mt-1">Score: {score} points</p>
   626→          <button onClick={startGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   627→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   628→          </button>
   629→        </div>
   630→      )}
   631→
   632→      {gameActive && (
   633→        <div className="grid grid-cols-3 gap-2 select-none">
   634→          {grid.map((item, i) => (
   635→            <button
   636→              key={i}
   637→              onClick={() => item ? handleWhack(i) : handleMiss()}
   638→              className="aspect-square rounded-lg transition-all duration-150 cursor-pointer border-2 flex items-center justify-center"
   639→              style={{

   651→                <motion.span
   652→                  initial={{ scale: 0 }}

   654→                  className="text-3xl md:text-4xl"
   655→                >
   656→                  {item.emoji}
   657→                </motion.span>

   659→            </button>

   661→        </div>
   662→      )}
   663→
   664→      {feedback && (
