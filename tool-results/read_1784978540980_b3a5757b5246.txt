   212→function getChalkboardCardBg(color: string) {
   213→  const map: Record<string, string> = {
   214→    yellow: 'bg-[#3d3a1a]/60 border-l-4 border-l-[#c9b458]',
   215→    pink: 'bg-[#3d1a2a]/60 border-l-4 border-l-[#c9588a]',
   216→    blue: 'bg-[#1a2a3d]/60 border-l-4 border-l-[#5890c9]',
   217→    green: 'bg-[#1a3d2a]/60 border-l-4 border-l-[#58c970]',
   218→    purple: 'bg-[#2a1a3d]/60 border-l-4 border-l-[#9058c9]',
   219→  };
   220→  return map[color] || map.yellow;
   221→}

   223→// ─── IDB helpers ──────────────────────────────────────
   224→
   225→async function getBoardMemos(): Promise<PinnedMemory[]> {
   226→  try {
   227→    const db = await getDB();
   228→    const record: any = await db.get('era_skin', BOARD_MEMOS_ID);
   229→    return record?.memos || [];
   230→  } catch {
   231→    return [];
   232→  }
   233→}
   234→
   235→async function saveBoardMemos(memos: PinnedMemory[]): Promise<void> {
   236→  try {
   237→    const db = await getDB();
   238→    await db.put('era_skin', { id: BOARD_MEMOS_ID, memos });
   239→  } catch {
   240→    // silently fail
   241→  }
   242→}
