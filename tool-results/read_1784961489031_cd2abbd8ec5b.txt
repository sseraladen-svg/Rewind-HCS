    36→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    37→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    38→  const [bootLines, setBootLines] = useState<string[]>([]);
    39→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    40→
    41→  const allBootLines = useMemo(() => [
    42→    'BIOS v2.4 — Nostalgia Systems Inc.',
    43→    'POST: Memory test... 640K OK',
    44→    'Detecting floppy drives... A: OK  B: OK',
    45→    'Loading REWIND.EXE...',
    46→    'Mounting childhood memories... 47% complete',
    47→    'Indexing nostalgia database...',
    48→    'Rewinding time...',
    49→    'Loading good old days... 83% complete',
    50→    'Ready.',
    51→  ], []);
    52→
    53→  useEffect(() => {
    54→    const timers: ReturnType<typeof setTimeout>[] = [];
    55→    // Reveal boot lines one by one
    56→    allBootLines.forEach((line, i) => {
    57→      timers.push(setTimeout(() => {
    58→        setBootLines(prev => [...prev, line]);
    59→        if (i === allBootLines.length - 1) {
    60→          // Last line — show title after a beat
    61→          timers.push(setTimeout(() => setPhase(1), 1000));
    62→        }
    63→      }, 500 + i * 500));
    64→    });
    65→
    66→    // Auto-dismiss after title phase
    67→    fadeTimerRef.current = setTimeout(() => {
    68→      setPhase(2);
    69→      setTimeout(onDone, 800);
    70→    }, 7000);
    71→
    72→    return () => {
    73→      timers.forEach(clearTimeout);
    74→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    75→    };
    76→  }, [allBootLines, onDone]);
    77→
    78→  return (
    79→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    80→      <div className="crt-scanlines" />
    81→      <div className="vhs-noise" />
    82→      <div className="vhs-tracking" />

    85→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
    86→        {/* Boot lines */}
    87→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
    88→          {bootLines.map((line, i) => (
    89→            <div
    90→              key={i}
    91→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    92→              style={{ animationDelay: '0s' }}
    93→            >
    94→              {line}
    95→              {i === bootLines.length - 1 && (
    96→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
    97→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
    98→                />
    99→              )}
   100→            </div>
   101→          ))}
   102→        </div>
   103→
   104→        {/* Title */}
   105→        {phase >= 1 && (
   106→          <motion.div
   107→            initial={{ opacity: 0, scale: 0.9 }}
   108→            animate={{ opacity: 1, scale: 1 }}
   109→            transition={{ duration: 1.0 }}

   319→function QuoteBanner({ quotes }: { quotes: string[] }) {

   328→      <motion.div
   329→        key={idx}
   330→        initial={{ opacity: 0, y: 10 }}

   335→      >

   388→function getDecadeTheme(birthYear: number): string {
   389→  const decade = Math.floor(birthYear / 10) * 10;
   390→  if (decade === 1980) return '80s';
   391→  if (decade === 1990) return '90s';
   392→  if (decade === 2000) return '2000s';
   393→  return '';
   394→}

   759→export default function Home() {
   760→  const [birthYear, setBirthYear] = useState<number | null>(null);
   761→  const [loading, setLoading] = useState(true);
   762→
   763→  // Run one-time localStorage → IndexedDB migration on first load
   764→  useEffect(() => {
   765→    if (typeof window !== 'undefined') {
   766→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   767→    }
   768→  }, []);
   769→
   770→  const handleLoadingDone = useCallback(() => {
   771→    setLoading(false);
   772→  }, []);
   773→
   774→  const handleSubmit = useCallback((year: number) => {
   775→    triggerOldLoader('Tuning into your era...');
   776→    setTimeout(() => {
   777→      setBirthYear(year);
   778→      window.scrollTo({ top: 0, behavior: 'smooth' });
   779→      hideOldLoader();
   780→    }, 1100);
   781→  }, []);
   782→
   783→  const handleBack = useCallback(() => {
   784→    triggerOldLoader('Rewinding...');
   785→    setTimeout(() => {
   786→      setBirthYear(null);
   787→      hideOldLoader();
   788→    }, 850);
   789→  }, []);
   790→
   791→  return (
   792→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   793→      {/* CRT Boot Loading Screen */}
   794→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   795→
   796→      <AnimatePresence mode="wait">
   797→        {birthYear === null ? (
   798→          <motion.div
   799→            key="hero"
   800→            initial={{ opacity: 0 }}
   801→            animate={{ opacity: 1 }}
   802→            exit={{ opacity: 0, y: -20 }}
   803→            transition={{ duration: 0.4 }}
   804→            style={{ opacity: loading ? 0 : 1 }}
   805→          >
   806→            <HeroSection onSubmit={handleSubmit} />
   807→          </motion.div>
   808→        ) : (
   809→          <motion.div
   810→            key="dashboard"
   811→            initial={{ opacity: 0, y: 20 }}
   812→            animate={{ opacity: 1, y: 0 }}
   813→            exit={{ opacity: 0 }}
   814→            transition={{ duration: 0.5 }}
   815→            className="flex flex-col min-h-screen"
   816→          >
   817→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   818→          </motion.div>
   819→        )}
   820→      </AnimatePresence>
   821→    </div>
   822→  );
