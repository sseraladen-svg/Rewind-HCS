     1→'use client';
     2→
     3→import { useState, useRef, useCallback, useEffect } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Camera, Video, X, Download, Trash2, Image as ImageIcon,
     7→  RotateCcw, SwitchCamera, Zap, Circle, Square, Play,
     8→} from 'lucide-react';
     9→import {
    10→  PIPELINE_PRESETS,
    11→  processFrame,
    12→  processPreviewFrame,
    13→  getGateFlickerEV,
    14→  type PipelineConfig,
    15→} from '@/lib/image-pipeline';
    16→import { getAllCameraPhotos, addCameraPhoto, deleteCameraPhoto } from '@/lib/idb-store';
    17→
    18→/* ═══════ TYPES ═══════ */
    19→interface CapturedItem {
    20→  id: string;
    21→  type: 'photo' | 'video';
    22→  url: string;
    23→  timestamp: number;
    24→  blob?: Blob;
    25→}
    26→
    27→type FilterPreset = 'none' | 'sepia70s' | 'faded80s' | 'warm90s' | 'bw60s' | 'polaroid' | 'vhs' | 'kodak';
    28→
    29→/* Real-time preview resolution (low for performance) */
    30→const PREVIEW_W = 360;
    31→const PREVIEW_H = 202;
    32→const TARGET_FPS = 24;
    33→const FRAME_INTERVAL = 1000 / TARGET_FPS;
    34→
    35→/* ═══════ VINTAGE CAMERA COMPONENT ═══════ */
    36→export default function VintageCamera({ onClose }: { onClose: () => void }) {
    37→  const videoRef = useRef<HTMLVideoElement>(null);
    38→  const captureCanvasRef = useRef<HTMLCanvasElement>(null);   // full-res for capture
    39→  const previewCanvasRef = useRef<HTMLCanvasElement>(null);   // live processed preview
    40→  const offscreenRef = useRef<HTMLCanvasElement | null>(null);
    41→  const streamRef = useRef<MediaStream | null>(null);
    42→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
    43→  const recordedChunksRef = useRef<Blob[]>([]);
    44→  const flashRef = useRef<HTMLDivElement>(null);
    45→  const rafRef = useRef<number>(0);
    46→  const lastFrameTimeRef = useRef<number>(0);
    47→  const frameCountRef = useRef<number>(0);
    48→  const flickerEvRef = useRef<number>(0);
    49→
    50→  const [cameraOpen, setCameraOpen] = useState(false);
    51→  const [facingMode, setFacingMode] = useState<'user' | 'environment'>('environment');
    52→  const [mode, setMode] = useState<'photo' | 'video'>('photo');
    53→  const [filter, setFilter] = useState<FilterPreset>('sepia70s');
    54→  const [recording, setRecording] = useState(false);
    55→  const [recordTime, setRecordTime] = useState(0);
    56→  const [flash, setFlash] = useState(false);
    57→  const [gallery, setGallery] = useState<CapturedItem[]>([]);
    58→  const [viewItem, setViewItem] = useState<CapturedItem | null>(null);
    59→  const [cameraError, setCameraError] = useState<string | null>(null);
    60→  const [showFilters, setShowFilters] = useState(false);
    61→  const [shutterAnim, setShutterAnim] = useState(false);
    62→  const [processing, setProcessing] = useState(false);
    63→
    64→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    65→  const dbObjectUrlsRef = useRef<string[]>([]);
    66→
    67→  /* ── Load persisted photos from IndexedDB on mount ── */
    68→  useEffect(() => {
    69→    let cancelled = false;
    70→    getAllCameraPhotos().then((photos) => {
    71→      if (cancelled) return;
    72→      const items: CapturedItem[] = photos.map((photo) => {
    73→        const url = URL.createObjectURL(photo.image_blob);
    74→        dbObjectUrlsRef.current.push(url);
    75→        return {
    76→          id: photo.id,
    77→          type: 'photo' as const,
    78→          url,
    79→          timestamp: photo.created_at,
    80→          blob: photo.image_blob,
    81→        };
    82→      });
    83→      if (items.length > 0) {
    84→        setGallery((prev) => {
    85→          // Avoid duplicates (e.g., if React StrictMode double-fires)
    86→          const existingIds = new Set(prev.map((i) => i.id));
    87→          const newItems = items.filter((i) => !existingIds.has(i.id));
    88→          return [...newItems, ...prev];
    89→        });
    90→      }
    91→    });
    92→    return () => { cancelled = true; };
    93→  }, []);
    94→
    95→  /* ── Get current preset config ── */
    96→  const currentConfig: PipelineConfig = PIPELINE_PRESETS[filter] || PIPELINE_PRESETS['none'];
    97→
    98→  /* ── Start Camera ── */
    99→  const startCamera = useCallback(async () => {
   100→    try {
   101→      if (streamRef.current) {
   102→        streamRef.current.getTracks().forEach(t => t.stop());
   103→      }
   104→      const stream = await navigator.mediaDevices.getUserMedia({
   105→        video: { facingMode, width: { ideal: 1280 }, height: { ideal: 720 } },
   106→        audio: mode === 'video',
   107→      });
   108→      streamRef.current = stream;
   109→      if (videoRef.current) {
   110→        videoRef.current.srcObject = stream;
   111→        await videoRef.current.play();
   112→      }
   113→      setCameraOpen(true);
   114→      setCameraError(null);
   115→    } catch (err) {
   116→      console.error('Camera error:', err);
   117→      setCameraError('Camera access denied or not available. Please allow camera permissions.');
   118→    }
   119→  }, [facingMode, mode]);
   120→
   121→  /* ── Stop Camera ── */
   122→  const stopCamera = useCallback(() => {
   123→    if (rafRef.current) {
   124→      cancelAnimationFrame(rafRef.current);
   125→      rafRef.current = 0;
   126→    }
   127→    if (streamRef.current) {
   128→      streamRef.current.getTracks().forEach(t => t.stop());
   129→      streamRef.current = null;
   130→    }
   131→    setCameraOpen(false);
   132→    if (recordTimerRef.current) {
   133→      clearInterval(recordTimerRef.current);
   134→      recordTimerRef.current = null;
   135→    }
   136→  }, []);
   137→
   138→  /* ── Switch Camera ── */
   139→  const handleSwitchCamera = useCallback(() => {
   140→    setFacingMode(prev => prev === 'user' ? 'environment' : 'user');
   141→  }, []);
   142→
   143→  /* ── Restart on facing/mode change ── */
   144→  useEffect(() => {
   145→    if (cameraOpen) {
   146→      startCamera();
   147→    }
   148→    return () => {};
   149→  }, [facingMode, mode, cameraOpen, startCamera]);
   150→
   151→  /* ── Real-time preview loop ── */
   152→  useEffect(() => {
   153→    if (!cameraOpen || !videoRef.current || !previewCanvasRef.current) return;
   154→
   155→    // Create offscreen canvas for drawing video frames
   156→    if (!offscreenRef.current) {
   157→      offscreenRef.current = document.createElement('canvas');
   158→    }
   159→    offscreenRef.current.width = PREVIEW_W;
   160→    offscreenRef.current.height = PREVIEW_H;
   161→    const offCtx = offscreenRef.current.getContext('2d', { willReadFrequently: true });
   162→    if (!offCtx) return;
   163→
   164→    const previewCanvas = previewCanvasRef.current;
   165→    previewCanvas.width = PREVIEW_W;
   166→    previewCanvas.height = PREVIEW_H;
   167→    const previewCtx = previewCanvas.getContext('2d', { willReadFrequently: true });
   168→    if (!previewCtx) return;
   169→
   170→    let running = true;
   171→
   172→    const renderLoop = (timestamp: number) => {
   173→      if (!running) return;
   174→
   175→      const elapsed = timestamp - lastFrameTimeRef.current;
   176→      if (elapsed >= FRAME_INTERVAL) {
   177→        lastFrameTimeRef.current = timestamp - (elapsed % FRAME_INTERVAL);
   178→        frameCountRef.current++;
   179→
   180→        const video = videoRef.current;
   181→        if (!video || video.readyState < 2) {
   182→          rafRef.current = requestAnimationFrame(renderLoop);
   183→          return;
   184→        }
   185→
   186→        // Draw video frame to offscreen canvas (mirrored if front camera)
   187→        offCtx.save();
   188→        if (facingMode === 'user') {
   189→          offCtx.translate(PREVIEW_W, 0);
   190→          offCtx.scale(-1, 1);
   191→        }
   192→        offCtx.drawImage(video, 0, 0, PREVIEW_W, PREVIEW_H);
   193→        offCtx.restore();
   194→
   195→        // Apply lightweight preview pipeline
   196→        try {
   197→          processPreviewFrame(offCtx, PREVIEW_W, PREVIEW_H, filter, frameCountRef.current);
   198→
   199→          // Gate flicker for video mode
   200→          if (mode === 'video' && currentConfig.gateFlicker.enabled) {
   201→            flickerEvRef.current = getGateFlickerEV(currentConfig.gateFlicker.intensity);
   202→            const imgData = offCtx.getImageData(0, 0, PREVIEW_W, PREVIEW_H);
   203→            const d = imgData.data;
   204→            const mult = 1 + flickerEvRef.current * 0.5;
   205→            for (let i = 0; i < d.length; i += 4) {
   206→              d[i] = Math.max(0, Math.min(255, Math.round(d[i] * mult)));
   207→              d[i + 1] = Math.max(0, Math.min(255, Math.round(d[i + 1] * mult)));
   208→              d[i + 2] = Math.max(0, Math.min(255, Math.round(d[i + 2] * mult)));
   209→            }
   210→            offCtx.putImageData(imgData, 0, 0);
   211→          }
   212→        } catch {
   213→          // silently skip if processing errors (e.g., tab not visible)
   214→        }
   215→
   216→        // Copy to visible preview canvas
   217→        previewCtx.drawImage(offscreenRef.current, 0, 0);
   218→      }
   219→
   220→      rafRef.current = requestAnimationFrame(renderLoop);
   221→    };
   222→
   223→    rafRef.current = requestAnimationFrame(renderLoop);
   224→
   225→    return () => {
   226→      running = false;
   227→      if (rafRef.current) {
   228→        cancelAnimationFrame(rafRef.current);
   229→        rafRef.current = 0;
   230→      }
   231→    };
   232→  }, [cameraOpen, filter, facingMode, mode, currentConfig.gateFlicker.enabled, currentConfig.gateFlicker.intensity]);
   233→
   234→  /* ── Capture Photo (Full Pipeline) ── */
   235→  const capturePhoto = useCallback(() => {
   236→    if (!videoRef.current || !captureCanvasRef.current) return;
   237→    const video = videoRef.current;
   238→    const canvas = captureCanvasRef.current;
   239→    const ctx = canvas.getContext('2d', { willReadFrequently: true });
   240→    if (!ctx) return;
   241→
   242→    setProcessing(true);
   243→
   244→    // Use full resolution
   245→    const w = video.videoWidth || 1280;
   246→    const h = video.videoHeight || 720;
   247→    canvas.width = w;
   248→    canvas.height = h;
   249→
   250→    // Mirror if front camera
   251→    if (facingMode === 'user') {
   252→      ctx.translate(w, 0);
   253→      ctx.scale(-1, 1);
   254→    }
   255→    ctx.drawImage(video, 0, 0, w, h);
   256→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   257→
   258→    // Apply full pipeline
   259→    try {
   260→      processFrame(ctx, w, h, filter, { addStamp: true });
   261→    } catch (e) {
   262→      console.error('Pipeline error:', e);
   263→    }
   264→
   265→    // Flash effect
   266→    setFlash(true);
   267→    setShutterAnim(true);
   268→    setTimeout(() => setFlash(false), 200);
   269→    setTimeout(() => setShutterAnim(false), 300);
   270→
   271→    // Save to gallery
   272→    canvas.toBlob((blob) => {
   273→      if (!blob) { setProcessing(false); return; }
   274→      const url = URL.createObjectURL(blob);
   275→      const id = `${Date.now()}-${Math.random().toString(36).slice(2)}`;
   276→      setGallery(prev => [{
   277→        id,
   278→        type: 'photo',
   279→        url,
   280→        timestamp: Date.now(),
   281→        blob,
   282→      }, ...prev]);
   283→      addCameraPhoto({ id, image_blob: blob, filter_used: filter, created_at: Date.now() });
   284→      setProcessing(false);
   285→    }, 'image/jpeg', 0.92);
   286→  }, [filter, facingMode]);
   287→
   288→  /* ── Start/Stop Video Recording ── */
   289→  const toggleRecording = useCallback(() => {
   290→    if (!recording && streamRef.current) {
   291→      const options = { mimeType: 'video/webm;codecs=vp9' };
   292→      const tryStart = (mimeType: string) => {
   293→        try {
   294→          const mediaRecorder = new MediaRecorder(streamRef.current!, { mimeType });
   295→          recordedChunksRef.current = [];
   296→          mediaRecorder.ondataavailable = (e) => {
   297→            if (e.data.size > 0) recordedChunksRef.current.push(e.data);
   298→          };
   299→          mediaRecorder.onstop = () => {
   300→            const blob = new Blob(recordedChunksRef.current, { type: 'video/webm' });
   301→            const url = URL.createObjectURL(blob);
   302→            setGallery(prev => [{
   303→              id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   304→              type: 'video',
   305→              url,
   306→              timestamp: Date.now(),
   307→              blob,
   308→            }, ...prev]);
   309→            recordedChunksRef.current = [];
   310→          };
   311→          mediaRecorder.start(100);
   312→          mediaRecorderRef.current = mediaRecorder;
   313→          setRecording(true);
   314→          setRecordTime(0);
   315→          recordTimerRef.current = setInterval(() => {
   316→            setRecordTime(prev => prev + 1);
   317→          }, 1000);
   318→        } catch {
   319→          if (mimeType.includes('vp9')) {
   320→            tryStart('video/webm;codecs=vp8');
   321→          } else {
   322→            setCameraError('Video recording not supported in this browser.');
   323→          }
   324→        }
   325→      };
   326→      tryStart(options.mimeType);
   327→    } else if (recording && mediaRecorderRef.current) {
   328→      mediaRecorderRef.current.stop();
   329→      mediaRecorderRef.current = null;
   330→      setRecording(false);
   331→      if (recordTimerRef.current) {
   332→        clearInterval(recordTimerRef.current);
   333→        recordTimerRef.current = null;
   334→      }
   335→    }
   336→  }, [recording]);
   337→
   338→  /* ── Delete from gallery ── */
   339→  const deleteItem = useCallback((id: string) => {
   340→    setGallery(prev => {
   341→      const item = prev.find(i => i.id === id);
   342→      if (item) {
   343→        URL.revokeObjectURL(item.url);
   344→        // Also remove from dbObjectUrlsRef tracking
   345→        dbObjectUrlsRef.current = dbObjectUrlsRef.current.filter(u => u !== item.url);
   346→      }
   347→      if (item?.type === 'photo') {
   348→        deleteCameraPhoto(id);
   349→      }
   350→      return prev.filter(i => i.id !== id);
   351→    });
   352→  }, []);
   353→
   354→  /* ── Download ── */
   355→  const downloadItem = useCallback((item: CapturedItem) => {
   356→    const a = document.createElement('a');
   357→    a.href = item.url;
   358→    a.download = `rewind_${item.type}_${new Date(item.timestamp).toISOString().replace(/[:.]/g, '-')}.${item.type === 'photo' ? 'jpg' : 'webm'}`;
   359→    document.body.appendChild(a);
   360→    a.click();
   361→    document.body.removeChild(a);
   362→  }, []);
   363→
   364→  /* ── Cleanup on close ── */
   365→  useEffect(() => {
   366→    return () => {
   367→      stopCamera();
   368→      gallery.forEach(item => URL.revokeObjectURL(item.url));
   369→      dbObjectUrlsRef.current.forEach(url => URL.revokeObjectURL(url));
   370→      dbObjectUrlsRef.current = [];
   371→    };
   372→  // eslint-disable-next-line react-hooks/exhaustive-deps
   373→  }, []);
   374→
   375→  /* ── Format time ── */
   376→  const formatTime = (s: number) => `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
   377→
   378→  /* ── Pipeline info for HUD ── */
   379→  const pipelineStages = [
   380→    currentConfig.chromaticAberration.enabled && 'CA',
   381→    currentConfig.toneCurve.blackLift > 0 && 'MATTE',
   382→    currentConfig.contrastReduce > 0 && 'FLAT',
   383→    currentConfig.antiClarity.enabled && 'SOFT',
   384→    currentConfig.grain.enabled && 'GRAIN',
   385→    currentConfig.halation.enabled && 'GLOW',
   386→    currentConfig.vignette.intensity > 0 && 'VIG',
   387→  ].filter(Boolean);
   388→
   389→  return (
   390→    <>
   391→      {/* Hidden canvases */}
   392→      <canvas ref={captureCanvasRef} className="hidden" />
   393→
   394→      <AnimatePresence>
   395→        <motion.div
   396→          initial={{ opacity: 0 }}
   397→          animate={{ opacity: 1 }}
   398→          exit={{ opacity: 0 }}
   399→          className="fixed inset-0 z-50 flex flex-col"
   400→          style={{ background: '#1a1208' }}
   401→        >
   402→          {/* ═══ HEADER - Camera Top Bar ═══ */}
   403→          <div className="sk-leather px-3 py-2.5 flex items-center justify-between z-10">
   404→            <button onClick={() => { stopCamera(); onClose(); }} className="sk-btn old-btn px-3 py-1.5 rounded-lg text-xs font-bold cursor-pointer flex items-center gap-1 text-[#faf6eb]">
   405→              <X className="h-4 w-4" /> Close
   406→            </button>
   407→            <h2 className="old-title font-black text-base text-[#faf6eb] tracking-wide" style={{ textShadow: '0 1px 2px rgba(0,0,0,0.3)' }}>
   408→              📷 Vintage Camera
   409→            </h2>
   410→            <div className="w-16" />
   411→          </div>
   412→
   413→          {/* ═══ VIEWFINDER ═══ */}
   414→          <div className="flex-1 relative overflow-hidden flex items-center justify-center">
   415→            {!cameraOpen && !cameraError && (
   416→              <div className="text-center px-6">
   417→                <motion.div
   418→                  animate={{ rotate: [0, 5, -5, 0] }}
   419→                  transition={{ duration: 4, repeat: Infinity }}
   420→                  className="text-6xl mb-4"
   421→                >📸</motion.div>
   422→                <p className="old-body text-[#d4a574] mb-2 text-sm">Open the camera to capture memories in vintage style</p>
   423→                <p className="old-typewriter text-[10px] text-[#8b7355] mb-6">
   424→                  Full image processing pipeline: chromatic aberration, tone curves, Perlin grain, halation glow
   425→                </p>
   426→                <button onClick={startCamera} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   427→                  <Camera className="h-5 w-5" /> Open Camera
   428→                </button>
   429→              </div>
   430→            )}
   431→
   432→            {cameraError && (
   433→              <div className="text-center px-6">
   434→                <div className="text-5xl mb-4">😵</div>
   435→                <p className="old-body text-[#d4a574] mb-4 text-sm">{cameraError}</p>
   436→                <button onClick={startCamera} className="sk-btn old-btn px-5 py-2.5 rounded-lg font-bold cursor-pointer flex items-center gap-2 mx-auto">
   437→                  <RotateCcw className="h-4 w-4" /> Try Again
   438→                </button>
   439→              </div>
   440→            )}
   441→
   442→            {cameraOpen && (
   443→              <>
   444→                {/* Hidden video source (never shown directly) */}
   445→                <video
   446→                  ref={videoRef}
   447→                  className="hidden"
   448→                  playsInline
   449→                  muted
   450→                />
   451→
   452→                {/* Real-time processed canvas preview */}
   453→                <canvas
   454→                  ref={previewCanvasRef}
   455→                  className="absolute inset-0 w-full h-full object-cover"
   456→                  style={{
   457→                    imageRendering: 'auto', // smooth upscaling for that vintage soft look
   458→                  }}
   459→                />
   460→
   461→                {/* Halation glow overlay (CSS-based, adds dreamy ambient light) */}
   462→                {currentConfig.halation.enabled && (
   463→                  <div
   464→                    className="absolute inset-0 pointer-events-none mix-blend-screen"
   465→                    style={{
   466→                      background: `radial-gradient(ellipse at 50% 50%, rgba(255,240,200,${currentConfig.halation.intensity * 0.06}) 0%, transparent 70%)`,
   467→                    }}
   468→                  />
   469→                )}
   470→
   471→                {/* Scanline overlay (VHS era feel) */}
   472→                <div className="absolute inset-0 pointer-events-none opacity-[0.07]" style={{
   473→                  background: 'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(0,0,0,0.2) 2px, rgba(0,0,0,0.2) 4px)',
   474→                }} />
   475→
   476→                {/* Film border overlay - viewfinder frame */}
   477→                <div className="absolute inset-0 pointer-events-none border-[6px] border-[#1a1208] rounded-none" />
   478→
   479→                {/* Corner brackets */}
   480→                <div className="absolute top-3 left-3 w-8 h-8 border-t-2 border-l-2 border-[#d4a574]/50 pointer-events-none" />
   481→                <div className="absolute top-3 right-3 w-8 h-8 border-t-2 border-r-2 border-[#d4a574]/50 pointer-events-none" />
   482→                <div className="absolute bottom-3 left-3 w-8 h-8 border-b-2 border-l-2 border-[#d4a574]/50 pointer-events-none" />
   483→                <div className="absolute bottom-3 right-3 w-8 h-8 border-b-2 border-r-2 border-[#d4a574]/50 pointer-events-none" />
   484→
   485→                {/* Center crosshair */}
   486→                <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 pointer-events-none">
   487→                  <div className="w-px h-6 bg-[#d4a574]/30" />
   488→                  <div className="h-px w-6 bg-[#d4a574]/30 absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />
   489→                </div>
   490→
   491→                {/* Recording indicator */}
   492→                {recording && (
   493→                  <motion.div
   494→                    initial={{ opacity: 0, scale: 0.8 }}
   495→                    animate={{ opacity: 1, scale: 1 }}
   496→                    className="absolute top-4 left-1/2 -translate-x-1/2 flex items-center gap-2 sk-vhs-label px-3 py-1 rounded-full z-10"
   497→                  >
   498→                    <motion.div
   499→                      animate={{ opacity: [1, 0.3, 1] }}
   500→                      transition={{ duration: 1, repeat: Infinity }}
   501→                      className="w-2.5 h-2.5 rounded-full bg-red-500"
   502→                    />
   503→                    <span className="old-typewriter text-xs font-bold text-red-600">REC {formatTime(recordTime)}</span>
   504→                  </motion.div>
   505→                )}
   506→
   507→                {/* Filter label + Pipeline HUD */}
   508→                <div className="absolute top-4 right-4 z-10 flex flex-col items-end gap-1">
   509→                  <div className="sk-vhs-label px-2 py-1 rounded">
   510→                    <span className="old-typewriter text-[10px] font-bold text-[#5c4033]">
   511→                      {currentConfig.emoji} {currentConfig.name}
   512→                    </span>
   513→                  </div>
   514→                  {/* Pipeline stages HUD */}
   515→                  {pipelineStages.length > 0 && (
   516→                    <div className="sk-vhs-label px-2 py-0.5 rounded opacity-70">
