   186→interface MatchCard {
   187→  id: number;
   188→  emoji: string;
   189→  label: string;
   190→  isFlipped: boolean;
   191→  isMatched: boolean;
   192→}

   194→function shuffle<T>(arr: T[]): T[] {
   195→  const a = [...arr];
   196→  for (let i = a.length - 1; i > 0; i--) {
   197→    const j = Math.floor(Math.random() * (i + 1));
   198→    [a[i], a[j]] = [a[j], a[i]];
   199→  }
   200→  return a;
   201→}
   202→
   203→function createMatchCards(data: DecadeData): MatchCard[] {
   204→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   205→  const picked = shuffle(pool).slice(0, 6);
   206→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   207→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   208→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   209→  ]);
   210→  return shuffle(initial);
   211→}

   213→function MemoryMatchGame({ data }: { data: DecadeData }) {
   214→  const [cards, setCards] = useState<MatchCard[]>([]);
   215→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   216→  const [moves, setMoves] = useState(0);
   217→  const [matches, setMatches] = useState(0);
   218→  const [gameWon, setGameWon] = useState(false);
   219→  const lockRef = useRef(false);
   220→
   221→  useEffect(() => {
   222→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   223→    return () => cancelAnimationFrame(id);
   224→  }, [data]);
   225→
   226→  const resetGame = useCallback(() => {
   227→    setCards(createMatchCards(data));
   228→    setFlippedIds([]);
   229→    setMoves(0);
   230→    setMatches(0);
   231→    setGameWon(false);
   232→    lockRef.current = false;
   233→  }, [data]);
   234→
   235→  const handleFlip = useCallback((id: number) => {
   236→    if (lockRef.current) return;
   237→    setCards(prev => {
   238→      const card = prev.find(c => c.id === id);
   239→      if (!card || card.isFlipped || card.isMatched) return prev;
   240→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   241→    });
   242→
   243→    setFlippedIds(prev => {
   244→      if (prev.includes(id) || prev.length >= 2) return prev;
   245→      const newFlipped = [...prev, id];
   246→
   247→      if (newFlipped.length === 2) {
   248→        setMoves(m => m + 1);
   249→        lockRef.current = true;
   250→
   251→        // Use a timeout to read the latest cards state
   252→        setTimeout(() => {
   253→          setCards(currentCards => {
   254→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   255→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   256→
   257→            if (first.label === second.label) {
   258→              // Match!
   259→              const updated = currentCards.map(c =>
   260→                c.id === first.id || c.id === second.id
   261→                  ? { ...c, isMatched: true, isFlipped: true }
   262→                  : c
   263→              );
   264→              setMatches(m => {
   265→                const newM = m + 1;
   266→                if (newM === 6) setGameWon(true);
   267→                return newM;
   268→              });
   269→              setFlippedIds([]);
   270→              lockRef.current = false;
   271→              return updated;
   272→            } else {
   273→              // No match — flip back
   274→              setTimeout(() => {
   275→                setCards(cc => cc.map(c =>
   276→                  c.id === first.id || c.id === second.id
   277→                    ? { ...c, isFlipped: false }
   278→                    : c
   279→                ));
   280→                setFlippedIds([]);
   281→                lockRef.current = false;
   282→              }, 700);
   283→              return currentCards;
   284→            }
   285→          });
   286→        }, 50);
   287→      }
   288→
   289→      return newFlipped;
   290→    });
   291→  }, []);
   292→
   293→  return (
   294→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   295→      <div className="flex items-center justify-between mb-4">
   296→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   297→          <span className="text-xl">🃏</span> Memory Match
   298→        </h4>
   299→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   300→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   301→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   302→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   303→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   304→          </button>
   305→        </div>
   306→      </div>
   307→
   308→      {gameWon && (
   309→        <motion.div
   310→          initial={{ scale: 0.8, opacity: 0 }}
   311→          animate={{ scale: 1, opacity: 1 }}
   312→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   313→        >
   314→          <span className="text-5xl mb-2">🎉</span>
   315→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   316→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   317→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   318→            Play Again
   319→          </button>
   320→        </motion.div>
   321→      )}
   322→
   323→      <div className="grid grid-cols-4 gap-2">
   324→        {cards.map((card) => {
   325→          const showFront = card.isFlipped || card.isMatched;
   326→          return (
   327→            <button
   328→              key={card.id}
   329→              onClick={() => handleFlip(card.id)}
   330→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   331→              disabled={card.isMatched}
   332→              style={{
   333→                background: showFront
   334→                  ? '#faf6eb'
   335→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   336→                borderColor: showFront
   337→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')
   338→                  : '#5c4033',
   339→                boxShadow: showFront
   340→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   341→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   342→                opacity: card.isMatched ? 0.6 : 1,
   343→              }}
   344→            >
   345→              {showFront ? (
   346→                <>
   347→                  <span className="text-2xl md:text-3xl">{card.emoji}</span>
   348→                  <span className="old-typewriter text-[7px] md:text-[9px] font-bold text-[#5c4033] mt-0.5 text-center leading-tight" style={{ maxWidth: '100%', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{card.label}</span>
   349→                </>
   350→              ) : (
   351→                <span className="text-xl text-[#d4a574] font-bold old-title">?</span>
   352→              )}
   353→            </button>
   354→          );
   355→        })}
   356→      </div>
   357→    </div>
   358→  );
   359→}

   361→/* ═══════════════════════════════════════════════════

   364→interface TriviaQ {
   365→  question: string;
   366→  emoji: string;
   367→  correct: number;
   368→  options: string[];
   369→}
   370→
   371→function buildTrivia(data: DecadeData): TriviaQ[] {
   372→  const decadeStart = parseInt(data.years.split('–')[0]);
   373→  const qs: TriviaQ[] = [];
   374→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   375→  const shuffled = shuffle(all).slice(0, 8);
   376→
   377→  for (const item of shuffled) {
   378→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   379→    const wrongYears = [
   380→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   381→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   382→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   383→    ].filter(y => y !== realYear).slice(0, 3);
   384→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   385→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   386→    qs.push({
   387→      question: `When did "${item.title}" first become popular?`,
   388→      emoji: item.emoji,
   389→      correct: options.indexOf(realYear),
   390→      options: options.map(String),
   391→    });
   392→  }
   393→  return qs;
   394→}

   396→function GuessTheYearGame({ data }: { data: DecadeData }) {
   397→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   398→  const [qIdx, setQIdx] = useState(0);
   399→  const [score, setScore] = useState(0);
   400→  const [selected, setSelected] = useState<number | null>(null);
   401→  const [showResult, setShowResult] = useState(false);
   402→  const [done, setDone] = useState(false);
   403→
   404→  useEffect(() => {
   405→    setQuestions(buildTrivia(data));
   406→  }, [data]);
   407→
   408→  const q = questions[qIdx];
   409→
   410→  const handleAnswer = (optIdx: number) => {
   411→    if (showResult || !q) return;
   412→    setSelected(optIdx);
   413→    setShowResult(true);
   414→    if (optIdx === q.correct) setScore(s => s + 1);
   415→    setTimeout(() => {
   416→      if (qIdx + 1 < questions.length) {
   417→        setQIdx(i => i + 1);
   418→        setSelected(null);
   419→        setShowResult(false);
   420→      } else {
   421→        setDone(true);
   422→      }
   423→    }, 1500);
   424→  };
   425→
   426→  const resetGame = () => {
   427→    const fresh = buildTrivia(data);
   428→    setQuestions(fresh);
   429→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   430→  };
   431→
   432→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   433→
   434→  if (done) {
   435→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   436→    return (
   437→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   438→        <div className="text-center py-4">
   439→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   440→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
   441→          <p className="old-body text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   442→          <p className="old-quote text-xs text-[#8b7355] mt-2">
   443→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   444→          </p>
   445→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   446→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   447→          </button>
   448→        </div>
   449→      </div>
   450→    );
   451→  }
   452→
   453→  return (
   454→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   455→      <div className="flex items-center justify-between mb-4">
   456→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   457→          <span className="text-xl">📅</span> Guess the Year
   458→        </h4>
   459→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   460→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{qIdx + 1}/{questions.length}</span>
   461→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Score: {score}</span>
   462→        </div>
   463→      </div>
   464→
   465→      {/* Progress bar */}
   466→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   467→        <div
   468→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   469→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   470→        />
   471→      </div>
   472→
   473→      <div className="text-center mb-4">
   474→        <span className="text-5xl block mb-2">{q.emoji}</span>
   475→        <p className="old-body text-sm font-semibold text-[#5c4033]">{q.question}</p>
   476→      </div>
   477→
   478→      <div className="grid grid-cols-2 gap-2">
   479→        {q.options.map((opt, i) => {
   480→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 old-typewriter ';

   516→function WhackGame({ data }: { data: DecadeData }) {
   517→  const GRID = 9;
   518→  const allItems = useMemo(() => [
   519→    ...data.toys, ...data.snacks, ...data.cartoons, ...data.games,
   520→  ], [data]);
   521→
   522→  const [grid, setGrid] = useState<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   523→  const [score, setScore] = useState(0);
   524→  const [timeLeft, setTimeLeft] = useState(20);
   525→  const [gameActive, setGameActive] = useState(false);
   526→  const [gameOver, setGameOver] = useState(false);
   527→  const [feedback, setFeedback] = useState<string | null>(null);
   528→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   529→  const spawnRef = useRef<ReturnType<typeof setInterval> | null>(null);
   530→  const gridRef = useRef<(NostalgiaItem | null)[]>(Array(GRID).fill(null));
   531→  const activeRef = useRef(false);
   532→
   533→  const spawnItem = useCallback(() => {
   534→    if (!activeRef.current) return;
   535→    const empty = gridRef.current.reduce<number[]>((acc, v, i) => v === null ? [...acc, i] : acc, []);
   536→    if (empty.length === 0) return;
   537→    const pos = empty[Math.floor(Math.random() * empty.length)];
   538→    const item = allItems[Math.floor(Math.random() * allItems.length)];
   539→    const newGrid = [...gridRef.current];
   540→    newGrid[pos] = item;
   541→    gridRef.current = newGrid;
   542→    setGrid([...newGrid]);
   543→    setTimeout(() => {
   544→      if (gridRef.current[pos] === item) {
   545→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);
   546→      }
   547→    }, 1200);
   548→  }, [allItems]);

   584→      if (timerRef.current) clearInterval(timerRef.current);

   594→    setTimeout(() => setFeedback(null), 300);

   605→      <div className="flex items-center justify-between mb-4">
   606→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   607→          <span className="text-xl">🔨</span> Tap the Memory

   609→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   610→          <span className="sk-inset px-2 py-1 rounded old-typewriter flex items-center gap-1">
   611→            <Timer className="h-3 w-3" />{timeLeft}s
   612→          </span>
   613→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{score}pts</span>
   614→        </div>
   615→      </div>
   616→
   617→      {!gameActive && !gameOver && (
   618→        <div className="text-center py-6">
   619→          <p className="old-body text-sm text-[#8b7355] mb-4">Tap nostalgic items as they pop up! You have 20 seconds.</p>
   620→          <button onClick={startGame} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   621→            <Play className="h-5 w-5" /> Start Game
   622→          </button>
   623→        </div>

   627→        <div className="text-center py-4">
   628→          <span className="text-5xl mb-2 block">{score >= 80 ? '🏆' : score >= 40 ? '🎉' : '💪'}</span>
   629→          <p className="old-title text-lg font-bold text-[#5c4033]">Time&apos;s Up!</p>
   630→          <p className="old-body text-sm text-[#8b7355] mt-1">Score: {score} points</p>
