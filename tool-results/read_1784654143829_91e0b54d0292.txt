   142→// useDBState — IndexedDB-backed state
   143→// ═══════════════════════════════════════════════════════════════
   144→
   145→function useDBState<T>(
   146→  loader: () => Promise<T>,
   147→  saver: (val: T) => Promise<void>,
   148→  initial: T,
   149→): [T, (updater: T | ((prev: T) => T)) => void] {
   150→  const [value, setValue] = useState<T>(initial);
   151→
   152→  useEffect(() => {
   153→    loader().then((loaded) => setValue(loaded)).catch(() => {});
   154→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
   155→
   156→  const set = useCallback(
   157→    (updater: T | ((prev: T) => T)) => {
   158→      setValue((prev) => {
   159→        const next = typeof updater === 'function' ? (updater as (prev: T) => T)(prev) : updater;
   160→        saver(next).catch(() => {});
   161→        return next;
   162→      });
   163→    },
   164→    [saver], // eslint-disable-line react-hooks/exhaustive-deps
   165→  );
   166→
   167→  return [value, set];
   168→}

   180→  return count;
   181→}
