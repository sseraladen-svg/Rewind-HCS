   175→   175→interface MatchCard {
   176→   176→  id: number;
   177→   177→  emoji: string;

   180→   180→  isMatched: boolean;

   183→   183→function shuffle<T>(arr: T[]): T[] {
   184→   184→  const a = [...arr];
   185→   185→  for (let i = a.length - 1; i > 0; i--) {
   186→   186→    const j = Math.floor(Math.random() * (i + 1));
   187→   187→    [a[i], a[j]] = [a[j], a[i]];
   188→   188→  }
   189→   189→  return a;
   190→   190→}
   191→   191→
   192→   192→function createMatchCards(data: DecadeData): MatchCard[] {
   193→   193→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   194→   194→  const picked = shuffle(pool).slice(0, 6);
   195→   195→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   196→   196→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   197→   197→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   198→   198→  ]);
   199→   199→  return shuffle(initial);
   200→   200→}

   202→   202→function MemoryMatchGame({ data }: { data: DecadeData }) {
   203→   203→  const [cards, setCards] = useState<MatchCard[]>([]);
   204→   204→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   205→   205→  const [moves, setMoves] = useState(0);
   206→   206→  const [matches, setMatches] = useState(0);
   207→   207→  const [gameWon, setGameWon] = useState(false);
   208→   208→  const lockRef = useRef(false);
   209→   209→
   210→   210→  useEffect(() => {
   211→   211→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   212→   212→    return () => cancelAnimationFrame(id);
   213→   213→  }, [data]);
   214→   214→
   215→   215→  const resetGame = useCallback(() => {
   216→   216→    setCards(createMatchCards(data));
   217→   217→    setFlippedIds([]);
   218→   218→    setMoves(0);
   219→   219→    setMatches(0);
   220→   220→    setGameWon(false);
   221→   221→    lockRef.current = false;
   222→   222→  }, [data]);
   223→   223→
   224→   224→  const handleFlip = useCallback((id: number) => {
   225→   225→    if (lockRef.current) return;
   226→   226→    setCards(prev => {
   227→   227→      const card = prev.find(c => c.id === id);
   228→   228→      if (!card || card.isFlipped || card.isMatched) return prev;
   229→   229→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   230→   230→    });
   231→   231→
   232→   232→    setFlippedIds(prev => {
   233→   233→      if (prev.includes(id) || prev.length >= 2) return prev;
   234→   234→      const newFlipped = [...prev, id];
   235→   235→
   236→   236→      if (newFlipped.length === 2) {
   237→   237→        setMoves(m => m + 1);
   238→   238→        lockRef.current = true;
   239→   239→
   240→   240→        // Use a timeout to read the latest cards state
   241→   241→        setTimeout(() => {
   242→   242→          setCards(currentCards => {
   243→   243→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   244→   244→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   245→   245→
   246→   246→            if (first.label === second.label) {
   247→   247→              // Match!
   248→   248→              const updated = currentCards.map(c =>
   249→   249→                c.id === first.id || c.id === second.id
   250→   250→                  ? { ...c, isMatched: true, isFlipped: true }
   251→   251→                  : c
   252→   252→              );
   253→   253→              setMatches(m => {
   254→   254→                const newM = m + 1;
   255→   255→                if (newM === 6) setGameWon(true);
   256→   256→                return newM;
   257→   257→              });
   258→   258→              setFlippedIds([]);
   259→   259→              lockRef.current = false;
   260→   260→              return updated;
   261→   261→            } else {
   262→   262→              // No match — flip back
   263→   263→              setTimeout(() => {
   264→   264→                setCards(cc => cc.map(c =>
   265→   265→                  c.id === first.id || c.id === second.id
   266→   266→                    ? { ...c, isFlipped: false }
   267→   267→                    : c
   268→   268→                ));
   269→   269→                setFlippedIds([]);
   270→   270→                lockRef.current = false;
   271→   271→              }, 700);
   272→   272→              return currentCards;
   273→   273→            }
   274→   274→          });
   275→   275→        }, 50);
   276→   276→      }
   277→   277→
   278→   278→      return newFlipped;
   279→   279→    });
   280→   280→  }, []);
   281→   281→
   282→   282→  return (

   284→   284→      <div className="flex items-center justify-between mb-4">
   285→   285→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   286→   286→          <span className="text-xl">🃏</span> Memory Match
   287→   287→        </h4>
   288→   288→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   289→   289→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   290→   290→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   291→   291→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   292→   292→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   293→   293→          </button>
   294→   294→        </div>
   295→   295→      </div>
   296→   296→
   297→   297→      {gameWon && (
   298→   298→        <motion.div
   299→   299→          initial={{ scale: 0.8, opacity: 0 }}
   300→   300→          animate={{ scale: 1, opacity: 1 }}
   301→   301→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   302→   302→        >
   303→   303→          <span className="text-5xl mb-2">🎉</span>
   304→   304→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   305→   305→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   306→   306→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   307→   307→            Play Again
   308→   308→          </button>
   309→   309→        </motion.div>
   310→   310→      )}
   311→   311→
   312→   312→      <div className="grid grid-cols-4 gap-2">
   313→   313→        {cards.map((card) => {
   314→   314→          const showFront = card.isFlipped || card.isMatched;
   315→   315→          return (
   316→   316→            <button
   317→   317→              key={card.id}
   318→   318→              onClick={() => handleFlip(card.id)}
   319→   319→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   320→   320→              disabled={card.isMatched}
   321→   321→              style={{
   322→   322→                background: showFront
   323→   323→                  ? '#faf6eb'
   324→   324→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',

   327→   327→                  : '#5c4033',
   328→   328→                boxShadow: showFront
   329→   329→                  ? '0 2px 6px rgba(0,0,0,0.1), inset 0 1px 0 rgba(255,255,255,0.5)'
   330→   330→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',

   334→   334→              {showFront ? (

   353→   353→interface TriviaQ {
   354→   354→  question: string;
   355→   355→  emoji: string;
   356→   356→  correct: number;
   357→   357→  options: string[];
   358→   358→}
   359→   359→
   360→   360→function buildTrivia(data: DecadeData): TriviaQ[] {
   361→   361→  const decadeStart = parseInt(data.years.split('–')[0]);
   362→   362→  const qs: TriviaQ[] = [];
   363→   363→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   364→   364→  const shuffled = shuffle(all).slice(0, 8);
   365→   365→
   366→   366→  for (const item of shuffled) {
   367→   367→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   368→   368→    const wrongYears = [
   369→   369→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   370→   370→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   371→   371→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   372→   372→    ].filter(y => y !== realYear).slice(0, 3);
   373→   373→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   374→   374→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   375→   375→    qs.push({
   376→   376→      question: `When did "${item.title}" first become popular?`,
   377→   377→      emoji: item.emoji,
   378→   378→      correct: options.indexOf(realYear),
   379→   379→      options: options.map(String),
   380→   380→    });
   381→   381→  }
   382→   382→  return qs;
   383→   383→}

   385→   385→function GuessTheYearGame({ data }: { data: DecadeData }) {
   386→   386→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   387→   387→  const [qIdx, setQIdx] = useState(0);
   388→   388→  const [score, setScore] = useState(0);
   389→   389→  const [selected, setSelected] = useState<number | null>(null);
   390→   390→  const [showResult, setShowResult] = useState(false);
   391→   391→  const [done, setDone] = useState(false);
   392→   392→
   393→   393→  useEffect(() => {
   394→   394→    setQuestions(buildTrivia(data));
   395→   395→  }, [data]);
   396→   396→
   397→   397→  const q = questions[qIdx];
   398→   398→
   399→   399→  const handleAnswer = (optIdx: number) => {
   400→   400→    if (showResult || !q) return;
   401→   401→    setSelected(optIdx);
   402→   402→    setShowResult(true);
   403→   403→    if (optIdx === q.correct) setScore(s => s + 1);
   404→   404→    setTimeout(() => {
   405→   405→      if (qIdx + 1 < questions.length) {
   406→   406→        setQIdx(i => i + 1);
   407→   407→        setSelected(null);
   408→   408→        setShowResult(false);
   409→   409→      } else {
   410→   410→        setDone(true);
   411→   411→      }
   412→   412→    }, 1500);
   413→   413→  };
   414→   414→
   415→   415→  const resetGame = () => {
   416→   416→    const fresh = buildTrivia(data);
   417→   417→    setQuestions(fresh);
   418→   418→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   419→   419→  };
   420→   420→
   421→   421→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   422→   422→
   423→   423→  if (done) {
   424→   424→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   425→   425→    return (
   426→   426→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   427→   427→        <div className="text-center py-4">
   428→   428→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   429→   429→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
