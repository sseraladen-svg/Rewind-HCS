    23→    23→function formatTime(seconds: number): string {
    24→    24→  const m = Math.floor(seconds / 60);
    25→    25→  const s = Math.floor(seconds % 60);
    26→    26→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    27→    27→}
    28→    28→
    29→    29→function generateId(): string {
    30→    30→  return `vm_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
    31→    31→}

    33→    33→function formatDate(iso: string): string {
    34→    34→  const d = new Date(iso);
    35→    35→  const month = d.toLocaleString('en-US', { month: 'short' }).toUpperCase();
    36→    36→  const day = d.getDate();
    37→    37→  const hour = d.getHours();
    38→    38→  const ampm = hour >= 12 ? 'PM' : 'AM';
    39→    39→  const h12 = hour % 12 || 12;
    40→    40→  const min = String(d.getMinutes()).padStart(2, '0');
    41→    41→  return `${month} ${day}, ${h12}:${min} ${ampm}`;
    42→    42→}

    44→    44→/** Play a short beep using Web Audio API */
    45→    45→function playBeep(frequency = 1000, duration = 0.15, type: OscillatorType = 'square') {
    46→    46→  if (typeof window === 'undefined') return;
    47→    47→  try {
    48→    48→    const ctx = new AudioContext();
    49→    49→    const osc = ctx.createOscillator();
    50→    50→    const gain = ctx.createGain();
    51→    51→    osc.type = type;
    52→    52→    osc.frequency.value = frequency;
    53→    53→    gain.gain.value = 0.12;
    54→    54→    osc.connect(gain);
    55→    55→    gain.connect(ctx.destination);
    56→    56→    osc.start(ctx.currentTime);
    57→    57→    osc.stop(ctx.currentTime + duration);
    58→    58→    osc.onended = () => ctx.close();
    59→    59→  } catch {
    60→    60→    // silently ignore audio errors
    61→    61→  }
    62→    62→}

    67→    67→}
