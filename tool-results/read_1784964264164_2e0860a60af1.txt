    40→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    41→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    42→  const [bootLines, setBootLines] = useState<string[]>([]);
    43→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    44→
    45→  const allBootLines = useMemo(() => [
    46→    'BIOS v2.4 — Nostalgia Systems Inc.',
    47→    'POST: Memory test... 640K OK',
    48→    'Detecting floppy drives... A: OK  B: OK',
    49→    'Loading REWIND.EXE...',
    50→    'Mounting childhood memories... 47% complete',
    51→    'Indexing nostalgia database...',
    52→    'Rewinding time...',
    53→    'Loading good old days... 83% complete',
    54→    'Ready.',
    55→  ], []);
    56→
    57→  useEffect(() => {
    58→    const timers: ReturnType<typeof setTimeout>[] = [];
    59→    // Reveal boot lines one by one
    60→    allBootLines.forEach((line, i) => {
    61→      timers.push(setTimeout(() => {
    62→        setBootLines(prev => [...prev, line]);
    63→        if (i === allBootLines.length - 1) {
    64→          // Last line — show title after a beat
    65→          timers.push(setTimeout(() => setPhase(1), 1000));
    66→        }
    67→      }, 500 + i * 500));
    68→    });
    69→
    70→    // Auto-dismiss after title phase
    71→    fadeTimerRef.current = setTimeout(() => {
    72→      setPhase(2);
    73→      setTimeout(onDone, 800);
    74→    }, 7000);
    75→
    76→    return () => {
    77→      timers.forEach(clearTimeout);
    78→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    79→    };
    80→  }, [allBootLines, onDone]);
    81→
    82→  return (
    83→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    84→      <div className="crt-scanlines" />
    85→      <div className="vhs-noise" />

    90→        {/* Boot lines */}

    92→          {bootLines.map((line, i) => (
    93→            <div
    94→              key={i}
    95→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    96→              style={{ animationDelay: '0s' }}
    97→            >
    98→              {line}
    99→              {i === bootLines.length - 1 && (
   100→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
   101→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
   102→                />
   103→              )}
   104→            </div>
   105→          ))}
   106→        </div>
   107→
   108→        {/* Title */}
   109→        {phase >= 1 && (
   110→          <motion.div
   111→            initial={{ opacity: 0, scale: 0.9 }}
   112→            animate={{ opacity: 1, scale: 1 }}

   397→  return '';

   755→export default function Home() {
   756→  const [birthYear, setBirthYear] = useState<number | null>(null);
   757→  const [loading, setLoading] = useState(true);
   758→  const [showOnboarding, setShowOnboarding] = useState(false);
   759→  const [onboardingChecked, setOnboardingChecked] = useState(false);
   760→
   761→  // Run one-time localStorage → IndexedDB migration on first load
   762→  useEffect(() => {
   763→    if (typeof window !== 'undefined') {
   764→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   765→      // Check if onboarding was completed
   766→      (async () => {
   767→        try {
   768→          const db = await getDB();
   769→          const record: any = await db.get('era_skin', 'onboarding');
   770→          if (record?.complete) {
   771→            // Restore persisted birth year
   772→            if (record.birthYear) setBirthYear(record.birthYear);
   773→          } else {
   774→            setShowOnboarding(true);
   775→          }
   776→        } catch {
   777→          setShowOnboarding(true);
   778→        }
   779→        setOnboardingChecked(true);
   780→      })();
   781→    }
   782→  }, []);
   783→
   784→  const handleLoadingDone = useCallback(() => {
   785→    setLoading(false);
   786→  }, []);
   787→
   788→  const handleSubmit = useCallback((year: number) => {
   789→    triggerOldLoader('Tuning into your era...');
   790→    setTimeout(() => {
   791→      setBirthYear(year);
   792→      window.scrollTo({ top: 0, behavior: 'smooth' });
   793→      hideOldLoader();
   794→    }, 1100);
   795→  }, []);
   796→
   797→  const handleOnboardingComplete = useCallback((data: { birthYear: number; eraSkin: string }) => {
   798→    setShowOnboarding(false);
   799→    triggerOldLoader('Tuning into your era...');
   800→    setTimeout(() => {
   801→      setBirthYear(data.birthYear);
   802→      hideOldLoader();
   803→    }, 1100);
   804→  }, []);
   805→
   806→  const handleBack = useCallback(() => {
   807→    triggerOldLoader('Rewinding...');
   808→    setTimeout(() => {
   809→      setBirthYear(null);
   810→      hideOldLoader();
   811→    }, 850);
   812→  }, []);
   813→
   814→  return (
   815→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   816→      {/* CRT Boot Loading Screen */}
   817→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   818→
   819→      {/* Onboarding (first visit only) */}
   820→      <AnimatePresence>
   821→        {showOnboarding && onboardingChecked && (
   822→          <OnboardingFlow onComplete={handleOnboardingComplete} />
   823→        )}
   824→      </AnimatePresence>
   825→
   826→      <AnimatePresence mode="wait">
   827→        {!showOnboarding && birthYear === null && onboardingChecked ? (
   828→          <motion.div
   829→            key="hero"
   830→            initial={{ opacity: 0 }}
   831→            animate={{ opacity: 1 }}
   832→            exit={{ opacity: 0, y: -20 }}
   833→            transition={{ duration: 0.4 }}
   834→            style={{ opacity: loading ? 0 : 1 }}
   835→          >
   836→            <HeroSection onSubmit={handleSubmit} />
   837→          </motion.div>
   838→        ) : birthYear !== null ? (
   839→          <motion.div
   840→            key="dashboard"
   841→            initial={{ opacity: 0, y: 20 }}
   842→            animate={{ opacity: 1, y: 0 }}
   843→            exit={{ opacity: 0 }}
   844→            transition={{ duration: 0.5 }}
   845→            className="flex flex-col min-h-screen"
   846→          >
   847→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   848→          </motion.div>
   849→        ) : null}
   850→      </AnimatePresence>
   851→    </div>
   852→  );
   853→}
