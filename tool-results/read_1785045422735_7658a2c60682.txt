   107→   107→function loadImage(url: string): Promise<HTMLImageElement> {
   108→   108→  return new Promise((resolve, reject) => {
   109→   109→    const img = new Image();
   110→   110→    img.crossOrigin = 'anonymous';
   111→   111→    img.onload = () => resolve(img);
   112→   112→    img.onerror = reject;
   113→   113→    img.src = url;
   114→   114→  });
   115→   115→}

   117→   117→/** Draw a rounded rectangle path (no fill/stroke) */
   118→   118→function roundRectPath(
   119→   119→  ctx: CanvasRenderingContext2D,
   120→   120→  x: number, y: number, w: number, h: number,
   121→   121→  r: number,
   122→   122→) {
   123→   123→  ctx.beginPath();
   124→   124→  ctx.moveTo(x + r, y);
   125→   125→  ctx.lineTo(x + w - r, y);
   126→   126→  ctx.arcTo(x + w, y, x + w, y + r, r);
   127→   127→  ctx.lineTo(x + w, y + h - r);
   128→   128→  ctx.arcTo(x + w, y + h, x + w - r, y + h, r);
   129→   129→  ctx.lineTo(x + r, y + h);
   130→   130→  ctx.arcTo(x, y + h, x, y + h - r, r);
   131→   131→  ctx.lineTo(x, y + r);
   132→   132→  ctx.arcTo(x, y, x + r, y, r);
   133→   133→  ctx.closePath();
   134→   134→}

   231→   231→/**
   232→   232→ * Render a VHS-style collage to a canvas.
   233→   233→ * Photos are arranged in a grid, processed with the VHS filter,
   234→   234→ * overlaid with scanlines, glitch bands, timestamp, and CRT vignette.
   235→   235→ */
   236→   236→async function renderVHSCollage(
   237→   237→  canvas: HTMLCanvasElement,
   238→   238→  photos: PhotoItem[],
   239→   239→): Promise<HTMLCanvasElement> {
   240→   240→  const cols = photos.length <= 4 ? 2 : 3;
   241→   241→  const rows = Math.ceil(photos.length / cols);
   242→   242→  const cellW = 400;
   243→   243→  const cellH = 300;
   244→   244→  const padding = 12;
   245→   245→  const headerH = 60;
   246→   246→
   247→   247→  canvas.width = cols * cellW + (cols + 1) * padding;
   248→   248→  canvas.height = rows * cellH + (rows + 1) * padding + headerH;
   249→   249→  const ctx = canvas.getContext('2d')!;
   250→   250→
   251→   251→  // Dark CRT background
   252→   252→  ctx.fillStyle = '#1a1a1a';
   253→   253→  ctx.fillRect(0, 0, canvas.width, canvas.height);
   254→   254→
   255→   255→  // Header
   256→   256→  ctx.save();
   257→   257→  ctx.globalAlpha = 0.7;
   258→   258→  ctx.fillStyle = '#ff3333';
   259→   259→  ctx.beginPath();
   260→   260→  ctx.arc(28, headerH / 2, 6, 0, Math.PI * 2);
   261→   261→  ctx.fill();
   262→   262→  ctx.font = 'bold 22px monospace';
   263→   263→  ctx.fillStyle = '#fff';
   264→   264→  ctx.fillText('⏪ REWIND', 44, headerH / 2 + 8);
   265→   265→  ctx.font = '14px monospace';
   266→   266→  ctx.globalAlpha = 0.5;
   267→   267→  ctx.textAlign = 'right';
   268→   268→  ctx.fillText(`PLAY ▶  ${photos.length} MEMORIES`, canvas.width - 20, headerH / 2 + 6);
   269→   269→  ctx.restore();
   270→   270→
   271→   271→  // Draw each photo
   272→   272→  for (let i = 0; i < photos.length; i++) {
   273→   273→    const col = i % cols;
   274→   274→    const row = Math.floor(i / cols);
   275→   275→    const x = padding + col * (cellW + padding);
   276→   276→    const y = headerH + padding + row * (cellH + padding);
   277→   277→
   278→   278→    // Create temp canvas for processing
   279→   279→    const tmp = document.createElement('canvas');
   280→   280→    tmp.width = cellW;
   281→   281→    tmp.height = cellH;
   282→   282→    const tmpCtx = tmp.getContext('2d')!;
   283→   283→
   284→   284→    // Load and draw image
   285→   285→    const img = await loadImage(photos[i].url);
   286→   286→    const scale = Math.max(cellW / img.width, cellH / img.height);
   287→   287→    const sw = cellW / scale;
   288→   288→    const sh = cellH / scale;
   289→   289→    const sx = (img.width - sw) / 2;
   290→   290→    const sy = (img.height - sh) / 2;
   291→   291→    tmpCtx.drawImage(img, sx, sy, sw, sh, 0, 0, cellW, cellH);
   292→   292→
   293→   293→    // Apply VHS pipeline
   294→   294→    processFrame(tmpCtx, cellW, cellH, 'vhs');
   295→   295→
   296→   296→    // Draw to main canvas
   297→   297→    ctx.drawImage(tmp, x, y, cellW, cellH);
   298→   298→
   299→   299→    // Slight glow/border
   300→   300→    ctx.save();
   301→   301→    ctx.strokeStyle = 'rgba(255,255,255,0.08)';
   302→   302→    ctx.lineWidth = 1;
   303→   303→    ctx.strokeRect(x, y, cellW, cellH);
   304→   304→    ctx.restore();
   305→   305→  }
   306→   306→
   307→   307→  // Overlays on top
   308→   308→  drawScanlines(ctx, canvas.width, canvas.height, 0.06);
   309→   309→  drawTrackingGlitch(ctx, canvas.width, canvas.height);
   310→   310→  drawVHSTimestamp(ctx, canvas.width, canvas.height);
   311→   311→
   312→   312→  // CRT vignette
   313→   313→  ctx.save();
   314→   314→  const vg = ctx.createRadialGradient(
   315→   315→    canvas.width / 2, canvas.height / 2, canvas.width * 0.25,
   316→   316→    canvas.width / 2, canvas.height / 2, canvas.width * 0.65,
   317→   317→  );
   318→   318→  vg.addColorStop(0, 'rgba(0,0,0,0)');
   319→   319→  vg.addColorStop(1, 'rgba(0,0,0,0.45)');
   320→   320→  ctx.fillStyle = vg;
   321→   321→  ctx.fillRect(0, 0, canvas.width, canvas.height);
   322→   322→  ctx.restore();
   323→   323→
   324→   324→  drawWatermark(ctx, canvas.width, canvas.height);
   325→   325→
   326→   326→  return canvas;
   327→   327→}

   329→   329→/**
   330→   330→ * Render a Polaroid-style collage.
   331→   331→ * Photos are shown as Polaroids with white borders, slightly
   332→   332→ * rotated and stacked with drop shadows.

   334→   334→async function renderPolaroidCollage(
   335→   335→  canvas: HTMLCanvasElement,
   336→   336→  photos: PhotoItem[],
   337→   337→): Promise<HTMLCanvasElement> {
   338→   338→  const polaroidW = 320;
   339→   339→  const polaroidH = 380;
   340→   340→  const photoInsetTop = 28;
   341→   341→  const photoInsetSide = 24;
   342→   342→  const photoW = polaroidW - photoInsetSide * 2;
   343→   343→  const photoH = polaroidH - photoInsetTop - 70;
   344→   344→  const bottomLabelH = 50;
   345→   345→
   346→   346→  canvas.width = photos.length <= 3
   347→   347→    ? photos.length * polaroidW + (photos.length + 1) * 30
   348→   348→    : Math.ceil(photos.length / 2) * (polaroidW + 30) + 30;
   349→   349→  canvas.height = photos.length <= 3
   350→   350→    ? polaroidH + 100
   351→   351→    : 2 * (polaroidH + 30) + 30;
   352→   352→  const ctx = canvas.getContext('2d')!;
   353→   353→
   354→   354→  // Warm paper background
   355→   355→  ctx.fillStyle = '#f5edd6';
   356→   356→  ctx.fillRect(0, 0, canvas.width, canvas.height);
   357→   357→
   358→   358→  // Subtle paper texture
   359→   359→  ctx.save();
   360→   360→  ctx.globalAlpha = 0.03;
   361→   361→  for (let i = 0; i < 4000; i++) {
   362→   362→    ctx.fillStyle = Math.random() > 0.5 ? '#000' : '#8b7355';
   363→   363→    ctx.fillRect(

   367→   367→    );
   368→   368→  }
   369→   369→  ctx.restore();
   370→   370→
   371→   371→  const rotations = [-3, 2, -1.5, 2.5, -2, 1.5];
   372→   372→  const cols = photos.length <= 3 ? photos.length : Math.ceil(photos.length / 2);
   373→   373→
   374→   374→  for (let i = 0; i < photos.length; i++) {
   375→   375→    const col = i % cols;
   376→   376→    const row = Math.floor(i / cols);
   377→   377→    const cx = 30 + col * (polaroidW + 30) + polaroidW / 2;
   378→   378→    const cy = 30 + row * (polaroidH + 30) + polaroidH / 2;
   379→   379→    const rotation = (rotations[i % rotations.length] * Math.PI) / 180;
   380→   380→

   387→   387→    ctx.shadowColor = 'rgba(0,0,0,0.25)';
   388→   388→    ctx.shadowBlur = 18;
   389→   389→    ctx.shadowOffsetX = 4;
   390→   390→    ctx.shadowOffsetY = 6;
   391→   391→    ctx.fillStyle = '#fff';
   392→   392→    ctx.fillRect(-polaroidW / 2, -polaroidH / 2, polaroidW, polaroidH);
   393→   393→    ctx.restore();
   394→   394→
   395→   395→    // White border
   396→   396→    ctx.fillStyle = '#faf6eb';
   397→   397→    ctx.fillRect(-polaroidW / 2, -polaroidH / 2, polaroidW, polaroidH);

   399→   399→    // Draw photo

   405→   405→    const img = await loadImage(photos[i].url);
   406→   406→    const scale = Math.max(photoW / img.width, photoH / img.height);
   407→   407→    const sw = photoW / scale;
   408→   408→    const sh = photoH / scale;
   409→   409→    const sx = (img.width - sw) / 2;
   410→   410→    const sy = (img.height - sh) / 2;
   411→   411→    tmpCtx.drawImage(img, sx, sy, sw, sh, 0, 0, photoW, photoH);
   412→   412→
   413→   413→    // Apply polaroid filter
   414→   414→    processFrame(tmpCtx, photoW, photoH, 'polaroid');
   415→   415→
   416→   416→    ctx.drawImage(tmp, -polaroidW / 2 + photoInsetSide, -polaroidH / 2 + photoInsetTop, photoW, photoH);
   417→   417→
   418→   418→    // Subtle date label at bottom
   419→   419→    ctx.save();
   420→   420→    ctx.globalAlpha = 0.45;
   421→   421→    ctx.font = '12px "Special Elite", monospace';
   422→   422→    ctx.fillStyle = '#8b7355';
   423→   423→    ctx.textAlign = 'center';
   424→   424→    const dateStr = new Date(photos[i].created_at).toLocaleDateString('en-US', {
   425→   425→      month: 'short',
   426→   426→      day: 'numeric',
   427→   427→      year: 'numeric',
   428→   428→    });
   429→   429→    ctx.fillText(dateStr, 0, polaroidH / 2 - 18);
   430→   430→    ctx.restore();
   431→   431→
   432→   432→    ctx.restore();
   433→   433→  }
   434→   434→
   435→   435→  drawWatermark(ctx, canvas.width, canvas.height);
   436→   436→  return canvas;
   437→   437→}

   493→   493→    stripW - 28,
   494→   494→    filmBorder,
   495→   495→    canvas.height - filmBorder,
   496→   496→    20,
   497→   497→    sprocketH,
   498→   498→    sprocketGap,
   499→   499→    '#2a2018',
   500→   500→  );
   501→   501→
   502→   502→  // Draw each photo frame
   503→   503→  for (let i = 0; i < photos.length; i++) {
   504→   504→    const y = filmBorder + i * (photoH + frameGap);
   505→   505→
   506→   506→    // Frame number
   507→   507→    ctx.save();
   508→   508→    ctx.globalAlpha = 0.35;
   509→   509→    ctx.font = '9px monospace';
   510→   510→    ctx.fillStyle = '#d4a574';
   511→   511→    ctx.textAlign = 'right';
   512→   512→    ctx.fillText(`${String(i + 1).padStart(2, '0')}A`, stripW - 6, y + photoH - 4);
   513→   513→    ctx.restore();
   514→   514→
   515→   515→    // Photo
   516→   516→    const tmp = document.createElement('canvas');
   517→   517→    tmp.width = photoW;
   518→   518→    tmp.height = photoH;
   519→   519→    const tmpCtx = tmp.getContext('2d')!;
   520→   520→
   521→   521→    const img = await loadImage(photos[i].url);
   522→   522→    const scale = Math.max(photoW / img.width, photoH / img.height);
   523→   523→    const sw = photoW / scale;
   524→   524→    const sh = photoH / scale;
   525→   525→    const sx = (img.width - sw) / 2;
   526→   526→    const sy = (img.height - sh) / 2;
   527→   527→    tmpCtx.drawImage(img, sx, sy, sw, sh, 0, 0, photoW, photoH);
   528→   528→
   529→   529→    // Apply kodak filter
   530→   530→    processFrame(tmpCtx, photoW, photoH, 'kodak');
   531→   531→
   532→   532→    ctx.drawImage(tmp, filmBorder, y, photoW, photoH);
   533→   533→
   534→   534→    // Frame divider line
   535→   535→    if (i < photos.length - 1) {
   536→   536→      ctx.save();
   537→   537→      ctx.strokeStyle = 'rgba(200,180,140,0.15)';
   538→   538→      ctx.lineWidth = 1;
   539→   539→      ctx.beginPath();
   540→   540→      ctx.moveTo(filmBorder, y + photoH + frameGap / 2);
   541→   541→      ctx.lineTo(stripW - filmBorder, y + photoH + frameGap / 2);
   542→   542→      ctx.stroke();
   543→   543→      ctx.restore();
   544→   544→    }
   545→   545→  }
   546→   546→
   547→   547→  drawWatermark(ctx, stripW, canvas.height);
   548→   548→  return canvas;
   549→   549→}
