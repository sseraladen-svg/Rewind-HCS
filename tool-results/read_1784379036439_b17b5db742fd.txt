   427→export function getDecadeForYear(birthYear: number): string {
   428→  if (birthYear >= 1970 && birthYear <= 1979) return "1970s";
   429→  if (birthYear >= 1980 && birthYear <= 1989) return "1980s";
   430→  if (birthYear >= 1990 && birthYear <= 1999) return "1990s";
   431→  if (birthYear >= 2000 && birthYear <= 2009) return "2000s";
   432→  if (birthYear >= 2010 && birthYear <= 2019) return "2010s";
   433→  return "1990s"; // default
   434→}
   435→
   436→export function getAgeAtDecadeEnd(birthYear: number, targetDecade: string): number | null {
   437→  const decadeStart: Record<string, number> = {
   438→    "1970s": 1970, "1980s": 1980, "1990s": 1990, "2000s": 2000, "2010s": 2010,
   439→  };
   440→  const start = decadeStart[targetDecade];
   441→  if (start === undefined) return null;
   442→  // Was the person born before or during this decade?
   443→  if (birthYear > start + 9) return null; // born after decade ended
   444→  // Age at the END of the decade
   445→  return (start + 9) - birthYear;
   446→}

   448→export function getPersonalTimeline(birthYear: number): { decade: string; age: string; label: string }[] {
   449→  const timeline: { decade: string; age: string; label: string }[] = [];
   450→  const decades = ["1970s", "1980s", "1990s", "2000s", "2010s"];
   451→
   452→  for (const decade of decades) {
   453→    const ageAtEnd = getAgeAtDecadeEnd(birthYear, decade);
   454→    if (ageAtEnd === null || ageAtEnd < 0) continue;
   455→
   456→    // Only show decades where the person was a kid/teen (age 0-18 at decade end)
   457→    if (ageAtEnd > 18) continue;
   458→
   459→    // Calculate age at the start of the decade for the label
   460→    const decadeStart: Record<string, number> = {
   461→      "1970s": 1970, "1980s": 1980, "1990s": 1990, "2000s": 2000, "2010s": 2010,
   462→    };
   463→    const ageAtStart = Math.max(0, birthYear - decadeStart[decade]);
   464→    const ageRange = ageAtStart === ageAtEnd ? `${ageAtEnd}` : `${ageAtStart}–${ageAtEnd}`;
   465→
   466→    // Use the midpoint age for labeling the life stage
   467→    const midAge = Math.floor((ageAtStart + ageAtEnd) / 2);
   468→    let label = "";
   469→    if (ageAtEnd <= 1) label = "You were born! 🍼";
   470→    else if (midAge <= 3) label = "Your toddler years 👶";
   471→    else if (midAge <= 6) label = "Your early childhood 🧒";
   472→    else if (midAge <= 12) label = "Your school days 📚";
   473→    else if (midAge <= 15) label = "Your teenage years 🎸";
   474→    else label = "Your late teens 🎓";
   475→
   476→    timeline.push({
   477→      decade,
   478→      age: `Age ${ageRange}`,
   479→      label,
   480→    });
   481→  }
   482→  return timeline;
