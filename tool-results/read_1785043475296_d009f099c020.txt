   753→   753→/* ═══════════════════════════════════════════════════════════════
   754→   754→   Main OnboardingFlow Component
   755→   755→   ═══════════════════════════════════════════════════════════════ */
   756→   756→
   757→   757→export default function OnboardingFlow({ onComplete }: OnboardingFlowProps) {
   758→   758→  const [currentStep, setCurrentStep] = useState(0);
   759→   759→  const [direction, setDirection] = useState<number>(STEP_DIRECTION.forward);
   760→   760→  const [birthYear, setBirthYearState] = useState('');
   761→   761→  const [selectedEra, setSelectedEra] = useState<string | null>(null);
   762→   762→  const [currentFeature, setCurrentFeature] = useState(0);
   763→   763→  const [isLoaded, setIsLoaded] = useState(false);
   764→   764→  const [isCompleting, setIsCompleting] = useState(false);
   765→   765→
   766→   766→  /* ───── Check IndexedDB on mount ───── */
   767→   767→  useEffect(() => {
   768→   768→    let cancelled = false;
   769→   769→
   770→   770→    (async () => {
   771→   771→      try {
   772→   772→        const db = await getDB();
   773→   773→        if (cancelled) return;
   774→   774→
   775→   775→        const record: any = await db.get('era_skin', ONBOARDING_ID);
   776→   776→        if (cancelled) return;
   777→   777→
   778→   778→        if (record?.complete) {
   779→   779→          // Already onboarded — restore state and skip
   780→   780→          const savedBirthYear = record.birthYear ?? null;
   781→   781→          const savedEra = record.eraSkin ?? null;
   782→   782→
   783→   783→          if (savedBirthYear && savedEra) {
   784→   784→            onComplete({ birthYear: savedBirthYear, eraSkin: savedEra });
   785→   785→            return;
   786→   786→          }
   787→   787→        }
   788→   788→        setIsLoaded(true);
   789→   789→      } catch {
   790→   790→        // If IDB fails, still show onboarding
   791→   791→        setIsLoaded(true);
   792→   792→      }
   793→   793→    })();
   794→   794→
   795→   795→    return () => { cancelled = true; };
   796→   796→  }, [onComplete]);
   797→   797→
   798→   798→  /* ───── Persistence helpers ───── */
   799→   799→  const saveOnboardingData = useCallback(
   800→   800→    async (year: number, era: string) => {
   801→   801→      const db = await getDB();
   802→   802→      const record: OnboardingRecord = {
   803→   803→        id: ONBOARDING_ID,
   804→   804→        birthYear: year,
   805→   805→        eraSkin: era,
   806→   806→        complete: true,
   807→   807→      };
   808→   808→      await db.put('era_skin', record as any);
   809→   809→      // Also set the era skin in the standard location
   810→   810→      if (era !== 'mixed') {
   811→   811→        await setEraSkinId(era);
   812→   812→      }
   813→   813→    },
   814→   814→    [],
   815→   815→  );
   816→   816→
   817→   817→  /* ───── Navigation helpers ───── */
   818→   818→  const goNext = useCallback(() => {
   819→   819→    setDirection(STEP_DIRECTION.forward);
   820→   820→    setCurrentStep((s) => Math.min(s + 1, TOTAL_STEPS - 1));
   821→   821→  }, []);
   822→   822→
   823→   823→  const goBack = useCallback(() => {
   824→   824→    setDirection(STEP_DIRECTION.backward);
   825→   825→    setCurrentStep((s) => Math.max(s - 1, 0));
   826→   826→  }, []);
   827→   827→
   828→   828→  const setBirthYear = useCallback((val: string) => {
   829→   829→    setBirthYearState(val);
   830→   830→  }, []);
   831→   831→
   832→   832→  const canContinueBirthYear = useMemo(() => {
   833→   833→    const y = parseInt(birthYear);
   834→   834→    return y >= 1970 && y <= 2015;
   835→   835→  }, [birthYear]);
   836→   836→
   837→   837→  /* ───── Handle final completion ───── */
   838→   838→  const handleComplete = useCallback(() => {
   839→   839→    const year = parseInt(birthYear);
   840→   840→    const era = selectedEra || 'mixed';
   841→   841→    setIsCompleting(true);
   842→   842→
   843→   843→    // Small delay for the exit animation to feel natural
   844→   844→    setTimeout(async () => {
   845→   845→      try {
   846→   846→        await saveOnboardingData(year, era);
   847→   847→      } catch (err) {
   848→   848→        console.warn('[Onboarding] Failed to persist:', err);
   849→   849→      }
   850→   850→      onComplete({ birthYear: year, eraSkin: era });
   851→   851→    }, 600);
   852→   852→  }, [birthYear, selectedEra, saveOnboardingData, onComplete]);
   853→   853→
   854→   854→  /* ───── Loading state while checking IDB ───── */
   855→   855→  if (!isLoaded) {
   856→   856→    return (
   857→   857→      <div className="min-h-screen flex items-center justify-center bg-[#f5edd6]">
   858→   858→        <motion.div
   859→   859→          initial={{ opacity: 0 }}
   860→   860→          animate={{ opacity: 1 }}
   861→   861→          className="text-center"
   862→   862→        >
   863→   863→          <div className="text-4xl mb-3">📼</div>
   864→   864→          <p className="old-typewriter text-sm text-[#8b7355]">Loading...</p>

   867→   867→    );
   868→   868→  }

   871→   871→  const renderStep = () => {

   875→   875→      case 1:
   876→   876→        return (
   877→   877→          <BirthYearStep
   878→   878→            birthYear={birthYear}
   879→   879→            setBirthYear={setBirthYear}
   880→   880→            onNext={goNext}
   881→   881→            onBack={goBack}
   882→   882→            canContinue={canContinueBirthYear}
   883→   883→          />
   884→   884→        );
   885→   885→      case 2:
   886→   886→        return (
   887→   887→          <EraSkinStep
   888→   888→            selectedEra={selectedEra}
   889→   889→            setSelectedEra={setSelectedEra}
   890→   890→            onNext={goNext}
   891→   891→            onBack={goBack}
   892→   892→          />
   893→   893→        );
   894→   894→      case 3:
   895→   895→        return (
   896→   896→          <FeatureTourStep
   897→   897→            currentFeature={currentFeature}
   898→   898→            setCurrentFeature={setCurrentFeature}
   899→   899→            onNext={handleComplete}
   900→   900→            onBack={goBack}
   901→   901→          />
   902→   902→        );
   903→   903→      default:
   904→   904→        return null;
   905→   905→    }
   906→   906→  };
   907→   907→
   908→   908→  return (
   909→   909→    <AnimatePresence mode="wait">
   910→   910→      {!isCompleting ? (
   911→   911→        <motion.div
   912→   912→          key="onboarding-container"
   913→   913→          initial={{ opacity: 0 }}
   914→   914→          animate={{ opacity: 1 }}
   915→   915→          exit={{ opacity: 0, scale: 0.95, y: -20 }}
   916→   916→          transition={{ duration: 0.6, ease: 'easeInOut' }}
   917→   917→          className="min-h-screen flex flex-col"
   918→   918→          style={{
   919→   919→            background:
   920→   920→              'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)',
   921→   921→          }}
