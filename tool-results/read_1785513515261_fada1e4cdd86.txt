    25→function formatTime(seconds: number): string {
    26→  const m = Math.floor(seconds / 60);
    27→  const s = Math.floor(seconds % 60);
    28→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    29→}
    30→
    31→function generateId(): string {
    32→  return `vm_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
    33→}
    34→
    35→function formatDate(iso: string): string {
    36→  const d = new Date(iso);
    37→  const month = d.toLocaleString('en-US', { month: 'short' }).toUpperCase();
    38→  const day = d.getDate();
    39→  const hour = d.getHours();
    40→  const ampm = hour >= 12 ? 'PM' : 'AM';
    41→  const h12 = hour % 12 || 12;
    42→  const min = String(d.getMinutes()).padStart(2, '0');
    43→  return `${month} ${day}, ${h12}:${min} ${ampm}`;
    44→}
    45→
    46→/** Play a short beep using Web Audio API */
    47→function playBeep(frequency = 1000, duration = 0.15, type: OscillatorType = 'square') {
    48→  if (typeof window === 'undefined') return;
    49→  try {
    50→    const ctx = new AudioContext();
    51→    const osc = ctx.createOscillator();
    52→    const gain = ctx.createGain();
    53→    osc.type = type;
    54→    osc.frequency.value = frequency;
    55→    gain.gain.value = 0.12;
    56→    osc.connect(gain);
    57→    gain.connect(ctx.destination);
    58→    osc.start(ctx.currentTime);
    59→    osc.stop(ctx.currentTime + duration);
    60→    osc.onended = () => ctx.close();
    61→  } catch {
    62→    // silently ignore audio errors
    63→  }
    64→}
    65→
    66→/** Generate fake waveform bar heights for visual display */
    67→function generateWaveformBars(count = 32): number[] {
    68→  return Array.from({ length: count }, () => Math.random() * 0.7 + 0.15);
    69→}

    81→  const [errorMsg, setErrorMsg] = useState<string | null>(null);
    82→

    86→  const audioElRef = useRef<HTMLAudioElement | null>(null);

    88→  const streamRef = useRef<MediaStream | null>(null);
    89→
    90→  const MAX_RECORDING_SECONDS = 60;
    91→  const MAX_SAVED_MESSAGES = 5;
    92→  const objectUrlRef = useRef<string | null>(null);
    93→
    94→  // ── Load messages from IndexedDB ──
    95→  useEffect(() => {
    96→    if (typeof window === 'undefined') return;
    97→    getAllVoicemails()
    98→      .then((records) => {
    99→        // records come sorted by timestamp ascending; reverse for newest-first
   100→        setMessages(
   101→          records.reverse().map((r) => ({
   102→            id: r.id,
   103→            audioBlob: r.audio_blob,
   104→            duration: r.duration,
   105→            timestamp: r.timestamp,
   106→          }))
   107→        );
   108→      })
   109→      .catch(() => {
   110→        // IDB unavailable, start fresh
   111→      });
   112→  }, []);
   113→
   114→  // ── Check browser support ──
   115→  useEffect(() => {
   116→    if (typeof window === 'undefined') return;
   117→    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
   118→      setStatus('error-unsupported');
   119→    }
   120→  }, []);
   121→
   122→  // ── Cleanup on unmount ──
   123→  useEffect(() => {
   124→    return () => {
   125→      if (recordingTimerRef.current) clearInterval(recordingTimerRef.current);
   126→      if (streamRef.current) {
   127→        streamRef.current.getTracks().forEach((t) => t.stop());
   128→      }
   129→      if (audioElRef.current) {
   130→        audioElRef.current.pause();
   131→        audioElRef.current = null;
   132→      }
   133→      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);
   134→      if (objectUrlRef.current) {
   135→        URL.revokeObjectURL(objectUrlRef.current);
   136→        objectUrlRef.current = null;
   137→      }
   138→    };
   139→  }, []);
   140→
   141→  // ── Start Recording ──
   142→  const startRecording = useCallback(async () => {
   143→    try {
   144→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   145→      streamRef.current = stream;
   146→
   147→      const recorder = new MediaRecorder(stream);
   148→      mediaRecorderRef.current = recorder;
   149→      audioChunksRef.current = [];
   150→
   151→      recorder.ondataavailable = (e) => {
   152→        if (e.data.size > 0) audioChunksRef.current.push(e.data);
   153→      };
   154→
   155→      recorder.onstop = () => {
   156→        const blob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
   157→        const id = generateId();
   158→        const duration = recordingTime || 1;
   159→        const timestamp = new Date().toISOString();
   160→
   161→        addVoicemail({ id, audio_blob: blob, duration, timestamp })
   162→          .then(() => {
   163→            const newMsg: VoicemailMessage = { id, audioBlob: blob, duration, timestamp };
   164→            setMessages((prev) => {
   165→              const updated = [newMsg, ...prev];
   166→              if (updated.length > MAX_SAVED_MESSAGES) {
   167→                // Delete overflow messages (oldest at the end)
   168→                const overflow = updated.slice(MAX_SAVED_MESSAGES);
   169→                overflow.forEach((m) => deleteVoicemail(m.id).catch(() => {}));
   170→                return updated.slice(0, MAX_SAVED_MESSAGES);
   171→              }
   172→              return updated;
   173→            });
   174→          })
   175→          .catch(() => {});
   176→
   177→        // Clean up stream
   178→        if (streamRef.current) {
   179→          streamRef.current.getTracks().forEach((t) => t.stop());
   180→          streamRef.current = null;
   181→        }
   182→      };
   183→
   184→      recorder.start();
   185→      soundRecordStart();
   186→      playBeep(800, 0.1, 'square');
   187→      setStatus('recording');
   188→      setRecordingTime(0);
   189→
   190→      recordingTimerRef.current = setInterval(() => {
   191→        setRecordingTime((prev) => {
   192→          const next = prev + 1;
   193→          if (next >= MAX_RECORDING_SECONDS) {
   194→            stopRecording();
   195→            return prev;
   196→          }
   197→          return next;
   198→        });
   199→      }, 1000);
   200→    } catch (err: unknown) {
   201→      const name = (err as DOMException)?.name;
   202→      if (name === 'NotAllowedError' || name === 'PermissionDeniedError') {
   203→        setStatus('error-denied');
   204→        setErrorMsg('Microphone access denied. Please allow microphone access.');
   205→      } else {
   206→        setStatus('error-unsupported');
   207→        setErrorMsg('Audio recording not supported in this browser... try Chrome or Firefox!');
   208→      }
   209→    }
   210→  }, [recordingTime]);
   211→
   212→  // ── Stop Recording ──
   213→  const stopRecording = useCallback(() => {
   214→    if (recordingTimerRef.current) {
   215→      clearInterval(recordingTimerRef.current);
   216→      recordingTimerRef.current = null;
   217→    }
   218→    if (mediaRecorderRef.current && mediaRecorderRef.current.state !== 'inactive') {
   219→      mediaRecorderRef.current.stop();
   220→    }

   243→      setPlayingId(null);
   244→      setPlayProgress(0);
   245→      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);

   248→      if (audioElRef.current) {
   249→        audioElRef.current.pause();
   250→        audioElRef.current = null;
   251→      }
   252→      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);

   254→        URL.revokeObjectURL(objectUrlRef.current);
   255→      }

   259→      const audio = new Audio(url);
   260→      audioElRef.current = audio;
   261→

   264→          setPlayProgress(audio.currentTime / audio.duration);
   265→        }
   266→        animFrameRef.current = requestAnimationFrame(updateProgress);
   267→      };

   269→      audio.onplay = () => {
   270→        setStatus('playing');
   271→        setPlayingId(msg.id);
   272→        animFrameRef.current = requestAnimationFrame(updateProgress);
   273→      };
   274→
   275→      audio.onended = () => {
   276→        setPlayingId(null);
   277→        setPlayProgress(0);
   278→        setStatus('ready');
   279→        if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);
   280→        if (objectUrlRef.current) {
   281→          URL.revokeObjectURL(objectUrlRef.current);
   282→          objectUrlRef.current = null;
   283→        }
   284→      };
   285→
   286→      audio.onerror = () => {
   287→        setPlayingId(null);
   288→        setPlayProgress(0);
   289→        setStatus('ready');
   290→        if (objectUrlRef.current) {
   291→          URL.revokeObjectURL(objectUrlRef.current);
   292→          objectUrlRef.current = null;
   293→        }
   294→      };
   295→
   296→      audio.play().catch(() => {
   297→        setPlayingId(null);
   298→        setStatus('ready');
   299→        if (objectUrlRef.current) {
   300→          URL.revokeObjectURL(objectUrlRef.current);
   301→          objectUrlRef.current = null;
   302→        }
   303→      });
   304→    }
   305→  }, [playingId]);
   306→
   307→  // ── Delete a message ──

   313→        audioElRef.current = null;
   314→        setPlayingId(null);
   315→        setStatus('ready');
   316→      }
   317→      setMessages((prev) => prev.filter((m) => m.id !== id));
   318→      deleteVoicemail(id).catch(() => {});
   319→    },
   320→    [playingId]
   321→  );
   322→

   324→

   330→      {/* Section Header */}
   331→      <motion.div
   332→        initial={{ opacity: 0, y: 12 }}
   333→        whileInView={{ opacity: 1, y: 0 }}
   334→        viewport={{ once: true }}
   335→        transition={{ duration: 0.6 }}
   336→        className="text-center mb-8"
   337→      >
   338→        <div className="inline-flex items-center justify-center w-14 h-14 rounded-full sk-leather mb-4">
   339→          <span className="text-2xl">🎙️</span>
   340→        </div>

   342→          ✦ VOICE MESSAGES ✦

   344→        <p className="old-body text-muted-foreground text-sm md:text-base max-w-md mx-auto">
   345→          Leave a message after the beep... just like the old answering machine.

   349→      {/* Error States */}
   350→      {(status === 'error-unsupported' || status === 'error-denied') && (
   351→        <div className="sk-paper rounded-xl p-6 text-center mb-6">
   352→          <p className="old-typewriter text-sm text-red-800">
   353→            {errorMsg || 'Audio recording not supported in this browser... try Chrome or Firefox!'}
   354→          </p>

   357→
   358→      {/* ── Answering Machine Body ── */}
   359→      <div className="sk-leather rounded-2xl p-5 md:p-8 max-w-lg mx-auto shadow-[inset_0_2px_4px_rgba(0,0,0,0.3),0_8px_24px_rgba(0,0,0,0.3)] relative overflow-hidden">
   360→        <div className="absolute inset-0 sk-noise pointer-events-none opacity-30" />
   361→
   362→        {/* Brand label */}
   363→        <div className="text-center mb-4 relative">
   364→          <p className="old-label text-[9px] tracking-[0.3em] text-[#c9a96e] uppercase">
   365→            REWIND™ ANSWERING MACHINE

   539→          >

   541→              <div
   542→                className="absolute inset-0 rounded-full bg-red-500/30"

   560→              MAX {MAX_RECORDING_SECONDS}s
   561→            </p>
   562→          </div>
   563→        </div>

   565→        {/* ── New Messages Indicator ── */}
   566→        <AnimatePresence>

   569→              initial={{ opacity: 0, y: -8 }}

   583→      <div className="max-w-lg mx-auto mt-6 space-y-3">

   603→                  <div>

   605→                      ✉ VOICEMAIL #{messages.indexOf(msg) + 1}
   606→                    </p>

   620→                    const isCurrent = isThisPlaying && i === progressIdx;

   622→                      <div
   623→                        key={i}
   624→                        className="flex-1 rounded-sm transition-colors duration-150 min-w-[2px]"

   639→                {/* Controls */}
   640→                <div className="flex items-center gap-3">

   642→                  <button

   647→                    {isThisPlaying ? (

   649→                        <span className="inline-block w-2.5 h-3 bg-[#f5edd6] rounded-[1px]" />

   651→                      </>

   654→                        <span className="inline-block w-0 h-0 border-l-[6px] border-l-[#f5edd6] border-t-[4px] border-t-transparent border-b-[4px] border-b-transparent" />
   655→                        PLAY
   656→                      </>
   657→                    )}
   658→                  </button>
   659→
   660→                  {/* Progress bar */}
   661→                  <div className="flex-1 h-1.5 bg-[#e0d5b8] rounded-full overflow-hidden">
   662→                    <div
   663→                      className="h-full bg-[#8b5e3c] rounded-full transition-all duration-100"
   664→                      style={{ width: isThisPlaying ? `${playProgress * 100}%` : '0%' }}
   665→                    />
   666→                  </div>
   667→
   668→                  {/* Delete */}
   669→                  <button
   670→                    onClick={() => { soundButtonClick(); deleteMessage(msg.id); }}
   671→                    className="text-[#8b7355] hover:text-red-700 transition-colors opacity-60 group-hover:opacity-100 focus:outline-none focus-visible:ring-2 focus-visible:ring-red-400 rounded p-1"
   672→                    aria-label="Delete message"
   673→                  >
   674→                    <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   675→                      <polyline points="3 6 5 6 21 6" />
   676→                      <path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2" />
   677→                    </svg>
   678→                  </button>
   679→                </div>
   680→              </motion.div>
   681→            );
   682→          })}
   683→        </AnimatePresence>
   684→
   685→        {messages.length === 0 && status !== 'recording' && (
   686→          <p className="text-center old-body text-sm text-muted-foreground py-6 opacity-60">
   687→            No messages yet... press record to leave one! 🎙️
   688→          </p>
   689→        )}
   690→      </div>
   691→
   692→      {/* ── Inline Keyframes ── */}
   693→      <style jsx global>{`
   694→        @keyframes blink {
   695→          0%, 100% { opacity: 1; }
   696→          50% { opacity: 0.15; }
   697→        }
   698→        @keyframes pulse-ring {
   699→          0% { transform: scale(1); opacity: 0.4; }
   700→          50% { transform: scale(1.15); opacity: 0.1; }
   701→          100% { transform: scale(1); opacity: 0.4; }
   702→        }
   703→      `}</style>
   704→    </section>
