   144→): [T, (updater: T | ((prev: T) => T)) => void] {
   145→  const [value, setValue] = useState<T>(initial);
   146→
   147→  useEffect(() => {
   148→    loader().then((loaded) => setValue(loaded)).catch(() => {});
   149→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
   150→
   151→  const set = useCallback(
   152→    (updater: T | ((prev: T) => T)) => {
   153→      setValue((prev) => {
   154→        const next = typeof updater === 'function' ? (updater as (prev: T) => T)(prev) : updater;
   155→        saver(next).catch(() => {});
   156→        return next;
   157→      });
   158→    },
   159→    [saver], // eslint-disable-line react-hooks/exhaustive-deps
   160→  );
   161→
   162→  return [value, set];
   163→}
