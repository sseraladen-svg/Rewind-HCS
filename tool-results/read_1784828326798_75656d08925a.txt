    24→function formatTime(seconds: number): string {
    25→  const m = Math.floor(seconds / 60);
    26→  const s = Math.floor(seconds % 60);
    27→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    28→}
    29→
    30→function generateId(): string {
    31→  return `vm_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
    32→}
    33→
    34→function formatDate(iso: string): string {
    35→  const d = new Date(iso);
    36→  const month = d.toLocaleString('en-US', { month: 'short' }).toUpperCase();
    37→  const day = d.getDate();
    38→  const hour = d.getHours();
    39→  const ampm = hour >= 12 ? 'PM' : 'AM';
    40→  const h12 = hour % 12 || 12;
    41→  const min = String(d.getMinutes()).padStart(2, '0');
    42→  return `${month} ${day}, ${h12}:${min} ${ampm}`;
    43→}
    44→
    45→/** Play a short beep using Web Audio API */
    46→function playBeep(frequency = 1000, duration = 0.15, type: OscillatorType = 'square') {
    47→  if (typeof window === 'undefined') return;
    48→  try {
    49→    const ctx = new AudioContext();
    50→    const osc = ctx.createOscillator();
    51→    const gain = ctx.createGain();
    52→    osc.type = type;
    53→    osc.frequency.value = frequency;
    54→    gain.gain.value = 0.12;
    55→    osc.connect(gain);
    56→    gain.connect(ctx.destination);
    57→    osc.start(ctx.currentTime);
    58→    osc.stop(ctx.currentTime + duration);
    59→    osc.onended = () => ctx.close();
    60→  } catch {
    61→    // silently ignore audio errors
    62→  }
    63→}
    64→
    65→/** Generate fake waveform bar heights for visual display */
    66→function generateWaveformBars(count = 32): number[] {
    67→  return Array.from({ length: count }, () => Math.random() * 0.7 + 0.15);
    68→}

   150→      recorder.ondataavailable = (e) => {

   152→      };

   154→      recorder.onstop = () => {
   155→        const blob = new Blob(audioChunksRef.current, { type: 'audio/webm' });

   160→        addVoicemail({ id, audio_blob: blob, duration, timestamp })

   166→                // Delete overflow messages (oldest at the end)
   167→                const overflow = updated.slice(MAX_SAVED_MESSAGES);
   168→                overflow.forEach((m) => deleteVoicemail(m.id).catch(() => {}));
   169→                return updated.slice(0, MAX_SAVED_MESSAGES);
   170→              }
   171→              return updated;
   172→            });
   173→          })
   174→          .catch(() => {});
   175→
   176→        // Clean up stream

   178→          streamRef.current.getTracks().forEach((t) => t.stop());

   287→          URL.revokeObjectURL(objectUrlRef.current);
   288→          objectUrlRef.current = null;
   289→        }
   290→      };
   291→
   292→      audio.play().catch(() => {
   293→        setPlayingId(null);
   294→        setStatus('ready');
   295→        if (objectUrlRef.current) {
   296→          URL.revokeObjectURL(objectUrlRef.current);
   297→          objectUrlRef.current = null;
   298→        }
   299→      });
   300→    }

   333→        <div className="inline-flex items-center justify-center w-14 h-14 rounded-full sk-leather mb-4">
   334→          <span className="text-2xl">🎙️</span>

   354→      <div className="sk-leather rounded-2xl p-5 md:p-8 max-w-lg mx-auto shadow-[inset_0_2px_4px_rgba(0,0,0,0.3),0_8px_24px_rgba(0,0,0,0.3)] relative overflow-hidden">
   355→        <div className="absolute inset-0 sk-noise pointer-events-none opacity-30" />

   539→                  animation: 'pulse-ring 1.5s ease-in-out infinite',
   540→                }}
   541→              />
   542→            )}
   543→            <div className="flex items-center justify-center w-full h-full">
   544→              {isRecording ? (
   545→                <div className="w-5 h-5 rounded-sm bg-[#1a0a0a]" />

   547→                <div className="w-6 h-6 rounded-full bg-[#1a0a0a] border border-[#5a1010]" />

   552→          <div className="text-center">
   553→            <p className="old-label text-[9px] text-[#c9a96e] tracking-wider">RECORD</p>
   554→            <p className="old-typewriter text-[7px] text-[#8b7355] mt-0.5">
   555→              MAX {MAX_RECORDING_SECONDS}s
   556→            </p>
   557→          </div>
   558→        </div>
   559→
   560→        {/* ── New Messages Indicator ── */}
   561→        <AnimatePresence>
   562→          {messages.length > 0 && (
   563→            <motion.div
   564→              initial={{ opacity: 0, y: -8 }}
   565→              animate={{ opacity: 1, y: 0 }}
   566→              exit={{ opacity: 0, y: -8 }}
   567→              className="text-center mb-4"
   568→            >
   569→              <p className="old-typewriter text-xs text-amber-600" style={{ textShadow: '0 0 6px rgba(217,119,6,0.2)' }}>

   575→      </div>

   577→      {/* ── Message List ── */}
   578→      <div className="max-w-lg mx-auto mt-6 space-y-3">
   579→        <AnimatePresence>
   580→          {messages.map((msg) => {

   582→            const bars = generateWaveformBars(32);
   583→            const progressIdx = isThisPlaying ? Math.floor(playProgress * bars.length) : 0;
   584→
   585→            return (

   587→                key={msg.id}
   588→                initial={{ opacity: 0, y: 16, scale: 0.96 }}

   590→                exit={{ opacity: 0, x: -40, scale: 0.9 }}

   597→                <div className="flex items-center justify-between mb-3">
   598→                  <div>
   599→                    <p className="old-label text-[9px] text-[#8b7355] tracking-wider">
   600→                      ✉ VOICEMAIL #{messages.indexOf(msg) + 1}
   601→                    </p>

   603→                      {formatDate(msg.timestamp)}
   604→                    </p>
   605→                  </div>

   613→                  {bars.map((height, i) => {
   614→                    const isPast = isThisPlaying && i < progressIdx;
   615→                    const isCurrent = isThisPlaying && i === progressIdx;
   616→                    return (
   617→                      <div
   618→                        key={i}
   619→                        className="flex-1 rounded-sm transition-colors duration-150 min-w-[2px]"
   620→                        style={{
   621→                          height: `${height * 100}%`,
   622→                          backgroundColor: isCurrent
   623→                            ? '#b91c1c'
   624→                            : isPast
   625→                            ? '#8b5e3c'

   627→                          opacity: isPast || isCurrent ? 1 : 0.5,
   628→                        }}
   629→                      />
   630→                    );
   631→                  })}
   632→                </div>
   633→
   634→                {/* Controls */}
   635→                <div className="flex items-center gap-3">
   636→                  {/* Play / Pause */}
   637→                  <button

   639→                    className="flex items-center gap-1.5 px-3 py-1.5 rounded bg-[#3d2b1f] text-[#f5edd6] old-typewriter text-[10px] tracking-wider hover:bg-[#5a4030] transition-colors focus:outline-none focus-visible:ring-2 focus-visible:ring-amber-400"
   640→                    aria-label={isThisPlaying ? 'Pause' : 'Play'}
   641→                  >
   642→                    {isThisPlaying ? (
   643→                      <>
   644→                        <span className="inline-block w-2.5 h-3 bg-[#f5edd6] rounded-[1px]" />
   645→                        PAUSE
   646→                      </>
   647→                    ) : (
   648→                      <>
   649→                        <span className="inline-block w-0 h-0 border-l-[6px] border-l-[#f5edd6] border-t-[4px] border-t-transparent border-b-[4px] border-b-transparent" />
   650→                        PLAY
   651→                      </>
   652→                    )}
   653→                  </button>
   654→
   655→                  {/* Progress bar */}
   656→                  <div className="flex-1 h-1.5 bg-[#e0d5b8] rounded-full overflow-hidden">
   657→                    <div
   658→                      className="h-full bg-[#8b5e3c] rounded-full transition-all duration-100"
   659→                      style={{ width: isThisPlaying ? `${playProgress * 100}%` : '0%' }}
   660→                    />
   661→                  </div>
   662→
   663→                  {/* Delete */}
   664→                  <button
   665→                    onClick={() => deleteMessage(msg.id)}
   666→                    className="text-[#8b7355] hover:text-red-700 transition-colors opacity-60 group-hover:opacity-100 focus:outline-none focus-visible:ring-2 focus-visible:ring-red-400 rounded p-1"
   667→                    aria-label="Delete message"
   668→                  >
   669→                    <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   670→                      <polyline points="3 6 5 6 21 6" />
   671→                      <path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2" />
   672→                    </svg>
   673→                  </button>
   674→                </div>
   675→              </motion.div>
   676→            );
   677→          })}
   678→        </AnimatePresence>
   679→
   680→        {messages.length === 0 && status !== 'recording' && (
   681→          <p className="text-center old-body text-sm text-muted-foreground py-6 opacity-60">
   682→            No messages yet... press record to leave one! 🎙️
   683→          </p>
   684→        )}
   685→      </div>
   686→
   687→      {/* ── Inline Keyframes ── */}
   688→      <style jsx global>{`
   689→        @keyframes blink {
   690→          0%, 100% { opacity: 1; }
   691→          50% { opacity: 0.15; }
   692→        }
   693→        @keyframes pulse-ring {
   694→          0% { transform: scale(1); opacity: 0.4; }
   695→          50% { transform: scale(1.15); opacity: 0.1; }
   696→          100% { transform: scale(1); opacity: 0.4; }
   697→        }
   698→      `}</style>
   699→    </section>
   700→  );
