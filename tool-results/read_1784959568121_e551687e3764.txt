   199→   199→/* ── fetch from TMDB ──────────────────────── */
   200→   200→async function fetchFromTmdb(
   201→   201→  birthYear: number,
   202→   202→  maxYear: number,
   203→   203→  category: string,
   204→   204→  page: number
   205→   205→): Promise<{ items: MovieItem[]; totalPages: number } | null> {
   206→   206→  const apiKey = process.env.TMDB_API_KEY;
   207→   207→  if (!apiKey) return null;
   208→   208→
   209→   209→  const items: MovieItem[] = [];
   210→   210→  let totalPages = 1;
   211→   211→
   212→   212→  try {
   213→   213→    if (category === 'cartoon') {
   214→   214→      // Fetch both movies and TV with animation genre
   215→   215→      const [movieRes, tvRes] = await Promise.all([
   216→   216→        fetch(`${TMDB_BASE}/discover/movie?api_key=${apiKey}&with_genres=16&primary_release_date.gte=${birthYear}-01-01&primary_release_date.lte=${maxYear}-12-31&sort_by=popularity.desc&page=${page}`),
   217→   217→        fetch(`${TMDB_BASE}/discover/tv?api_key=${apiKey}&with_genres=16&first_air_date.gte=${birthYear}-01-01&first_air_date.lte=${maxYear}-12-31&sort_by=popularity.desc&page=${page}`),
   218→   218→      ]);
   219→   219→
   220→   220→      const [movieData, tvData] = await Promise.all([movieRes.json(), tvRes.json()]);
   221→   221→      totalPages = Math.max(movieData.total_pages || 1, tvData.total_pages || 1);
   222→   222→
   223→   223→      for (const m of (movieData.results || []).slice(0, 10)) {
   224→   224→        items.push(formatTmdbItem(m, 'movie'));
   225→   225→      }
   226→   226→      for (const t of (tvData.results || []).slice(0, 10)) {
   227→   227→        items.push(formatTmdbItem(t, 'tv'));
   228→   228→      }
   229→   229→    } else {
   230→   230→      const isMovie = category === 'movie';
   231→   231→      const dateField = isMovie ? 'primary_release_date' : 'first_air_date';
   232→   232→      const endpoint = isMovie ? 'movie' : 'tv';
   233→   233→      const res = await fetch(
   234→   234→        `${TMDB_BASE}/discover/${endpoint}?api_key=${apiKey}&${dateField}.gte=${birthYear}-01-01&${dateField}.lte=${maxYear}-12-31&sort_by=popularity.desc&page=${page}&vote_count.gte=50`
   235→   235→      );
   236→   236→      const data = await res.json();
   237→   237→      totalPages = data.total_pages || 1;
   238→   238→
   239→   239→      for (const item of (data.results || []).slice(0, 20)) {
   240→   240→        items.push(formatTmdbItem(item, endpoint));
   241→   241→      }
   242→   242→    }
   243→   243→  } catch (err) {
   244→   244→    console.error('[api/movies] TMDB fetch error:', err);
   245→   245→    return null;
   246→   246→  }
   247→   247→
   248→   248→  return { items, totalPages: Math.min(totalPages, 10) };
   249→   249→}
   250→   250→
   251→   251→function formatTmdbItem(item: TmdbMovie, mediaType: string): MovieItem {
   252→   252→  const dateStr = item.release_date || item.first_air_date || '';
   253→   253→  const year = dateStr ? parseInt(dateStr.substring(0, 4)) : 0;
   254→   254→  const genres = (item.genre_ids || []).map((g) => GENRE_MAP[g] || '').filter(Boolean);
   255→   255→  const isCartoon = genres.includes('Animation') || genres.includes('Kids');
   256→   256→
   257→   257→  return {
   258→   258→    id: String(item.id),
   259→   259→    title: item.title || item.name || 'Unknown',
   260→   260→    year,
   261→   261→    type: isCartoon ? 'cartoon' : (mediaType === 'tv' ? 'tv' : 'movie'),
   262→   262→    poster: item.poster_path ? `${TMDB_IMG}${item.poster_path}` : '',
   263→   263→    backdrop: item.backdrop_path ? `${TMDB_ORIG}${item.backdrop_path}` : '',
   264→   264→    rating: Math.round(item.vote_average * 10) / 10,
   265→   265→    genres,
   266→   266→    overview: item.overview || '',
   267→   267→    mediaType,
   268→   268→  };
   269→   269→}

   271→   271→/* ── response type ────────────────────────── */
   272→   272→interface MovieItem {
   273→   273→  id: string;
   274→   274→  title: string;
   275→   275→  year: number;
   276→   276→  type: 'movie' | 'tv' | 'cartoon';
   277→   277→  poster: string;
   278→   278→  backdrop?: string;
   279→   279→  rating: number;
   280→   280→  genres: string[];
   281→   281→  overview: string;
   282→   282→  mediaType?: string;
   283→   283→  watchProviders?: { name: string; logo: string }[];
   284→   284→}

   286→   286→/* ── GET handler ──────────────────────────── */
   287→   287→export async function GET(req: Request) {
   288→   288→  const { searchParams } = new URL(req.url);
   289→   289→  const birthYear = parseInt(searchParams.get('birthYear') || '1990');
   290→   290→  const category = searchParams.get('category') || 'all'; // all, movie, tv, cartoon
   291→   291→  const page = parseInt(searchParams.get('page') || '1');
   292→   292→
   293→   293→  // Core age filter: birth year to 18 years old
   294→   294→  const minYear = birthYear;
   295→   295→  const maxYear = birthYear + 18;
   296→   296→
   297→   297→  // Try TMDB first
   298→   298→  const tmdbResult = await fetchFromTmdb(minYear, maxYear, category, page);
   299→   299→  if (tmdbResult) {
   300→   300→    return NextResponse.json({
   301→   301→      items: tmdbResult.items,
   302→   302→      totalPages: tmdbResult.totalPages,
   303→   303→      minYear,
   304→   304→      maxYear,
   305→   305→      source: 'tmdb',
   306→   306→    });
   307→   307→  }
   308→   308→
   309→   309→  // Fallback: curated data
   310→   310→  const allItems: MovieItem[] = [];
   311→   311→  const startDecade = Math.floor(minYear / 10) * 10;
   312→   312→  const endDecade = Math.floor(maxYear / 10) * 10;
   313→   313→
   314→   314→  for (let d = startDecade; d <= endDecade; d += 10) {
   315→   315→    const key = `${d}s`;
   316→   316→    const decadeItems = FALLBACK[key] || [];
   317→   317→    for (const item of decadeItems) {
   318→   318→      if (item.year >= minYear && item.year <= maxYear) {
   319→   319→        if (category === 'all' || item.type === category) {
   320→   320→          allItems.push({
   321→   321→            id: `fb-${d}-${item.title.replace(/\s+/g, '-').toLowerCase()}`,
   322→   322→            title: item.title,
   323→   323→            year: item.year,
   324→   324→            type: item.type,
   325→   325→            poster: item.poster,
   326→   326→            rating: item.rating,
   327→   327→            genres: item.genres,
   328→   328→            overview: item.overview,
   329→   329→          });
   330→   330→        }
   331→   331→      }
   332→   332→    }
   333→   333→  }
   334→   334→
   335→   335→  // Sort by year desc, then rating desc
   336→   336→  allItems.sort((a, b) => b.year - a.year || b.rating - a.rating);
   337→   337→
   338→   338→  return NextResponse.json({
   339→   339→    items: allItems,
   340→   340→    totalPages: 1,
   341→   341→    minYear,
   342→   342→    maxYear,
   343→   343→    source: 'curated',
   344→   344→  });
   345→   345→}
