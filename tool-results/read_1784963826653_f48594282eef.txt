    39→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    40→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    41→  const [bootLines, setBootLines] = useState<string[]>([]);
    42→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    43→
    44→  const allBootLines = useMemo(() => [
    45→    'BIOS v2.4 — Nostalgia Systems Inc.',
    46→    'POST: Memory test... 640K OK',
    47→    'Detecting floppy drives... A: OK  B: OK',
    48→    'Loading REWIND.EXE...',
    49→    'Mounting childhood memories... 47% complete',
    50→    'Indexing nostalgia database...',
    51→    'Rewinding time...',
    52→    'Loading good old days... 83% complete',
    53→    'Ready.',
    54→  ], []);
    55→
    56→  useEffect(() => {
    57→    const timers: ReturnType<typeof setTimeout>[] = [];
    58→    // Reveal boot lines one by one
    59→    allBootLines.forEach((line, i) => {
    60→      timers.push(setTimeout(() => {
    61→        setBootLines(prev => [...prev, line]);
    62→        if (i === allBootLines.length - 1) {
    63→          // Last line — show title after a beat
    64→          timers.push(setTimeout(() => setPhase(1), 1000));
    65→        }
    66→      }, 500 + i * 500));
    67→    });
    68→
    69→    // Auto-dismiss after title phase
    70→    fadeTimerRef.current = setTimeout(() => {
    71→      setPhase(2);
    72→      setTimeout(onDone, 800);
    73→    }, 7000);
    74→
    75→    return () => {
    76→      timers.forEach(clearTimeout);
    77→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    78→    };
    79→  }, [allBootLines, onDone]);
    80→
    81→  return (
    82→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    83→      <div className="crt-scanlines" />
    84→      <div className="vhs-noise" />

    89→        {/* Boot lines */}

    91→          {bootLines.map((line, i) => (
    92→            <div
    93→              key={i}
    94→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    95→              style={{ animationDelay: '0s' }}
    96→            >
    97→              {line}
    98→              {i === bootLines.length - 1 && (
    99→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
   100→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
   101→                />
   102→              )}
   103→            </div>
   104→          ))}
   105→        </div>
   106→
   107→        {/* Title */}
   108→        {phase >= 1 && (
   109→          <motion.div
   110→            initial={{ opacity: 0, scale: 0.9 }}
   111→            animate={{ opacity: 1, scale: 1 }}

   396→  return '';

   748→export default function Home() {
   749→  const [birthYear, setBirthYear] = useState<number | null>(null);
   750→  const [loading, setLoading] = useState(true);
   751→  const [showOnboarding, setShowOnboarding] = useState(false);
   752→  const [onboardingChecked, setOnboardingChecked] = useState(false);
   753→
   754→  // Run one-time localStorage → IndexedDB migration on first load
   755→  useEffect(() => {
   756→    if (typeof window !== 'undefined') {
   757→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   758→      // Check if onboarding was completed
   759→      (async () => {
   760→        try {
   761→          const db = await getDB();
   762→          const record: any = await db.get('era_skin', 'onboarding');
   763→          if (record?.complete) {
   764→            // Restore persisted birth year
   765→            if (record.birthYear) setBirthYear(record.birthYear);
   766→          } else {
   767→            setShowOnboarding(true);
   768→          }
   769→        } catch {
   770→          setShowOnboarding(true);
   771→        }
   772→        setOnboardingChecked(true);
   773→      })();
   774→    }
   775→  }, []);
   776→
   777→  const handleLoadingDone = useCallback(() => {
   778→    setLoading(false);
   779→  }, []);
   780→
   781→  const handleSubmit = useCallback((year: number) => {
   782→    triggerOldLoader('Tuning into your era...');
   783→    setTimeout(() => {
   784→      setBirthYear(year);
   785→      window.scrollTo({ top: 0, behavior: 'smooth' });
   786→      hideOldLoader();
   787→    }, 1100);
   788→  }, []);
   789→
   790→  const handleOnboardingComplete = useCallback((data: { birthYear: number; eraSkin: string }) => {
   791→    setShowOnboarding(false);
   792→    triggerOldLoader('Tuning into your era...');
   793→    setTimeout(() => {
   794→      setBirthYear(data.birthYear);
   795→      hideOldLoader();
   796→    }, 1100);
   797→  }, []);
   798→
   799→  const handleBack = useCallback(() => {
   800→    triggerOldLoader('Rewinding...');
   801→    setTimeout(() => {
   802→      setBirthYear(null);
   803→      hideOldLoader();
   804→    }, 850);
   805→  }, []);
   806→
   807→  return (
   808→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   809→      {/* CRT Boot Loading Screen */}
   810→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   811→
   812→      {/* Onboarding (first visit only) */}
   813→      <AnimatePresence>
   814→        {showOnboarding && onboardingChecked && (
   815→          <OnboardingFlow onComplete={handleOnboardingComplete} />
   816→        )}
   817→      </AnimatePresence>
   818→
   819→      <AnimatePresence mode="wait">
   820→        {!showOnboarding && birthYear === null && onboardingChecked ? (
   821→          <motion.div
   822→            key="hero"
   823→            initial={{ opacity: 0 }}
   824→            animate={{ opacity: 1 }}
   825→            exit={{ opacity: 0, y: -20 }}
   826→            transition={{ duration: 0.4 }}
   827→            style={{ opacity: loading ? 0 : 1 }}
   828→          >
   829→            <HeroSection onSubmit={handleSubmit} />
   830→          </motion.div>
   831→        ) : birthYear !== null ? (
   832→          <motion.div
   833→            key="dashboard"
   834→            initial={{ opacity: 0, y: 20 }}
   835→            animate={{ opacity: 1, y: 0 }}
   836→            exit={{ opacity: 0 }}
   837→            transition={{ duration: 0.5 }}
   838→            className="flex flex-col min-h-screen"
   839→          >
   840→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   841→          </motion.div>
   842→        ) : null}
   843→      </AnimatePresence>
   844→    </div>
   845→  );
