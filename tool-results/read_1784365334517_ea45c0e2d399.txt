    10→/* ═══════ TYPES ═══════ */

    24→  faded80s: { label: "80's Fade", emoji: '🟡', css: 'sepia(0.25) contrast(0.9) brightness(1.08) saturate(0.75) hue-rotate(-8deg)' },
    25→  warm90s: { label: "90's Warm", emoji: '🟠', css: 'sepia(0.35) contrast(1.1) brightness(1.02) saturate(1.25) hue-rotate(-5deg)' },
    26→  bw60s: { label: "60's B&W", emoji: '⬛', css: 'grayscale(1) contrast(1.25) brightness(0.9)' },
    27→  polaroid: { label: 'Polaroid', emoji: '📸', css: 'sepia(0.2) contrast(0.95) brightness(1.12) saturate(1.35) hue-rotate(-3deg)' },
    28→  vhs: { label: 'VHS', emoji: '📼', css: 'sepia(0.15) contrast(1.2) brightness(0.88) saturate(0.85) hue-rotate(-15deg)' },
    29→  kodak: { label: 'Kodak', emoji: '🎞️', css: 'sepia(0.1) contrast(1.08) brightness(1.05) saturate(1.5) hue-rotate(5deg)' },
    30→};

    32→/* ═══════ VINTAGE CAMERA COMPONENT ═══════ */
    33→export default function VintageCamera({ onClose }: { onClose: () => void }) {
    34→  const videoRef = useRef<HTMLVideoElement>(null);
    35→  const canvasRef = useRef<HTMLCanvasElement>(null);
    36→  const streamRef = useRef<MediaStream | null>(null);
    37→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
    38→  const recordedChunksRef = useRef<Blob[]>([]);
    39→  const flashRef = useRef<HTMLDivElement>(null);
    40→
    41→  const [cameraOpen, setCameraOpen] = useState(false);
    42→  const [facingMode, setFacingMode] = useState<'user' | 'environment'>('environment');
    43→  const [mode, setMode] = useState<'photo' | 'video'>('photo');
    44→  const [filter, setFilter] = useState<FilterPreset>('sepia70s');
    45→  const [recording, setRecording] = useState(false);
    46→  const [recordTime, setRecordTime] = useState(0);
    47→  const [flash, setFlash] = useState(false);
    48→  const [gallery, setGallery] = useState<CapturedItem[]>([]);
    49→  const [viewItem, setViewItem] = useState<CapturedItem | null>(null);
    50→  const [cameraError, setCameraError] = useState<string | null>(null);
    51→  const [showFilters, setShowFilters] = useState(false);
    52→  const [shutterAnim, setShutterAnim] = useState(false);
    53→
    54→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    55→
    56→  /* ── Start Camera ── */
    57→  const startCamera = useCallback(async () => {
    58→    try {
    59→      if (streamRef.current) {
    60→        streamRef.current.getTracks().forEach(t => t.stop());
    61→      }
    62→      const stream = await navigator.mediaDevices.getUserMedia({
    63→        video: { facingMode, width: { ideal: 1280 }, height: { ideal: 720 } },
    64→        audio: mode === 'video',
    65→      });
    66→      streamRef.current = stream;
    67→      if (videoRef.current) {
    68→        videoRef.current.srcObject = stream;
    69→        await videoRef.current.play();
    70→      }
    71→      setCameraOpen(true);
    72→      setCameraError(null);
    73→    } catch (err) {
    74→      console.error('Camera error:', err);
    75→      setCameraError('Camera access denied or not available. Please allow camera permissions.');
    76→    }
    77→  }, [facingMode, mode]);
    78→
    79→  /* ── Stop Camera ── */
    80→  const stopCamera = useCallback(() => {
    81→    if (streamRef.current) {
    82→      streamRef.current.getTracks().forEach(t => t.stop());
    83→      streamRef.current = null;
    84→    }
    85→    setCameraOpen(false);
    86→    if (recordTimerRef.current) {
    87→      clearInterval(recordTimerRef.current);
    88→      recordTimerRef.current = null;
    89→    }
    90→  }, []);
    91→
    92→  /* ── Switch Camera ── */
    93→  const handleSwitchCamera = useCallback(() => {
    94→    setFacingMode(prev => prev === 'user' ? 'environment' : 'user');
    95→  }, []);
    96→
    97→  /* ── Restart on facing/mode change ── */
    98→  useEffect(() => {
    99→    if (cameraOpen) {
   100→      startCamera();
   101→    }
   102→    return () => {
   103→      // cleanup on unmount
   104→    };
   105→  }, [facingMode, mode, cameraOpen, startCamera]);
   106→
   107→  /* ── Apply vintage canvas filters ── */
   108→  const applyVintageFilter = useCallback((ctx: CanvasRenderingContext2D, w: number, h: number, preset: FilterPreset) => {
   109→    if (preset === 'none') return;
   110→
   111→    // Get image data
   112→    const imageData = ctx.getImageData(0, 0, w, h);
   113→    const data = imageData.data;
   114→
   115→    // Per-pixel vintage processing
   116→    for (let i = 0; i < data.length; i += 4) {
   117→      let r = data[i], g = data[i + 1], b = data[i + 2];
   118→
   119→      switch (preset) {
   120→        case 'sepia70s': {
   121→          // Heavy warm sepia + slightly desaturate
   122→          const gray = 0.299 * r + 0.587 * g + 0.114 * b;
   123→          r = gray * 1.15 + 35;
   124→          g = gray * 1.0 + 15;
   125→          b = gray * 0.85;
   126→          // Reduce contrast slightly
   127→          r = 128 + (r - 128) * 0.92;
   128→          g = 128 + (g - 128) * 0.92;
   129→          b = 128 + (b - 128) * 0.92;
   130→          break;
   131→        }
   132→        case 'faded80s': {
   133→          // Washed out, lifted blacks, warm shift
   134→          r = r * 0.88 + 45;
   135→          g = g * 0.85 + 40;
   136→          b = b * 0.8 + 50;
   137→          // Lift shadows (add to dark areas)
   138→          const lum = (r + g + b) / 3;
   139→          if (lum < 100) {
   140→            const lift = (100 - lum) * 0.3;
   141→            r += lift;
   142→            g += lift;
   143→            b += lift + 10;
   144→          }
   145→          break;
   146→        }
   147→        case 'warm90s': {
   148→          // Warm golden tones, slightly saturated
   149→          r = r * 1.08 + 12;
   150→          g = g * 0.98 + 8;
   151→          b = b * 0.88;
   152→          // Warm color cast
   153→          r = Math.min(255, r + 15);
   154→          b = Math.max(0, b - 10);
   155→          break;
   156→        }
   157→        case 'bw60s': {
   158→          // Classic high-contrast B&W with slight grain look
   159→          const gray = 0.299 * r + 0.587 * g + 0.114 * b;
   160→          const contrast = 1.25;
   161→          const newGray = 128 + (gray - 128) * contrast;
   162→          r = g = b = newGray;
   163→          // Add subtle random grain
   164→          const grain = (Math.random() - 0.5) * 12;
   165→          r += grain; g += grain; b += grain;
   166→          break;
   167→        }
   168→        case 'polaroid': {
   169→          // Soft, bright, slightly oversaturated with cool shadows
   170→          r = r * 1.05 + 10;
   171→          g = g * 1.08 + 8;
   172→          b = b * 0.95 + 15;
   173→          // Slight green cast in shadows
   174→          const lum = (r + g + b) / 3;
   175→          if (lum < 120) {
   176→            g = Math.min(255, g + 8);
   177→            b = Math.max(0, b - 5);
   178→          }
   179→          break;
   180→        }
   181→        case 'vhs': {
   182→          // Desaturated, slight color shift, low contrast
   183→          r = r * 0.85 + 30;
   184→          g = g * 0.82 + 28;
   185→          b = b * 0.88 + 25;
   186→          // Slight purple/magenta shift
   187→          r = Math.min(255, r + 8);
   188→          b = Math.min(255, b + 5);
   189→          // Lower contrast
   190→          r = 128 + (r - 128) * 0.85;
   191→          g = 128 + (g - 128) * 0.85;
   192→          b = 128 + (b - 128) * 0.85;
   193→          break;
   194→        }
   195→        case 'kodak': {
   196→          // Vibrant, warm, rich
   197→          r = r * 1.1 + 5;
   198→          g = g * 1.05 + 5;
   199→          b = b * 0.92;
   200→          // Slight warm lift
   201→          const lum2 = (r + g + b) / 3;
   202→          if (lum2 > 150) {
   203→            r = Math.min(255, r * 1.05);
   204→            g = Math.min(255, g * 0.98);
   205→          }
   206→          break;
   207→        }
   208→      }
   209→
   210→      data[i] = Math.max(0, Math.min(255, r));
   211→      data[i + 1] = Math.max(0, Math.min(255, g));
   212→      data[i + 2] = Math.max(0, Math.min(255, b));
   213→    }
   214→
   215→    ctx.putImageData(imageData, 0, 0);
   216→
   217→    // Add vignette overlay
   218→    addVignette(ctx, w, h, preset);
   219→
   220→    // Add film grain overlay
   221→    if (preset === 'bw60s' || preset === 'vhs' || preset === 'sepia70s') {
   222→      addGrain(ctx, w, h, preset === 'bw60s' ? 25 : 15);
   223→    }
   224→
   225→    // Add date stamp for some presets
   226→    if (preset === 'polaroid' || preset === 'sepia70s') {
   227→      addDateStamp(ctx, w, h);
   228→    }
   229→  }, []);
   230→
   231→  /* ── Vignette ── */
   232→  const addVignette = (ctx: CanvasRenderingContext2D, w: number, h: number, preset: FilterPreset) => {
   233→    const intensity = preset === 'bw60s' ? 0.6 : preset === 'vhs' ? 0.5 : 0.35;
   234→    const gradient = ctx.createRadialGradient(w / 2, h / 2, w * 0.25, w / 2, h / 2, w * 0.75);
   235→    gradient.addColorStop(0, 'rgba(0,0,0,0)');
   236→    gradient.addColorStop(1, `rgba(0,0,0,${intensity})`);
   237→    ctx.fillStyle = gradient;
   238→    ctx.fillRect(0, 0, w, h);
   239→  };
   240→
   241→  /* ── Film Grain ── */
   242→  const addGrain = (ctx: CanvasRenderingContext2D, w: number, h: number, amount: number) => {
   243→    const imageData = ctx.getImageData(0, 0, w, h);
   244→    const data = imageData.data;
   245→    for (let i = 0; i < data.length; i += 4) {
   246→      const noise = (Math.random() - 0.5) * amount;
   247→      data[i] += noise;
   248→      data[i + 1] += noise;
   249→      data[i + 2] += noise;
   250→    }
   251→    ctx.putImageData(imageData, 0, 0);
   252→  };
   253→
   254→  /* ── Date Stamp (like old film photos) ── */
   255→  const addDateStamp = (ctx: CanvasRenderingContext2D, w: number, h: number) => {
   256→    const now = new Date();
   257→    const dateStr = `${now.getFullYear()}.${String(now.getMonth() + 1).padStart(2, '0')}.${String(now.getDate()).padStart(2, '0')}`;
   258→    ctx.save();
   259→    ctx.font = 'bold 14px "Courier New", monospace';
   260→    ctx.fillStyle = 'rgba(255, 80, 80, 0.75)';
   261→    ctx.shadowColor = 'rgba(0,0,0,0.5)';
   262→    ctx.shadowBlur = 1;
   263→    ctx.textAlign = 'right';
   264→    ctx.fillText(dateStr, w - 12, h - 12);
   265→    ctx.restore();
   266→  };
   267→
   268→  /* ── Capture Photo ── */
   269→  const capturePhoto = useCallback(() => {
   270→    if (!videoRef.current || !canvasRef.current) return;
   271→    const video = videoRef.current;
   272→    const canvas = canvasRef.current;
   273→    const ctx = canvas.getContext('2d');
   274→    if (!ctx) return;
   275→
   276→    canvas.width = video.videoWidth;
   277→    canvas.height = video.videoHeight;
   278→
   279→    // Mirror if front camera
   280→    if (facingMode === 'user') {
   281→      ctx.translate(canvas.width, 0);
   282→      ctx.scale(-1, 1);
   283→    }
   284→
   285→    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
   286→
   287→    // Reset transform
   288→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   289→
   290→    // Apply vintage filter
   291→    applyVintageFilter(ctx, canvas.width, canvas.height, filter);
   292→
   293→    // Flash effect
   294→    setFlash(true);
   295→    setShutterAnim(true);
   296→    setTimeout(() => setFlash(false), 200);
   297→    setTimeout(() => setShutterAnim(false), 300);
   298→
   299→    // Save to gallery
   300→    canvas.toBlob((blob) => {
   301→      if (!blob) return;
   302→      const url = URL.createObjectURL(blob);
   303→      setGallery(prev => [{
   304→        id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   305→        type: 'photo',
   306→        url,
   307→        timestamp: Date.now(),
   308→        blob,
   309→      }, ...prev]);
   310→    }, 'image/jpeg', 0.92);
   311→  }, [filter, facingMode, applyVintageFilter]);
   312→
   313→  /* ── Start/Stop Video Recording ── */
   314→  const toggleRecording = useCallback(() => {
   315→    if (!recording && streamRef.current) {
   316→      // Start recording
   317→      const options = { mimeType: 'video/webm;codecs=vp9' };
   318→      try {
   319→        const mediaRecorder = new MediaRecorder(streamRef.current, options);
   320→        recordedChunksRef.current = [];
   321→        mediaRecorder.ondataavailable = (e) => {
   322→          if (e.data.size > 0) recordedChunksRef.current.push(e.data);
   323→        };
   324→        mediaRecorder.onstop = () => {
   325→          const blob = new Blob(recordedChunksRef.current, { type: 'video/webm' });
   326→          const url = URL.createObjectURL(blob);
   327→          setGallery(prev => [{
   328→            id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   329→            type: 'video',
   330→            url,
   331→            timestamp: Date.now(),
   332→            blob,
   333→          }, ...prev]);
   334→          recordedChunksRef.current = [];
   335→        };
   336→        mediaRecorder.start(100);
   337→        mediaRecorderRef.current = mediaRecorder;
   338→        setRecording(true);
   339→        setRecordTime(0);
   340→        recordTimerRef.current = setInterval(() => {
   341→          setRecordTime(prev => prev + 1);
   342→        }, 1000);
   343→      } catch {
   344→        // Fallback: try vp8
   345→        try {
   346→          const mediaRecorder = new MediaRecorder(streamRef.current, { mimeType: 'video/webm;codecs=vp8' });
   347→          recordedChunksRef.current = [];
   348→          mediaRecorder.ondataavailable = (e) => {
   349→            if (e.data.size > 0) recordedChunksRef.current.push(e.data);
   350→          };
   351→          mediaRecorder.onstop = () => {
   352→            const blob = new Blob(recordedChunksRef.current, { type: 'video/webm' });
   353→            const url = URL.createObjectURL(blob);
   354→            setGallery(prev => [{
   355→              id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   356→              type: 'video',
   357→              url,
   358→              timestamp: Date.now(),
   359→              blob,
   360→            }, ...prev]);
   361→            recordedChunksRef.current = [];
   362→          };
   363→          mediaRecorder.start(100);
   364→          mediaRecorderRef.current = mediaRecorder;
   365→          setRecording(true);
   366→          setRecordTime(0);
   367→          recordTimerRef.current = setInterval(() => {
   368→            setRecordTime(prev => prev + 1);
   369→          }, 1000);
   370→        } catch {
   371→          setCameraError('Video recording not supported in this browser.');
   372→        }
   373→      }
   374→    } else if (recording && mediaRecorderRef.current) {
   375→      // Stop recording
   376→      mediaRecorderRef.current.stop();
   377→      mediaRecorderRef.current = null;
   378→      setRecording(false);
   379→      if (recordTimerRef.current) {
   380→        clearInterval(recordTimerRef.current);
   381→        recordTimerRef.current = null;
   382→      }
   383→    }
   384→  }, [recording]);
   385→
   386→  /* ── Delete from gallery ── */
   387→  const deleteItem = useCallback((id: string) => {
   388→    setGallery(prev => {
   389→      const item = prev.find(i => i.id === id);
   390→      if (item) URL.revokeObjectURL(item.url);
   391→      return prev.filter(i => i.id !== id);
   392→    });
   393→  }, []);
   394→
   395→  /* ── Download ── */
   396→  const downloadItem = useCallback((item: CapturedItem) => {
   397→    const a = document.createElement('a');
   398→    a.href = item.url;
   399→    a.download = `rewind_${item.type}_${new Date(item.timestamp).toISOString().replace(/[:.]/g, '-')}.${item.type === 'photo' ? 'jpg' : 'webm'}`;
   400→    document.body.appendChild(a);
   401→    a.click();
   402→    document.body.removeChild(a);
   403→  }, []);
   404→
   405→  /* ── Cleanup on close ── */
   406→  useEffect(() => {
   407→    return () => {
   408→      stopCamera();
   409→      gallery.forEach(item => URL.revokeObjectURL(item.url));
   410→    };
   411→  // eslint-disable-next-line react-hooks/exhaustive-deps
   412→  }, []);
   413→
   414→  /* ── Format time ── */
   415→  const formatTime = (s: number) => `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
   416→
   417→  return (
   418→    <>
   419→      {/* Hidden canvas for processing */}
   420→      <canvas ref={canvasRef} className="hidden" />
   421→
   422→      <AnimatePresence>
   423→        <motion.div
   424→          initial={{ opacity: 0 }}
   425→          animate={{ opacity: 1 }}
   426→          exit={{ opacity: 0 }}
   427→          className="fixed inset-0 z-50 flex flex-col"
   428→          style={{ background: '#1a1208' }}
   429→        >
   430→          {/* ═══ HEADER - Camera Top Bar ═══ */}
   431→          <div className="sk-leather px-3 py-2.5 flex items-center justify-between z-10">
   432→            <button onClick={() => { stopCamera(); onClose(); }} className="sk-btn old-btn px-3 py-1.5 rounded-lg text-xs font-bold cursor-pointer flex items-center gap-1 text-[#faf6eb]">
   433→              <X className="h-4 w-4" /> Close
   434→            </button>
   435→            <h2 className="old-title font-black text-base text-[#faf6eb] tracking-wide" style={{ textShadow: '0 1px 2px rgba(0,0,0,0.3)' }}>
   436→              📷 Vintage Camera
   437→            </h2>
   438→            <div className="w-16" />
   439→          </div>
   440→
   441→          {/* ═══ VIEWFINDER ═══ */}
   442→          <div className="flex-1 relative overflow-hidden flex items-center justify-center">
   443→            {!cameraOpen && !cameraError && (
   444→              <div className="text-center px-6">
   445→                <motion.div
   446→                  animate={{ rotate: [0, 5, -5, 0] }}
   447→                  transition={{ duration: 4, repeat: Infinity }}
   448→                  className="text-6xl mb-4"
   449→                >📸</motion.div>
   450→                <p className="old-body text-[#d4a574] mb-6 text-sm">Open the camera to capture memories in vintage style</p>
   451→                <button onClick={startCamera} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   452→                  <Camera className="h-5 w-5" /> Open Camera
   453→                </button>
   454→              </div>
   455→            )}
   456→
   457→            {cameraError && (
   458→              <div className="text-center px-6">
   459→                <div className="text-5xl mb-4">😵</div>
   460→                <p className="old-body text-[#d4a574] mb-4 text-sm">{cameraError}</p>
   461→                <button onClick={startCamera} className="sk-btn old-btn px-5 py-2.5 rounded-lg font-bold cursor-pointer flex items-center gap-2 mx-auto">
   462→                  <RotateCcw className="h-4 w-4" /> Try Again
   463→                </button>
   464→              </div>
   465→            )}
   466→
   467→            {cameraOpen && (
   468→              <>
   469→                {/* Video Feed */}
   470→                <video
   471→                  ref={videoRef}
   472→                  className={`absolute inset-0 w-full h-full object-cover transition-transform duration-300 ${facingMode === 'user' ? '-scale-x-100' : ''}`}
   473→                  style={{ filter: FILTERS[filter].css }}
   474→                  playsInline
   475→                  muted
   476→                />
   477→
   478→                {/* Vignette overlay on viewfinder */}
   479→                <div className="absolute inset-0 pointer-events-none" style={{
   480→                  background: 'radial-gradient(ellipse at center, transparent 50%, rgba(0,0,0,0.4) 100%)',
   481→                }} />
   482→
   483→                {/* Scanline overlay */}
   484→                <div className="absolute inset-0 pointer-events-none opacity-10" style={{
   485→                  background: 'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(0,0,0,0.15) 2px, rgba(0,0,0,0.15) 4px)',
   486→                }} />
   487→
   488→                {/* Film border overlay - viewfinder frame */}
   489→                <div className="absolute inset-0 pointer-events-none border-[6px] border-[#1a1208] rounded-none" />
   490→
   491→                {/* Corner brackets */}
   492→                <div className="absolute top-3 left-3 w-8 h-8 border-t-2 border-l-2 border-[#d4a574]/50 pointer-events-none" />
   493→                <div className="absolute top-3 right-3 w-8 h-8 border-t-2 border-r-2 border-[#d4a574]/50 pointer-events-none" />
   494→                <div className="absolute bottom-3 left-3 w-8 h-8 border-b-2 border-l-2 border-[#d4a574]/50 pointer-events-none" />
