    46→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    47→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    48→  const [bootLines, setBootLines] = useState<string[]>([]);
    49→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    50→
    51→  const allBootLines = useMemo(() => [
    52→    'BIOS v2.4 — Nostalgia Systems Inc.',
    53→    'POST: Memory test... 640K OK',
    54→    'Detecting floppy drives... A: OK  B: OK',
    55→    'Loading REWIND.EXE...',
    56→    'Mounting childhood memories... 47% complete',
    57→    'Indexing nostalgia database...',
    58→    'Rewinding time...',
    59→    'Loading good old days... 83% complete',
    60→    'Ready.',
    61→  ], []);
    62→
    63→  useEffect(() => {
    64→    const timers: ReturnType<typeof setTimeout>[] = [];
    65→    // Reveal boot lines one by one
    66→    allBootLines.forEach((line, i) => {
    67→      timers.push(setTimeout(() => {
    68→        setBootLines(prev => [...prev, line]);
    69→        if (i === allBootLines.length - 1) {
    70→          // Last line — show title after a beat
    71→          timers.push(setTimeout(() => setPhase(1), 1000));
    72→        }
    73→      }, 500 + i * 500));
    74→    });
    75→
    76→    // Auto-dismiss after title phase
    77→    fadeTimerRef.current = setTimeout(() => {
    78→      setPhase(2);
    79→      setTimeout(onDone, 800);
    80→    }, 7000);
    81→
    82→    return () => {
    83→      timers.forEach(clearTimeout);
    84→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    85→    };
    86→  }, [allBootLines, onDone]);
    87→
    88→  return (
    89→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    90→      <div className="crt-scanlines" />
    91→      <div className="vhs-noise" />

    96→        {/* Boot lines */}

    98→          {bootLines.map((line, i) => (

   100→              key={i}
   101→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
   102→              style={{ animationDelay: '0s' }}
   103→            >
   104→              {line}
   105→              {i === bootLines.length - 1 && (
   106→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
   107→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}

   109→              )}
   110→            </div>
   111→          ))}
   112→        </div>
   113→
   114→        {/* Title */}
   115→        {phase >= 1 && (
   116→          <motion.div
   117→            initial={{ opacity: 0, scale: 0.9 }}

   398→function getDecadeTheme(birthYear: number): string {

   403→  return '';

   788→export default function Home() {
   789→  const [birthYear, setBirthYear] = useState<number | null>(null);
   790→  const [loading, setLoading] = useState(true);
   791→  const [showOnboarding, setShowOnboarding] = useState(false);
   792→  const [onboardingChecked, setOnboardingChecked] = useState(false);
   793→
   794→  // Run one-time localStorage → IndexedDB migration on first load
   795→  useEffect(() => {
   796→    if (typeof window !== 'undefined') {
   797→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   798→      // Check if onboarding was completed
   799→      (async () => {
   800→        try {
   801→          const db = await getDB();
   802→          const record: any = await db.get('era_skin', 'onboarding');
   803→          if (record?.complete) {
   804→            // Restore persisted birth year
   805→            if (record.birthYear) setBirthYear(record.birthYear);
   806→          } else {
   807→            setShowOnboarding(true);
   808→          }
   809→        } catch {
   810→          setShowOnboarding(true);
   811→        }
   812→        setOnboardingChecked(true);
   813→      })();
   814→    }
   815→  }, []);
   816→
   817→  const handleLoadingDone = useCallback(() => {
   818→    setLoading(false);
   819→  }, []);
   820→
   821→  const handleSubmit = useCallback((year: number) => {
   822→    triggerOldLoader('Tuning into your era...');

   824→      setBirthYear(year);
   825→      window.scrollTo({ top: 0, behavior: 'smooth' });
   826→      hideOldLoader();
   827→    }, 1100);
   828→  }, []);

   830→  const handleOnboardingComplete = useCallback((data: { birthYear: number; eraSkin: string }) => {
   831→    setShowOnboarding(false);

   833→    setTimeout(() => {
   834→      setBirthYear(data.birthYear);
   835→      hideOldLoader();
   836→    }, 1100);
   837→  }, []);
   838→
   839→  const handleBack = useCallback(() => {
   840→    triggerOldLoader('Rewinding...');
   841→    setTimeout(() => {
   842→      setBirthYear(null);
   843→      hideOldLoader();
   844→    }, 850);
   845→  }, []);
   846→
   847→  return (
   848→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>

   850→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   851→
   852→      {/* Onboarding (first visit only) */}
   853→      <AnimatePresence>
   854→        {showOnboarding && onboardingChecked && (
   855→          <OnboardingFlow onComplete={handleOnboardingComplete} />
   856→        )}
   857→      </AnimatePresence>
   858→
   859→      <AnimatePresence mode="wait">

   861→          <motion.div
   862→            key="hero"

   869→            <HeroSection onSubmit={handleSubmit} />
   870→          </motion.div>
   871→        ) : birthYear !== null ? (
   872→          <motion.div
   873→            key="dashboard"
   874→            initial={{ opacity: 0, y: 20 }}
   875→            animate={{ opacity: 1, y: 0 }}

   883→      </AnimatePresence>
   884→    </div>
   885→  );
