   243→function getDecadeKey(year: number): string {
   244→  const decade = Math.floor(year / 10) * 10;
   245→  const key = `${decade}s`;
   246→  return decadeDataMap[key] ? key : '1970s';
   247→}

   249→function getEntriesForDate(data: HistoryEntry[], month: number, day: number): HistoryEntry[] {
   250→  const exact = data.filter(e => e.month === month && e.day === day);
   251→  if (exact.length > 0) return exact;
   252→
   253→  const monthEntries = data.filter(e => e.month === month);
   254→  return monthEntries.length > 0 ? monthEntries : data.slice(0, 5);
   255→}

   257→export async function GET(request: NextRequest) {
   258→  const { searchParams } = new URL(request.url);
   259→  const birthYear = parseInt(searchParams.get('birthYear') || '1985', 10);
   260→
   261→  const clampedYear = Math.max(1970, Math.min(2015, birthYear));
   262→
   263→  const decadeKey = getDecadeKey(clampedYear);
   264→  const pool = decadeDataMap[decadeKey] || data1980s;
   265→
   266→  const now = new Date();
   267→  const month = now.getMonth() + 1;
   268→  const day = now.getDate();
   269→
   270→  const entries = getEntriesForDate(pool, month, day);
   271→
   272→  const seed = clampedYear + month;
   273→  const shuffled = [...entries].sort((a, b) => {
   274→    const ha = ((a.year * 31 + a.day * 7 + seed) % 100) / 100;
   275→    const hb = ((b.year * 31 + b.day * 7 + seed) % 100) / 100;
