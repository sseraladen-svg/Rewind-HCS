   120→// ═══════════════════════════════════════════════════════════════
   121→// useStoredState — localStorage-backed state via useSyncExternalStore
   122→// ═══════════════════════════════════════════════════════════════
   123→
   124→function useStoredState<T>(
   125→  key: string,
   126→  initial: T,
   127→): [T, (updater: T | ((prev: T) => T)) => void] {
   128→  const subscribe = useCallback(
   129→    (onStoreChange: () => void) => {
   130→      const handler = (e: StorageEvent) => {
   131→        if (e.key === key || e.key === null) onStoreChange();
   132→      };
   133→      window.addEventListener('storage', handler);
   134→      return () => window.removeEventListener('storage', handler);
   135→    },
   136→    [key],
   137→  );
   138→
   139→  const getSnapshot = useCallback(() => {
   140→    return localStorage.getItem(key);
   141→  }, [key]);
   142→
   143→  const getServerSnapshot = useCallback(() => null, []);
   144→
   145→  const raw = useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
   146→
   147→  const value = useMemo<T>(() => {
   148→    if (raw === null) return initial;
   149→    try {
   150→      return JSON.parse(raw) as T;
   151→    } catch {
   152→      return initial;
   153→    }
   154→  }, [raw, initial]);
   155→
   156→  const setValue = useCallback(
   157→    (updater: T | ((prev: T) => T)) => {
   158→      try {
   159→        const currentRaw = localStorage.getItem(key);
   160→        const current: T =
   161→          currentRaw !== null ? (JSON.parse(currentRaw) as T) : initial;
   162→        const next =
   163→          typeof updater === 'function'
   164→            ? (updater as (prev: T) => T)(current)
   165→            : updater;
   166→        localStorage.setItem(key, JSON.stringify(next));
   167→        window.dispatchEvent(new StorageEvent('storage', { key }));
   168→      } catch {
   169→        /* ignore storage errors */
   170→      }
   171→    },
   172→    [key, initial],
   173→  );
   174→
   175→  return [value, setValue];
   176→}

   179→// useVisitCounter — auto-incrementing visit counter
   180→// ═══════════════════════════════════════════════════════════════

   182→// Only called when isClient is true, so localStorage is available.
   183→// The useState lazy initializer runs exactly once on first mount.
   184→function useVisitCounter(): number {
   185→  const [count] = useState(() => {
   186→    const stored = localStorage.getItem('rewind_visit_count');
   187→    let c = 1;
   188→    if (stored) {
   189→      const parsed = parseInt(stored, 10);
   190→      if (!isNaN(parsed)) c = parsed + 1;
   191→    }
   192→    localStorage.setItem('rewind_visit_count', String(c));
   193→    return c;
   194→  });
   195→  return count;
   196→}
