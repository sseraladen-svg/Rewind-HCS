    41→/* ═══════════════════════════════════════════════════

    44→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    45→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    46→  const [bootLines, setBootLines] = useState<string[]>([]);
    47→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    48→
    49→  const allBootLines = useMemo(() => [
    50→    'BIOS v2.4 — Nostalgia Systems Inc.',
    51→    'POST: Memory test... 640K OK',
    52→    'Detecting floppy drives... A: OK  B: OK',
    53→    'Loading REWIND.EXE...',
    54→    'Mounting childhood memories... 47% complete',
    55→    'Indexing nostalgia database...',
    56→    'Rewinding time...',
    57→    'Loading good old days... 83% complete',
    58→    'Ready.',
    59→  ], []);
    60→
    61→  useEffect(() => {
    62→    const timers: ReturnType<typeof setTimeout>[] = [];
    63→    // Initialize audio on first boot
    64→    timers.push(setTimeout(() => initAudio(), 100));
    65→    // Reveal boot lines one by one with beep sounds
    66→    allBootLines.forEach((line, i) => {
    67→      timers.push(setTimeout(() => {
    68→        setBootLines(prev => [...prev, line]);
    69→        soundBootBeep();
    70→        if (i === allBootLines.length - 1) {
    71→          // Last line — show title after a beat
    72→          timers.push(setTimeout(() => {
    73→            setPhase(1);
    74→            soundBootReady();
    75→          }, 1000));
    76→        }
    77→      }, 500 + i * 500));
    78→    });
    79→
    80→    // Auto-dismiss after title phase
    81→    fadeTimerRef.current = setTimeout(() => {
    82→      setPhase(2);
    83→      setTimeout(onDone, 800);
    84→    }, 7000);
    85→
    86→    return () => {
    87→      timers.forEach(clearTimeout);
    88→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    89→    };
    90→  }, [allBootLines, onDone]);
    91→
    92→  return (
    93→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    94→      <div className="crt-scanlines" />
    95→      <div className="vhs-noise" />
    96→      <div className="vhs-tracking" />
    97→      <div className="vhs-tracking" style={{ animationDelay: '1.2s', opacity: 0.5 }} />
    98→
    99→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
   100→        {/* Boot lines */}
   101→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
   102→          {bootLines.map((line, i) => (
   103→            <div
   104→              key={i}
   105→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
   106→              style={{ animationDelay: '0s' }}
   107→            >
   108→              {line}
   109→              {i === bootLines.length - 1 && (
   110→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
   111→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
   112→                />
   113→              )}
   114→            </div>
   115→          ))}
   116→        </div>
   117→
   118→        {/* Title */}
   119→        {phase >= 1 && (
   120→          <motion.div
   121→            initial={{ opacity: 0, scale: 0.9 }}
   122→            animate={{ opacity: 1, scale: 1 }}
   123→            transition={{ duration: 1.0 }}
   124→            className="text-center mt-1 sm:mt-2 w-full"

   131→            </div>
   132→            <p className="old-typewriter text-[8px] sm:text-[10px] mt-2 sm:mt-3" style={{ color: 'rgba(196,149,106,0.5)' }}>

   410→function getDecadeTheme(birthYear: number): string {

   415→  return '';

   803→export default function Home() {
