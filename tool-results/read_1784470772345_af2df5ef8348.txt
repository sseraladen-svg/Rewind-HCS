    23→function formatTime(seconds: number): string {
    24→  const m = Math.floor(seconds / 60);
    25→  const s = Math.floor(seconds % 60);
    26→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    27→}
    28→
    29→function generateId(): string {
    30→  return `vm_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
    31→}
    32→
    33→function formatDate(iso: string): string {
    34→  const d = new Date(iso);
    35→  const month = d.toLocaleString('en-US', { month: 'short' }).toUpperCase();
    36→  const day = d.getDate();
    37→  const hour = d.getHours();
    38→  const ampm = hour >= 12 ? 'PM' : 'AM';
    39→  const h12 = hour % 12 || 12;
    40→  const min = String(d.getMinutes()).padStart(2, '0');
    41→  return `${month} ${day}, ${h12}:${min} ${ampm}`;
    42→}
    43→
    44→/** Play a short beep using Web Audio API */
    45→function playBeep(frequency = 1000, duration = 0.15, type: OscillatorType = 'square') {
    46→  if (typeof window === 'undefined') return;
    47→  try {
    48→    const ctx = new AudioContext();
    49→    const osc = ctx.createOscillator();
    50→    const gain = ctx.createGain();
    51→    osc.type = type;
    52→    osc.frequency.value = frequency;
    53→    gain.gain.value = 0.12;
    54→    osc.connect(gain);
    55→    gain.connect(ctx.destination);
    56→    osc.start(ctx.currentTime);
    57→    osc.stop(ctx.currentTime + duration);
    58→    osc.onended = () => ctx.close();
    59→  } catch {
    60→    // silently ignore audio errors
    61→  }
    62→}
    63→
    64→/** Generate fake waveform bar heights for visual display */
    65→function generateWaveformBars(count = 32): number[] {
    66→  return Array.from({ length: count }, () => Math.random() * 0.7 + 0.15);
    67→}

    98→        const parsed: VoicemailMessage[] = JSON.parse(raw);

   106→  // ── Persist messages ──

   108→    (msgs: VoicemailMessage[]) => {

   115→        try {
   116→          localStorage.setItem(STORAGE_KEY, JSON.stringify(trimmed));
   117→        } catch {
   118→          // give up silently
   119→        }
   120→      }
   121→    },
   122→    []
   123→  );
   124→
   125→  // ── Check browser support ──

   127→    if (typeof window === 'undefined') return;

   141→        audioElRef.current.pause();

   143→      }

   160→      };

   336→      {/* Error States */}

   338→        <div className="sk-paper rounded-xl p-6 text-center mb-6">

   347→        <div className="absolute inset-0 sk-noise pointer-events-none opacity-30" />
   348→
   349→        {/* Brand label */}
   350→        <div className="text-center mb-4 relative">

   352→            REWIND™ ANSWERING MACHINE

   545→            <p className="old-label text-[9px] text-[#c9a96e] tracking-wider">RECORD</p>
   546→            <p className="old-typewriter text-[7px] text-[#8b7355] mt-0.5">
   547→              MAX {MAX_RECORDING_SECONDS}s
   548→            </p>
   549→          </div>
   550→        </div>
   551→
   552→        {/* ── New Messages Indicator ── */}
   553→        <AnimatePresence>
   554→          {messages.length > 0 && (
   555→            <motion.div
   556→              initial={{ opacity: 0, y: -8 }}
   557→              animate={{ opacity: 1, y: 0 }}
   558→              exit={{ opacity: 0, y: -8 }}
   559→              className="text-center mb-4"
   560→            >
   561→              <p className="old-typewriter text-xs text-amber-600" style={{ textShadow: '0 0 6px rgba(217,119,6,0.2)' }}>
   562→                📼 You have <span className="font-bold">{messages.length}</span> saved message{messages.length !== 1 ? 's' : ''}
   563→              </p>
   564→            </motion.div>
   565→          )}
   566→        </AnimatePresence>
   567→      </div>
   568→
   569→      {/* ── Message List ── */}
   570→      <div className="max-w-lg mx-auto mt-6 space-y-3">
   571→        <AnimatePresence>
   572→          {messages.map((msg) => {
   573→            const isThisPlaying = playingId === msg.id;
   574→            const bars = generateWaveformBars(32);
   575→            const progressIdx = isThisPlaying ? Math.floor(playProgress * bars.length) : 0;
   576→
   577→            return (
   578→              <motion.div
   579→                key={msg.id}
   580→                initial={{ opacity: 0, y: 16, scale: 0.96 }}
   581→                animate={{ opacity: 1, y: 0, scale: 1 }}
   582→                exit={{ opacity: 0, x: -40, scale: 0.9 }}
   583→                transition={{ duration: 0.35 }}
   584→                className="sk-paper rounded-xl p-4 relative group shadow-sm"
   585→              >
   586→                <div className="absolute inset-0 sk-noise pointer-events-none opacity-10" />
   587→
   588→                {/* Tape Label Header */}
   589→                <div className="flex items-center justify-between mb-3">
   590→                  <div>
   591→                    <p className="old-label text-[9px] text-[#8b7355] tracking-wider">
   592→                      ✉ VOICEMAIL #{messages.indexOf(msg) + 1}
   593→                    </p>
   594→                    <p className="old-typewriter text-[8px] text-[#a08060] mt-0.5">
   595→                      {formatDate(msg.timestamp)}
   596→                    </p>
   597→                  </div>
   598→                  <span className="old-typewriter text-[10px] text-[#6b5b3e] bg-[#f5edd6] px-2 py-0.5 rounded border border-[#d4c9a8]">
   599→                    {formatTime(msg.duration)}
   600→                  </span>
   601→                </div>
   602→
   603→                {/* Waveform Visual */}
   604→                <div className="flex items-end gap-[2px] h-10 mb-3 px-1 bg-[#f5edd6]/50 rounded border border-[#e0d5b8] overflow-hidden">
   605→                  {bars.map((height, i) => {
   606→                    const isPast = isThisPlaying && i < progressIdx;
   607→                    const isCurrent = isThisPlaying && i === progressIdx;
   608→                    return (
   609→                      <div
   610→                        key={i}
   611→                        className="flex-1 rounded-sm transition-colors duration-150 min-w-[2px]"
   612→                        style={{
   613→                          height: `${height * 100}%`,
   614→                          backgroundColor: isCurrent
   615→                            ? '#b91c1c'
   616→                            : isPast
   617→                            ? '#8b5e3c'
   618→                            : '#c9a96e',
   619→                          opacity: isPast || isCurrent ? 1 : 0.5,
   620→                        }}
   621→                      />
   622→                    );
   623→                  })}
   624→                </div>
   625→
   626→                {/* Controls */}
   627→                <div className="flex items-center gap-3">
   628→                  {/* Play / Pause */}
   629→                  <button
   630→                    onClick={() => togglePlay(msg)}
   631→                    className="flex items-center gap-1.5 px-3 py-1.5 rounded bg-[#3d2b1f] text-[#f5edd6] old-typewriter text-[10px] tracking-wider hover:bg-[#5a4030] transition-colors focus:outline-none focus-visible:ring-2 focus-visible:ring-amber-400"
   632→                    aria-label={isThisPlaying ? 'Pause' : 'Play'}
   633→                  >
   634→                    {isThisPlaying ? (
   635→                      <>
   636→                        <span className="inline-block w-2.5 h-3 bg-[#f5edd6] rounded-[1px]" />
   637→                        PAUSE
   638→                      </>
   639→                    ) : (
   640→                      <>
   641→                        <span className="inline-block w-0 h-0 border-l-[6px] border-l-[#f5edd6] border-t-[4px] border-t-transparent border-b-[4px] border-b-transparent" />
   642→                        PLAY
   643→                      </>
   644→                    )}
   645→                  </button>
   646→
   647→                  {/* Progress bar */}
   648→                  <div className="flex-1 h-1.5 bg-[#e0d5b8] rounded-full overflow-hidden">
   649→                    <div
   650→                      className="h-full bg-[#8b5e3c] rounded-full transition-all duration-100"
   651→                      style={{ width: isThisPlaying ? `${playProgress * 100}%` : '0%' }}
   652→                    />
   653→                  </div>
   654→
   655→                  {/* Delete */}
   656→                  <button
   657→                    onClick={() => deleteMessage(msg.id)}
   658→                    className="text-[#8b7355] hover:text-red-700 transition-colors opacity-60 group-hover:opacity-100 focus:outline-none focus-visible:ring-2 focus-visible:ring-red-400 rounded p-1"
   659→                    aria-label="Delete message"
   660→                  >
   661→                    <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   662→                      <polyline points="3 6 5 6 21 6" />
   663→                      <path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2" />
   664→                    </svg>
   665→                  </button>
   666→                </div>
   667→              </motion.div>
   668→            );
   669→          })}
   670→        </AnimatePresence>
   671→
   672→        {messages.length === 0 && status !== 'recording' && (
   673→          <p className="text-center old-body text-sm text-muted-foreground py-6 opacity-60">
   674→            No messages yet... press record to leave one! 🎙️
   675→          </p>
   676→        )}
   677→      </div>
   678→
   679→      {/* ── Inline Keyframes ── */}
   680→      <style jsx global>{`
   681→        @keyframes blink {
   682→          0%, 100% { opacity: 1; }
   683→          50% { opacity: 0.15; }
   684→        }
   685→        @keyframes pulse-ring {
   686→          0% { transform: scale(1); opacity: 0.4; }
   687→          50% { transform: scale(1.15); opacity: 0.1; }
   688→          100% { transform: scale(1); opacity: 0.4; }
   689→        }
   690→      `}</style>
   691→    </section>
   692→  );
