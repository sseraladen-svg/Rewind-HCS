   687→export default function Home() {
   688→  const [birthYear, setBirthYear] = useState<number | null>(null);
   689→  const [loading, setLoading] = useState(true);
   690→
   691→  // Run one-time localStorage → IndexedDB migration on first load
   692→  useEffect(() => {
   693→    if (typeof window !== 'undefined') {
   694→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   695→    }
   696→  }, []);
   697→
   698→  const handleLoadingDone = useCallback(() => {
   699→    setLoading(false);
   700→  }, []);
   701→
   702→  const handleSubmit = useCallback((year: number) => {
   703→    triggerOldLoader('Tuning into your era...');
   704→    setTimeout(() => {
   705→      setBirthYear(year);
   706→      window.scrollTo({ top: 0, behavior: 'smooth' });
   707→      hideOldLoader();
   708→    }, 1100);
   709→  }, []);
   710→
   711→  const handleBack = useCallback(() => {
   712→    triggerOldLoader('Rewinding...');
   713→    setTimeout(() => {
   714→      setBirthYear(null);
   715→      hideOldLoader();
   716→    }, 850);
   717→  }, []);
   718→
   719→  return (
   720→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   721→      {/* CRT Boot Loading Screen */}
   722→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   723→
   724→      <AnimatePresence mode="wait">
   725→        {birthYear === null ? (
   726→          <motion.div
   727→            key="hero"
   728→            initial={{ opacity: 0 }}
   729→            animate={{ opacity: 1 }}
   730→            exit={{ opacity: 0, y: -20 }}
   731→            transition={{ duration: 0.4 }}
   732→            style={{ opacity: loading ? 0 : 1 }}
   733→          >
   734→            <HeroSection onSubmit={handleSubmit} />
   735→          </motion.div>
   736→        ) : (
   737→          <motion.div
   738→            key="dashboard"
   739→            initial={{ opacity: 0, y: 20 }}
   740→            animate={{ opacity: 1, y: 0 }}
   741→            exit={{ opacity: 0 }}
   742→            transition={{ duration: 0.5 }}
   743→            className="flex flex-col min-h-screen"
   744→          >
   745→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   746→          </motion.div>
   747→        )}
   748→      </AnimatePresence>
   749→    </div>
   750→  );
   751→}
