   121→   121→// useStoredState — localStorage-backed state via useSyncExternalStore
   122→   122→// ═══════════════════════════════════════════════════════════════
   123→   123→
   124→   124→function useStoredState<T>(
   125→   125→  key: string,
   126→   126→  initial: T,
   127→   127→): [T, (updater: T | ((prev: T) => T)) => void] {
   128→   128→  const subscribe = useCallback(
   129→   129→    (onStoreChange: () => void) => {
   130→   130→      const handler = (e: StorageEvent) => {
   131→   131→        if (e.key === key || e.key === null) onStoreChange();
   132→   132→      };
   133→   133→      window.addEventListener('storage', handler);
   134→   134→      return () => window.removeEventListener('storage', handler);
   135→   135→    },
   136→   136→    [key],
   137→   137→  );
   138→   138→
   139→   139→  const getSnapshot = useCallback(() => {
   140→   140→    return localStorage.getItem(key);
   141→   141→  }, [key]);
   142→   142→
   143→   143→  const getServerSnapshot = useCallback(() => null, []);
   144→   144→
   145→   145→  const raw = useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
   146→   146→
   147→   147→  const value = useMemo<T>(() => {
   148→   148→    if (raw === null) return initial;
   149→   149→    try {
   150→   150→      return JSON.parse(raw) as T;
   151→   151→    } catch {
   152→   152→      return initial;
   153→   153→    }
   154→   154→  }, [raw, initial]);
   155→   155→
   156→   156→  const setValue = useCallback(
   157→   157→    (updater: T | ((prev: T) => T)) => {
   158→   158→      try {
   159→   159→        const currentRaw = localStorage.getItem(key);
   160→   160→        const current: T =
   161→   161→          currentRaw !== null ? (JSON.parse(currentRaw) as T) : initial;
   162→   162→        const next =
   163→   163→          typeof updater === 'function'
   164→   164→            ? (updater as (prev: T) => T)(current)
   165→   165→            : updater;
   166→   166→        localStorage.setItem(key, JSON.stringify(next));
   167→   167→        window.dispatchEvent(new StorageEvent('storage', { key }));
   168→   168→      } catch {
   169→   169→        /* ignore storage errors */
   170→   170→      }
   171→   171→    },
   172→   172→    [key, initial],
   173→   173→  );
   174→   174→
   175→   175→  return [value, setValue];
   176→   176→}

   179→   179→// useVisitCounter — auto-incrementing visit counter
   180→   180→// ═══════════════════════════════════════════════════════════════
   181→   181→
   182→   182→// Only called when isClient is true, so localStorage is available.
   183→   183→// The useState lazy initializer runs exactly once on first mount.
   184→   184→function useVisitCounter(): number {
   185→   185→  const [count] = useState(() => {
   186→   186→    const stored = localStorage.getItem('rewind_visit_count');
   187→   187→    let c = 1;
   188→   188→    if (stored) {
   189→   189→      const parsed = parseInt(stored, 10);
   190→   190→      if (!isNaN(parsed)) c = parsed + 1;
   191→   191→    }
   192→   192→    localStorage.setItem('rewind_visit_count', String(c));
   193→   193→    return c;
   194→   194→  });
   195→   195→  return count;
   196→   196→}
