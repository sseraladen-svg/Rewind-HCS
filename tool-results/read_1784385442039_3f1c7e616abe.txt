    22→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    23→  const [visibleLines, setVisibleLines] = useState(0);
    24→  const [fadingOut, setFadingOut] = useState(false);
    25→  const [showBar, setShowBar] = useState(false);
    26→
    27→  const bootLines = useMemo(() => [
    28→    '> Initializing nostalgia buffer...',
    29→    '> Loading childhood memories...',
    30→    '> Dusting off the VHS tapes...',
    31→    '> Warming up the CRT display...',
    32→    '> Searching for good times...',
    33→    '> Indexing mixtape tracks...',
    34→    '> Polishing polaroid photos...',
    35→    '> Rewinding cassette deck...',
    36→    '> Ready.',
    37→  ], []);
    38→
    39→  useEffect(() => {
    40→    const timers: ReturnType<typeof setTimeout>[] = [];
    41→
    42→    // Reveal boot lines one by one at 500ms intervals
    43→    bootLines.forEach((_, i) => {
    44→      timers.push(setTimeout(() => {
    45→        setVisibleLines(i + 1);
    46→      }, 500 * (i + 1)));
    47→    });
    48→
    49→    // Show progress bar after all lines
    50→    const barStart = 500 * (bootLines.length + 1);
    51→    timers.push(setTimeout(() => setShowBar(true), barStart));
    52→
    53→    // Fade out (CRT turn-off) after bar fills (3s animation)
    54→    const fadeStart = barStart + 3200;
    55→    timers.push(setTimeout(() => setFadingOut(true), fadeStart));
    56→    timers.push(setTimeout(onDone, fadeStart + 1000));
    57→
    58→    return () => timers.forEach(clearTimeout);
    59→  }, [bootLines, onDone]);
    60→
    61→  return (
    62→    <div
    63→      className={`fixed inset-0 z-[9999] flex flex-col items-center justify-center ${fadingOut ? 'crt-fadeout' : ''}`}
    64→      style={{
    65→        background: '#1a1208',
    66→      }}
    67→    >
    68→      {/* CRT scanlines overlay */}
    69→      <div className="absolute inset-0 pointer-events-none" style={{
    70→        background: 'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(0,0,0,0.06) 2px, rgba(0,0,0,0.06) 4px)',
    71→      }} />
    72→
    73→      {/* CRT screen glow */}
    74→      <div className="absolute inset-0 pointer-events-none" style={{
    75→        background: 'radial-gradient(ellipse at center, rgba(196,149,106,0.03) 0%, transparent 70%)',
    76→      }} />
    77→
    78→      {/* Boot text lines */}
    79→      <div className="relative z-10 flex flex-col gap-2 px-8 max-w-md">
    80→        {bootLines.slice(0, visibleLines).map((line, i) => (
    81→          <div
    82→            key={i}
    83→            className="boot-line"
    84→            style={{ animationDelay: '0s' }}
    85→          >
    86→            {line}
    87→            {i === visibleLines - 1 && !showBar && (
    88→              <span className="inline-block w-[2px] h-[1em] bg-[#c4956a]/70 ml-0.5 align-middle" style={{ animation: 'blink-cursor 0.6s step-end infinite' }} />
    89→            )}
    90→          </div>
    91→        ))}
    92→      </div>
    93→
    94→      {/* Loading bar */}
    95→      {showBar && (
    96→        <div className="relative z-10 mt-8">
    97→          <div className="loading-bar-track">
    98→            <div className="loading-bar-fill" />
    99→          </div>
   100→        </div>
   101→      )}
   102→    </div>
   103→  );
   104→}

   155→interface MatchCard {
   156→  id: number;
   157→  emoji: string;
   158→  label: string;
   159→  isFlipped: boolean;
   160→  isMatched: boolean;

   162→
   163→function shuffle<T>(arr: T[]): T[] {
   164→  const a = [...arr];
   165→  for (let i = a.length - 1; i > 0; i--) {
   166→    const j = Math.floor(Math.random() * (i + 1));
   167→    [a[i], a[j]] = [a[j], a[i]];
   168→  }
   169→  return a;
   170→}
   171→
   172→function createMatchCards(data: DecadeData): MatchCard[] {
   173→  const pool = [...data.cartoons, ...data.toys, ...data.snacks, ...data.games];
   174→  const picked = shuffle(pool).slice(0, 6);
   175→  const initial: MatchCard[] = picked.flatMap((item, i) => [
   176→    { id: i * 2, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   177→    { id: i * 2 + 1, emoji: item.emoji, label: item.title, isFlipped: false, isMatched: false },
   178→  ]);
   179→  return shuffle(initial);
   180→}

   182→function MemoryMatchGame({ data }: { data: DecadeData }) {
   183→  const [cards, setCards] = useState<MatchCard[]>([]);
   184→  const [flippedIds, setFlippedIds] = useState<number[]>([]);
   185→  const [moves, setMoves] = useState(0);
   186→  const [matches, setMatches] = useState(0);
   187→  const [gameWon, setGameWon] = useState(false);
   188→  const lockRef = useRef(false);
   189→
   190→  useEffect(() => {
   191→    const id = requestAnimationFrame(() => setCards(createMatchCards(data)));
   192→    return () => cancelAnimationFrame(id);
   193→  }, [data]);
   194→
   195→  const resetGame = useCallback(() => {
   196→    setCards(createMatchCards(data));
   197→    setFlippedIds([]);
   198→    setMoves(0);
   199→    setMatches(0);
   200→    setGameWon(false);
   201→    lockRef.current = false;
   202→  }, [data]);
   203→
   204→  const handleFlip = useCallback((id: number) => {
   205→    if (lockRef.current) return;
   206→    setCards(prev => {
   207→      const card = prev.find(c => c.id === id);
   208→      if (!card || card.isFlipped || card.isMatched) return prev;
   209→      return prev.map(c => c.id === id ? { ...c, isFlipped: true } : c);
   210→    });
   211→
   212→    setFlippedIds(prev => {
   213→      if (prev.includes(id) || prev.length >= 2) return prev;
   214→      const newFlipped = [...prev, id];
   215→
   216→      if (newFlipped.length === 2) {
   217→        setMoves(m => m + 1);
   218→        lockRef.current = true;
   219→
   220→        // Use a timeout to read the latest cards state
   221→        setTimeout(() => {
   222→          setCards(currentCards => {
   223→            const first = currentCards.find(c => c.id === newFlipped[0])!;
   224→            const second = currentCards.find(c => c.id === newFlipped[1])!;
   225→
   226→            if (first.label === second.label) {
   227→              // Match!
   228→              const updated = currentCards.map(c =>
   229→                c.id === first.id || c.id === second.id
   230→                  ? { ...c, isMatched: true, isFlipped: true }
   231→                  : c
   232→              );
   233→              setMatches(m => {
   234→                const newM = m + 1;
   235→                if (newM === 6) setGameWon(true);
   236→                return newM;
   237→              });
   238→              setFlippedIds([]);
   239→              lockRef.current = false;
   240→              return updated;
   241→            } else {
   242→              // No match — flip back
   243→              setTimeout(() => {
   244→                setCards(cc => cc.map(c =>
   245→                  c.id === first.id || c.id === second.id
   246→                    ? { ...c, isFlipped: false }
   247→                    : c
   248→                ));
   249→                setFlippedIds([]);
   250→                lockRef.current = false;
   251→              }, 700);
   252→              return currentCards;
   253→            }
   254→          });
   255→        }, 50);
   256→      }
   257→
   258→      return newFlipped;
   259→    });
   260→  }, []);
   261→
   262→  return (
   263→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   264→      <div className="flex items-center justify-between mb-4">
   265→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   266→          <span className="text-xl">🃏</span> Memory Match
   267→        </h4>
   268→        <div className="flex items-center gap-3 text-xs font-semibold text-[#8b7355]">
   269→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Moves: {moves}</span>
   270→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Pairs: {matches}/6</span>
   271→          <button onClick={resetGame} className="sk-btn old-btn px-3 py-1 rounded text-xs cursor-pointer">
   272→            <RotateCcw className="h-3 w-3 inline mr-1" />Reset
   273→          </button>
   274→        </div>
   275→      </div>
   276→
   277→      {gameWon && (
   278→        <motion.div
   279→          initial={{ scale: 0.8, opacity: 0 }}
   280→          animate={{ scale: 1, opacity: 1 }}
   281→          className="absolute inset-0 bg-[#faf6eb]/95 z-10 flex flex-col items-center justify-center rounded-lg"
   282→        >
   283→          <span className="text-5xl mb-2">🎉</span>
   284→          <p className="old-title text-lg font-bold text-[#5c4033]">You found them all!</p>
   285→          <p className="old-body text-sm text-[#8b7355] mt-1">Completed in {moves} moves</p>
   286→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-3 cursor-pointer">
   287→            Play Again
   288→          </button>
   289→        </motion.div>
   290→      )}
   291→
   292→      <div className="grid grid-cols-4 gap-2">
   293→        {cards.map((card) => {
   294→          const showFront = card.isFlipped || card.isMatched;
   295→          return (
   296→            <button
   297→              key={card.id}
   298→              onClick={() => handleFlip(card.id)}
   299→              className="aspect-square rounded-md transition-all duration-300 cursor-pointer border-2 flex flex-col items-center justify-center p-1"
   300→              disabled={card.isMatched}
   301→              style={{
   302→                background: showFront
   303→                  ? '#faf6eb'
   304→                  : 'linear-gradient(135deg, #7a4f30, #8b5e3c, #6b4226)',
   305→                borderColor: showFront
   306→                  ? (card.isMatched ? '#a8d08d' : data.color + '80')

   308→                boxShadow: showFront

   310→                  : '0 3px 8px rgba(0,0,0,0.25), inset 0 1px 0 rgba(255,255,255,0.1)',
   311→                opacity: card.isMatched ? 0.6 : 1,
   312→              }}
   313→            >

   330→/* ═══════════════════════════════════════════════════

   333→interface TriviaQ {
   334→  question: string;
   335→  emoji: string;
   336→  correct: number;
   337→  options: string[];
   338→}
   339→
   340→function buildTrivia(data: DecadeData): TriviaQ[] {
   341→  const decadeStart = parseInt(data.years.split('–')[0]);
   342→  const qs: TriviaQ[] = [];
   343→  const all = [...data.cartoons, ...data.toys, ...data.music, ...data.movies, ...data.snacks, ...data.games, ...data.tech, ...data.trends];
   344→  const shuffled = shuffle(all).slice(0, 8);
   345→
   346→  for (const item of shuffled) {
   347→    const realYear = decadeStart + Math.floor(Math.random() * 8) + 1;
   348→    const wrongYears = [
   349→      Math.max(1970, realYear - 3 - Math.floor(Math.random() * 5)),
   350→      Math.min(2019, realYear + 2 + Math.floor(Math.random() * 5)),
   351→      Math.max(1970, realYear - 6 - Math.floor(Math.random() * 4)),
   352→    ].filter(y => y !== realYear).slice(0, 3);
   353→    while (wrongYears.length < 3) wrongYears.push(Math.max(1970, realYear + 7));
   354→    const options = shuffle([realYear, ...wrongYears.slice(0, 3)]);
   355→    qs.push({
   356→      question: `When did "${item.title}" first become popular?`,
   357→      emoji: item.emoji,
   358→      correct: options.indexOf(realYear),
   359→      options: options.map(String),
   360→    });
   361→  }
   362→  return qs;
   363→}

   365→function GuessTheYearGame({ data }: { data: DecadeData }) {
   366→  const [questions, setQuestions] = useState<TriviaQ[]>([]);
   367→  const [qIdx, setQIdx] = useState(0);
   368→  const [score, setScore] = useState(0);
   369→  const [selected, setSelected] = useState<number | null>(null);
   370→  const [showResult, setShowResult] = useState(false);
   371→  const [done, setDone] = useState(false);
   372→
   373→  useEffect(() => {
   374→    setQuestions(buildTrivia(data));
   375→  }, [data]);
   376→
   377→  const q = questions[qIdx];
   378→
   379→  const handleAnswer = (optIdx: number) => {
   380→    if (showResult || !q) return;
   381→    setSelected(optIdx);
   382→    setShowResult(true);
   383→    if (optIdx === q.correct) setScore(s => s + 1);
   384→    setTimeout(() => {
   385→      if (qIdx + 1 < questions.length) {
   386→        setQIdx(i => i + 1);
   387→        setSelected(null);
   388→        setShowResult(false);
   389→      } else {
   390→        setDone(true);
   391→      }
   392→    }, 1500);
   393→  };
   394→
   395→  const resetGame = () => {
   396→    const fresh = buildTrivia(data);
   397→    setQuestions(fresh);
   398→    setQIdx(0); setScore(0); setSelected(null); setShowResult(false); setDone(false);
   399→  };
   400→
   401→  if (!q && !done) return <div className="sk-paper p-5 rounded-lg sk-noise text-center old-typewriter text-sm text-[#8b7355]">Loading...</div>;
   402→
   403→  if (done) {
   404→    const pct = questions.length > 0 ? Math.round((score / questions.length) * 100) : 0;
   405→    return (
   406→      <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   407→        <div className="text-center py-4">
   408→          <span className="text-5xl mb-3 block">{pct >= 70 ? '🏆' : pct >= 40 ? '👏' : '😅'}</span>
   409→          <p className="old-title text-lg font-bold text-[#5c4033]">Game Over!</p>
   410→          <p className="old-body text-sm text-[#8b7355] mt-1">You scored {score}/{questions.length} ({pct}%)</p>
   411→          <p className="old-quote text-xs text-[#8b7355] mt-2">
   412→            {pct >= 70 ? "You truly remember your era!" : pct >= 40 ? "Not bad — some things fade with time" : "Hey, it's been a while! No shame in that."}
   413→          </p>
   414→          <button onClick={resetGame} className="sk-btn old-btn px-5 py-2 rounded-lg font-semibold mt-4 cursor-pointer">
   415→            <RotateCcw className="h-3 w-3 inline mr-1" />Play Again
   416→          </button>
   417→        </div>
   418→      </div>
   419→    );
   420→  }
   421→
   422→  return (
   423→    <div className="sk-paper p-5 rounded-lg sk-noise relative overflow-hidden">
   424→      <div className="flex items-center justify-between mb-4">
   425→        <h4 className="old-heading font-bold text-base text-[#5c4033] flex items-center gap-2">
   426→          <span className="text-xl">📅</span> Guess the Year
   427→        </h4>
   428→        <div className="flex items-center gap-2 text-xs font-semibold text-[#8b7355]">
   429→          <span className="sk-inset px-2 py-1 rounded old-typewriter">{qIdx + 1}/{questions.length}</span>
   430→          <span className="sk-inset px-2 py-1 rounded old-typewriter">Score: {score}</span>
   431→        </div>
   432→      </div>
   433→
   434→      {/* Progress bar */}
   435→      <div className="w-full h-2 sk-wood rounded-full mb-4 overflow-hidden">
   436→        <div
   437→          className="h-full bg-gradient-to-r from-[#8b5e3c] to-[#c4956a] rounded-full transition-all duration-500"
   438→          style={{ width: `${((qIdx) / questions.length) * 100}%` }}
   439→        />
   440→      </div>
   441→
   442→      <div className="text-center mb-4">
   443→        <span className="text-5xl block mb-2">{q.emoji}</span>
   444→        <p className="old-body text-sm font-semibold text-[#5c4033]">{q.question}</p>
   445→      </div>
   446→
   447→      <div className="grid grid-cols-2 gap-2">
   448→        {q.options.map((opt, i) => {
   449→          let cls = 'sk-paper p-3 rounded-lg text-center font-bold text-sm cursor-pointer transition-all duration-200 border-2 old-typewriter ';
   450→          if (showResult) {
   451→            if (i === q.correct) cls += 'border-green-500 bg-green-50 sk-sticker';
   452→            else if (i === selected) cls += 'border-red-400 bg-red-50 opacity-60';
   453→            else cls += 'border-[#d4c5a9] opacity-40';
   454→          } else {
   455→            cls += 'border-[#d4c5a9] hover:border-[#8b5e3c] hover:shadow-md active:translate-y-0.5';
   456→          }
   457→          return (
   458→            <button key={`${qIdx}-${i}`} onClick={() => handleAnswer(i)} className={cls} disabled={showResult}>
   459→              {opt}
   460→            </button>
   461→          );
   462→        })}
   463→      </div>
   464→
   465→      {showResult && (
   466→        <motion.div
   467→          initial={{ opacity: 0, y: 5 }}
   468→          animate={{ opacity: 1, y: 0 }}
   469→          className="text-center mt-3 text-xs font-semibold flex items-center justify-center gap-1 old-body"
   470→        >
   471→          {selected === q.correct ? (
   472→            <><CheckCircle2 className="h-4 w-4 text-green-600" /><span className="text-green-700">Correct! You remembered!</span></>
   473→          ) : (
   474→            <><XCircle className="h-4 w-4 text-red-500" /><span className="text-red-600">It was {q.options[q.correct]}</span></>
   475→          )}
   476→        </motion.div>
   477→      )}
   478→    </div>
   479→  );
   480→}

   505→    if (empty.length === 0) return;

   509→    newGrid[pos] = item;
   510→    gridRef.current = newGrid;
   511→    setGrid([...newGrid]);
   512→    setTimeout(() => {
   513→      if (gridRef.current[pos] === item) {
   514→        const g = [...gridRef.current]; g[pos] = null; gridRef.current = g; setGrid([...g]);

   516→    }, 1200);
   517→  }, [allItems]);
