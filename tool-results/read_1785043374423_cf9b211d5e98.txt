     1→'use client';
     2→
     3→import { useState, useRef, useCallback, useEffect } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Camera, Video, X, Download, Trash2, Image as ImageIcon,
     7→  RotateCcw, SwitchCamera, Zap, Circle, Square, Play,
     8→} from 'lucide-react';
     9→import {
    10→  PIPELINE_PRESETS,
    11→  processFrame,
    12→  processPreviewFrame,
    13→  getGateFlickerEV,
    14→  type PipelineConfig,
    15→} from '@/lib/image-pipeline';
    16→import { getAllCameraPhotos, addCameraPhoto, deleteCameraPhoto } from '@/lib/idb-store';
    17→
    18→/* ═══════ TYPES ═══════ */
    19→interface CapturedItem {
    20→  id: string;
    21→  type: 'photo' | 'video';
    22→  url: string;
    23→  timestamp: number;
    24→  blob?: Blob;
    25→}
    26→
    27→type FilterPreset = 'none' | 'sepia70s' | 'faded80s' | 'warm90s' | 'bw60s' | 'polaroid' | 'vhs' | 'kodak';
    28→
    29→/* Real-time preview resolution (low for performance) */
    30→const PREVIEW_W = 360;
    31→const PREVIEW_H = 202;
    32→const TARGET_FPS = 24;
    33→const FRAME_INTERVAL = 1000 / TARGET_FPS;
    34→
    35→/* ═══════ VINTAGE CAMERA COMPONENT ═══════ */
    36→export default function VintageCamera({ onClose }: { onClose: () => void }) {
    37→  const videoRef = useRef<HTMLVideoElement>(null);
    38→  const captureCanvasRef = useRef<HTMLCanvasElement>(null);   // full-res for capture
    39→  const previewCanvasRef = useRef<HTMLCanvasElement>(null);   // live processed preview
    40→  const offscreenRef = useRef<HTMLCanvasElement | null>(null);
    41→  const streamRef = useRef<MediaStream | null>(null);
    42→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
    43→  const recordedChunksRef = useRef<Blob[]>([]);
    44→  const flashRef = useRef<HTMLDivElement>(null);
    45→  const rafRef = useRef<number>(0);
    46→  const lastFrameTimeRef = useRef<number>(0);
    47→  const frameCountRef = useRef<number>(0);
    48→  const flickerEvRef = useRef<number>(0);
    49→
    50→  const [cameraOpen, setCameraOpen] = useState(false);
    51→  const [facingMode, setFacingMode] = useState<'user' | 'environment'>('environment');
    52→  const [mode, setMode] = useState<'photo' | 'video'>('photo');
    53→  const [filter, setFilter] = useState<FilterPreset>('sepia70s');
    54→  const [recording, setRecording] = useState(false);
    55→  const [recordTime, setRecordTime] = useState(0);
    56→  const [flash, setFlash] = useState(false);
    57→  const [gallery, setGallery] = useState<CapturedItem[]>([]);
    58→  const [viewItem, setViewItem] = useState<CapturedItem | null>(null);
    59→  const [cameraError, setCameraError] = useState<string | null>(null);
    60→  const [showFilters, setShowFilters] = useState(false);
    61→  const [shutterAnim, setShutterAnim] = useState(false);
    62→  const [processing, setProcessing] = useState(false);
    63→
    64→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    65→  const dbObjectUrlsRef = useRef<string[]>([]);
    66→
    67→  /* ── Load persisted photos from IndexedDB on mount ── */
    68→  useEffect(() => {
    69→    let cancelled = false;
    70→    getAllCameraPhotos().then((photos) => {
    71→      if (cancelled) return;
    72→      const items: CapturedItem[] = photos.map((photo) => {
    73→        const url = URL.createObjectURL(photo.image_blob);
    74→        dbObjectUrlsRef.current.push(url);
    75→        return {
    76→          id: photo.id,
    77→          type: 'photo' as const,
    78→          url,
    79→          timestamp: photo.created_at,
    80→          blob: photo.image_blob,
    81→        };
    82→      });
    83→      if (items.length > 0) {
    84→        setGallery((prev) => {
    85→          // Avoid duplicates (e.g., if React StrictMode double-fires)
    86→          const existingIds = new Set(prev.map((i) => i.id));
    87→          const newItems = items.filter((i) => !existingIds.has(i.id));
    88→          return [...newItems, ...prev];
    89→        });
    90→      }
    91→    });
    92→    return () => { cancelled = true; };
    93→  }, []);
    94→
    95→  /* ── Get current preset config ── */
    96→  const currentConfig: PipelineConfig = PIPELINE_PRESETS[filter] || PIPELINE_PRESETS['none'];
    97→
    98→  /* ── Start Camera ── */
    99→  const startCamera = useCallback(async () => {
   100→    try {
   101→      if (streamRef.current) {
   102→        streamRef.current.getTracks().forEach(t => t.stop());
   103→      }
   104→      const stream = await navigator.mediaDevices.getUserMedia({
   105→        video: { facingMode, width: { ideal: 1280 }, height: { ideal: 720 } },
   106→        audio: mode === 'video',
   107→      });
   108→      streamRef.current = stream;
   109→      if (videoRef.current) {
   110→        videoRef.current.srcObject = stream;
   111→        await videoRef.current.play();
   112→      }
   113→      setCameraOpen(true);
   114→      setCameraError(null);
   115→    } catch (err) {
   116→      console.error('Camera error:', err);
   117→      setCameraError('Camera access denied or not available. Please allow camera permissions.');
   118→    }
   119→  }, [facingMode, mode]);
   120→
   121→  /* ── Stop Camera ── */
   122→  const stopCamera = useCallback(() => {
   123→    if (rafRef.current) {
   124→      cancelAnimationFrame(rafRef.current);
   125→      rafRef.current = 0;
   126→    }
   127→    if (streamRef.current) {
   128→      streamRef.current.getTracks().forEach(t => t.stop());
   129→      streamRef.current = null;
   130→    }
   131→    setCameraOpen(false);
   132→    if (recordTimerRef.current) {
   133→      clearInterval(recordTimerRef.current);
   134→      recordTimerRef.current = null;
   135→    }
   136→  }, []);
   137→
   138→  /* ── Switch Camera ── */
   139→  const handleSwitchCamera = useCallback(() => {
   140→    setFacingMode(prev => prev === 'user' ? 'environment' : 'user');
   141→  }, []);
   142→
   143→  /* ── Restart on facing/mode change ── */
   144→  useEffect(() => {
   145→    if (cameraOpen) {
   146→      startCamera();
   147→    }
   148→    return () => {};
   149→  }, [facingMode, mode, cameraOpen, startCamera]);
   150→
   151→  /* ── Real-time preview loop ── */
   152→  useEffect(() => {
   153→    if (!cameraOpen || !videoRef.current || !previewCanvasRef.current) return;
   154→
   155→    // Create offscreen canvas for drawing video frames
   156→    if (!offscreenRef.current) {
   157→      offscreenRef.current = document.createElement('canvas');
   158→    }
   159→    offscreenRef.current.width = PREVIEW_W;
   160→    offscreenRef.current.height = PREVIEW_H;
   161→    const offCtx = offscreenRef.current.getContext('2d', { willReadFrequently: true });
   162→    if (!offCtx) return;
   163→
   164→    const previewCanvas = previewCanvasRef.current;
   165→    previewCanvas.width = PREVIEW_W;
   166→    previewCanvas.height = PREVIEW_H;
   167→    const previewCtx = previewCanvas.getContext('2d', { willReadFrequently: true });
   168→    if (!previewCtx) return;
   169→
   170→    let running = true;
   171→
   172→    const renderLoop = (timestamp: number) => {
   173→      if (!running) return;
   174→
   175→      const elapsed = timestamp - lastFrameTimeRef.current;
   176→      if (elapsed >= FRAME_INTERVAL) {
   177→        lastFrameTimeRef.current = timestamp - (elapsed % FRAME_INTERVAL);
   178→        frameCountRef.current++;
   179→
   180→        const video = videoRef.current;
   181→        if (!video || video.readyState < 2) {
   182→          rafRef.current = requestAnimationFrame(renderLoop);
   183→          return;
   184→        }
   185→
   186→        // Draw video frame to offscreen canvas (mirrored if front camera)
   187→        offCtx.save();
   188→        if (facingMode === 'user') {
   189→          offCtx.translate(PREVIEW_W, 0);
   190→          offCtx.scale(-1, 1);
   191→        }
   192→        offCtx.drawImage(video, 0, 0, PREVIEW_W, PREVIEW_H);
   193→        offCtx.restore();
   194→
   195→        // Apply lightweight preview pipeline
   196→        try {
   197→          processPreviewFrame(offCtx, PREVIEW_W, PREVIEW_H, filter, frameCountRef.current);
   198→
   199→          // Gate flicker for video mode
   200→          if (mode === 'video' && currentConfig.gateFlicker.enabled) {
   201→            flickerEvRef.current = getGateFlickerEV(currentConfig.gateFlicker.intensity);
   202→            const imgData = offCtx.getImageData(0, 0, PREVIEW_W, PREVIEW_H);
   203→            const d = imgData.data;
   204→            const mult = 1 + flickerEvRef.current * 0.5;
   205→            for (let i = 0; i < d.length; i += 4) {
   206→              d[i] = Math.max(0, Math.min(255, Math.round(d[i] * mult)));
   207→              d[i + 1] = Math.max(0, Math.min(255, Math.round(d[i + 1] * mult)));
   208→              d[i + 2] = Math.max(0, Math.min(255, Math.round(d[i + 2] * mult)));
   209→            }
   210→            offCtx.putImageData(imgData, 0, 0);
   211→          }
   212→        } catch {
   213→          // silently skip if processing errors (e.g., tab not visible)
   214→        }
   215→
   216→        // Copy to visible preview canvas
   217→        if (offscreenRef.current) {
   218→          previewCtx.drawImage(offscreenRef.current, 0, 0);
   219→        }
   220→      }
   221→
   222→      rafRef.current = requestAnimationFrame(renderLoop);
   223→    };
   224→
   225→    rafRef.current = requestAnimationFrame(renderLoop);
   226→
   227→    return () => {
   228→      running = false;
   229→      if (rafRef.current) {
   230→        cancelAnimationFrame(rafRef.current);
   231→        rafRef.current = 0;
   232→      }
   233→    };
   234→  }, [cameraOpen, filter, facingMode, mode, currentConfig.gateFlicker.enabled, currentConfig.gateFlicker.intensity]);
   235→
   236→  /* ── Capture Photo (Full Pipeline) ── */
   237→  const capturePhoto = useCallback(() => {
   238→    if (!videoRef.current || !captureCanvasRef.current) return;
   239→    const video = videoRef.current;
   240→    const canvas = captureCanvasRef.current;
   241→    const ctx = canvas.getContext('2d', { willReadFrequently: true });
   242→    if (!ctx) return;
   243→
   244→    setProcessing(true);
   245→
   246→    // Use full resolution
   247→    const w = video.videoWidth || 1280;
   248→    const h = video.videoHeight || 720;
   249→    canvas.width = w;
   250→    canvas.height = h;
   251→
   252→    // Mirror if front camera
   253→    if (facingMode === 'user') {
   254→      ctx.translate(w, 0);
   255→      ctx.scale(-1, 1);
   256→    }
   257→    ctx.drawImage(video, 0, 0, w, h);
   258→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   259→
   260→    // Apply full pipeline
   261→    try {
   262→      processFrame(ctx, w, h, filter, { addStamp: true });
   263→    } catch (e) {
   264→      console.error('Pipeline error:', e);
   265→    }
   266→
   267→    // Flash effect
   268→    setFlash(true);
   269→    setShutterAnim(true);
   270→    setTimeout(() => setFlash(false), 200);
   271→    setTimeout(() => setShutterAnim(false), 300);
   272→
   273→    // Save to gallery
   274→    canvas.toBlob((blob) => {
   275→      if (!blob) { setProcessing(false); return; }
   276→      const url = URL.createObjectURL(blob);
   277→      const id = `${Date.now()}-${Math.random().toString(36).slice(2)}`;
   278→      setGallery(prev => [{
   279→        id,
   280→        type: 'photo',
   281→        url,
   282→        timestamp: Date.now(),
   283→        blob,
   284→      }, ...prev]);
   285→      addCameraPhoto({ id, image_blob: blob, filter_used: filter, created_at: Date.now() });
   286→      setProcessing(false);
   287→    }, 'image/jpeg', 0.92);
   288→  }, [filter, facingMode]);
   289→
   290→  /* ── Start/Stop Video Recording ── */
   291→  const toggleRecording = useCallback(() => {
   292→    if (!recording && streamRef.current) {
   293→      const options = { mimeType: 'video/webm;codecs=vp9' };
   294→      const tryStart = (mimeType: string) => {
   295→        try {
   296→          const mediaRecorder = new MediaRecorder(streamRef.current!, { mimeType });
   297→          recordedChunksRef.current = [];
   298→          mediaRecorder.ondataavailable = (e) => {
   299→            if (e.data.size > 0) recordedChunksRef.current.push(e.data);
   300→          };
   301→          mediaRecorder.onstop = () => {
   302→            const blob = new Blob(recordedChunksRef.current, { type: 'video/webm' });
   303→            const url = URL.createObjectURL(blob);
   304→            setGallery(prev => [{
   305→              id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   306→              type: 'video',
   307→              url,
   308→              timestamp: Date.now(),
   309→              blob,
   310→            }, ...prev]);
   311→            recordedChunksRef.current = [];
   312→          };
   313→          mediaRecorder.start(100);
   314→          mediaRecorderRef.current = mediaRecorder;
   315→          setRecording(true);
   316→          setRecordTime(0);
   317→          recordTimerRef.current = setInterval(() => {
   318→            setRecordTime(prev => prev + 1);
   319→          }, 1000);
   320→        } catch {
   321→          if (mimeType.includes('vp9')) {
   322→            tryStart('video/webm;codecs=vp8');
   323→          } else {
   324→            setCameraError('Video recording not supported in this browser.');
   325→          }
   326→        }
   327→      };
   328→      tryStart(options.mimeType);
   329→    } else if (recording && mediaRecorderRef.current) {
   330→      mediaRecorderRef.current.stop();
   331→      mediaRecorderRef.current = null;
   332→      setRecording(false);
   333→      if (recordTimerRef.current) {
   334→        clearInterval(recordTimerRef.current);
   335→        recordTimerRef.current = null;
   336→      }
   337→    }
   338→  }, [recording]);
   339→
   340→  /* ── Delete from gallery ── */
   341→  const deleteItem = useCallback((id: string) => {
   342→    setGallery(prev => {
   343→      const item = prev.find(i => i.id === id);
   344→      if (item) {
   345→        URL.revokeObjectURL(item.url);
   346→        // Also remove from dbObjectUrlsRef tracking
   347→        dbObjectUrlsRef.current = dbObjectUrlsRef.current.filter(u => u !== item.url);
   348→      }
   349→      if (item?.type === 'photo') {
   350→        deleteCameraPhoto(id);
   351→      }
   352→      return prev.filter(i => i.id !== id);
   353→    });
   354→  }, []);
   355→
   356→  /* ── Download ── */
   357→  const downloadItem = useCallback((item: CapturedItem) => {
   358→    const a = document.createElement('a');
   359→    a.href = item.url;
   360→    a.download = `rewind_${item.type}_${new Date(item.timestamp).toISOString().replace(/[:.]/g, '-')}.${item.type === 'photo' ? 'jpg' : 'webm'}`;
   361→    document.body.appendChild(a);
   362→    a.click();
   363→    document.body.removeChild(a);
   364→  }, []);
   365→
   366→  /* ── Cleanup on close ── */
   367→  useEffect(() => {
   368→    return () => {
   369→      stopCamera();
   370→      gallery.forEach(item => URL.revokeObjectURL(item.url));
   371→      dbObjectUrlsRef.current.forEach(url => URL.revokeObjectURL(url));
   372→      dbObjectUrlsRef.current = [];
   373→    };
   374→  }, []);
   375→
   376→  /* ── Format time ── */
   377→  const formatTime = (s: number) => `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
   378→
   379→  /* ── Pipeline info for HUD ── */
   380→  const pipelineStages = [
   381→    currentConfig.chromaticAberration.enabled && 'CA',
   382→    currentConfig.toneCurve.blackLift > 0 && 'MATTE',
   383→    currentConfig.contrastReduce > 0 && 'FLAT',
   384→    currentConfig.antiClarity.enabled && 'SOFT',
   385→    currentConfig.grain.enabled && 'GRAIN',
   386→    currentConfig.halation.enabled && 'GLOW',
   387→    currentConfig.vignette.intensity > 0 && 'VIG',
   388→  ].filter(Boolean);
   389→
   390→  return (
   391→    <>
   392→      {/* Hidden canvases */}
   393→      <canvas ref={captureCanvasRef} className="hidden" />
   394→
   395→      <AnimatePresence>
   396→        <motion.div
   397→          initial={{ opacity: 0 }}
   398→          animate={{ opacity: 1 }}
   399→          exit={{ opacity: 0 }}
   400→          className="fixed inset-0 z-50 flex flex-col"
   401→          style={{ background: '#1a1208' }}
   402→        >
   403→          {/* ═══ HEADER - Camera Top Bar ═══ */}
   404→          <div className="sk-leather px-3 py-2.5 flex items-center justify-between z-10">
   405→            <button onClick={() => { stopCamera(); onClose(); }} className="sk-btn old-btn px-3 py-1.5 rounded-lg text-xs font-bold cursor-pointer flex items-center gap-1 text-[#faf6eb]">
   406→              <X className="h-4 w-4" /> Close
   407→            </button>
   408→            <h2 className="old-title font-black text-base text-[#faf6eb] tracking-wide" style={{ textShadow: '0 1px 2px rgba(0,0,0,0.3)' }}>
   409→              📷 Vintage Camera
   410→            </h2>
   411→            <div className="w-16" />
   412→          </div>
   413→
   414→          {/* ═══ VIEWFINDER ═══ */}
   415→          <div className="flex-1 relative overflow-hidden flex items-center justify-center">
   416→            {!cameraOpen && !cameraError && (
   417→              <div className="text-center px-6">
   418→                <motion.div
   419→                  animate={{ rotate: [0, 5, -5, 0] }}
   420→                  transition={{ duration: 4, repeat: Infinity }}
   421→                  className="text-6xl mb-4"
   422→                >📸</motion.div>
   423→                <p className="old-body text-[#d4a574] mb-2 text-sm">Open the camera to capture memories in vintage style</p>
   424→                <p className="old-typewriter text-[10px] text-[#8b7355] mb-6">
   425→                  Full image processing pipeline: chromatic aberration, tone curves, Perlin grain, halation glow
   426→                </p>
   427→                <button onClick={startCamera} className="sk-btn old-btn px-6 py-3 rounded-xl font-bold text-base cursor-pointer flex items-center gap-2 mx-auto">
   428→                  <Camera className="h-5 w-5" /> Open Camera
   429→                </button>
   430→              </div>
   431→            )}
   432→
   433→            {cameraError && (
   434→              <div className="text-center px-6">
   435→                <div className="text-5xl mb-4">😵</div>
   436→                <p className="old-body text-[#d4a574] mb-4 text-sm">{cameraError}</p>
   437→                <button onClick={startCamera} className="sk-btn old-btn px-5 py-2.5 rounded-lg font-bold cursor-pointer flex items-center gap-2 mx-auto">
   438→                  <RotateCcw className="h-4 w-4" /> Try Again
   439→                </button>
   440→              </div>
   441→            )}
   442→
   443→            {cameraOpen && (
   444→              <>
   445→                {/* Hidden video source (never shown directly) */}
   446→                <video
   447→                  ref={videoRef}
   448→                  className="hidden"
   449→                  playsInline
   450→                  muted
   451→                />
   452→
   453→                {/* Real-time processed canvas preview */}
   454→                <canvas
   455→                  ref={previewCanvasRef}
   456→                  className="absolute inset-0 w-full h-full object-cover"
   457→                  style={{
   458→                    imageRendering: 'auto', // smooth upscaling for that vintage soft look
   459→                  }}
   460→                />
   461→
   462→                {/* Halation glow overlay (CSS-based, adds dreamy ambient light) */}
   463→                {currentConfig.halation.enabled && (
   464→                  <div
   465→                    className="absolute inset-0 pointer-events-none mix-blend-screen"
   466→                    style={{
   467→                      background: `radial-gradient(ellipse at 50% 50%, rgba(255,240,200,${currentConfig.halation.intensity * 0.06}) 0%, transparent 70%)`,
   468→                    }}
   469→                  />
   470→                )}
   471→
   472→                {/* Scanline overlay (VHS era feel) */}
   473→                <div className="absolute inset-0 pointer-events-none opacity-[0.07]" style={{
   474→                  background: 'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(0,0,0,0.2) 2px, rgba(0,0,0,0.2) 4px)',
   475→                }} />
   476→
   477→                {/* Film border overlay - viewfinder frame */}
   478→                <div className="absolute inset-0 pointer-events-none border-[6px] border-[#1a1208] rounded-none" />
   479→
   480→                {/* Corner brackets */}
   481→                <div className="absolute top-3 left-3 w-8 h-8 border-t-2 border-l-2 border-[#d4a574]/50 pointer-events-none" />
   482→                <div className="absolute top-3 right-3 w-8 h-8 border-t-2 border-r-2 border-[#d4a574]/50 pointer-events-none" />
   483→                <div className="absolute bottom-3 left-3 w-8 h-8 border-b-2 border-l-2 border-[#d4a574]/50 pointer-events-none" />
   484→                <div className="absolute bottom-3 right-3 w-8 h-8 border-b-2 border-r-2 border-[#d4a574]/50 pointer-events-none" />
   485→
   486→                {/* Center crosshair */}
   487→                <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 pointer-events-none">
   488→                  <div className="w-px h-6 bg-[#d4a574]/30" />
   489→                  <div className="h-px w-6 bg-[#d4a574]/30 absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />
   490→                </div>
   491→
   492→                {/* Recording indicator */}
   493→                {recording && (
   494→                  <motion.div
   495→                    initial={{ opacity: 0, scale: 0.8 }}
   496→                    animate={{ opacity: 1, scale: 1 }}
   497→                    className="absolute top-4 left-1/2 -translate-x-1/2 flex items-center gap-2 sk-vhs-label px-3 py-1 rounded-full z-10"
   498→                  >
   499→                    <motion.div
   500→                      animate={{ opacity: [1, 0.3, 1] }}
   501→                      transition={{ duration: 1, repeat: Infinity }}
   502→                      className="w-2.5 h-2.5 rounded-full bg-red-500"
   503→                    />
   504→                    <span className="old-typewriter text-xs font-bold text-red-600">REC {formatTime(recordTime)}</span>
   505→                  </motion.div>
   506→                )}
   507→
   508→                {/* Filter label + Pipeline HUD */}
   509→                <div className="absolute top-4 right-4 z-10 flex flex-col items-end gap-1">
   510→                  <div className="sk-vhs-label px-2 py-1 rounded">
   511→                    <span className="old-typewriter text-[10px] font-bold text-[#5c4033]">
   512→                      {currentConfig.emoji} {currentConfig.name}
   513→                    </span>
   514→                  </div>
   515→                  {/* Pipeline stages HUD */}
   516→                  {pipelineStages.length > 0 && (
   517→                    <div className="sk-vhs-label px-2 py-0.5 rounded opacity-70">
