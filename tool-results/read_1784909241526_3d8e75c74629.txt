    33→function CrtLoadingScreen({ onDone }: { onDone: () => void }) {
    34→  const [phase, setPhase] = useState(0); // 0=boot, 1=title, 2=fading
    35→  const [bootLines, setBootLines] = useState<string[]>([]);
    36→  const fadeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    37→
    38→  const allBootLines = useMemo(() => [
    39→    'BIOS v2.4 — Nostalgia Systems Inc.',
    40→    'POST: Memory test... 640K OK',
    41→    'Detecting floppy drives... A: OK  B: OK',
    42→    'Loading REWIND.EXE...',
    43→    'Mounting childhood memories... 47% complete',
    44→    'Indexing nostalgia database...',
    45→    'Rewinding time...',
    46→    'Loading good old days... 83% complete',
    47→    'Ready.',
    48→  ], []);
    49→
    50→  useEffect(() => {
    51→    const timers: ReturnType<typeof setTimeout>[] = [];
    52→    // Reveal boot lines one by one
    53→    allBootLines.forEach((line, i) => {
    54→      timers.push(setTimeout(() => {
    55→        setBootLines(prev => [...prev, line]);
    56→        if (i === allBootLines.length - 1) {
    57→          // Last line — show title after a beat
    58→          timers.push(setTimeout(() => setPhase(1), 1000));
    59→        }
    60→      }, 500 + i * 500));
    61→    });
    62→
    63→    // Auto-dismiss after title phase
    64→    fadeTimerRef.current = setTimeout(() => {
    65→      setPhase(2);
    66→      setTimeout(onDone, 800);
    67→    }, 7000);
    68→
    69→    return () => {
    70→      timers.forEach(clearTimeout);
    71→      if (fadeTimerRef.current) clearTimeout(fadeTimerRef.current);
    72→    };
    73→  }, [allBootLines, onDone]);
    74→
    75→  return (
    76→    <div className={`crt-screen ${phase === 2 ? 'crt-fadeout' : 'crt-flicker'}`}>
    77→      <div className="crt-scanlines" />
    78→      <div className="vhs-noise" />
    79→      <div className="vhs-tracking" />
    80→      <div className="vhs-tracking" style={{ animationDelay: '1.2s', opacity: 0.5 }} />
    81→
    82→      <div className="relative z-10 flex flex-col items-center gap-4 px-4 w-full max-w-xl">
    83→        {/* Boot lines */}
    84→        <div className="space-y-1 sm:space-y-1.5 min-h-[120px] sm:min-h-[180px] w-full">
    85→          {bootLines.map((line, i) => (
    86→            <div
    87→              key={i}
    88→              className="boot-line text-[10px] sm:text-xs md:text-[0.75rem]"
    89→              style={{ animationDelay: '0s' }}
    90→            >
    91→              {line}
    92→              {i === bootLines.length - 1 && (
    93→                <span className="inline-block w-1.5 h-3 sm:w-2 sm:h-3.5 bg-[#c4956a] ml-0.5 align-middle"
    94→                  style={{ animation: 'blink-cursor 0.6s step-end infinite' }}
    95→                />
    96→              )}
    97→            </div>
    98→          ))}
    99→        </div>
   100→
   101→        {/* Title */}
   102→        {phase >= 1 && (
   103→          <motion.div
   104→            initial={{ opacity: 0, scale: 0.9 }}
   105→            animate={{ opacity: 1, scale: 1 }}
   106→            transition={{ duration: 1.0 }}
   107→            className="text-center mt-1 sm:mt-2 w-full"

   114→            </div>

   165→    <div className="pointer-events-none fixed inset-0 overflow-hidden z-0" aria-hidden>

   167→        <span
   168→          key={m.id}

   316→function QuoteBanner({ quotes }: { quotes: string[] }) {
   317→  const [idx, setIdx] = useState(0);
   318→  useEffect(() => {
   319→    const t = setInterval(() => setIdx((i) => (i + 1) % quotes.length), 8000);
   320→    return () => clearInterval(t);
   321→  }, [quotes.length]);
   322→
   323→  return (
   324→    <AnimatePresence mode="wait">
   325→      <motion.div
   326→        key={idx}
   327→        initial={{ opacity: 0, y: 10 }}
   328→        animate={{ opacity: 1, y: 0 }}
   329→        exit={{ opacity: 0, y: -10 }}
   330→        transition={{ duration: 0.5 }}
   331→        className="sk-lined p-4 rounded-lg old-quote text-[#5c4033] text-sm md:text-base text-center"
   332→      >
   333→        &ldquo;{quotes[idx]}&rdquo;
   334→      </motion.div>
   335→    </AnimatePresence>
   336→  );

   385→function getDecadeTheme(birthYear: number): string {
   386→  const decade = Math.floor(birthYear / 10) * 10;
   387→  if (decade === 1980) return '80s';
   388→  if (decade === 1990) return '90s';
   389→  if (decade === 2000) return '2000s';
   390→  return '';
   391→}

   676→export default function Home() {
   677→  const [birthYear, setBirthYear] = useState<number | null>(null);
   678→  const [loading, setLoading] = useState(true);
   679→
   680→  // Run one-time localStorage → IndexedDB migration on first load
   681→  useEffect(() => {
   682→    if (typeof window !== 'undefined') {
   683→      runMigration().catch(err => console.warn('[rewind] Migration error:', err));
   684→    }
   685→  }, []);
   686→
   687→  const handleLoadingDone = useCallback(() => {
   688→    setLoading(false);
   689→  }, []);
   690→
   691→  const handleSubmit = useCallback((year: number) => {
   692→    triggerOldLoader('Tuning into your era...');
   693→    setTimeout(() => {
   694→      setBirthYear(year);
   695→      window.scrollTo({ top: 0, behavior: 'smooth' });
   696→      hideOldLoader();
   697→    }, 1100);
   698→  }, []);
   699→
   700→  const handleBack = useCallback(() => {
   701→    triggerOldLoader('Rewinding...');
   702→    setTimeout(() => {
   703→      setBirthYear(null);
   704→      hideOldLoader();
   705→    }, 850);
   706→  }, []);
   707→
   708→  return (
   709→    <div className="min-h-screen flex flex-col" style={{ background: 'linear-gradient(180deg, #f5edd6 0%, #ede0c8 40%, #e8d8b8 100%)' }}>
   710→      {/* CRT Boot Loading Screen */}
   711→      {loading && <CrtLoadingScreen onDone={handleLoadingDone} />}
   712→
   713→      <AnimatePresence mode="wait">
   714→        {birthYear === null ? (
   715→          <motion.div
   716→            key="hero"
   717→            initial={{ opacity: 0 }}
   718→            animate={{ opacity: 1 }}
   719→            exit={{ opacity: 0, y: -20 }}
   720→            transition={{ duration: 0.4 }}
   721→            style={{ opacity: loading ? 0 : 1 }}
   722→          >
   723→            <HeroSection onSubmit={handleSubmit} />
   724→          </motion.div>
   725→        ) : (
   726→          <motion.div
   727→            key="dashboard"
   728→            initial={{ opacity: 0, y: 20 }}
   729→            animate={{ opacity: 1, y: 0 }}
   730→            exit={{ opacity: 0 }}
   731→            transition={{ duration: 0.5 }}
   732→            className="flex flex-col min-h-screen"
   733→          >
   734→            <Dashboard birthYear={birthYear} onBack={handleBack} />
   735→          </motion.div>
   736→        )}
   737→      </AnimatePresence>
   738→    </div>
   739→  );
   740→}
